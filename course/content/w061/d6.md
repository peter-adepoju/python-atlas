:::exercise id=mix_retry points=10 title="Retrying a block"
Write `attempts(times, exceptions)`: a generator yielding a context manager for each attempt, so that callers write `for attempt in attempts(3, (ConnectionError,)): with attempt: do_something()`. Each attempt's manager suppresses the listed exceptions **unless** it's the last attempt; after an attempt whose block succeeds, the generator stops (so the loop ends). Each manager exposes `number` (from 1).
@@starter

@@tests
def test_attempts():
    "generators and context managers"
    calls = []
    replies = iter([ConnectionError("a"), ConnectionError("b"), "ok"])
    for attempt in attempts(5, (ConnectionError,)):
        with attempt:
            calls.append(attempt.number)
            r = next(replies)
            if isinstance(r, Exception):
                raise r
    assert calls == [1, 2, 3]
    tries = []
    try:
        for attempt in attempts(2, (ConnectionError,)):
            with attempt:
                tries.append(attempt.number)
                raise ConnectionError("down")
        assert False
    except ConnectionError:
        pass
    assert tries == [1, 2]
    try:
        for attempt in attempts(3, (ConnectionError,)):
            with attempt:
                raise KeyError
        assert False
    except KeyError:
        pass
@@solution
from contextlib import contextmanager

class _Attempt:
    def __init__(self, number, last, exceptions):
        self.number, self.last, self.exceptions = number, last, exceptions
        self.ok = False

    def __enter__(self):
        return self

    def __exit__(self, exc_type, exc, tb):
        if exc_type is None:
            self.ok = True
            return False
        return not self.last and issubclass(exc_type, self.exceptions)

def attempts(times, exceptions):
    for n in range(1, times + 1):
        attempt = _Attempt(n, n == times, exceptions)
        yield attempt
        if attempt.ok:
            return

from contextlib import ExitStack
from itertools import batched

def split_lines(lines, size, prefix):
    names = []
    with ExitStack() as stack:
        for n, group in enumerate(batched(lines, size), 1):
            name = f"{prefix}{n}.txt"
            f = stack.enter_context(open(name, "w"))
            for line in group:
                f.write(line + "\n")
            names.append(name)
    return names

## Forge: a job runner

A job runner runs named jobs and promises to clean up after each one, however it ends. Write `JobRunner(workdir, clock=time.perf_counter)`:

- `@runner.job` or `@runner.job(name=...)` registers a function (returned unchanged); `runner.jobs()` lists names, sorted.
- `runner.run(name, **params)` (`KeyError` for an unknown job) calls `job(ctx, **params)` with a fresh **context** and returns a **result**, also appended to `runner.history`.
- Each run gets a new temporary folder **inside** `workdir` (create `workdir` if needed; `tempfile.TemporaryDirectory(dir=workdir)`), removed at the end of the run.
- The context offers: `ctx.scratch` (that folder's path); `ctx.path(name)`; `ctx.open(name, mode="r")`, opening a file in the scratch folder that's closed automatically at the end of the run; `ctx.on_cleanup(func, *args)`, to register cleanups; and `ctx.publish(name, text)`, writing `workdir/outputs/name` **atomically** (a `.tmp` file, then `os.replace`) and returning its path.
- Everything the job **prints** is captured.
- If the job raises an `Exception`, the run doesn't raise: it's recorded as failed, and every file the job published during that run is **removed**.
- The result has `name`, `status` (`"ok"` or `"failed"`), `value` (the job's return value, or `None`), `error` (the exception's type name, or `None`), `log` (the captured lines) and `seconds` (by the clock).

    runner = make("runs1")
    seen = {}
    @runner.job
    def count_words(ctx, text):

            f.write(text)

@@hint
Inside `run`: `with tempfile.TemporaryDirectory(dir=self.workdir) as scratch:`, then `try:` a `with ExitStack() as inner, redirect_stdout(buffer):` in which you build the context (giving it `inner` for `open` and `on_cleanup`, and a `published` list it appends to) and call the job; `except Exception` records the failure and deletes every path in `published`.
@@solution
import io, os, tempfile, time
from contextlib import ExitStack, redirect_stdout

class Context:
    def __init__(self, scratch, outputs, stack, published):
        self.scratch = scratch
        self._outputs = outputs
        self._stack = stack
        self.published = published

    def path(self, name):
        return os.path.join(self.scratch, name)

    def open(self, name, mode="r"):
        return self._stack.enter_context(open(self.path(name), mode))

    def on_cleanup(self, func, *args):
        self._stack.callback(func, *args)

    def publish(self, name, text):
        os.makedirs(self._outputs, exist_ok=True)
        final = os.path.join(self._outputs, name)
        tmp = final + ".tmp"
        with open(tmp, "w") as f:
            f.write(text)
        os.replace(tmp, final)
        self.published.append(final)
        return final

class Result:
    def __init__(self, name, status, value, error, log, seconds):
        self.name, self.status, self.value = name, status, value
        self.error, self.log, self.seconds = error, log, seconds

class JobRunner:
    def __init__(self, workdir, clock=time.perf_counter):
        self.workdir = workdir
        self.clock = clock
        self._jobs = {}
        self.history = []

    def job(self, func=None, *, name=None):
        if func is None:
            return lambda f: self.job(f, name=name)
        self._jobs[name or func.__name__] = func
        return func

    def jobs(self):
        return sorted(self._jobs)

    def run(self, name, **params):
        if name not in self._jobs:
            raise KeyError(name)
        outputs = os.path.join(self.workdir, "outputs")
        buffer = io.StringIO()
        start = self.clock()
        status, value, error = "ok", None, None
        published = []
        os.makedirs(self.workdir, exist_ok=True)
        with ExitStack() as outer:
            scratch = outer.enter_context(tempfile.TemporaryDirectory(dir=self.workdir))
            try:
                with ExitStack() as inner, redirect_stdout(buffer):
                    ctx = Context(scratch, outputs, inner, published)
                    value = self._jobs[name](ctx, **params)
            except Exception as e:
                status, error = "failed", type(e).__name__
                for path in published:
                    if os.path.exists(path):
                        os.remove(path)
        result = Result(name, status, value, error, buffer.getvalue().splitlines(), self.clock() - start)
        self.history.append(result)
        return result
:::

:::think title="Where the guarantees come from"
The runner promises four things: files closed, scratch removed, output captured, failures rolled back. None of them is written as cleanup code in the job; each comes from a context manager wrapped around it. The ExitStack closes whatever the job opened, the TemporaryDirectory removes the folder, redirect_stdout restores printing, and the rollback runs in the except clause before the temporary folder goes. Because every guarantee lives in a with statement, it holds on every path, including the ones nobody thought to test. That's the lesson of the week: put cleanup where the language guarantees it runs.
:::

## Forge bonus: layered settings

:::exercise id=forge2 points=30 title="Forge bonus: layered settings"
Programs often need settings that tests or parts of a program change temporarily. Write `Settings(**defaults)`:

- `get(key)` and `settings[key]`: the current value (`KeyError` for a key that has no default).
- `override(**changes)`: a context manager (that also works as a **decorator**, via `ContextDecorator`) pushing a layer of changes for the block, and popping it at the end, even after an error. Layers nest; the newest wins. Entering raises `KeyError` for an unknown key, and `RuntimeError("settings are frozen")` while frozen. Entering returns the settings.
- `frozen()`: a context manager (reentrant) during which no override can be entered.
- `snapshot()`: a dict of every key's current value; `depth()`: how many layers are active.
@@starter
from contextlib import contextmanager, ContextDecorator

class Settings:
    def __init__(self, **defaults):
        ___
@@tests
def test_layers():
    "nested overrides"
    s = Settings(colour="blue", size=10, debug=False)
    assert s.get("colour") == "blue" and s["size"] == 10
    with s.override(colour="red") as same:
        assert same is s and s["colour"] == "red" and s.depth() == 1
        with s.override(size=12, colour="green"):
            assert s.snapshot() == {"colour": "green", "size": 12, "debug": False}
        assert s.snapshot() == {"colour": "red", "size": 10, "debug": False}
    assert s.depth() == 0 and s["colour"] == "blue"
    try:
        with s.override(debug=True):
            raise ValueError
    except ValueError:
        pass
    assert s["debug"] is False
    try:
        with s.override(font="mono"):
            pass
        assert False
    except KeyError:
        pass
    assert s.depth() == 0

def test_decorator_and_frozen():
    "decorators and freezing"
    s = Settings(debug=False, level=1)
    @s.override(debug=True)
    def check():
        return s["debug"]
    assert check() is True and s["debug"] is False and check() is True
    with s.frozen():
        try:
            with s.override(level=2):
                pass
            assert False
        except RuntimeError as e:
            assert str(e) == "settings are frozen"
        with s.frozen():
            pass
        try:
            s.override(level=3).__enter__()
            assert False
        except RuntimeError:
            pass
    with s.override(level=5):
        assert s["level"] == 5
@@solution
from contextlib import contextmanager, ContextDecorator

class _Override(ContextDecorator):
    def __init__(self, settings, changes):
        self.settings, self.changes = settings, changes

    def __enter__(self):
        for key in self.changes:
            if key not in self.settings._defaults:
                raise KeyError(key)
        if self.settings._frozen:
            raise RuntimeError("settings are frozen")
        self.settings._layers.append(dict(self.changes))
        return self.settings

    def __exit__(self, *exc):
        self.settings._layers.pop()
        return False

class Settings:
    def __init__(self, **defaults):
        self._defaults = dict(defaults)
        self._layers = []
        self._frozen = 0

    def get(self, key):
        for layer in reversed(self._layers):
            if key in layer:
                return layer[key]
        return self._defaults[key]

    def __getitem__(self, key):
        return self.get(key)

    def snapshot(self):
        return {k: self.get(k) for k in self._defaults}

    def depth(self):
        return len(self._layers)

    def override(self, **changes):
        return _Override(self, changes)

    @contextmanager
    def frozen(self):
        self._frozen += 1
        try:
            yield self
        finally:
            self._frozen -= 1
