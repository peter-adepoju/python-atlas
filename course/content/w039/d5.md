class Interval:
    def __init__(self, lo, hi):
        if lo > hi:
            raise ValueError(f"lo must not be above hi: {lo} > {hi}")
        if lo == hi:
            lo = hi = 0
        self.lo, self.hi = lo, hi

    def __repr__(self):
        return f"Interval({self.lo}, {self.hi})"

    def __len__(self):
        return self.hi - self.lo

    def __contains__(self, n):
        return self.lo <= n < self.hi

    def __iter__(self):
        return iter(range(self.lo, self.hi))

    def __eq__(self, other):
        if not isinstance(other, Interval):
            return NotImplemented
        return (self.lo, self.hi) == (other.lo, other.hi)

    def __hash__(self):
        return hash((self.lo, self.hi))

    def __and__(self, other):
        if not isinstance(other, Interval):
            return NotImplemented
        lo, hi = max(self.lo, other.lo), min(self.hi, other.hi)
        return Interval(lo, hi) if lo < hi else Interval(0, 0)

from functools import total_ordering

class CurrencyError(ValueError):
    pass

@total_ordering
class Money:
    def __init__(self, pence, currency):
        self.pence = pence
        self.currency = currency.upper()

    def _same(self, other):
        if self.currency != other.currency:
            raise CurrencyError(f"can't mix {self.currency} and {other.currency}")

    def __add__(self, other):
        if not isinstance(other, Money):
            return NotImplemented
        self._same(other)
        return Money(self.pence + other.pence, self.currency)

    def __sub__(self, other):
        if not isinstance(other, Money):
            return NotImplemented
        self._same(other)
        return Money(self.pence - other.pence, self.currency)

    def __eq__(self, other):
        if not isinstance(other, Money):
            return NotImplemented
        return (self.pence, self.currency) == (other.pence, other.currency)

    def __hash__(self):
        return hash((self.pence, self.currency))

    def __lt__(self, other):
        if not isinstance(other, Money):
            return NotImplemented
        self._same(other)
        return self.pence < other.pence

    def __str__(self):
        sign = "-" if self.pence < 0 else ""
        pounds, rest = divmod(abs(self.pence), 100)
        return f"{self.currency} {sign}{pounds}.{rest:02d}"

Write `BadMove(ValueError)` and `Order`, a state machine with a transition table `MOVES` from `(state, event)` to the new state: placed + pay → paid; placed + cancel → cancelled; paid + ship → shipped; paid + cancel → refunded; shipped + deliver → delivered. @@solution
class BadMove(ValueError):
    pass

class Order:
    MOVES = {
        ("placed", "pay"): "paid",
        ("placed", "cancel"): "cancelled",
        ("paid", "ship"): "shipped",
        ("paid", "cancel"): "refunded",
        ("shipped", "deliver"): "delivered",
    }

    def __init__(self):
        self.state = "placed"
        self.history = []

    def handle(self, event):
        new = self.MOVES.get((self.state, event))
        if new is None:
            raise BadMove(f"can't {event} a {self.state} order")
        self.history.append(self.state)
        self.state = new
        return new

    def undo(self):
        if not self.history:
            return False
        self.state = self.history.pop()
        return True

@@solution
class Feed:
    def __init__(self):
        self._posts = []
        self._listeners = []

    def subscribe(self, listener):
        self._listeners.append(listener)

    def unsubscribe(self, listener):
        self._listeners.remove(listener)

    def post(self, text):
        if not text.strip():
            raise ValueError("a post can't be blank")
        self._posts.append(text)
        for listener in list(self._listeners):
            listener(text)

    def __iter__(self):
        return iter(self._posts[::-1])

    def __len__(self):
        return len(self._posts)

:::exercise id=b6 points=15 title="B6: Typed helpers" exam=yes
Write two generic functions with full type hints (use the `def name[T](...)` form):

- `group_by[T, K](items: list[T], key: Callable[[T], K]) -> dict[K, list[T]]`: the items grouped by their key, keys in the order first seen, items in their original order;
- `first_where[T](items: list[T], test: Callable[[T], bool]) -> T | None`: the first item passing the test, or `None`.
@@starter

___
@@tests
import typing
def test_values():
    "behaviour"
    assert group_by(["apple", "fig", "avocado", "kiwi"], lambda w: w[0]) == {"a": ["apple", "avocado"], "f": ["fig"], "k": ["kiwi"]}
    assert list(group_by([3, 1, 2], lambda n: n % 2)) == [1, 0] and group_by([], len) == {}
    assert first_where([1, 4, 6], lambda n: n % 2 == 0) == 4 and first_where([1, 3], lambda n: n > 5) is None

    "type hints"
    T, K = group_by.__type_params__
    a = group_by.__annotations__

@@solution
from typing import Callable

def group_by[T, K](items: list[T], key: Callable[[T], K]) -> dict[K, list[T]]:
    groups: dict[K, list[T]] = {}
    for item in items:
        groups.setdefault(key(item), []).append(item)
    return groups

def first_where[T](items: list[T], test: Callable[[T], bool]) -> T | None:
    for item in items:
        if test(item):
            return item
    return None

## Part C: a larger program

:::exercise id=c1 points=50 title="C: A league table" exam=yes
A five-a-side league records results typed by a volunteer, like `Red Lions 3-1 Blue Tigers`. Build it in layers.

**Errors.** `LeagueError(Exception)` is the base; `UnknownTeam(LeagueError, LookupError)`; `BadResult(LeagueError, ValueError)`.

**Domain.** `Result`, a frozen dataclass with `home`, `away`, `home_goals` and `away_goals`, and `__str__` giving `Red Lions 3-1 Blue Tigers`. `parse_result(line)` reads that form: the score is the one word made of digits, a `-`, and digits; the home team is the words before it and the away team the words after (both stripped, with single spaces). It raises `BadResult(f"bad result: {line!r}")` if there's no such word, either team is missing, or the two teams are the same.

**Points are a strategy**: a function `(goals_for, goals_against) -> points`. `standard_points` gives 3 for a win, 1 for a draw and 0 for a loss.

**Service.** `League(teams, points=standard_points)`, where `teams` is a list of names:

- `record(line)` parses the line, checks both teams are in the league (otherwise `UnknownTeam(f"no team {name!r}")`, home team checked first), stores the result and returns it;
- `undo()` removes and returns the most recent result, or raises `LeagueError("nothing to undo")`;
- `table()` returns one tuple per team, `(team, played, won, drawn, lost, goals_for, goals_against, points)`, sorted by points (most first), then goal difference (most first), then goals for (most first), then name (A to Z). Every team in the league appears, even with no games.

**Interface.** `handle(line, league)`: `table` gives one line per team, `1. Red Lions P2 W1 D1 L0 F4 A2 Pts4` (the position, then the figures); `undo` gives `removed Red Lions 3-1 Blue Tigers`; anything else is a result, giving `recorded Red Lions 3-1 Blue Tigers`. A `LeagueError` becomes `error: ` and its message.

from dataclasses import dataclass


class LeagueError(Exception):
    pass


class UnknownTeam(LeagueError, LookupError):
    pass


class BadResult(LeagueError, ValueError):
    pass


@dataclass(frozen=True)
class Result:
    home: str
    away: str
    home_goals: int
    away_goals: int

    def __str__(self):
        return f"{self.home} {self.home_goals}-{self.away_goals} {self.away}"


def _is_score(word):
    left, dash, right = word.partition("-")
    return bool(dash) and all(p.isascii() and p.isdigit() for p in (left, right))


def parse_result(line):
    words = line.split()
    scores = [i for i, w in enumerate(words) if _is_score(w)]
    if len(scores) != 1:
        raise BadResult(f"bad result: {line!r}")
    i = scores[0]
    home, away = " ".join(words[:i]), " ".join(words[i + 1:])
    if not home or not away or home == away:
        raise BadResult(f"bad result: {line!r}")
    left, _, right = words[i].partition("-")
    return Result(home, away, int(left), int(right))


def standard_points(goals_for, goals_against):
    if goals_for > goals_against:
        return 3
    return 1 if goals_for == goals_against else 0


class League:
    def __init__(self, teams, points=standard_points):
        self.teams = list(teams)
        self.points = points
        self.results = []

    def record(self, line):
        result = parse_result(line)
        for name in (result.home, result.away):
            if name not in self.teams:
                raise UnknownTeam(f"no team {name!r}")
        self.results.append(result)
        return result

    def undo(self):
        if not self.results:
            raise LeagueError("nothing to undo")
        return self.results.pop()

    def table(self):
        rows = {t: [0, 0, 0, 0, 0, 0, 0] for t in self.teams}    # played, won, drawn, lost, for, against, points
        for r in self.results:
            for team, scored, conceded in ((r.home, r.home_goals, r.away_goals), (r.away, r.away_goals, r.home_goals)):
                row = rows[team]
                row[0] += 1
                row[1 if scored > conceded else 2 if scored == conceded else 3] += 1
                row[4] += scored
                row[5] += conceded
                row[6] += self.points(scored, conceded)
        table = [(team, *row) for team, row in rows.items()]
        return sorted(table, key=lambda t: (-t[7], -(t[5] - t[6]), -t[5], t[0]))


def handle(line, league):
    try:
        command = line.strip()
        if command == "table":
            return "\n".join(f"{i}. {t} P{p} W{w} D{d} L{l} F{f} A{a} Pts{pts}"
                             for i, (t, p, w, d, l, f, a, pts) in enumerate(league.table(), start=1))
        if command == "undo":
            return f"removed {league.undo()}"
        return f"recorded {league.record(line)}"
    except LeagueError as err:
        return f"error: {err}"
