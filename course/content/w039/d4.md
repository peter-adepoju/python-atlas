import math

class Vector:
    def __init__(self, x, y):
        self.x, self.y = x, y

    def __repr__(self):
        return f"Vector({self.x}, {self.y})"

    def __eq__(self, other):
        if not isinstance(other, Vector):
            return NotImplemented
        return (self.x, self.y) == (other.x, other.y)

    def __hash__(self):
        return hash((self.x, self.y))

    def __add__(self, other):
        if not isinstance(other, Vector):
            return NotImplemented
        return Vector(self.x + other.x, self.y + other.y)

    def __sub__(self, other):
        if not isinstance(other, Vector):
            return NotImplemented
        return Vector(self.x - other.x, self.y - other.y)

    def __mul__(self, n):
        if not isinstance(n, (int, float)):
            return NotImplemented
        return Vector(self.x * n, self.y * n)

    __rmul__ = __mul__

    def __abs__(self):
        return math.hypot(self.x, self.y)

    def __bool__(self):
        return bool(self.x or self.y)

class Stock:
    def __init__(self):
        self._counts = {}

    def __getitem__(self, name):
        return self._counts.get(name, 0)

    def __setitem__(self, name, n):
        if n < 0:
            raise ValueError(f"count can't be negative: {n}")
        if n == 0:
            self._counts.pop(name, None)
        else:
            self._counts[name] = n

    def __len__(self):
        return len(self._counts)

    def __contains__(self, name):
        return name in self._counts

    def __iter__(self):
        return iter(sorted(self._counts))

    def add(self, other):
        result = Stock()
        for source in (self, other):
            for name in source:
                result[name] = result[name] + source[name]
        return result

Inside one `try`, unpack `line.split("|")` into three names (a `ValueError` if there aren't three), convert the id with `int`, look the priority up with `Priority[name.upper()]` (a `KeyError` if unknown), and raise your own `ValueError` for an empty title. @@solution
from dataclasses import dataclass
from enum import Enum

class Priority(Enum):
    LOW = 1
    MEDIUM = 2
    HIGH = 3

@dataclass(frozen=True)
class Ticket:
    ticket_id: int
    priority: Priority
    title: str

class TicketError(Exception):
    pass

def triage(lines):
    tickets = []
    for n, line in enumerate(lines, start=1):
        if not line.strip():
            continue
        try:
            raw_id, raw_priority, title = line.split("|")
            if not title.strip():
                raise ValueError("no title")
            tickets.append(Ticket(int(raw_id), Priority[raw_priority.strip().upper()], title.strip()))
        except (ValueError, KeyError) as err:
            raise TicketError(f"line {n}: {line!r}") from err
    return sorted(tickets, key=lambda t: (-t.priority.value, t.ticket_id))

@@solution
from typing import Protocol, runtime_checkable

@runtime_checkable
class Discount(Protocol):
    def apply(self, total: int) -> int: ...

class PercentOff:
    def __init__(self, percent):
        if not 1 <= percent <= 100:
            raise ValueError(f"percent must be from 1 to 100, not {percent}")
        self.percent = percent

    def apply(self, total):
        return total * (100 - self.percent) // 100

class AmountOff:
    def __init__(self, pence):
        self.pence = pence

    def apply(self, total):
        return max(0, total - self.pence)

class OverThreshold:
    def __init__(self, threshold, pence):
        self.threshold, self.pence = threshold, pence

    def apply(self, total):
        return total - self.pence if total >= self.threshold else total

def best_price(total, discounts):
    prices = [d.apply(total) for d in discounts if isinstance(d, Discount)]
    return min([total, *prices])

:::exercise id=c1 points=30 title="C: A car park"
Write a small car park system, in layers.

**Domain.** `ParkError(Exception)` is the base, with `Full(ParkError)` and `NotParked(ParkError, LookupError)`. `Stay`, a dataclass with `plate`, `arrived` (minutes since midnight) and `left` (default `None`). Fees are a **strategy**: a function `(minutes) -> pence`. Write `standard_fee`: free for the first 30 minutes, then 150p for each hour or part of an hour after that (so 31 minutes costs 150, 90 minutes costs 150, 91 minutes costs 300), at most 1200p.

**Service.** `CarPark(spaces, fee=standard_fee, clock)`, where `clock` is a function returning the current minute (a keyword argument, required). It keeps the stays, and has:

- `arrive(plate)`: the plate is upper-cased with spaces removed; a car already inside raises `ParkError("AB12CDE is already parked")`; no free space raises `Full("no spaces")`. Records a stay and notifies observers with `("arrived", plate)`.
- `leave(plate)`: the same plate cleaning; a car not inside raises `NotParked("AB12CDE isn't parked")`. Sets `left`, notifies `("left", plate)`, and returns the fee for the stay.
- `free()`: the number of free spaces; `subscribe(listener)` adds an observer.

**Interface.** `handle(line, park)` for lines `in PLATE`, `out PLATE` and `free`, replying `AB12CDE parked`, `AB12CDE paid £1.50` and `3 spaces free`. A `ParkError` becomes `error: ` and its message; anything else typed gets `unknown command`.

Keep the stays in a list and find the open stay for a plate with a loop (`left is None`). `free()` is `spaces` minus the number of open stays. For the fee, `hours = -(-(minutes - 30) // 60)` rounds up. In `handle`, split the line once with `line.split(maxsplit=1)`; a plate can contain spaces.
@@solution
from dataclasses import dataclass


class ParkError(Exception):
    pass


class Full(ParkError):
    pass


class NotParked(ParkError, LookupError):
    pass


@dataclass
class Stay:
    plate: str
    arrived: int
    left: int | None = None


def standard_fee(minutes):
    if minutes <= 30:
        return 0
    hours = -(-(minutes - 30) // 60)
    return min(150 * hours, 1200)


def clean(plate):
    return plate.replace(" ", "").upper()


class CarPark:
    def __init__(self, spaces, fee=standard_fee, *, clock):
        self.spaces = spaces
        self.fee = fee
        self.clock = clock
        self.stays = []
        self._listeners = []

    def subscribe(self, listener):
        self._listeners.append(listener)

    def _tell(self, kind, plate):
        for listener in list(self._listeners):
            listener(kind, plate)

    def _open_stay(self, plate):
        for stay in self.stays:
            if stay.plate == plate and stay.left is None:
                return stay
        return None

    def free(self):
        return self.spaces - sum(1 for s in self.stays if s.left is None)

    def arrive(self, plate):
        plate = clean(plate)
        if self._open_stay(plate):
            raise ParkError(f"{plate} is already parked")
        if self.free() == 0:
            raise Full("no spaces")
        self.stays.append(Stay(plate, self.clock()))
        self._tell("arrived", plate)

    def leave(self, plate):
        plate = clean(plate)
        stay = self._open_stay(plate)
        if stay is None:
            raise NotParked(f"{plate} isn't parked")
        stay.left = self.clock()
        self._tell("left", plate)
        return self.fee(stay.left - stay.arrived)


def handle(line, park):
    words = line.split(maxsplit=1)
    try:
        if words == ["free"]:
            return f"{park.free()} spaces free"
        if len(words) == 2 and words[0] == "in":
            park.arrive(words[1])
            return f"{clean(words[1])} parked"
        if len(words) == 2 and words[0] == "out":
            pence = park.leave(words[1])
            return f"{clean(words[1])} paid £{pence // 100}.{pence % 100:02d}"
    except ParkError as err:
        return f"error: {err}"
    return "unknown command"
