        divide(1, 0)
        assert False
    except ZeroDivisionError:
        pass

@@solution
import functools

def spy(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        try:
            result = func(*args, **kwargs)
        except Exception as e:
            wrapper.calls.append((args, kwargs, e))
            raise
        wrapper.calls.append((args, kwargs, result))
        return result
    wrapper.calls = []
    return wrapper

import random

def disagreement(fast, slow, seed=0, runs=500):
    rng = random.Random(seed)
    for _ in range(runs):
        xs = [rng.randint(0, 5) for _ in range(rng.randint(0, 6))]
        if fast(xs) != slow(xs):
            improved = True
            while improved:
                improved = False
                for i in range(len(xs)):
                    shorter = xs[:i] + xs[i + 1:]
                    if fast(shorter) != slow(shorter):
                        xs, improved = shorter, True
                        break
            return xs
    return None

## Forge: a booking system's test suite

`booking.py` (in the exercise) is a small room-booking service, built the Day 5 way: `Bookings(repo, clock, mailer)` takes its storage, clock and mailer as arguments. Its rules:

- `book(room, start, hours, who)` books `room` from `start` for `hours` whole hours. A booking lasts **1 to 3 hours**; it can't start in the **past** (before `clock.now()`); it must start at or after **8:00** and finish by **20:00** on the same day; and it can't **overlap** another booking of the same room (one ending at 10:00 and another starting at 10:00 don't overlap). Any broken rule raises `BookingError`. A successful booking is stored with `repo.add`, confirmed with `mailer.send(who, "Booked Oak at 10:00")` (room and start time), and returned.
- `cancel(booking, who)` removes a booking, but only for the person who made it (`BookingError` otherwise), and sends `"Cancelled Oak at 10:00"`.
- `free_hours(room, day)` lists the whole hours from 8 to 19 at which an hour-long booking of that room on that day would be free.
- `MemoryRepo` is the in-memory repository, with `add`, `remove` and `for_room(room)` (that room's bookings, sorted by start).

:::exercise id=forge points=50 title="Forge: a booking system's test suite"
Write a pytest suite for `booking.py` that passes on the correct code and catches **all eleven** broken versions. Use fixtures for the parts (a fixed clock, a mailer double, a repository, and the `Bookings` service built from them) and `parametrize` for families of cases. Work out every expected value from the rules above, not from the code.

=== booking.py
from dataclasses import dataclass
from datetime import datetime, timedelta


class BookingError(Exception):
    pass


@dataclass(frozen=True)
class Booking:
    room: str
    start: datetime
    end: datetime
    who: str


class MemoryRepo:
    def __init__(self):
        self._bookings = []

    def add(self, booking):
        self._bookings.append(booking)

    def remove(self, booking):
        self._bookings.remove(booking)

    def for_room(self, room):
        return sorted((b for b in self._bookings if b.room == room), key=lambda b: b.start)


class Bookings:
    def __init__(self, repo, clock, mailer, opening=8, closing=20, max_hours=3):
        self.repo, self.clock, self.mailer = repo, clock, mailer
        self.opening, self.closing, self.max_hours = opening, closing, max_hours

    def book(self, room, start, hours, who):
        if hours <= 0 or hours > self.max_hours:
            raise BookingError("a booking lasts from 1 to 3 hours")
        if start < self.clock.now():
            raise BookingError("that time has passed")
        end = start + timedelta(hours=hours)
        if start.hour < self.opening or end > start.replace(hour=self.closing, minute=0):
            raise BookingError("outside opening hours")
        for other in self.repo.for_room(room):
            if start < other.end and other.start < end:
                raise BookingError("that room is already booked")
        booking = Booking(room, start, end, who)
        self.repo.add(booking)
        self.mailer.send(who, f"Booked {room} at {start:%H:%M}")
        return booking

    def cancel(self, booking, who):
        if booking.who != who:
            raise BookingError("you can only cancel your own bookings")
        self.repo.remove(booking)
        self.mailer.send(who, f"Cancelled {booking.room} at {booking.start:%H:%M}")

    def free_hours(self, room, day):
        taken = self.repo.for_room(room)
        free = []
        for hour in range(self.opening, self.closing):
            slot_start = datetime(day.year, day.month, day.day, hour)
            slot_end = slot_start + timedelta(hours=1)
            if not any(slot_start < b.end and b.start < slot_end for b in taken):
                free.append(hour)
        return free

:::exercise id=forge2 points=30 title="Forge bonus: finding flaky tests"
Write two detectors. `flaky(history)`: `history` is a list of runs, each a dict with `"commit"` and `"results"` (a dict of test name to `"pass"` or `"fail"`). A test is **flaky** if it both passed and failed on the **same** commit. Return the sorted list of flaky tests. Return a sorted list of `(name, how)` pairs for each test whose result differs from its result alone, where `how` is `"in order"` or `"reversed"` (a test can appear twice).

def test_flaky():
    "same commit, different results"
    history = [
        {"commit": "a1", "results": {"test_x": "pass", "test_y": "pass", "test_z": "fail"}},
        {"commit": "a1", "results": {"test_x": "fail", "test_y": "pass", "test_z": "fail"}},

def test_order():
    "order-dependent tests"
    state = {"items": [], "logged_in": False}
    def reset():
        state["items"] = []
        state["logged_in"] = False
    def test_add():
        state["items"].append(1)
        assert len(state["items"]) == 1
    def test_login():
        state["logged_in"] = True
    def test_dashboard():
        assert state["logged_in"]
    def test_empty():
        assert state["items"] == []
    found = order_dependent([test_add, test_login, test_dashboard, test_empty], reset)
    assert found == [("test_dashboard", "in order"), ("test_empty", "in order")], found
    assert order_dependent([test_empty, test_add], reset) == [("test_empty", "reversed")]
    assert order_dependent([test_login], reset) == []
@@hint
For `flaky`, collect a set of outcomes per `(commit, test)` and pick the tests where any set has both. For `order_dependent`, write `outcome(test)` returning `"pass"` or `"fail"`, build the alone results, then compare each sequence's results with them.
@@solution
def flaky(history):
    seen = {}
    for run in history:
        for test, result in run["results"].items():
            seen.setdefault((run["commit"], test), set()).add(result)
    return sorted({test for (_, test), results in seen.items() if len(results) > 1})

def order_dependent(tests, reset):
    def outcome(test):
        try:
            test()
            return "pass"
        except AssertionError:
            return "fail"

    alone = {}
    for test in tests:
        reset()
        alone[test.__name__] = outcome(test)
    found = []
    for how, sequence in (("in order", tests), ("reversed", list(reversed(tests)))):
        reset()
        for test in sequence:
            if outcome(test) != alone[test.__name__]:
                found.append((test.__name__, how))
    return sorted(found)
