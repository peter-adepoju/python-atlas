import random
def brute(fuel, cost):
    n = len(fuel)
    for s in range(n):
        tank = 0
        for k in range(n):
            i = (s + k) % n
            tank += fuel[i] - cost[i]
            if tank < 0:
                break
        else:
            return s
    return -1
def test_fuel():
    "one pass"

    assert start_station([2, 3, 4], [3, 4, 3]) == -1 and start_station([5], [5]) == 0
    rng = random.Random(1)
    for _ in range(300):
        n = rng.randint(1, 7)
        fuel = [rng.randint(0, 6) for _ in range(n)]
        cost = [rng.randint(0, 6) for _ in range(n)]
        assert start_station(fuel, cost) == brute(fuel, cost), (fuel, cost)
@@solution
def start_station(fuel, cost):
    total = tank = start = 0
    for i in range(len(fuel)):
        gain = fuel[i] - cost[i]
        total += gain
        tank += gain
        if tank < 0:
            start, tank = i + 1, 0
    return start if total >= 0 else -1

    if len(path) != len(word) or len(set(path)) != len(path):

def find_word(grid, word):
    if not word:
        return []
    rows, cols = len(grid), len(grid[0])
    path, used = [], set()

    def search(r, c, k):
        if not (0 <= r < rows and 0 <= c < cols) or (r, c) in used or grid[r][c] != word[k]:
            return False
        path.append((r, c))
        used.add((r, c))
        if k == len(word) - 1:
            return True
        for nr, nc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            if search(nr, nc, k + 1):
                return True
        path.pop()
        used.discard((r, c))
        return False

    for r in range(rows):
        for c in range(cols):
            if search(r, c, 0):
                return path
    return None

## Forge: an exam timetabler

A school has to fit its exams into a few time slots so that no student sits two exams at once. Two exams **clash** if any student takes both; the clashes form a graph, and a timetable is a colouring of that graph with the slots as colours. Finding the fewest colours is a famously hard problem, so the timetabler drafts quickly with a greedy rule, then searches properly with backtracking.

Write `ExamTimetable(enrolments, slots)`, where `enrolments` maps each student to a list of exams and slots are numbered `0` to `slots - 1`:

- Raise `ValueError` if `slots` is less than 1. Store `slots`, and the clash graph as `graph`, a dict from every exam (anything a student takes) to the **set** of exams it clashes with. Unknown exams given to any method raise `KeyError`.
- `exams()` (sorted) and `clashes(exam)` (sorted).
- `fix(exam, slot)`: the exam must go in that slot (`ValueError` if the slot is out of range). Fixed exams are kept in `fixed`, a dict.
- `greedy()`: the Welsh-Powell draft: take exams in order of most clashes first (ties alphabetical), and give each the lowest slot no clashing exam already has. Return a dict; it may use more slots than allowed, and ignores fixed exams.
- `is_valid(timetable, slots=None)`: every exam is placed in a slot from 0 to slots − 1 (default `self.slots`), no clashing exams share a slot, and every fixed exam is where it was fixed.
- `schedule(slots=None)`: a valid timetable by backtracking, or `None`. Start from the fixed exams (return `None` if they clash). Each step, choose the unplaced exam whose clashing exams already use the **most different slots** (ties: more clashes, then alphabetical), and try its free slots in increasing order. When nothing is fixed, break symmetry: a new exam may use at most one slot beyond the highest slot used so far. Count every call of the search in `nodes`.
- `fewest_slots()`: the smallest number of slots a valid timetable needs (with the fixed exams; `0` with no exams).
- `back_to_back(timetable)`: the number of times a student has exams in two consecutive slots, summed over students (one count per pair of consecutive slots a student has).

Build the graph by joining every pair of exams on each student's list. In `schedule`, keep `timetable` as a dict that grows and shrinks: `pick()` scans the unplaced exams for the best key `(-len(used_slots), -len(clashes), name)`, and `place()` returns `True` when nothing is left to pick. The symmetry limit is `max(timetable.values(), default=-1) + 2`. `fewest_slots` can call `schedule(k)` for k = 1, 2, 3, ... (starting above the highest fixed slot).
@@solution
class ExamTimetable:
    def __init__(self, enrolments, slots):
        if slots < 1:
            raise ValueError("at least one slot is needed")
        self.slots = slots
        self.enrolments = {s: sorted(set(es)) for s, es in enrolments.items()}
        self.graph = {}
        for exams in self.enrolments.values():
            for e in exams:
                self.graph.setdefault(e, set())
                for f in exams:
                    if f != e:
                        self.graph[e].add(f)
        self.fixed = {}
        self.nodes = 0

    def _known(self, exam):
        if exam not in self.graph:
            raise KeyError(exam)

    def exams(self):
        return sorted(self.graph)

    def clashes(self, exam):
        self._known(exam)
        return sorted(self.graph[exam])

    def fix(self, exam, slot):
        self._known(exam)
        if not 0 <= slot < self.slots:
            raise ValueError(f"slot {slot} is outside 0 to {self.slots - 1}")
        self.fixed[exam] = slot

    def greedy(self):
        order = sorted(self.graph, key=lambda e: (-len(self.graph[e]), e))
        timetable = {}
        for e in order:
            taken = {timetable[f] for f in self.graph[e] if f in timetable}
            slot = 0
            while slot in taken:
                slot += 1
            timetable[e] = slot
        return timetable

    def is_valid(self, timetable, slots=None):
        slots = self.slots if slots is None else slots
        if set(timetable) != set(self.graph):
            return False
        for e, s in timetable.items():
            if not 0 <= s < slots:
                return False
            if any(timetable[f] == s for f in self.graph[e]):
                return False
        return all(timetable[e] == s for e, s in self.fixed.items())

    def schedule(self, slots=None):
        slots = self.slots if slots is None else slots
        self.nodes = 0
        timetable = dict(self.fixed)
        for e, s in self.fixed.items():
            if s >= slots or any(timetable.get(f) == s for f in self.graph[e]):
                return None
        symmetric = not self.fixed

        def pick():
            best, best_key = None, None
            for e in self.graph:
                if e in timetable:
                    continue
                used = {timetable[f] for f in self.graph[e] if f in timetable}
                key = (-len(used), -len(self.graph[e]), e)
                if best_key is None or key < best_key:
                    best, best_key = e, key
            return best

        def place():
            self.nodes += 1
            exam = pick()
            if exam is None:
                return True
            used = {timetable[f] for f in self.graph[exam] if f in timetable}
            limit = slots
            if symmetric:
                limit = min(slots, max(timetable.values(), default=-1) + 2)
            for s in range(limit):
                if s in used:
                    continue
                timetable[exam] = s
                if place():
                    return True
                del timetable[exam]
            return False

        return dict(timetable) if place() else None

    def fewest_slots(self):
        if not self.graph:
            return 0
        k = max(self.fixed.values(), default=0) + 1
        while self.schedule(k) is None:
            k += 1
        return k

    def back_to_back(self, timetable):
        count = 0
        for exams in self.enrolments.values():
            taken = {timetable[e] for e in exams}
            count += sum(1 for s in taken if s + 1 in taken)
        return count

:::think title="Draft, then search"
The timetabler works in two stages, and real schedulers usually do the same. A greedy rule gives a decent answer almost instantly and tells you roughly how many slots you need. A backtracking search then finds a valid answer with fewer slots, or proves there isn't one, and it's only fast because of this week's pruning ideas: placing the most constrained exam first (as with sudoku cells), refusing slots that clash the moment they're tried, and treating slots as interchangeable so the same timetable with relabelled slots is never searched twice. With hundreds of exams even that gives out, and solvers turn to heuristics that improve a timetable step by step without promising the best.
:::

## Forge bonus: a coin box

:::exercise id=forge2 points=30 title="Forge bonus: a coin box"
A till holds a limited number of each coin. Greedy change-giving (largest coin first) works for full tills of most currencies, but with a limited supply it can fail even when change is possible. Write `CoinBox(coins)`, where `coins` maps a coin's value to how many the till holds:

- Raise `ValueError` for a value that isn't positive or a count that's negative. Store the coins with a positive count in `coins`. Any method given a negative amount raises `ValueError`.
- `total()`: the value of everything in the till.
- `greedy_change(amount)`: largest coins first, as many as fit and are available; a dict of value to count (only coins used), or `None` if it doesn't reach the amount exactly.
- `best_change(amount)`: the change in the **fewest coins**, by branch and bound: take coin values largest first, try each count from the most possible down to zero, and prune when the coins used so far plus ⌈amount left ÷ current value⌉ can't beat the best found. `{}` for 0, `None` if impossible. Count every call of the search in `nodes`.
- `pay(amount)`: take the best change out of the till and return it; `ValueError` (with the till unchanged) if it can't be made.
- `ways(amount)`: how many different collections of coins from the till make the amount exactly (dynamic programming over the coin values, a preview of next week).

import itertools, random, time

def brute(coins, amount):
    values = sorted(coins)
    best, n_ways = None, 0
    for combo in itertools.product(*[range(coins[v] + 1) for v in values]):
        if sum(c * v for c, v in zip(combo, values)) == amount:
            n_ways += 1
            if best is None or sum(combo) < best:
                best = sum(combo)
    return best, n_ways

class CoinBox:
    def __init__(self, coins):
        for value, count in coins.items():
            if value <= 0 or count < 0:
                raise ValueError(f"bad coins: {count} of {value}")
        self.coins = {v: c for v, c in coins.items() if c > 0}
        self.nodes = 0

    def total(self):
        return sum(v * c for v, c in self.coins.items())

    def _check(self, amount):
        if amount < 0:
            raise ValueError("amounts can't be negative")

    def greedy_change(self, amount):
        self._check(amount)
        out = {}
        for v in sorted(self.coins, reverse=True):
            take = min(self.coins[v], amount // v)
            if take:
                out[v] = take
                amount -= take * v
        return out if amount == 0 else None

    def best_change(self, amount):
        self._check(amount)
        values = sorted(self.coins, reverse=True)
        best = [None, None]
        current = {}
        self.nodes = 0

        def search(i, left, used):
            self.nodes += 1
            if left == 0:
                if best[0] is None or used < best[0]:
                    best[0], best[1] = used, dict(current)
                return
            if i == len(values):
                return
            v = values[i]
            if best[0] is not None and used + -(-left // v) >= best[0]:
                return                          # even all coins this size can't do better
            for take in range(min(self.coins[v], left // v), -1, -1):
                if take:
                    current[v] = take
                search(i + 1, left - take * v, used + take)
                current.pop(v, None)

        search(0, amount, 0)
        return best[1]

    def pay(self, amount):
        change = self.best_change(amount)
        if change is None:
            raise ValueError(f"can't make {amount} exactly")
        for v, c in change.items():
            self.coins[v] -= c
            if self.coins[v] == 0:
                del self.coins[v]
        return change

    def ways(self, amount):
        self._check(amount)
        values = sorted(self.coins)
        counts = [0] * (amount + 1)
        counts[0] = 1
        for v in values:
            new = [0] * (amount + 1)
            for a in range(amount + 1):
                for k in range(self.coins[v] + 1):
                    if k * v > a:
                        break
                    new[a] += counts[a - k * v]
            counts = new
        return counts[amount]
