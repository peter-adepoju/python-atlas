def products_except_self(nums):
    out = [1] * len(nums)
    left = 1
    for i, x in enumerate(nums):
        out[i] = left
        left *= x
    right = 1
    for i in range(len(nums) - 1, -1, -1):
        out[i] *= right
        right *= nums[i]
    return out

from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self._d = OrderedDict()

    def get(self, key):
        if key not in self._d:
            return None
        self._d.move_to_end(key)
        return self._d[key]

    def put(self, key, value):
        self._d[key] = value
        self._d.move_to_end(key)
        if len(self._d) > self.capacity:
            self._d.popitem(last=False)

    def keys(self):
        return list(self._d)

from collections import deque

def _window(nums, k, better):
    out, dq = [], deque()
    for i, x in enumerate(nums):
        while dq and not better(nums[dq[-1]], x):
            dq.pop()
        dq.append(i)
        if dq[0] <= i - k:
            dq.popleft()
        if i >= k - 1:
            out.append(nums[dq[0]])
    return out

def window_max(nums, k):
    return _window(nums, k, lambda a, b: a > b)

def window_ranges(nums, k):
    highs = _window(nums, k, lambda a, b: a > b)
    lows = _window(nums, k, lambda a, b: a < b)
    return [h - l for h, l in zip(highs, lows)]

from collections import deque

class T:
    def __init__(self, v, l=None, r=None):
        self.v, self.l, self.r = v, l, r

def save(root):
    out, queue = [], deque([root])
    while queue:
        n = queue.popleft()
        if n is None:
            out.append("#")
        else:
            out.append(str(n.v))
            queue.append(n.l)
            queue.append(n.r)
    while out and out[-1] == "#":
        out.pop()
    return ",".join(out)

def load(text):
    if not text:
        return None
    items = text.split(",")
    root = T(int(items[0]))
    queue, i = deque([root]), 1
    while queue and i < len(items):
        node = queue.popleft()
        for side in ("l", "r"):
            if i < len(items) and items[i] != "#":
                child = T(int(items[i]))
                setattr(node, side, child)
                queue.append(child)
            i += 1
    return root

import random, time
def slow_has(grid, w):
    R, C = len(grid), len(grid[0]) if grid else 0
    def go(r, c, i, used):
        if grid[r][c] != w[i]:
            return False
        if i == len(w) - 1:
            return True
        used.add((r, c))
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < R and 0 <= nc < C and (nr, nc) not in used and go(nr, nc, i + 1, used):
                used.discard((r, c))
                return True
        used.discard((r, c))
        return False

def find_words(grid, words):
    root = {}
    for w in words:
        node = root
        for ch in w:
            node = node.setdefault(ch, {})
        node["$"] = w
    R = len(grid)
    C = len(grid[0]) if R else 0
    found = set()

    def search(r, c, node, used):
        node = node.get(grid[r][c])
        if node is None:
            return
        if "$" in node:
            found.add(node["$"])
        used.add((r, c))
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < R and 0 <= nc < C and (nr, nc) not in used:
                search(nr, nc, node, used)
        used.discard((r, c))

    for r in range(R):
        for c in range(C):
            search(r, c, root, set())
    return sorted(found)

:::bughunt id=b6 points=15 title="B6: Three bugs" exam=yes
`top_words(text, k)` should return the k most common words (lower-cased runs of letters), most common first, ties alphabetical, keeping a heap of at most k entries. It has three bugs: it counts upper- and lower-case forms separately, it keeps the k **least** common words, and it breaks ties the wrong way round. Fix them so this prints `['the', 'and', 'cat']`.
@@starter
import heapq, re
from collections import Counter

class Entry:
    def __init__(self, count, word):
        self.count, self.word = count, word
    def __lt__(self, other):
        # "less" means "worse": fewer uses, or the same uses and later in the alphabet
        if self.count != other.count:
            return self.count < other.count
        return self.word < other.word

def top_words(text, k):
    counts = Counter(re.findall(r"[a-zA-Z]+", text))
    heap = []
    for word, count in counts.items():
        e = Entry(count, word)
        if len(heap) < k:
            heapq.heappush(heap, e)
        elif e < heap[0]:
            heapq.heapreplace(heap, e)
    return [e.word for e in sorted(heap, reverse=True)]
