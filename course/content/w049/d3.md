@@hint
Sort by start. Keep a list of merged intervals; for each interval, if it starts at or before the last merged interval's end, extend that end; otherwise start a new merged interval.
@@solution
def merge_intervals(intervals):
    merged = []
    for start, end in sorted(intervals):
        if merged and start <= merged[-1][1]:
            merged[-1] = (merged[-1][0], max(merged[-1][1], end))
        else:
            merged.append((start, end))
    return merged

def total_covered(intervals):
    return sum(end - start for start, end in merge_intervals(intervals))

@@solution
class MinStack:
    def __init__(self):
        self._items, self._mins = [], []

    def __len__(self):
        return len(self._items)

    def push(self, x):
        self._items.append(x)
        self._mins.append(x if not self._mins else min(x, self._mins[-1]))

    def pop(self):
        if not self._items:
            raise IndexError("pop from an empty stack")
        self._mins.pop()
        return self._items.pop()

    def top(self):
        if not self._items:
            raise IndexError("empty stack")
        return self._items[-1]

    def get_min(self):
        if not self._mins:
            raise IndexError("empty stack")
        return self._mins[-1]

@@solution
class Node:
    def __init__(self, value, next=None, other=None):
        self.value, self.next, self.other = value, next, other

def deep_copy(head):
    copies, n = {}, head
    while n:
        copies[n] = Node(n.value)
        n = n.next
    n = head
    while n:
        copies[n].next = copies.get(n.next)
        copies[n].other = copies.get(n.other)
        n = n.next
    return copies.get(head)

Heap entries `(value, row, col)`. For the staircase, start at the last row, first column: if the value is ≤ x, the whole column above it is too, so add `row + 1` and move right; otherwise move up.
@@solution
import heapq

def kth_smallest(grid, k):
    n = len(grid)
    if not 1 <= k <= n * n:
        raise IndexError("k out of range")
    heap = [(grid[r][0], r, 0) for r in range(n)]
    heapq.heapify(heap)
    for _ in range(k - 1):
        value, r, c = heapq.heappop(heap)
        if c + 1 < n:
            heapq.heappush(heap, (grid[r][c + 1], r, c + 1))
    return heap[0][0]

def count_at_most(grid, x):
    n = len(grid)
    r, c, count = n - 1, 0, 0
    while r >= 0 and c < n:
        if grid[r][c] <= x:
            count += r + 1
            c += 1
        else:
            r -= 1
    return count

        want = 0

                want = dist
                break
            nxt = []

                for v in words:

                        seen.add(v)
                        nxt.append(v)

@@hint
Build `patterns = {}` mapping e.g. `"h*t"` to every word matching it. In the search, the neighbours of w are all words in `patterns[w[:i] + "*" + w[i + 1:]]` for each i. Mark words as seen when you add them to the queue, not when you take them out.
@@solution
from collections import deque

def _search(start, end, words):
    if end not in words:
        return {}
    patterns = {}
    for w in set(words):
        for i in range(len(w)):
            patterns.setdefault(w[:i] + "*" + w[i + 1:], []).append(w)
    parent = {start: None}
    queue = deque([start])
    while queue:
        w = queue.popleft()
        if w == end:
            return parent
        for i in range(len(w)):
            for v in patterns.get(w[:i] + "*" + w[i + 1:], ()):
                if v not in parent:
                    parent[v] = w
                    queue.append(v)
    return {}

def ladder(start, end, words):
    parent = _search(start, end, words)
    if end not in parent:
        return []
    path, w = [], end
    while w is not None:
        path.append(w)
        w = parent[w]
    return path[::-1]

def ladder_length(start, end, words):
    return len(ladder(start, end, words))
