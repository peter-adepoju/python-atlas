import random
def brute(nums):
    n = len(nums)
    return max(sum(nums[(s + k) % n] for k in range(L)) for s in range(n) for L in range(1, n + 1))
def test_ring():
    "wrapping runs"
    assert best_circular([1, -2, 3, -2]) == 3 and best_circular([5, -3, 5]) == 10
    assert best_circular([-3, -2, -3]) == -2 and best_circular([7]) == 7
    rng = random.Random(1)

@@solution
def best_circular(nums):
    best_here = best = nums[0]
    low_here = low = nums[0]
    for x in nums[1:]:
        best_here = max(x, best_here + x)
        best = max(best, best_here)
        low_here = min(x, low_here + x)
        low = min(low, low_here)
    if best < 0:
        return best
    return max(best, sum(nums) - low)

def paint_ways(n, k):
    if n == 0:
        return 0
    same, different = 0, k             # after one post
    for _ in range(1, n):
        same, different = different, (same + different) * (k - 1)
    return same + different

## Forge: a festival planner

A town festival has one stage and a budget for fees. Each act has a fee, an expected crowd, and a time slot from `start` to `end` (whole hours; an act ending at 14 and one starting at 14 don't clash). The organisers want the biggest total crowd from acts that don't overlap, within the budget.

It's two of this week's ideas at once. Sort acts by **end time**: the weighted interval schedule (Week 52 Day 2's challenge) says f(i) chooses between leaving act i out and taking it, which jumps back to the last act that ends by its start. The budget adds the knapsack's second dimension.

Write `FestivalPlanner(acts, budget)`, where `acts` maps a name to `(fee, crowd, start, end)`:

- Raise `ValueError` for a negative budget, fee or crowd, or an act whose end isn't after its start.
- `best_lineup()`: `(crowd, names)`, the largest total crowd from non-overlapping acts whose fees fit the budget, with the names in order of start time. Use a table best(i, b) over the first i acts in end order and budget b, with `bisect` to find how many acts end by each act's start; walk back to find the acts.
- `best_unlimited()`: the largest crowd ignoring the budget (one-dimensional).
- `cheapest_for(target)`: the least total fee for non-overlapping acts with a total crowd of **at least** `target`, or `None`. Swap the roles: a table of the least fee over (acts considered, crowd still needed), where the crowd still needed never drops below 0.
- `count_lineups()`: how many sets of non-overlapping acts (including the empty set) fit the budget.

In `__init__`, sort the names by `(end, name)` and store, for each act in that order, `bisect.bisect_right(ends, start)`: the number of acts finishing by its start. Then every method is a table over i from 0 to n: leave act i − 1 out (row i − 1) or take it (row `before[i - 1]`). For the walk back, when taking the act, jump to row `before[i - 1]` rather than i − 1.
@@solution
import bisect

class FestivalPlanner:
    def __init__(self, acts, budget):
        if budget < 0:
            raise ValueError("the budget can't be negative")
        for name, (fee, crowd, start, end) in acts.items():
            if fee < 0 or crowd < 0 or end <= start:
                raise ValueError(f"bad act {name!r}")
        self.budget = budget
        self.acts = dict(acts)
        self.order = sorted(self.acts, key=lambda a: (self.acts[a][3], a))
        ends = [self.acts[a][3] for a in self.order]
        # before[i]: how many acts (in end order) finish by the time act i starts
        self.before = [bisect.bisect_right(ends, self.acts[a][2]) for a in self.order]

    def best_lineup(self):
        n, B = len(self.order), self.budget
        best = [[0] * (B + 1) for _ in range(n + 1)]
        for i in range(1, n + 1):
            fee, crowd, _, _ = self.acts[self.order[i - 1]]
            p = self.before[i - 1]
            for b in range(B + 1):
                best[i][b] = best[i - 1][b]
                if fee <= b and best[p][b - fee] + crowd > best[i][b]:
                    best[i][b] = best[p][b - fee] + crowd
        names, i, b = [], n, B
        while i > 0:
            if best[i][b] == best[i - 1][b]:
                i -= 1
            else:
                name = self.order[i - 1]
                names.append(name)
                b -= self.acts[name][0]
                i = self.before[i - 1]
        names.sort(key=lambda a: self.acts[a][2])
        return best[n][B], names

    def best_unlimited(self):
        n = len(self.order)
        best = [0] * (n + 1)
        for i in range(1, n + 1):
            crowd = self.acts[self.order[i - 1]][1]
            best[i] = max(best[i - 1], best[self.before[i - 1]] + crowd)
        return best[n]

    def cheapest_for(self, target):
        if target <= 0:
            return 0
        INF = float("inf")
        n = len(self.order)
        fee_for = [[INF] * (target + 1) for _ in range(n + 1)]
        for i in range(n + 1):
            fee_for[i][0] = 0
        for i in range(1, n + 1):
            fee, crowd, _, _ = self.acts[self.order[i - 1]]
            p = self.before[i - 1]
            for k in range(1, target + 1):
                fee_for[i][k] = min(fee_for[i - 1][k], fee_for[p][max(0, k - crowd)] + fee)
        answer = fee_for[n][target]
        return None if answer == INF else answer

    def count_lineups(self):
        n, B = len(self.order), self.budget
        count = [[1] * (B + 1)] + [[0] * (B + 1) for _ in range(n)]
        for i in range(1, n + 1):
            fee = self.acts[self.order[i - 1]][0]
            p = self.before[i - 1]
            for b in range(B + 1):
                count[i][b] = count[i - 1][b] + (count[p][b - fee] if fee <= b else 0)
        return count[n][B]
:::

:::think title="Two dimensions from two constraints"
The festival table has one dimension for each thing the question constrains: which acts have been considered (because of the time slots) and how much budget is left. That's the general rule for designing a DP state when a problem has several limits: each independent limit becomes a coordinate, and the table's size is the product of their ranges. It's why a third limit (say, a cap on the number of acts) is affordable when its range is small, and why limits measured in millions make DP tables impossible, sending you back to last week's branch and bound.
:::

## Forge bonus: a climbing wall

:::exercise id=forge2 points=30 title="Forge bonus: a climbing wall"
A climbing wall is a grid of holds, given as strings from the **top** row to the bottom: a digit is the effort of using that hold, and `#` is no hold. A climber starts on any hold in the bottom row and moves up one row at a time, to the hold straight above or diagonally up-left or up-right, until reaching the top row. Every hold used, including the first and last, adds its effort. Write `ClimbingWall(rows)`:

- Raise `ValueError` if the rows are empty, of different lengths, or contain anything but digits and `#`.
- `easiest(max_sideways=None)`: the least total effort to climb the wall, or `None` if it can't be climbed. With `max_sideways`, at most that many of the moves may be diagonal; add the number of diagonal moves used to the state.
- `route()`: the columns of the holds on one easiest climb, from the bottom row to the top, or `None`.
- `routes()`: the number of different climbs, using one row of counts.
@@starter

import itertools, random

WALL = ["31#5", "2914", "5121", "9#13"]

def all_routes(rows):
    H, W = len(rows), len(rows[0])
    out = []
    def go(r, c, path):
        if rows[r][c] == "#":
            return
        path = path + [c]
        if r == 0:
            out.append(path)
            return
        for nc in (c - 1, c, c + 1):
            if 0 <= nc < W:
                go(r - 1, nc, path)
    for c in range(W):
        go(H - 1, c, [])
    return out

def effort(rows, path):
    H = len(rows)
    return sum(int(rows[H - 1 - i][c]) for i, c in enumerate(path))

def sideways(path):
    return sum(1 for a, b in zip(path, path[1:]) if a != b)

def test_checks():
    "walls"
    for bad in ([], ["12", "1"], ["1a"]):
        try:
            ClimbingWall(bad)
            assert False, "ValueError expected"
        except ValueError:

def test_easiest():
    "least effort"
    wall = ClimbingWall(WALL)
    assert wall.easiest() == 4 and wall.route() in all_routes(WALL) and effort(WALL, wall.route()) == 4
    assert wall.routes() == len(all_routes(WALL)) == 32
    assert wall.easiest(0) == 13 and wall.easiest(1) == 5
    assert ClimbingWall(["#"]).easiest() is None and ClimbingWall(["#"]).route() is None
    assert ClimbingWall(["7"]).route() == [0] and ClimbingWall(["7"]).routes() == 1

def test_random():

        H, W = rng.randint(1, 5), rng.randint(1, 4)

        wall = ClimbingWall(rows)

        assert wall.routes() == len(paths)
        if not paths:

        low = min(effort(rows, p) for p in paths)
        assert wall.easiest() == low
        route = wall.route()

        for k in range(0, 3):

            assert wall.easiest(k) == (min(ok) if ok else None)

@@solution
class ClimbingWall:
    def __init__(self, rows):
        if not rows or len({len(r) for r in rows}) != 1:
            raise ValueError("the wall must be a non-empty rectangle")
        self.rows = list(rows)
        self.height, self.width = len(rows), len(rows[0])
        for row in rows:
            for ch in row:
                if ch != "#" and not ch.isdigit():
                    raise ValueError(f"bad hold {ch!r}")

    def _cost(self, r, c):
        ch = self.rows[r][c]
        return None if ch == "#" else int(ch)

    def _table(self, max_sideways=None):
        """best[r][c][s]: least effort from the bottom to (r, c) using s sideways moves."""
        H, W = self.height, self.width
        S = (H if max_sideways is None else max_sideways) + 1
        INF = float("inf")
        best = [[[INF] * S for _ in range(W)] for _ in range(H)]
        for c in range(W):
            cost = self._cost(H - 1, c)
            if cost is not None:
                best[H - 1][c][0] = cost
        for r in range(H - 2, -1, -1):
            for c in range(W):
                cost = self._cost(r, c)
                if cost is None:
                    continue
                for s in range(S):
                    options = [best[r + 1][c][s]]
                    if s > 0:
                        for pc in (c - 1, c + 1):
                            if 0 <= pc < W:
                                options.append(best[r + 1][pc][s - 1])
                    low = min(options)
                    if low < INF:
                        best[r][c][s] = low + cost
        return best

    def easiest(self, max_sideways=None):
        best = self._table(max_sideways)
        low = min(min(cell) for cell in best[0])
        return None if low == float("inf") else low

    def route(self):
        best = self._table()
        INF = float("inf")
        low = self.easiest()
        if low is None:
            return None
        c = min(range(self.width), key=lambda c: (min(best[0][c]), c))
        path = [c]
        for r in range(0, self.height - 1):
            need = min(best[r][c]) - self._cost(r, c)
            for pc in (c, c - 1, c + 1):
                if 0 <= pc < self.width and min(best[r + 1][pc]) == need:
                    c = pc
                    break
            path.append(c)
        return path[::-1]

    def routes(self):
        H, W = self.height, self.width
        ways = [1 if self._cost(H - 1, c) is not None else 0 for c in range(W)]
        for r in range(H - 2, -1, -1):
            ways = [0 if self._cost(r, c) is None else
                    sum(ways[pc] for pc in (c - 1, c, c + 1) if 0 <= pc < W)
                    for c in range(W)]
        return sum(ways)
