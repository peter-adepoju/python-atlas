def longest_mountain(a):
    n = len(a)
    up, down = [1] * n, [1] * n
    for i in range(n):
        for j in range(i):
            if a[j] < a[i]:
                up[i] = max(up[i], up[j] + 1)
    for i in range(n - 1, -1, -1):
        for j in range(i + 1, n):
            if a[j] < a[i]:
                down[i] = max(down[i], down[j] + 1)
    return max((up[i] + down[i] - 1 for i in range(n)), default=0)

def longest_chain(words):

## Forge: a document comparer

A writing group swaps drafts and wants to see what changed between two versions, word by word. Write `DocumentCompare(old_text, new_text)`:

- Split both texts into words with `split()` (any whitespace separates words), kept as `old` and `new`. Build the LCS table over the words once, in `__init__`.
- `common()`: the number of words in a longest common subsequence.
- `similarity()`: `2 × common / (len(old) + len(new))`, rounded to 3 places; `1.0` when both are empty.
- `word_diff()`: a list of `(mark, word)` pairs, with marks `"="` (kept), `"-"` (removed) and `"+"` (added), from walking back through the table: a match goes diagonally; otherwise an addition if moving left keeps at least as much LCS as moving up, else a removal (as in Day 1).
- `changes()`: the diff grouped into runs of non-kept words, each `(kind, removed, added)` with the words joined by spaces, where kind is `"replace"` (both), `"delete"` (only removed) or `"insert"` (only added).
- `render()`: the new text with changes marked like Git's word diff: removed runs as `[-...-]`, added runs as `{+...+}` (removed first), everything separated by single spaces.
- `edit_distance()`: the word-level edit distance, with two rows.
- `shared_passage()`: the longest run of consecutive words appearing in both texts, joined by spaces (the first one found scanning the old text's end positions in order); `""` if none.

def lcs_len(a, b):
    prev = [0] * (len(b) + 1)
    for i in range(1, len(a) + 1):
        cur = [0] * (len(b) + 1)
        for j in range(1, len(b) + 1):
            cur[j] = prev[j - 1] + 1 if a[i - 1] == b[j - 1] else max(prev[j], cur[j - 1])
        prev = cur
    return prev[-1]

Build `changes` and `render` on top of `word_diff`: walk the pairs, collecting removed and added words into two lists, and "flush" them (emit a change or the marked-up text) whenever a kept word arrives, and once more at the end.
@@solution
class DocumentCompare:
    def __init__(self, old_text, new_text):
        self.old = old_text.split()
        self.new = new_text.split()
        n, m = len(self.old), len(self.new)
        L = [[0] * (m + 1) for _ in range(n + 1)]
        for i in range(1, n + 1):
            for j in range(1, m + 1):
                if self.old[i - 1] == self.new[j - 1]:
                    L[i][j] = L[i - 1][j - 1] + 1
                else:
                    L[i][j] = max(L[i - 1][j], L[i][j - 1])
        self.table = L

    def common(self):
        return self.table[-1][-1]

    def similarity(self):
        total = len(self.old) + len(self.new)
        if total == 0:
            return 1.0
        return round(2 * self.common() / total, 3)

    def word_diff(self):
        L, old, new = self.table, self.old, self.new
        out, i, j = [], len(old), len(new)
        while i or j:
            if i and j and old[i - 1] == new[j - 1]:
                out.append(("=", old[i - 1]))
                i, j = i - 1, j - 1
            elif j and (not i or L[i][j - 1] >= L[i - 1][j]):
                out.append(("+", new[j - 1]))
                j -= 1
            else:
                out.append(("-", old[i - 1]))
                i -= 1
        return out[::-1]

    def changes(self):
        out, removed, added = [], [], []
        def flush():
            if removed or added:
                kind = "replace" if removed and added else ("delete" if removed else "insert")
                out.append((kind, " ".join(removed), " ".join(added)))
                removed.clear()
                added.clear()
        for mark, word in self.word_diff():
            if mark == "=":
                flush()
            elif mark == "-":
                removed.append(word)
            else:
                added.append(word)
        flush()
        return out

    def render(self):
        parts, removed, added = [], [], []
        def flush():
            if removed:
                parts.append("[-" + " ".join(removed) + "-]")
            if added:
                parts.append("{+" + " ".join(added) + "+}")
            removed.clear()
            added.clear()
        for mark, word in self.word_diff():
            if mark == "=":
                flush()
                parts.append(word)
            elif mark == "-":
                removed.append(word)
            else:
                added.append(word)
        flush()
        return " ".join(parts)

    def edit_distance(self):
        a, b = self.old, self.new
        prev = list(range(len(b) + 1))
        for i in range(1, len(a) + 1):
            cur = [i] + [0] * len(b)
            for j in range(1, len(b) + 1):
                cur[j] = min(prev[j - 1] + (a[i - 1] != b[j - 1]), prev[j] + 1, cur[j - 1] + 1)
            prev = cur
        return prev[-1]

    def shared_passage(self):
        a, b = self.old, self.new
        best_len, best_end = 0, 0
        prev = [0] * (len(b) + 1)
        for i in range(1, len(a) + 1):
            cur = [0] * (len(b) + 1)
            for j in range(1, len(b) + 1):
                if a[i - 1] == b[j - 1]:
                    cur[j] = prev[j - 1] + 1
                    if cur[j] > best_len:
                        best_len, best_end = cur[j], i
            prev = cur
        return " ".join(a[best_end - best_len:best_end])
:::

:::think title="One table, many answers"
The comparer builds a single LCS table and reads five different things from it: a number, a ratio, a diff, a grouped summary and marked-up text. That's typical of DP in practice. The expensive part is filling the table; walking back through it costs only O(n + m). Real tools add refinements on top: Git compares lines first and only then words within changed lines, ignores whitespace changes on request, and uses Myers' algorithm, which is fast when the two versions are mostly the same.
:::

:::exercise id=forge2 points=30 title="Forge bonus: a delivery drone"
A drone delivers parcels from a depot. Each **flight** leaves the depot, visits some drops, and returns; its length (Manhattan distance, in grid steps) must not exceed the battery. Write `DronePlanner(depot, drops, battery)`, where `drops` maps names to `(x, y)`:

- Raise `ValueError` if the battery isn't positive or there are more than 12 drops. In `__init__`, precompute with a Held-Karp DP the shortest flight for **every subset** of the drops (start each single drop at its distance from the depot; a subset's flight is its best path plus the way home), and how to rebuild its order.
- `flight(names)`: `(length, order)`, the shortest single flight visiting those drops, with the visiting order; `(0, [])` for none; `KeyError` for an unknown name.
- `plan()`: `(total, flights)`, the least total distance to deliver everything in flights that each fit the battery, with each flight's visiting order and the list of flights sorted; `None` if some drop can't be reached. A submask DP: the last flight is a submask containing the mask's lowest drop.
- `fewest_flights()`: the fewest flights that deliver everything within the battery (`None` if impossible, `0` with no drops).

import itertools, random, time

DROPS = {"bakery": (2, 0), "clinic": (0, 3), "farm": (5, 5), "school": (3, 1), "mill": (6, 2)}

def dist(a, b):
    return abs(a[0] - b[0]) + abs(a[1] - b[1])

def flight_length(depot, where, order):
    points = [depot] + [where[n] for n in order] + [depot]
    return sum(dist(a, b) for a, b in zip(points, points[1:]))

def best_flight(depot, where, names):
    if not names:
        return 0
    return min(flight_length(depot, where, p) for p in itertools.permutations(names))

def partitions(items):
    if not items:
        yield []
        return
    first, rest = items[0], items[1:]
    for smaller in partitions(rest):
        yield [[first]] + smaller
        for i in range(len(smaller)):
            yield smaller[:i] + [[first] + smaller[i]] + smaller[i + 1:]

class DronePlanner:
    def __init__(self, depot, drops, battery):
        if battery <= 0:
            raise ValueError("the battery must hold some charge")
        if len(drops) > 12:
            raise ValueError("too many drops for an exact plan")
        self.depot = depot
        self.names = sorted(drops)
        self.where = dict(drops)
        self.battery = battery
        n = len(self.names)
        points = [self.where[name] for name in self.names] + [depot]
        d = [[abs(a[0] - b[0]) + abs(a[1] - b[1]) for b in points] for a in points]
        INF = float("inf")
        home = n
        # best[mask][last]: shortest flight from the depot through mask, ending at last
        best = [[INF] * n for _ in range(1 << n)]
        parent = [[None] * n for _ in range(1 << n)]
        for i in range(n):
            best[1 << i][i] = d[home][i]
        for mask in range(1, 1 << n):
            for last in range(n):
                here = best[mask][last]
                if here == INF:
                    continue
                for nxt in range(n):
                    if not mask >> nxt & 1:
                        new = mask | (1 << nxt)
                        if here + d[last][nxt] < best[new][nxt]:
                            best[new][nxt] = here + d[last][nxt]
                            parent[new][nxt] = last
        self.cost = [0] + [INF] * ((1 << n) - 1)
        self.end = [None] * (1 << n)
        for mask in range(1, 1 << n):
            for last in range(n):
                total = best[mask][last] + d[last][home]
                if total < self.cost[mask]:
                    self.cost[mask], self.end[mask] = total, last
        self.parent = parent

    def _mask(self, names):
        mask = 0
        for name in names:
            if name not in self.where:
                raise KeyError(name)
            mask |= 1 << self.names.index(name)
        return mask

    def _order(self, mask):
        last, order = self.end[mask], []
        while last is not None:
            order.append(self.names[last])
            prev = self.parent[mask][last]
            mask &= ~(1 << last)
            last = prev
        return order[::-1]

    def flight(self, names):
        mask = self._mask(names)
        if mask == 0:
            return 0, []
        return self.cost[mask], self._order(mask)

    def plan(self):
        n = len(self.names)
        INF = float("inf")
        full = (1 << n) - 1
        total = [INF] * (1 << n)
        choice = [None] * (1 << n)
        total[0] = 0
        for mask in range(1, 1 << n):
            low = mask & -mask
            sub = mask
            while sub:
                if sub & low and self.cost[sub] <= self.battery:
                    option = total[mask & ~sub] + self.cost[sub]
                    if option < total[mask]:
                        total[mask], choice[mask] = option, sub
                sub = (sub - 1) & mask
        if total[full] == INF:
            return None
        flights, mask = [], full
        while mask:
            flights.append(self._order(choice[mask]))
            mask &= ~choice[mask]
        flights.sort()
        return total[full], flights

    def fewest_flights(self):
        n = len(self.names)
        INF = float("inf")
        count = [INF] * (1 << n)
        count[0] = 0
        for mask in range(1, 1 << n):
            low = mask & -mask
            sub = mask
            while sub:
                if sub & low and self.cost[sub] <= self.battery:
                    count[mask] = min(count[mask], count[mask & ~sub] + 1)
                sub = (sub - 1) & mask
        answer = count[(1 << n) - 1]
        return None if answer == INF else answer
