from urllib.parse import urlsplit, urlunsplit, parse_qsl, urlencode

DEFAULT = {"http": 80, "https": 443}

def canonical(url):
    parts = urlsplit(url)
    scheme = parts.scheme.lower()
    host = (parts.hostname or "").lower()
    netloc = host if parts.port in (None, DEFAULT.get(scheme)) else f"{host}:{parts.port}"
    pairs = sorted(parse_qsl(parts.query, keep_blank_values=True), key=lambda kv: kv[0])
    return urlunsplit((scheme, netloc, parts.path or "/", urlencode(pairs), ""))

def same_resource(a, b):
    return canonical(a) == canonical(b)

def messages(chunks):
    buffer = b""
    for chunk in chunks:
        buffer += chunk
        while b"\n" in buffer:
            line, buffer = buffer.split(b"\n", 1)
            yield line.decode("utf-8")
    if buffer:
        raise ValueError("unfinished message")

## Forge: a web framework

Build a small WSGI web framework. The pieces, all tested together through a test client:

- **`HTTPError(status, message=None)`** (the message defaults to the status's phrase) and **`abort(status, message=None)`**, which raises it.
- **`Request(environ)`**: `method` (upper case), `path`, `query` (a dict of the **first** value of each query parameter), `headers` (lower-case names with `-`, built from the `HTTP_...` keys, plus `content-type` from `CONTENT_TYPE`), `body` (the `CONTENT_LENGTH` bytes from `wsgi.input`), `params` (the route's path parameters), and `json()` (the decoded body; an empty body is `None`; invalid JSON aborts with `400`).
- **`Response(body=b"", status=200, headers=None, content_type="text/plain; charset=utf-8")`**: a `str` body is UTF-8 encoded, and a `Content-Type` is added if there's a body and no type was given; class methods **`Response.json(data, status=200, headers=None)`** and **`Response.redirect(location, status=303)`**.
- **`App`** with:
  - `@app.route(pattern, methods=("GET",))`, where patterns use `<name>` and `<int:name>` as on Day 5, and handlers are called as `handler(request, **params)`;
  - `@app.before` (hooks called with the request; if one returns a response, it's used without routing) and `@app.after` (hooks called with the request and response, returning the response to use);
  - `handle(request)`: route and call the handler, turning a returned dict or list into `Response.json`, a `str` or `bytes` into `Response`; `405` with an `Allow` header (sorted, `", "`-separated) for a path matched with the wrong method; an `HTTPError` into `Response.json({"error": message}, status)` (so a missing route gives `{"error": "Not Found"}` with `404`); any other exception into `{"error": "internal error"}` with `500`; after hooks run in every case;
  - `__call__(environ, start_response)`, making it a WSGI application, adding `Content-Length` (except for 204 and 304).
- **`TestClient(app)`** with `request(method, path, json_body=None, headers=None)` (the path may include a query string; a JSON body sets `CONTENT_TYPE`; headers become `HTTP_...` keys) and shortcuts `get`, `post(path, json=None)` and `delete`, each returning a `TestResponse` with `status_code`, `headers` (a dict with lower-case names), `body`, `text` and `json()`.

@@solution
import io, json, re
from http import HTTPStatus
from urllib.parse import parse_qs, urlsplit, urlencode


class HTTPError(Exception):
    def __init__(self, status, message=None):
        super().__init__(message or HTTPStatus(status).phrase)
        self.status = status
        self.message = message or HTTPStatus(status).phrase


def abort(status, message=None):
    raise HTTPError(status, message)


class Request:
    def __init__(self, environ, params=None):
        self.environ = environ
        self.method = environ["REQUEST_METHOD"].upper()
        self.path = environ.get("PATH_INFO", "/") or "/"
        self.query = {k: v[0] for k, v in parse_qs(environ.get("QUERY_STRING", "")).items()}
        self.headers = {}
        for key, value in environ.items():
            if key.startswith("HTTP_"):
                self.headers[key[5:].replace("_", "-").lower()] = value
        if environ.get("CONTENT_TYPE"):
            self.headers["content-type"] = environ["CONTENT_TYPE"]
        length = int(environ.get("CONTENT_LENGTH") or 0)
        self.body = environ["wsgi.input"].read(length) if length else b""
        self.params = params or {}

    def json(self):
        try:
            return json.loads(self.body or b"null")
        except ValueError:
            abort(400, "invalid JSON")


class Response:
    def __init__(self, body=b"", status=200, headers=None, content_type="text/plain; charset=utf-8"):
        if isinstance(body, str):
            body = body.encode("utf-8")
        self.body, self.status = body, status
        self.headers = dict(headers or {})
        if body and "Content-Type" not in self.headers:
            self.headers["Content-Type"] = content_type

    @classmethod
    def json(cls, data, status=200, headers=None):
        return cls(json.dumps(data).encode("utf-8"), status, headers, "application/json")

    @classmethod
    def redirect(cls, location, status=303):
        return cls(b"", status, {"Location": location})


class App:
    def __init__(self):
        self.routes = []
        self.before_hooks, self.after_hooks = [], []

    def route(self, pattern, methods=("GET",)):
        def group(m):
            kind, name = m.groups()
            return rf"(?P<{name}>\d+)" if kind == "int" else rf"(?P<{name}>[^/]+)"
        parts = re.split(r"(<[^>]+>)", pattern)
        regex = "".join(re.sub(r"<(?:(int):)?(\w+)>", group, p) if p.startswith("<") else re.escape(p) for p in parts)
        compiled = re.compile(regex)
        ints = set(re.findall(r"<int:(\w+)>", pattern))

        def decorate(func):
            for method in methods:
                self.routes.append((method.upper(), compiled, ints, func))
            return func
        return decorate

    def before(self, func):
        self.before_hooks.append(func)
        return func

    def after(self, func):
        self.after_hooks.append(func)
        return func

    def _dispatch(self, request):
        allowed = set()
        path = request.path.rstrip("/") or "/"
        for method, regex, ints, func in self.routes:
            m = regex.fullmatch(path)
            if not m:
                continue
            if method == request.method:
                request.params = {k: int(v) if k in ints else v for k, v in m.groupdict().items()}
                return func(request, **request.params)
            allowed.add(method)
        if allowed:
            return Response.json({"error": "method not allowed"}, 405, {"Allow": ", ".join(sorted(allowed))})
        abort(404)

    def handle(self, request):
        try:
            for hook in self.before_hooks:
                early = hook(request)
                if early is not None:
                    return early
            result = self._dispatch(request)
        except HTTPError as e:
            result = Response.json({"error": e.message}, e.status)
        except Exception:
            result = Response.json({"error": "internal error"}, 500)
        if isinstance(result, (dict, list)):
            result = Response.json(result)
        elif isinstance(result, (str, bytes)):
            result = Response(result)
        for hook in self.after_hooks:
            result = hook(request, result) or result
        return result

    def __call__(self, environ, start_response):
        response = self.handle(Request(environ))
        headers = dict(response.headers)
        if response.status not in (204, 304):
            headers["Content-Length"] = str(len(response.body))
        start_response(f"{response.status} {HTTPStatus(response.status).phrase}", list(headers.items()))
        return [response.body] if response.body else []


class TestResponse:
    def __init__(self, status_line, headers, body):
        self.status_code = int(status_line.split()[0])
        self.headers = {k.lower(): v for k, v in headers}
        self.body = body

    @property
    def text(self):
        return self.body.decode("utf-8")

    def json(self):
        return json.loads(self.body)


class TestClient:
    def __init__(self, app):
        self.app = app

    def request(self, method, path, json_body=None, headers=None):
        url = urlsplit(path)
        body = b"" if json_body is None else json.dumps(json_body).encode("utf-8")
        environ = {"REQUEST_METHOD": method.upper(), "PATH_INFO": url.path, "QUERY_STRING": url.query,
                   "SERVER_NAME": "testserver", "SERVER_PORT": "80", "wsgi.input": io.BytesIO(body),
                   "CONTENT_LENGTH": str(len(body)), "wsgi.url_scheme": "http"}
        if json_body is not None:
            environ["CONTENT_TYPE"] = "application/json"
        for name, value in (headers or {}).items():
            environ["HTTP_" + name.upper().replace("-", "_")] = value
        captured = {}
        def start_response(status, response_headers, exc_info=None):
            captured["status"], captured["headers"] = status, response_headers
        body_out = b"".join(self.app(environ, start_response))
        return TestResponse(captured["status"], captured["headers"], body_out)

    def get(self, path, **kw):
        return self.request("GET", path, **kw)

    def post(self, path, json=None, **kw):
        return self.request("POST", path, json_body=json, **kw)

    def delete(self, path, **kw):
        return self.request("DELETE", path, **kw)

Count how much of the Forge was about the **application** (the notes handlers in the checks) and how much was **framework**. ## Forge bonus: an HTTP cache

:::exercise id=forge2 points=30 title="Forge bonus: an HTTP cache"
Write `caching(app, clock)`: WSGI middleware that caches successful (`200`) responses to `GET` requests that carry `Cache-Control: max-age=N` (and not `no-store` or `private`), keyed by path and query string. A later `GET` for the same key within `N` seconds (by `clock()`) is answered from the cache, without calling `app`, with an `Age` header (whole seconds since it was stored) added. If the request has `If-None-Match` equal to the cached (or fresh) response's `ETag`, answer `304 Not Modified` with the `ETag` and no body. Expired entries are dropped. Other methods pass straight through, and a successful `POST`, `PUT` or `DELETE` to a path **removes** every cached entry for that path. Also count `hits` and `misses` as attributes of the returned application.

def test_cache():

    import io
    now = [0]
    calls = []
    def app(environ, start_response):
        calls.append((environ["REQUEST_METHOD"], environ["PATH_INFO"]))
        path = environ["PATH_INFO"]
        if path == "/news":

            start_response("200 OK", [("Cache-Control", "private, max-age=60")])

        return []
    cached = caching(app, lambda: now[0])

    status, headers, body = call("GET", "/news", {"If-None-Match": '"n1"'})
    assert status == "304 Not Modified" and body == b"" and headers["ETag"] == '"n1"' and len(calls) == 1
    now[0] = 61
    call("GET", "/news")
    assert len(calls) == 2
    call("GET", "/private")
    call("GET", "/private")
    assert calls.count(("GET", "/private")) == 2
    call("POST", "/news")
    call("GET", "/news")

Keep a dict from `(path, query)` to `(stored_at, max_age, status, headers, body)`. To get the response from `app`, give it a `start_response` that records the status and headers, then join the body. Parse `max-age` with `re.search(r"max-age=(\d+)", value)`. Header names in `environ` are `HTTP_IF_NONE_MATCH` and so on.
@@solution
import re

def caching(app, clock):
    store = {}

    def find(headers, name):
        return next((v for k, v in headers if k.lower() == name), None)

    def wrapped(environ, start_response):
        method = environ["REQUEST_METHOD"].upper()
        key = (environ.get("PATH_INFO", "/"), environ.get("QUERY_STRING", ""))
        if method != "GET":
            captured = {}
            def recording(status, headers, exc_info=None):
                captured["status"] = status
                return start_response(status, headers)
            body = b"".join(app(environ, recording))
            if method in ("POST", "PUT", "DELETE") and captured.get("status", "").startswith("2"):
                for k in [k for k in store if k[0] == key[0]]:
                    del store[k]
            return [body]
        entry = store.get(key)
        if entry and clock() - entry[0] >= entry[1]:
            del store[key]
            entry = None
        if entry is None:
            wrapped.misses += 1
            captured = {}
            def recording(status, headers, exc_info=None):
                captured["status"], captured["headers"] = status, list(headers)
            body = b"".join(app(environ, recording))
            status, headers = captured["status"], captured["headers"]
            control = (find(headers, "cache-control") or "").lower()
            m = re.search(r"max-age=(\d+)", control)
            if status.startswith("200") and m and "no-store" not in control and "private" not in control:
                store[key] = (clock(), int(m.group(1)), status, headers, body)
            age = None
        else:
            wrapped.hits += 1
            stored_at, _, status, headers, body = entry
            age = int(clock() - stored_at)
        etag = find(headers, "etag")
        if etag is not None and environ.get("HTTP_IF_NONE_MATCH") == etag:
            start_response("304 Not Modified", [("ETag", etag)])
            return []
        out = list(headers) + ([("Age", str(age))] if age is not None else [])
        start_response(status, out)
        return [body]

    wrapped.hits = wrapped.misses = 0
    return wrapped
