"""Pretend TCP connections that run in the page, for practising sockets (Week 76).

a, b = socketpair()          two connected ends: bytes sent on one arrive at the other
a.send(data) -> int          sends SOME of data (perhaps not all), returning how many bytes
a.sendall(data)              (not provided: that's an exercise)
b.recv(n) -> bytes           returns between 1 and n bytes, or b"" once the other end has
                             closed and everything has been read
a.shutdown()                 says "I've finished sending" (like sock.shutdown(socket.SHUT_WR)):
                             the other end's recv then returns b"", but this end can still recv
a.close()                    closes this end completely

Like real TCP, the connection is a stream of bytes with no message boundaries: one send can
arrive as several recvs, and several sends as one. With chunk=k, sends and recvs move at most
k bytes at a time (chosen with a seeded random generator, so runs repeat exactly).
"""
import random
from collections import deque


class _Channel:
    def __init__(self):
        self.data = deque()
        self.closed = False


class FakeSocket:
    def __init__(self, inbox, outbox, chunk, rng, name):
        self._inbox, self._outbox = inbox, outbox
        self._chunk, self._rng, self.name = chunk, rng, name
        self.closed = False
        self.sent = 0

    def _size(self, n):
        if self._chunk is None:
            return n
        return max(1, min(n, self._rng.randint(1, self._chunk)))

    def send(self, data):
        if self.closed:
            raise OSError("send on a closed socket")
        if self._outbox.closed:
            raise BrokenPipeError("the other end has closed")
        data = bytes(data)
        if not data:
            return 0
        n = self._size(len(data))
        self._outbox.data.extend(data[:n])
        self.sent += n
        return n

    def recv(self, bufsize):
        if self.closed:
            raise OSError("recv on a closed socket")
        if bufsize < 1:
            raise ValueError("bufsize must be positive")
        if not self._inbox.data:
            if self._inbox.closed:
                return b""
            raise BlockingIOError("no data yet: in a real program this would wait")
        n = self._size(min(bufsize, len(self._inbox.data)))
        return bytes(self._inbox.data.popleft() for _ in range(n))

    def shutdown(self):
        """Finish sending: the other end reads b"" after the data already sent."""
        self._outbox.closed = True

    def close(self):
        if not self.closed:
            self.closed = True
            self._outbox.closed = True

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        self.close()

    def __repr__(self):
        return f"<FakeSocket {self.name}{' closed' if self.closed else ''}>"


def socketpair(chunk=None, seed=0):
    """Two connected FakeSockets. chunk limits how many bytes one send or recv moves."""
    rng = random.Random(seed)
    a_to_b, b_to_a = _Channel(), _Channel()
    return FakeSocket(b_to_a, a_to_b, chunk, rng, "a"), FakeSocket(a_to_b, b_to_a, chunk, rng, "b")

At heart, a web application is a **function from a request to a response**. Everything else in a server (sockets, parsing, threads, TLS) exists to call that function for each request that arrives and send back what it returns. Keeping that function separate from the machinery is what lets the same application run on a laptop, in a test, and on a production server.

## A server in four steps

For each connection, an HTTP server repeats:

1. **Read a request** from the socket: bytes up to the blank line (`\r\n\r\n`), then `Content-Length` bytes of body (Day 2), buffering whatever `recv` returns (Day 4).
2. **Route** it: find the handler for this method and path.
3. **Call the handler**, which returns a status, headers and a body.
4. **Write the response**, with a `Content-Length`, using `sendall`; then, if the connection is kept alive, go back to step 1 for the next request; otherwise close.

Servers also have to be robust in ways a tidy example isn't: a request too large to accept (`413`), a client that sends half a request and stops (a timeout), a handler that raises (a `500`, with the details logged rather than shown to the user), and many clients at once.

## Routing

A **router** maps `(method, path)` to a handler. Real paths carry data: `/books/42` means book 42, so routes are **patterns** with **parameters**: `/books/<int:book_id>` matches `/books/42` and passes `book_id=42`.

- Compile each pattern to a regular expression once: `<int:name>` becomes `(?P<name>\d+)`, `<name>` becomes `(?P<name>[^/]+)` (any characters except `/`), and everything else is matched literally.
- If no pattern matches the path, answer **`404 Not Found`**. If the path matches but not with this method, answer **`405 Method Not Allowed`** with an **`Allow`** header listing the methods that would work: that's the difference between "no such thing" and "you can't do that to it".
- Order matters when patterns overlap: `/books/new` must be tried before `/books/<name>`, or the literal route can never match.

Frameworks register routes with decorators, which you know how to build (Week 60): `@app.route("/books/<int:book_id>", methods=["GET"])`.

:::example
import re

class Router:
    def __init__(self):
        self.routes = []

    def add(self, method, pattern, handler):
        def group(m):
            kind, name = m.groups()
            return rf"(?P<{name}>\d+)" if kind == "int" else rf"(?P<{name}>[^/]+)"
        regex = re.sub(r"<(?:(int):)?(\w+)>", group, pattern)
        self.routes.append((method, re.compile(f"^{regex}$"), handler))

    def route(self, method, path):
        allowed = []
        for m, regex, handler in self.routes:
            match = regex.match(path)
            if match:
                if m == method:
                    params = {k: int(v) if v.isdigit() else v for k, v in match.groupdict().items()}
                    return 200, handler, params
                allowed.append(m)
        return (405, sorted(allowed), {}) if allowed else (404, None, {})

router = Router()
router.add("GET", "/books", lambda: "all books")
router.add("GET", "/books/<int:book_id>", lambda book_id: f"book {book_id}")
router.add("DELETE", "/books/<int:book_id>", lambda book_id: f"deleted {book_id}")
for method, path in [("GET", "/books/42"), ("PUT", "/books/42"), ("GET", "/authors")]:
    status, handler, params = router.route(method, path)
    print(method, path, status, handler(**params) if status == 200 else handler)

## WSGI

If every framework talked to every server in its own way, you couldn't run Django on a server written for Flask. **WSGI** (the Web Server Gateway Interface, PEP 3333) is the agreement between them. A WSGI **application** is any callable:

```python
def app(environ, start_response):
    start_response("200 OK", [("Content-Type", "text/plain; charset=utf-8")])
    return [b"Hello from WSGI\n"]
```

- **`environ`** is a dict describing the request: `REQUEST_METHOD`, `PATH_INFO` (the path), `QUERY_STRING`, `CONTENT_TYPE`, `CONTENT_LENGTH`, each request header as `HTTP_NAME` (upper case, `-` replaced by `_`: `HTTP_USER_AGENT`), `SERVER_NAME` and `SERVER_PORT`, `wsgi.url_scheme` (`http` or `https`), and **`wsgi.input`**, a file-like object to read the body from.
- **`start_response(status, headers)`** is called once with the status line text (`"404 Not Found"`, code **and** reason) and a list of `(name, value)` header pairs.
- The application **returns an iterable of bytes** (a list with one item is fine): the body. Not a string; bytes.

The server builds `environ`, calls the application, and writes what it gets back. **Middleware** is a WSGI application that wraps another, adding behaviour on the way in or out: logging every request, adding security headers, compressing responses, checking authentication. Because both sides have the same interface, middleware stacks like decorators.

The standard library's `wsgiref` has the pieces: `wsgiref.util.setup_testing_defaults(environ)` fills in a test environ, `wsgiref.validate.validator(app)` checks an application follows the rules, and `wsgiref.simple_server` is a small real server. (For `async` applications there's an equivalent interface, **ASGI**, which FastAPI uses: next week.)

:::example
import io, json
from wsgiref.util import setup_testing_defaults
from wsgiref.validate import validator

def app(environ, start_response):
    if environ["PATH_INFO"] == "/echo" and environ["REQUEST_METHOD"] == "POST":
        length = int(environ.get("CONTENT_LENGTH") or 0)
        data = json.loads(environ["wsgi.input"].read(length))
        body = json.dumps({"you sent": data, "agent": environ.get("HTTP_USER_AGENT")}).encode()
        start_response("200 OK", [("Content-Type", "application/json"), ("Content-Length", str(len(body)))])
        return [body]
    start_response("404 Not Found", [("Content-Type", "text/plain")])
    return [b"not found"]

body = b'{"n": 7}'
environ = {"REQUEST_METHOD": "POST", "SCRIPT_NAME": "", "PATH_INFO": "/echo", "QUERY_STRING": "", "CONTENT_LENGTH": str(len(body)),
           "CONTENT_TYPE": "application/json", "HTTP_USER_AGENT": "atlas/1.0", "wsgi.input": io.BytesIO(body)}
setup_testing_defaults(environ)
captured = {}
def start_response(status, headers, exc_info=None):
    captured["status"], captured["headers"] = status, headers
    return lambda data: None
result = validator(app)(environ, start_response)
print(captured["status"], b"".join(result))
result.close()

## The real servers

On your own computer, Python can serve straight away:

```bash
python -m http.server 8000
```

serves the current folder's files at `http://127.0.0.1:8000/` (for development only). To run a WSGI application:

```python
from wsgiref.simple_server import make_server

with make_server("127.0.0.1", 8000, app) as server:
    server.serve_forever()
```

For production, a WSGI application runs under a server built for it, such as **Gunicorn** (`gunicorn myapp:app --workers 4`), usually behind a **reverse proxy** such as nginx that handles TLS, static files and slow clients (Week 77). `http.server.BaseHTTPRequestHandler` also lets you write handlers directly (`do_GET`, `do_POST`), which is fine for small tools.

:::key
A server reads a request (head to the blank line, then Content-Length bytes), routes it, calls the handler, and writes the response with sendall, repeating while the connection is kept alive; it must also handle oversized requests, slow clients and handler errors (500, logged). Routers compile path patterns with parameters into regular expressions; no match is 404, a match with the wrong method is 405 with Allow; put literal routes before overlapping patterns. WSGI applications take environ (REQUEST_METHOD, PATH_INFO, QUERY_STRING, CONTENT_LENGTH, HTTP_* headers, wsgi.input) and start_response(status line, header pairs), and return an iterable of bytes; middleware wraps applications. wsgiref has testing defaults, a validator and a simple server; production uses servers like Gunicorn behind a reverse proxy.
:::

Build each regex piece by piece: split the pattern with `re.split(r"(<[^>]+>)", pattern)`, escape the literal pieces, and turn each `<...>` into its named group. Keep a list of `(method, regex, converters, handler)`.
@@solution
import re

class NotFound(Exception):
    pass

class MethodNotAllowed(Exception):
    def __init__(self, allowed):
        super().__init__(", ".join(allowed))
        self.allowed = allowed

class Router:
    def __init__(self):
        self.routes = []

    def add(self, method, pattern, handler):
        regex, ints = "", set()
        for piece in re.split(r"(<[^>]+>)", pattern):
            m = re.fullmatch(r"<(?:(int|path):)?(\w+)>", piece)
            if m:
                kind, name = m.groups()
                if kind == "int":
                    regex += rf"(?P<{name}>\d+)"
                    ints.add(name)
                elif kind == "path":
                    regex += rf"(?P<{name}>.+)"
                else:
                    regex += rf"(?P<{name}>[^/]+)"
            else:
                regex += re.escape(piece)
        self.routes.append((method.upper(), re.compile(regex), ints, handler))

    def route(self, pattern, methods=("GET",)):
        def decorate(func):
            for method in methods:
                self.add(method, pattern, func)
            return func
        return decorate

    def match(self, method, path):
        if len(path) > 1:
            path = path.rstrip("/")
        allowed = set()
        for m, regex, ints, handler in self.routes:
            found = regex.fullmatch(path)
            if not found:
                continue
            if m == method.upper():
                params = {k: int(v) if k in ints else v for k, v in found.groupdict().items()}
                return handler, params
            allowed.add(m)
        if allowed:
            raise MethodNotAllowed(sorted(allowed))
        raise NotFound(path)

import io
from urllib.parse import unquote

def make_environ(raw, scheme="http"):
    head, _, rest = raw.partition(b"\r\n\r\n")
    lines = head.decode("latin-1").split("\r\n")
    method, target, protocol = lines[0].split(" ")
    path, _, query = target.partition("?")
    headers = {}
    for line in lines[1:]:
        name, _, value = line.partition(":")
        key = name.strip().upper().replace("-", "_")
        value = value.strip()
        headers[key] = f"{headers[key]},{value}" if key in headers else value
    host, _, port = headers.get("HOST", "").partition(":")
    environ = {
        "REQUEST_METHOD": method,
        "SCRIPT_NAME": "",
        "PATH_INFO": unquote(path),
        "QUERY_STRING": query,
        "SERVER_PROTOCOL": protocol,
        "SERVER_NAME": host,
        "SERVER_PORT": port or ("443" if scheme == "https" else "80"),
        "wsgi.version": (1, 0),
        "wsgi.url_scheme": scheme,
        "wsgi.errors": io.StringIO(),
        "wsgi.multithread": False,
        "wsgi.multiprocess": False,
        "wsgi.run_once": False,
    }
    length = int(headers.get("CONTENT_LENGTH", "0") or 0)
    environ["wsgi.input"] = io.BytesIO(rest[:length])
    for key, value in headers.items():
        if key in ("CONTENT_TYPE", "CONTENT_LENGTH"):
            environ[key] = value
        else:
            environ[f"HTTP_{key}"] = value
    return environ

def call_app(app, environ):
    captured = {}
    written = []

    def start_response(status, headers, exc_info=None):
        captured["status"], captured["headers"] = status, list(headers)
        return written.append

    result = app(environ, start_response)
    try:
        chunks = [bytes(chunk) for chunk in result]
    finally:
        if hasattr(result, "close"):
            result.close()
    if "status" not in captured:
        raise RuntimeError("the application never called start_response")
    return captured["status"], captured["headers"], b"".join(written) + b"".join(chunks)

    path = unquote(url_path)

def log_requests(app, log):
    def wrapped(environ, start_response):
        def recording_start(status, headers, exc_info=None):
            log.append(f"{environ['REQUEST_METHOD']} {environ['PATH_INFO']} {status.split()[0]}")
            return start_response(status, headers, exc_info) if exc_info else start_response(status, headers)
        return app(environ, recording_start)

def security_headers(app):

def serve_connection(sock, app, max_body=1_000_000):
    buffer, served, ended = b"", 0, False

    def fill(enough):
        nonlocal buffer, ended
        while not enough() and not ended:
            try:
                chunk = sock.recv(4096)
            except BlockingIOError:
                chunk = b""
            if not chunk:
                ended = True
            buffer += chunk
        return enough()

    while True:
        if not fill(lambda: b"\r\n\r\n" in buffer):
            return served
        head_end = buffer.index(b"\r\n\r\n") + 4
        head = buffer[:head_end].decode("latin-1")
        fields = {}
        for line in head.split("\r\n")[1:]:
            name, _, value = line.partition(":")
            if name:
                fields[name.strip().lower()] = value.strip()
        length = int(fields.get("content-length", "0") or 0)
        close = fields.get("connection", "").lower() == "close"
        if not fill(lambda: len(buffer) >= head_end + length):
            return served
        raw, buffer = buffer[:head_end + length], buffer[head_end + length:]
        if length > max_body:
            _write(sock, "413 Payload Too Large", [("Content-Type", "text/plain")], b"too large", close)

            except Exception:
                status, headers, body = "500 Internal Server Error", [("Content-Type", "text/plain")], b"internal error"
            _write(sock, status, headers, body, close)

        if close:
