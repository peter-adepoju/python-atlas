## Using it for real

Using your own program for a day always turns up things the plan missed. For spend, these came up:

1. Typing `--file` every time is tedious, but the data shouldn't land in whatever folder you happen to be in either.
2. Rent was counted twice: once typed, once imported. The import couldn't know they were the same.
3. Imported payments it couldn't place are stuck as `other`, with no way to change them.
4. The summary shows one month; there's no way to see whether spending is going up.
5. `add_expense` and `set_budget` repeat the same `try`/`except` pattern again and again.
6. There's nothing telling a new user how to use it.

Each gets a step today. This is what **polish** means: not decoration, but fixing what real use reveals.

The modules so far are in the files below.

"""Reading and showing money and categories."""
import re

AMOUNT = re.compile(r"£?\s*(\d{1,3}(?:,\d{3})+|\d+)(?:\.(\d{1,2}))?")


def parse_amount(text):
    """Text such as '£1,250.50' as whole pence. Raises ValueError with a readable message."""
    text = text.strip()
    if not text.lstrip("£").strip():
        raise ValueError("missing amount")
    m = AMOUNT.fullmatch(text)
    if m is None:
        raise ValueError(f"not an amount: {text!r}")
    total = int(m.group(1).replace(",", "")) * 100 + int((m.group(2) or "").ljust(2, "0"))
    if total == 0:
        raise ValueError("amount must be more than £0")
    return total


def format_pence(pence):
    """1250 as '£12.50', -350 as '-£3.50'."""
    sign = "-" if pence < 0 else ""
    pounds, rest = divmod(abs(pence), 100)
    return f"{sign}£{pounds:,}.{rest:02d}"


def clean_category(text):
    """' Eating   Out ' as 'eating out'. Raises ValueError if it isn't words of letters."""
    shown = text.strip()
    if not shown:
        raise ValueError("missing category")
    cleaned = " ".join(shown.lower().split())
    if len(cleaned) > 20 or not re.fullmatch(r"[a-z]+( [a-z]+)*", cleaned):
        raise ValueError(f"bad category: {shown!r}")
    return cleaned

"""Reading dates and months that people type, and stepping through months."""
import re
from datetime import datetime, timedelta

FORMATS = ["%Y-%m-%d", "%d/%m/%Y"]


def parse_date(text, today):
    """'2026-11-02', '02/11/2026', 'today', 'yesterday' or '' as ISO text. No future dates."""
    shown = text.strip()
    word = shown.lower()
    if word in ("", "today"):
        when = today
    elif word == "yesterday":
        when = today - timedelta(days=1)
    else:
        when = None
        for fmt in FORMATS:
            try:
                when = datetime.strptime(word, fmt).date()
                break
            except ValueError:
                pass
        if when is None:
            raise ValueError(f"not a date: {shown!r}")
    if when > today:
        raise ValueError(f"date is in the future: {when.isoformat()}")
    return when.isoformat()


def parse_month(text):
    """'2026-11' or '11/2026' as '2026-11'. Raises ValueError otherwise."""
    shown = text.strip()
    m = re.fullmatch(r"(\d{4})-(\d{2})", shown)
    if m:
        year, month = int(m.group(1)), int(m.group(2))
    else:
        m = re.fullmatch(r"(\d{1,2})/(\d{4})", shown)
        if not m:
            raise ValueError(f"not a month: {shown!r}")
        month, year = int(m.group(1)), int(m.group(2))
    if not 1 <= month <= 12:
        raise ValueError(f"not a month: {shown!r}")
    return f"{year}-{month:02d}"


def next_month(month):
    """'2026-12' gives '2027-01'."""
    year, m = int(month[:4]), int(month[5:])
    if m == 12:
        return f"{year + 1}-01"
    return f"{year}-{m + 1:02d}"

"""The expense tracker's data and the operations on it."""
from money import parse_amount, clean_category
from dates import parse_date, next_month


def new_data():
    return {"expenses": [], "budgets": {}, "next_id": 1}


def add_expense(data, amount_text, category_text, note, date_text, today):
    """Check every field, then add the expense, or raise one ValueError listing every problem."""
    problems = []
    amount = category = when = None
    try:
        amount = parse_amount(amount_text)
    except ValueError as e:
        problems.append(str(e))
    try:
        category = clean_category(category_text)
    except ValueError as e:
        problems.append(str(e))
    try:
        when = parse_date(date_text, today)
    except ValueError as e:
        problems.append(str(e))
    note = note.strip()
    if len(note) > 60:
        problems.append("note is longer than 60 characters")
    if problems:
        raise ValueError("; ".join(problems))
    expense = {"id": data["next_id"], "date": when, "amount": amount, "category": category, "note": note}
    data["expenses"].append(expense)
    data["next_id"] += 1
    return expense


def remove_expense(data, expense_id):
    """Remove and return the expense with this id. KeyError if there isn't one."""
    for i, e in enumerate(data["expenses"]):
        if e["id"] == expense_id:
            return data["expenses"].pop(i)
    raise KeyError(f"no expense #{expense_id}")


def filter_expenses(data, month=None, category=None):
    """A new list of the matching expenses, newest first (by date, then id)."""
    chosen = [e for e in data["expenses"]
              if (month is None or e["date"][:7] == month)
              and (category is None or e["category"] == category)]
    return sorted(chosen, key=lambda e: (e["date"], e["id"]), reverse=True)


def total(expenses):
    return sum(e["amount"] for e in expenses)


def by_category(expenses):
    """(category, pence) pairs, largest first, alphabetical for ties."""
    totals = {}
    for e in expenses:
        totals[e["category"]] = totals.get(e["category"], 0) + e["amount"]
    return sorted(totals.items(), key=lambda pair: (-pair[1], pair[0]))


def month_summary(data, month):
    """Count, total, a row per category with spending or a budget, and the categories over budget."""
    expenses = filter_expenses(data, month=month)
    spent = dict(by_category(expenses))
    rows = []
    for category in set(spent) | set(data["budgets"]):
        budget = data["budgets"].get(category)
        used = spent.get(category, 0)
        rows.append({"category": category, "spent": used, "budget": budget,
                     "left": None if budget is None else budget - used})
    rows.sort(key=lambda r: (-r["spent"], r["category"]))
    return {"month": month, "count": len(expenses), "total": total(expenses), "rows": rows,
            "over": [r["category"] for r in rows if r["left"] is not None and r["left"] < 0]}


def monthly_totals(data):
    """(month, pence) for every month from the first with spending to the last, gaps as 0."""
    totals = {}
    for e in data["expenses"]:
        totals[e["date"][:7]] = totals.get(e["date"][:7], 0) + e["amount"]
    if not totals:
        return []
    out = []
    month, last = min(totals), max(totals)
    while month <= last:
        out.append((month, totals.get(month, 0)))
        month = next_month(month)
    return out


def set_budget(data, category_text, amount_text):
    """Check both, then set the category's monthly budget. Returns (category, pence)."""
    problems = []
    category = amount = None
    try:
        category = clean_category(category_text)
    except ValueError as e:
        problems.append(str(e))
    try:
        amount = parse_amount(amount_text)
    except ValueError as e:
        problems.append(str(e))
    if problems:
        raise ValueError("; ".join(problems))
    data["budgets"][category] = amount
    return category, amount


def remove_budget(data, category_text):
    """Remove a category's budget and return it. KeyError if it has none."""
    category = clean_category(category_text)
    if category not in data["budgets"]:
        raise KeyError(f"no budget for {category!r}")
    return data["budgets"].pop(category)

"""Saving and loading the data, and importing and exporting CSV."""
import csv
import io
import json
import os
import re
from collections import Counter
from datetime import datetime

from core import new_data
from money import parse_amount

KEYS = ["id", "date", "amount", "category", "note"]


def load_data(path):
    """The data saved at path, or new_data() if there's no file. ValueError if it's damaged."""
    if not os.path.exists(path):
        return new_data()
    with open(path, encoding="utf-8") as f:
        try:
            data = json.load(f)
        except json.JSONDecodeError as e:
            raise ValueError(f"{path} is damaged: {e.msg} at line {e.lineno}") from None
    if not (isinstance(data, dict) and isinstance(data.get("expenses"), list)
            and isinstance(data.get("budgets"), dict) and isinstance(data.get("next_id"), int)):
        raise ValueError(f"{path} is not a spend data file")
    for e in data["expenses"]:
        if not (isinstance(e, dict) and list(e) == KEYS and isinstance(e["amount"], int)):
            raise ValueError(f"{path} has a damaged expense: {e!r}")
    return data


def save_data(path, data):
    """Save safely: write a temporary file, then replace the old file with it in one step."""
    tmp = path + ".tmp"
    try:
        with open(tmp, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2, ensure_ascii=False)
        os.replace(tmp, path)
    finally:
        if os.path.exists(tmp):
            os.remove(tmp)


STATEMENT_AMOUNT = re.compile(r"(-?)(£?[\d,]+(?:\.\d{1,2})?)")


def read_statement(text):
    """A bank CSV (Date, Description, Amount) as (rows, problems). Only money out is kept."""
    rows, problems = [], []
    reader = csv.DictReader(io.StringIO(text))
    for row in reader:
        line = reader.line_num
        try:
            when = datetime.strptime(row["Date"].strip(), "%d/%m/%Y").date().isoformat()
        except ValueError:
            problems.append(f"line {line}: bad date {row['Date'].strip()!r}")
            continue
        m = STATEMENT_AMOUNT.fullmatch(row["Amount"].strip())
        try:
            if m is None:
                raise ValueError
            pence = parse_amount(m.group(2))
        except ValueError:
            problems.append(f"line {line}: bad amount {row['Amount'].strip()!r}")
            continue
        if m.group(1) != "-":
            continue
        rows.append({"date": when, "amount": pence, "description": " ".join(row["Description"].split())})
    return rows, problems


RULES = [
    (r"tesco|sainsbury|aldi|lidl|co-?op", "food"),
    (r"\btfl\b|trainline|uber|bus", "travel"),
    (r"netflix|spotify|cinema", "fun"),
    (r"rent|lettings", "rent"),
]


def guess_category(description, rules=RULES):
    for pattern, category in rules:
        if re.search(pattern, description, re.IGNORECASE):
            return category
    return "other"


def import_statement(data, text, rules=RULES):
    rows, problems = read_statement(text)
    existing = Counter((e["date"], e["amount"], e["note"]) for e in data["expenses"])
    added = skipped = 0
    for r in rows:
        note = r["description"][:60]
        key = (r["date"], r["amount"], note)
        if existing[key] > 0:
            existing[key] -= 1
            skipped += 1
            continue
        data["expenses"].append({"id": data["next_id"], "date": r["date"], "amount": r["amount"],
                                 "category": guess_category(r["description"], rules), "note": note})
        data["next_id"] += 1
        added += 1
    return {"added": added, "skipped": skipped, "problems": problems}


def export_csv(data, path):
    with open(path, "w", newline="", encoding="utf-8") as f:
        w = csv.writer(f)
        w.writerow(["id", "date", "amount", "category", "note"])
        for e in sorted(data["expenses"], key=lambda e: (e["date"], e["id"])):
            w.writerow([e["id"], e["date"], f"{e['amount'] // 100}.{e['amount'] % 100:02d}", e["category"], e["note"]])

## Step 26: where the data lives

Command-line tools usually find their settings in this order, most specific first:

1. an **option** on the command line (`--file`), for a one-off;
2. an **environment variable** (`SPEND_FILE`), set once in your shell's settings, for every run;
3. a **default** in your home folder, so it's the same file wherever you run it from.

`os.path.expanduser("~")` gives the home folder on any system. The function takes the environment as a parameter rather than reading `os.environ` itself, so tests can pass a plain dictionary instead of changing the real environment.

## Step 27: possible duplicates

The double rent can't be caught at import time without risking the opposite mistake: yesterday's two bus fares had the same date, amount and description, and both were real. No rule can tell those cases apart; a person can. So spend shouldn't guess. It should **point out** likely duplicates and let you remove the ones that really are.

"Likely" here: the same amount, dated at most 3 days apart. To find them, compare **every pair** of expenses once, with two loops where the inner one starts after the outer, as in selection sort (Week 18). For `n` expenses that's about n²/2 comparisons: fine for the few thousand expenses a person has (Week 17's lesson that O(n²) can be perfectly good for small n).

from datetime import date, timedelta

def possible_duplicates(data, days=3):
    """Pairs of ids (lower first) with the same amount, dated at most `days` apart, sorted."""
    pairs = []
    xs = sorted(data["expenses"], key=lambda e: e["id"])
    for i, a in enumerate(xs):
        for b in xs[i + 1:]:
            if a["amount"] == b["amount"]:
                gap = abs(date.fromisoformat(a["date"]) - date.fromisoformat(b["date"]))
                if gap <= timedelta(days=days):
                    pairs.append((a["id"], b["id"]))
    return pairs

## Step 29: a trend chart

A chart in a terminal can be made of characters. Each month gets a bar of `#`s whose length is its total **scaled** so the biggest month fills the width: length = total ÷ largest total × width, rounded. One detail matters for honesty: a month with a little spending rounds to 0 characters and looks like nothing at all, so any month above £0 gets at least one `#`.

```
2026-09      £512.40  #####
2026-10    £3,328.70  ##############################
2026-11    £3,262.10  #############################
```

from money import format_pence

def trend_lines(totals, width=30):
    """A text bar chart of (month, pence) pairs: the longest bar is `width` characters."""
    if not totals:
        return ["No expenses."]
    top = max(p for _, p in totals)
    lines = []
    for month, pence in totals:
        n = round(pence / top * width) if top else 0
        if pence and n == 0:
            n = 1
        lines.append(f"{month}  {format_pence(pence):>11}  {'#' * n}".rstrip())
    return lines

## Step 30: refactoring the checks

`add_expense` and `set_budget` repeat the same pattern: run a check inside `try`, collect its message on failure, then raise them all at once. Repetition like this tends to drift: one day someone fixes the pattern in one place and forgets the other. Day 2's tests mean you can now tidy it **safely**: change the structure, run the tests, and know straight away if behaviour changed (Week 24, Day 4).

The new helper, `collect`, takes `(check, argument)` pairs, runs every check, and returns all their results, or raises one `ValueError` listing every problem. from money import parse_amount, clean_category
from dates import parse_date


def check_note(text):
    note = text.strip()
    if len(note) > 60:
        raise ValueError("note is longer than 60 characters")
    return note


def collect(*checks):
    """Run each (function, argument) check. Return their results, or raise one ValueError listing every problem."""
    results, problems = [], []
    for check, argument in checks:
        try:
            results.append(check(argument))
        except ValueError as e:
            results.append(None)
            problems.append(str(e))
    if problems:
        raise ValueError("; ".join(problems))
    return results



def add_expense(data, amount_text, category_text, note, date_text, today):
    """Check every field, then add the expense, or raise one ValueError listing every problem."""
    amount, category, when, note = collect(
        (parse_amount, amount_text),
        (clean_category, category_text),
        (lambda text: parse_date(text, today), date_text),
        (check_note, note),
    )
    expense = {"id": data["next_id"], "date": when, "amount": amount, "category": category, "note": note}
    data["expenses"].append(expense)
    data["next_id"] += 1
    return expense


def set_budget(data, category_text, amount_text):
    """Check both, then set the category's monthly budget. Returns (category, pence)."""
    category, amount = collect((clean_category, category_text), (parse_amount, amount_text))
    data["budgets"][category] = amount
    return category, amount

import os
from datetime import date, timedelta
from money import clean_category, format_pence


def data_file(arg, environ):
    """The data file: --file if given, else $SPEND_FILE, else spend.json in the home folder."""
    if arg:
        return arg
    if environ.get("SPEND_FILE", "").strip():
        return environ["SPEND_FILE"].strip()
    return os.path.join(os.path.expanduser("~"), "spend.json")



def possible_duplicates(data, days=3):
    """Pairs of ids (lower first) with the same amount, dated at most `days` apart, sorted."""
    pairs = []
    xs = sorted(data["expenses"], key=lambda e: e["id"])
    for i, a in enumerate(xs):
        for b in xs[i + 1:]:
            if a["amount"] == b["amount"]:
                gap = abs(date.fromisoformat(a["date"]) - date.fromisoformat(b["date"]))
                if gap <= timedelta(days=days):
                    pairs.append((a["id"], b["id"]))
    return pairs



def recategorise(data, expense_id, category_text):
    """Change one expense's category. Returns (old, new). KeyError if there's no such expense."""
    category = clean_category(category_text)
    for e in data["expenses"]:
        if e["id"] == expense_id:
            old = e["category"]
            e["category"] = category
            return old, category
    raise KeyError(f"no expense #{expense_id}")



def trend_lines(totals, width=30):
    """A text bar chart of (month, pence) pairs: the longest bar is `width` characters."""
    if not totals:
        return ["No expenses."]
    top = max(p for _, p in totals)
    lines = []
    for month, pence in totals:
        n = round(pence / top * width) if top else 0
        if pence and n == 0:
            n = 1
        lines.append(f"{month}  {format_pence(pence):>11}  {'#' * n}".rstrip())
    return lines

def main(argv=None, today=None, environ=None):
    """Run one command. Returns the exit status: 0 for success, 1 for an error."""
    args = build_parser().parse_args(argv)
    today = today or date.today()
    path = data_file(args.file, os.environ if environ is None else environ)
    try:
