@@hint
Kahn's algorithm with a heap instead of a queue: count each course's unmet prerequisites, push every course with none, and repeatedly pop the smallest. If fewer than n courses come out, there's a cycle.
@@solution
import heapq

def course_order(n, prerequisites):
    # pattern: Kahn's topological sort with a heap; cost: O((n + e) log n)
    after = [[] for _ in range(n)]
    need = [0] * n
    for a, b in prerequisites:
        after[a].append(b)
        need[b] += 1
    ready = [c for c in range(n) if need[c] == 0]
    heapq.heapify(ready)
    order = []
    while ready:
        c = heapq.heappop(ready)
        order.append(c)
        for d in after[c]:
            need[d] -= 1
            if need[d] == 0:
                heapq.heappush(ready, d)
    return order if len(order) == n else None

@@hint
Run Dijkstra from the start with a heap of `(time, node)`, skipping stale entries. The answer is the largest distance; if any node is missing, -1.
@@solution
import heapq

def _times(n, links, start):
    adj = {v: [] for v in range(1, n + 1)}
    for u, v, t in links:
        adj[u].append((v, t))
    dist, heap = {start: 0}, [(0, start)]
    while heap:
        d, u = heapq.heappop(heap)
        if d > dist[u]:
            continue
        for v, t in adj[u]:
            if d + t < dist.get(v, float("inf")):
                dist[v] = d + t
                heapq.heappush(heap, (d + t, v))
    return dist

def time_to_reach_all(n, links, start):
    # pattern: Dijkstra (non-negative weights); cost: O(e log n)
    dist = _times(n, links, start)
    return max(dist.values()) if len(dist) == n else -1

def last_to_hear(n, links, start):
    dist = _times(n, links, start)
    if len(dist) < n:
        return []
    worst = max(dist.values())
    return sorted(v for v, d in dist.items() if d == worst)

@@solution
def arrow_positions(balloons):
    # pattern: greedy, sort by end; cost: O(n log n)
    arrows = []
    for start, end in sorted(balloons, key=lambda b: b[1]):
        if not arrows or start > arrows[-1]:
            arrows.append(end)
    return arrows

def fewest_arrows(balloons):
    return len(arrow_positions(balloons))

import itertools, random
def cost_of(L, order):
    pieces, total = [(0, L)], 0
    for c in order:
        for i, (a, b) in enumerate(pieces):
            if a < c < b:
                total += b - a
                pieces[i:i + 1] = [(a, c), (c, b)]
                break
    return total

    assert cheapest_cuts(10, []) == 0 and cheapest_cuts(10, [5]) == 10 and cut_order(10, [5]) == [5]

@@hint
Sort the cuts and add 0 and L as ends: points p₀ < p₁ < ... < p_{m+1}. Let f(i, j) be the cheapest way to make every cut strictly between pᵢ and pⱼ. If j = i + 1 it's 0; otherwise it's (pⱼ − pᵢ) + the minimum over k between them of f(i, k) + f(k, j). Fill by increasing j − i.
@@solution
def _tables(L, cuts):
    p = [0] + sorted(cuts) + [L]
    m = len(p)
    cost = [[0] * m for _ in range(m)]
    split = [[None] * m for _ in range(m)]
    for gap in range(2, m):
        for i in range(m - gap):
            j = i + gap
            best, where = None, None
            for k in range(i + 1, j):
                c = cost[i][k] + cost[k][j]
                if best is None or c < best:
                    best, where = c, k
            cost[i][j] = best + p[j] - p[i]
            split[i][j] = where
    return p, cost, split

def cheapest_cuts(L, cuts):
    # pattern: interval DP over cut positions; cost: O(m³)
    p, cost, _ = _tables(L, cuts)
    return cost[0][len(p) - 1]

def cut_order(L, cuts):
    p, cost, split = _tables(L, cuts)
    order = []
    def walk(i, j):
        k = split[i][j]
        if k is None:
            return
        order.append(p[k])
        walk(i, k)
        walk(k, j)
    walk(0, len(p) - 1)
    return order

@@solution
from functools import cache

def count_no_adjacent(n):
    # pattern: digit DP over bits; cost: O(bits)
    bits = [int(b) for b in bin(n)[2:]]

    @cache
    def go(pos, prev, tight):
        if pos == len(bits):
            return 1
        top = bits[pos] if tight else 1
        total = 0
        for b in range(top + 1):
            if b == 1 and prev == 1:
                continue
            total += go(pos + 1, b, tight and b == top)
        return total

    return go(0, 0, True)

def nth_no_adjacent(k):
    lo, hi = 0, 1
    while count_no_adjacent(hi) < k + 1:
        hi *= 2
    while lo < hi:
        mid = (lo + hi) // 2
        if count_no_adjacent(mid) >= k + 1:
            hi = mid
        else:
            lo = mid + 1
    return lo
