@@solution
def _lakes(grid):
    R = len(grid)
    C = len(grid[0]) if R else 0
    seen, sizes = set(), []
    for r in range(R):
        for c in range(C):
            if grid[r][c] != "W" or (r, c) in seen:
                continue
            size, inland, todo = 0, True, [(r, c)]
            seen.add((r, c))
            while todo:
                a, b = todo.pop()
                size += 1
                if a in (0, R - 1) or b in (0, C - 1):
                    inland = False
                for x, y in ((a + 1, b), (a - 1, b), (a, b + 1), (a, b - 1)):
                    if 0 <= x < R and 0 <= y < C and grid[x][y] == "W" and (x, y) not in seen:
                        seen.add((x, y))
                        todo.append((x, y))
            if inland:
                sizes.append(size)
    return sizes

def count_lakes(grid):
    # pattern: flood fill with a stack; cost: O(cells)
    return len(_lakes(grid))

def largest_lake(grid):
    return max(_lakes(grid), default=0)

import random
def brute(n, flights, src, dst, k):
    best = [None]
    def go(city, cost, legs, seen):
        if city == dst:
            best[0] = cost if best[0] is None else min(best[0], cost)
            return
        if legs == k + 1:
            return
        for a, b, p in flights:
            if a == city and b not in seen:
                go(b, cost + p, legs + 1, seen | {b})
    go(src, 0, 0, {src})
    return -1 if best[0] is None else best[0]

@@solution
def _rounds(n, flights, src, k):
    INF = float("inf")
    cost = [INF] * n
    cost[src] = 0
    history = [(list(cost), [None] * n)]
    for _ in range(k + 1):
        prev = history[-1][0]
        new, came = list(prev), [None] * n
        for a, b, p in flights:
            if prev[a] + p < new[b]:
                new[b], came[b] = prev[a] + p, a
        history.append((new, came))
    return history

def cheapest_with_stops(n, flights, src, dst, k):
    # pattern: Bellman-Ford limited to k + 1 rounds, each from the previous round's costs
    final = _rounds(n, flights, src, k)[-1][0][dst]
    return -1 if final == float("inf") else final

def route_with_stops(n, flights, src, dst, k):
    history = _rounds(n, flights, src, k)
    if history[-1][0][dst] == float("inf"):
        return None
    route, city, r = [dst], dst, len(history) - 1
    while r > 0:
        cost, came = history[r]
        if history[r - 1][0][city] == cost[city]:
            r -= 1                                # the same cost was already reachable a round earlier
            continue
        city = came[city]
        route.append(city)
        r -= 1
    return route[::-1]

@@solution
def combinations_summing(candidates, target):
    # pattern: backtracking, sorted, skipping equal values at one level; prune on size
    nums = sorted(candidates)
    out, path = [], []

    def go(start, left):
        if left == 0:
            out.append(list(path))
            return
        for i in range(start, len(nums)):
            if nums[i] > left:
                break
            if i > start and nums[i] == nums[i - 1]:
                continue
            path.append(nums[i])
            go(i + 1, left - nums[i])
            path.pop()

    go(0, target)
    return sorted(out)

import itertools, random
def brute(a, b, c):
    if len(a) + len(b) != len(c):
        return 0
    n = len(c)
    count = 0
    for picks in itertools.combinations(range(n), len(a)):

def riffle_choices(a, b, c):
    # pattern: DP over prefix lengths (i of a, j of b); cost: O(len(a) × len(b))
    n, m = len(a), len(b)
    if n + m != len(c):
        return 0
    ways = [[0] * (m + 1) for _ in range(n + 1)]
    ways[0][0] = 1
    for i in range(n + 1):
        for j in range(m + 1):
            if i and a[i - 1] == c[i + j - 1]:
                ways[i][j] += ways[i - 1][j]
            if j and b[j - 1] == c[i + j - 1]:
                ways[i][j] += ways[i][j - 1]
    return ways[n][m]

def is_interleaving(a, b, c):
    return riffle_choices(a, b, c) > 0

import itertools, random, time
def test_team():
    "bitmask DP over skills"
    skills = ["java", "nodejs", "reactjs"]
    people = [["java"], ["nodejs"], ["nodejs", "reactjs"]]
    assert smallest_team(skills, people) == [0, 2]
    assert smallest_team(["a"], [["b"]]) is None and smallest_team([], [["a"]]) == []
    rng = random.Random(5)
    names = [f"s{i}" for i in range(6)]
    for _ in range(60):
        need = rng.sample(names, rng.randint(1, 6))
        people = [rng.sample(names, rng.randint(0, 3)) for _ in range(rng.randint(1, 7))]
        got = smallest_team(need, people)
        best = None
        for k in range(len(people) + 1):
            for team in itertools.combinations(range(len(people)), k):
                if set(need) <= {s for i in team for s in people[i]}:
                    best = k
                    break
            if best is not None:
                break
        if best is None:
            assert got is None
        else:
            assert len(got) == best and set(need) <= {s for i in got for s in people[i]} and got == sorted(got)
    big_skills = [f"k{i}" for i in range(16)]
    big_people = [rng.sample(big_skills, rng.randint(1, 4)) for _ in range(60)]
    t0 = time.perf_counter()
    smallest_team(big_skills, big_people)
    assert time.perf_counter() - t0 < 8
@@solution
def smallest_team(skills, people):
    # pattern: DP over skill masks; cost: O(2^skills × people)
    index = {s: i for i, s in enumerate(skills)}
    full = (1 << len(skills)) - 1
    masks = []
    for p in people:
        m = 0
        for s in p:
            if s in index:
                m |= 1 << index[s]
        masks.append(m)
    best = {0: []}
    for mask in range(full + 1):
        if mask not in best:
            continue
        team = best[mask]
        for i, m in enumerate(masks):
            new = mask | m
            if new != mask and (new not in best or len(best[new]) > len(team) + 1):
                best[new] = team + [i]
    return sorted(best[full]) if full in best else None

:::bughunt id=c6 points=15 title="C6: Three bugs" exam=yes

    parent = list(range(n))
    def find(x):
        while parent[x] != x:
            x = parent[x]
        return x

    for a, b, cost in sorted(cables):

            total += cost
    return total

def cheapest_network(n, cables):
    parent = list(range(n))
    def find(x):
        while parent[x] != x:
            x = parent[x]
        return x
    total, joined = 0, 0
    for a, b, cost in sorted(cables, key=lambda c: c[2]):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[ra] = rb
            total += cost
            joined += 1
    return total if joined == n - 1 else None
