def lock_plan(transfers):
    return [tuple(sorted((a, b))) for a, b in transfers]

def could_deadlock(transfers):
    pairs = set()
    for a, b in transfers:
        if (b, a) in pairs:
            return True
        pairs.add((a, b))
    return False

import pickle

def chunk_payloads(items, chunksize):
    return [len(pickle.dumps(items[i:i + chunksize])) for i in range(0, len(items), chunksize)]

def overhead_ratio(items, chunksize):
    payloads = chunk_payloads(items, chunksize)
    if not payloads:
        return 0.0
    return round(sum(payloads) / len(pickle.dumps(items)), 2)

## Forge: an async job runner

Write `run_jobs(jobs, workers, timeout, max_attempts=3, base_delay=1)` using the event loop in the starter. Each job is a dict with a `name`, a list of `durations` (attempt `k` takes `durations[k - 1]`, or the last duration if there are fewer) and optionally `fails`, a collection of attempt numbers that fail even though they finish in time.

- `workers` worker tasks take jobs from an `AsyncQueue` (write it: `put` wakes one waiter; `get` waits while empty), one at a time.
- Each attempt is logged as `(loop.now, worker number, job name, attempt number)` when it starts.
- An attempt longer than `timeout` **times out**: the worker spends `timeout` units on it and it fails. Otherwise the worker spends its duration, and it succeeds unless its attempt number is in `fails`.
- A failed job with attempts left is **re-queued after a backoff** of `base_delay * 2 ** (attempt - 1)`, by a separate task that sleeps and then puts it back, so the worker moves straight on to other jobs.
- A job that succeeds, or fails its last attempt, is recorded as `{"status": "ok" or "failed", "attempts": n, "finished": loop.now}`. When every job is recorded, put one `None` per worker so they all stop.

Return `(report, loop.now, log)`, where `report` maps job names to their records.

        super().__init__()
        self.coro, self.done, self.result = coro, False, None
    def __await__(self):
        if not self.done:
            yield ("wait", self)
        return self.result

        task = Task(coro)
        self.ready.append(task)
        return task

        "Move waiting tasks (all, or up to count) back to the ready queue."
        while waitable.waiters and (count is None or count > 0):
            self.ready.append(waitable.waiters.popleft())
            if count is not None:

class AsyncQueue(Waitable):
    def __init__(self, loop):
        super().__init__()
        self.loop, self.items = loop, deque()

    def put(self, item):
        self.items.append(item)
        self.loop.wake(self, 1)

    async def get(self):
        while not self.items:
            await self.wait()
        return self.items.popleft()

def run_jobs(jobs, workers, timeout, max_attempts=3, base_delay=1):
    loop = Loop()
    queue = AsyncQueue(loop)
    report = {}
    remaining = [len(jobs)]
    attempts = {job["name"]: 0 for job in jobs}
    log = []

    async def requeue_later(job, delay):
        await Sleep(delay)
        queue.put(job)

    async def worker(number):
        while True:
            job = await queue.get()
            if job is None:
                return
            name = job["name"]
            attempts[name] += 1
            k = attempts[name]
            durations = job["durations"]
            duration = durations[min(k, len(durations)) - 1]
            log.append((loop.now, number, name, k))
            if duration > timeout:
                await Sleep(timeout)
                ok = False
            else:
                await Sleep(duration)
                ok = k not in job.get("fails", ())
            if ok or k >= max_attempts:
                report[name] = {"status": "ok" if ok else "failed", "attempts": k, "finished": loop.now}
                remaining[0] -= 1
                if remaining[0] == 0:
                    for _ in range(workers):
                        queue.put(None)
            else:
                loop.create_task(requeue_later(job, base_delay * 2 ** (k - 1)))

    async def main():
        tasks = [loop.create_task(worker(i)) for i in range(workers)]
        for job in jobs:
            queue.put(job)
        if not jobs:
            for _ in range(workers):
                queue.put(None)
        for t in tasks:
            await t
        return report

    result = loop.run_until_complete(main())
    return result, loop.now, log

:::think title="Why the backoff runs in its own task"
The tempting way to retry is for the worker to `await Sleep(backoff)` itself and then try again. It works, but while that worker sleeps it does nothing else, so a few failing jobs can tie up the whole pool while healthy jobs wait in the queue. Handing the delay to a separate small task, which just sleeps and puts the job back, keeps every worker busy with whatever is ready now. It's the same idea as Thursday's rule about blocking: in an event loop, any task that waits should wait without holding something others need. Real job systems (Celery, RQ, cloud task queues) schedule retries for later in exactly this way.
:::

## Forge bonus: a bounded pipeline

:::exercise id=forge2 points=30 title="Forge bonus: a bounded pipeline"
Using the loop in the starter, write `BoundedQueue(loop, maxsize)`: the coroutine `put(item)` waits while the queue holds `maxsize` items (wait on a second `Waitable`, `space`), then adds the item and wakes one getter; the coroutine `get()` waits while it's empty, then takes the oldest item and wakes one waiting putter; `peak` records the most items it has held. Then `run_pipeline(items, stages, maxsize)`, where `stages` is a list of `(seconds_per_item, workers)`: queues join the stages (one before each stage and one after the last), each stage's workers repeatedly take an item, `Sleep` for the stage's time, and put it into the next queue, and a collector records `(item, loop.now)` as items leave the last stage. Feed every item into the first queue, then shut the stages down in order: put one `None` per worker into each stage's input queue and await that stage's workers before moving to the next; finally put a `None` for the collector. Return `(done, loop.now, peaks)`, where `peaks` lists each queue's peak (the sentinels count).

        while self.ready or self.sleepers:

                self.now, _, task = heapq.heappop(self.sleepers)
                self.ready.append(task)
                continue
            self._step(self.ready.popleft())
        return main.result

class BoundedQueue(Waitable):
    def __init__(self, loop, maxsize):
        super().__init__()
        self.loop, self.maxsize = loop, maxsize
        self.items = deque()
        self.space = Waitable()
        self.peak = 0

    async def put(self, item):
        while len(self.items) >= self.maxsize:
            await self.space.wait()
        self.items.append(item)
        self.peak = max(self.peak, len(self.items))
        self.loop.wake(self, 1)

    async def get(self):
        while not self.items:
            await self.wait()
        item = self.items.popleft()
        self.loop.wake(self.space, 1)
        return item

def run_pipeline(items, stages, maxsize):
    loop = Loop()
    queues = [BoundedQueue(loop, maxsize) for _ in range(len(stages) + 1)]
    done = []

    async def stage_worker(index, seconds):
        source, target = queues[index], queues[index + 1]
        while True:
            item = await source.get()
            if item is None:
                return
            await Sleep(seconds)
            await target.put(item)

    async def collector():
        while True:
            item = await queues[-1].get()
            if item is None:
                return
            done.append((item, loop.now))

    async def main():
        workers = [[loop.create_task(stage_worker(i, seconds)) for _ in range(count)]
                   for i, (seconds, count) in enumerate(stages)]
        collect = loop.create_task(collector())
        for item in items:
            await queues[0].put(item)
        for i, group in enumerate(workers):
            for _ in group:
                await queues[i].put(None)
            for task in group:
                await task
        await queues[-1].put(None)
        await collect

    loop.run_until_complete(main())
    return done, loop.now, [q.peak for q in queues]
