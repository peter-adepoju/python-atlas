def possible(pushed, popped):
    stack = []
    i = 0
    for x in pushed:
        stack.append(x)
        while stack and i < len(popped) and stack[-1] == popped[i]:
            stack.pop()
            i += 1
    return i == len(popped)

def smallest_after_removing(number, k):
    stack = []

        while k and stack and stack[-1] > d:
            stack.pop()
            k -= 1

    if k:
        stack = stack[:-k]
    return "".join(stack).lstrip("0") or "0"

## Forge: a print shop

A small print shop takes jobs from customers. Jobs wait in a queue; an **urgent** job jumps to the front; the printer takes jobs from the front; a customer who changes their mind can **undo** their last submission, if it hasn't been printed yet; and the screen shows the last few jobs printed. Build `PrintShop(capacity=8, keep=5)`.

**Storage.** Waiting jobs are job ids in a **ring deque** on a fixed block of `capacity` slots (Day 5): normal jobs join the back, urgent ones the front. A dictionary maps each id to its `(name, pages)`. Printed ids go into a **ring log** of the last `keep` jobs (it may overwrite the oldest). Each submission is pushed onto an **undo stack** as `(id, end)`, where `end` is `"front"` or `"back"`.

**Operations.**

- `submit(name, pages, urgent=False)`: `pages` must be a whole number of at least 1 (else `ValueError`); a full queue raises `OverflowError("queue is full")`. Returns the new id: `"J1"`, `"J2"`, ... (ids are never reused, even after an undo).
- `print_next()`: removes the front job and returns `(id, name, pages)`; nothing waiting raises `IndexError("nothing to print")`.
- `undo()`: pops the undo stack. If that job is still waiting **at the end it was added to**, remove it and return its id. Otherwise (it's been printed), return `None`. With nothing to undo, return `None`.
- `waiting()`: the waiting ids, front to back. `recent()`: the printed ids, newest first (up to `keep`). `len(shop)`: how many are waiting.

**Interface.** `handle(line, shop)` returns a reply:

| Line | Reply |
|---|---|
| `submit NAME PAGES` | `queued J1` |
| `urgent NAME PAGES` | `queued J2 at the front` |
| `print` | `printed J2 poster (3 pages)` |
| `undo` | `removed J1`, or `nothing to undo` |
| `queue` | `queue: J3 J1` (front first), or `queue: empty` |
| `recent` | `recent: J2 J5` (newest first), or `recent: none` |

`PAGES` is read with `int`; any `ValueError`, `OverflowError` or `IndexError` from the shop becomes `error: ` and its message; anything else gets `unknown command`.

Keep the ring deque's head and size as in Day 5's `RingDeque`. `undo` needs to look at one end: for `"front"`, the job is at `slots[head]`; for `"back"`, at `slots[(head + size - 1) % capacity]`. If the id there matches, pop that end; otherwise return `None`. The history ring is Day 5's `RingLog`.
@@solution
class PrintShop:
    def __init__(self, capacity=8, keep=5):
        self._slots = [None] * capacity
        self._head = 0
        self._size = 0
        self._jobs = {}
        self._next_id = 1
        self._undo = []
        self._log = [None] * keep
        self._log_next = 0
        self._log_size = 0

    def __len__(self):
        return self._size

    def _back_index(self):
        return (self._head + self._size - 1) % len(self._slots)

    def submit(self, name, pages, urgent=False):
        if not isinstance(pages, int) or isinstance(pages, bool) or pages < 1:
            raise ValueError("pages must be a whole number of at least 1")
        if self._size == len(self._slots):
            raise OverflowError("queue is full")
        job_id = f"J{self._next_id}"
        self._next_id += 1
        self._jobs[job_id] = (name, pages)
        if urgent:
            self._head = (self._head - 1) % len(self._slots)
            self._slots[self._head] = job_id
        else:
            self._slots[(self._head + self._size) % len(self._slots)] = job_id
        self._size += 1
        self._undo.append((job_id, "front" if urgent else "back"))
        return job_id

    def _pop_front(self):
        job_id = self._slots[self._head]
        self._slots[self._head] = None
        self._head = (self._head + 1) % len(self._slots)
        self._size -= 1
        return job_id

    def _pop_back(self):
        i = self._back_index()
        job_id = self._slots[i]
        self._slots[i] = None
        self._size -= 1
        return job_id

    def print_next(self):
        if self._size == 0:
            raise IndexError("nothing to print")
        job_id = self._pop_front()
        self._log[self._log_next] = job_id
        self._log_next = (self._log_next + 1) % len(self._log)
        self._log_size = min(self._log_size + 1, len(self._log))
        name, pages = self._jobs[job_id]
        return job_id, name, pages

    def undo(self):
        if not self._undo:
            return None
        job_id, end = self._undo.pop()
        if self._size == 0:
            return None
        if end == "front" and self._slots[self._head] == job_id:
            return self._pop_front()
        if end == "back" and self._slots[self._back_index()] == job_id:
            return self._pop_back()
        return None

    def waiting(self):
        cap = len(self._slots)
        return [self._slots[(self._head + i) % cap] for i in range(self._size)]

    def recent(self):
        cap = len(self._log)
        return [self._log[(self._log_next - 1 - i) % cap] for i in range(self._log_size)]

def handle(line, shop):
    words = line.split()
    try:
        if len(words) == 3 and words[0] in ("submit", "urgent"):
            urgent = words[0] == "urgent"
            job_id = shop.submit(words[1], int(words[2]), urgent=urgent)
            return f"queued {job_id}" + (" at the front" if urgent else "")
        if words == ["print"]:
            job_id, name, pages = shop.print_next()
            return f"printed {job_id} {name} ({pages} pages)"
        if words == ["undo"]:
            job_id = shop.undo()
            return f"removed {job_id}" if job_id else "nothing to undo"
        if words == ["queue"]:
            ids = shop.waiting()
            return "queue: " + (" ".join(ids) if ids else "empty")
        if words == ["recent"]:
            ids = shop.recent()
            return "recent: " + (" ".join(ids) if ids else "none")
    except (ValueError, OverflowError, IndexError) as err:
        return f"error: {err}"
    return "unknown command"
:::

:::think title="Undo that knows its limits"
Week 36's undo assumed nothing else changes the document between a command and its undo. Here the printer works on the queue too, so `undo` can't blindly reverse: it checks that the world still looks the way it left it (the job is still waiting, at the end it joined) and refuses otherwise. That's the honest version of undo in any shared system: an undo is a request, not a guarantee, and it must check before it acts (Week 37's defensive programming). Notice also how little each structure had to do: the ring only ever touches its two ends, the undo stack only its top, the history only overwrites its oldest slot. Every operation is O(1), and the restrictions are what make that possible.
:::

:::exercise id=forge2 points=30 title="Forge bonus: two classics"
1. **The sum of the minimums.** Write `sum_of_minimums(nums)`: the sum, over every run of consecutive items (every subarray), of that run's smallest item. Do it in O(n) with monotonic stacks: each item `nums[i]` is the minimum of exactly `left[i] * right[i]` runs, where `left[i]` is how far you can go left before meeting a **strictly smaller** item, and `right[i]` how far right before meeting a **smaller or equal** one (counting `i` itself each way). from collections import deque

def sum_of_minimums(nums):
    n = len(nums)
    left, right = [0] * n, [0] * n
    stack = []
    for i in range(n):
        while stack and nums[stack[-1]] >= nums[i]:
            stack.pop()
        left[i] = i - (stack[-1] if stack else -1)
        stack.append(i)
    stack = []
    for i in range(n - 1, -1, -1):
        while stack and nums[stack[-1]] > nums[i]:
            stack.pop()
        right[i] = (stack[-1] if stack else n) - i
        stack.append(i)
    return sum(nums[i] * left[i] * right[i] for i in range(n))

class MaxQueue:
    def __init__(self):
        self._items = deque()
        self._maxes = deque()

    def enqueue(self, x):
        self._items.append(x)
        while self._maxes and self._maxes[-1] < x:
            self._maxes.pop()
        self._maxes.append(x)

    def dequeue(self):
        if not self._items:
            raise IndexError("dequeue from empty queue")
        x = self._items.popleft()
        if x == self._maxes[0]:
            self._maxes.popleft()
        return x

    def max(self):
        if not self._maxes:
            raise IndexError("max of empty queue")
        return self._maxes[0]

    def __len__(self):
        return len(self._items)
