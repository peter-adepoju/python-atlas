def rotate(nums, k):
    n = len(nums)
    if n == 0:
        return
    k %= n

    def flip(lo, hi):
        while lo < hi:
            nums[lo], nums[hi] = nums[hi], nums[lo]
            lo += 1
            hi -= 1

    flip(0, n - 1)
    flip(0, k - 1)
    flip(k, n - 1)

def moving_averages(values, k):
    if not 1 <= k <= len(values):
        raise ValueError("k must be from 1 to the number of values")
    total = sum(values[:k])
    out = [round(total / k, 2)]
    for i in range(k, len(values)):
        total += values[i] - values[i - k]
        out.append(round(total / k, 2))
    return out

## Forge: a sensor log

A weather station records one whole-number reading a minute. Volunteers ask questions about the log, and occasionally correct whole stretches of it after finding that a sensor was off. Build `SensorLog`, which answers every question quickly, even with hundreds of thousands of readings.

**Storage.** Readings go into your own dynamic array: a fixed block made with `[None] * capacity`, starting at capacity 1 and doubling when full, never appended to. Keep `resizes` (how many times the block grew) and a read-only property `capacity`. Keep **prefix sums** up to date as readings arrive, so range totals never need a loop.

- `add(value)` stores a reading; `len(log)` is how many there are; `log[i]` reads one (negative indexes count from the end; out of range raises `IndexError`).
- `total(i, j)`: the sum of readings `i` to `j - 1`, in O(1). `i` and `j` must satisfy `0 <= i <= j <= len`, else `IndexError`.

**Questions.**

- `best_window(k)`: the busiest `k` consecutive minutes, as `(start, total)`; earliest on a tie. `ValueError` if `k` isn't from 1 to `len`.
- `longest_between(lo, hi)`: the longest run of consecutive readings all from `lo` to `hi` inclusive, as `(start, end)` (half-open, earliest on a tie), or `None` if no reading is in range.
- `count_totals(k)`: how many runs of consecutive readings (length 1 or more) add up to exactly `k`. Readings may be negative.
- `closest_pair(target)`: two **different readings** (at different positions) whose sum is closest to `target`, as `(a, b)` with `a <= b`; on a tie, the smaller sum. Use a sorted copy and two pointers. Fewer than 2 readings raises `ValueError`.

**Corrections.** `calibrate(corrections)`, where each correction is `(i, j, delta)`: add `delta` to every reading from `i` to `j - 1`. Apply them all at once with a **difference array** in O(n + number of corrections), then bring the prefix sums back up to date (once, not once per correction).

The checks compare every answer with a simple, slow version on random data, and time the big cases.

Keep `self._prefix` as a plain Python list starting `[0]`, appending `prefix[-1] + value` in `add`: it's a summary, not the store. `best_window` and `count_totals` can then work from the prefix sums alone. For `longest_between`, a run of in-range readings is broken by any out-of-range one: track where the current run started. In `calibrate`, build the difference array, add its running total to each reading, then rebuild `self._prefix` once.
@@solution
class SensorLog:
    def __init__(self):
        self._size = 0
        self._capacity = 1
        self._slots = [None]
        self._prefix = [0]
        self.resizes = 0

    @property
    def capacity(self):
        return self._capacity

    def __len__(self):
        return self._size

    def add(self, value):
        if self._size == self._capacity:
            new = [None] * (2 * self._capacity)
            for i in range(self._size):
                new[i] = self._slots[i]
            self._slots, self._capacity = new, 2 * self._capacity
            self.resizes += 1
        self._slots[self._size] = value
        self._size += 1
        self._prefix.append(self._prefix[-1] + value)

    def __getitem__(self, i):
        if i < 0:
            i += self._size
        if not 0 <= i < self._size:
            raise IndexError("reading index out of range")
        return self._slots[i]

    def total(self, i, j):
        if not 0 <= i <= j <= self._size:
            raise IndexError(f"bad range {i}..{j}")
        return self._prefix[j] - self._prefix[i]

    def best_window(self, k):
        if not 1 <= k <= self._size:
            raise ValueError("k must be from 1 to the number of readings")
        best = (0, self.total(0, k))
        for start in range(1, self._size - k + 1):
            t = self._prefix[start + k] - self._prefix[start]
            if t > best[1]:
                best = (start, t)
        return best

    def longest_between(self, lo, hi):
        best = None
        run_start = None
        for i in range(self._size):
            if lo <= self._slots[i] <= hi:
                if run_start is None:
                    run_start = i
                if best is None or i + 1 - run_start > best[1] - best[0]:
                    best = (run_start, i + 1)
            else:
                run_start = None
        return best

    def count_totals(self, k):
        seen = {}
        count = 0
        for p in self._prefix:
            count += seen.get(p - k, 0)
            seen[p] = seen.get(p, 0) + 1
        return count

    def closest_pair(self, target):
        if self._size < 2:
            raise ValueError("need at least two readings")
        values = sorted(self._slots[i] for i in range(self._size))
        left, right = 0, len(values) - 1
        best = None
        while left < right:
            s = values[left] + values[right]
            key = (abs(s - target), s)
            if best is None or key < best[0]:
                best = (key, (values[left], values[right]))
            if s < target:
                left += 1
            elif s > target:
                right -= 1
            else:
                break
        return best[1]

    def calibrate(self, corrections):
        diff = [0] * (self._size + 1)
        for i, j, delta in corrections:
            diff[i] += delta
            diff[j] -= delta
        running = 0
        for i in range(self._size):
            running += diff[i]
            self._slots[i] += running
        self._prefix = [0]
        for i in range(self._size):
            self._prefix.append(self._prefix[-1] + self._slots[i])
:::

:::think title="Paying in advance"
Look at where `SensorLog` spends its time. Each `add` does a little extra work (one addition for the prefix sums, and occasionally a resize); in return, every `total` is instant, and `best_window` and `count_totals` never add up a range. `calibrate` is the one operation that disturbs the summary, so it rebuilds it once, at a cost of O(n), rather than patching it after every correction. That's the trade at the heart of data structure design: decide which operations must be fast, and pay for them, in advance or in memory, out of the operations that can afford it.
:::

def max_profit(prices):
    lowest = None
    best = 0
    for p in prices:
        if lowest is None or p < lowest:
            lowest = p
        else:
            best = max(best, p - lowest)
    return best

def products_below(nums, k):
    if k <= 1:
        return 0
    start, product, count = 0, 1, 0
    for end, x in enumerate(nums):
        product *= x
        while product >= k:
            product //= nums[start]
            start += 1
        count += end - start + 1
    return count

def start_station(petrol, cost):
    if sum(petrol) < sum(cost):
        return -1
    start, tank = 0, 0
    for i in range(len(petrol)):
        tank += petrol[i] - cost[i]
        if tank < 0:
            start, tank = i + 1, 0
    return start
