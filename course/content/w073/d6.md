import re, tomllib

def pin_report(pyproject):
    deps = tomllib.loads(pyproject).get("project", {}).get("dependencies", [])
    report = {}
    for dep in deps:
        spec = dep.partition(";")[0].strip()
        m = re.match(r"([A-Za-z0-9._-]+)\s*(\[[^\]]*\])?\s*(.*)", spec)
        name, clauses = m.group(1), [c.strip() for c in m.group(3).split(",") if c.strip()]
        if not clauses:
            kind = "open"
        elif len(clauses) == 1 and clauses[0].startswith("==") and "*" not in clauses[0]:
            kind = "pinned"
        else:
            kind = "ranged"
        report[re.sub(r"[-_.]+", "-", name).lower()] = kind
    return report

def is_library_friendly(report):
    return "pinned" not in report.values()

def _t(v):
    return tuple(int(x) for x in v.split("."))

def upgrade_risk(current, candidate):
    a, b = _t(current), _t(candidate)
    if b <= a:
        return "none"
    if a[0] != b[0] or (a[0] == 0 and a[1] != b[1]):
        return "breaking"
    if a[1] != b[1]:
        return "features"
    return "fixes"

def safe_upgrades(locked, available):
    result = {}
    for name, current in locked.items():
        ok = [v for v in available.get(name, []) if upgrade_risk(current, v) in ("fixes", "features")]
        result[name] = max(ok, key=_t) if ok else None
    return result

## Forge: a build backend

A build backend is the code that pip calls (through `[build-system]`) to turn a source tree into distributions. Real ones (hatchling, setuptools, flit) do more, but the core is what you'll build: read the project, check it, write the metadata, and pack the files.

A project is given as a dict of path to text, such as `{"pyproject.toml": "...", "README.md": "...", "src/text_kit/__init__.py": "...", ...}`. Write a class **`Project(files)`** with:

- **`metadata()`**: from `pyproject.toml`'s `[project]` table, a dict with `"name"`, `"version"`, `"summary"` (from `description`), `"requires_python"`, `"dependencies"` (a list), `"extras"` (a dict of extra to list), `"scripts"` (a dict) and `"readme"`, using `None`, `[]` or `{}` for missing fields. Raise `ValueError` if there's no `pyproject.toml`.
- **`problems()`**: a list of problems in this order: just `["no pyproject.toml"]` if it's missing; otherwise `"no name"` and `"no version"` if missing; `"readme not found: PATH"`; `"no package under src/"` if no file is inside a folder under `src/`; `"package NAME has no __init__.py"` for each top-level folder under `src/` (sorted) without one; and `"bad script: COMMAND"` for each script not of the form `module:function`.
- **`metadata_text()`**: the `METADATA` text: `Metadata-Version: 2.3`, `Name`, `Version`, then `Summary` and `Requires-Python` if set, then a `Requires-Dist` line per dependency, then for each extra (in order) `Provides-Extra: EXTRA` followed by its dependencies as `Requires-Dist: SPEC; extra == "EXTRA"`, or, if the dependency has its own marker, `Requires-Dist: SPEC; (MARKER) and extra == "EXTRA"`. End the headers with a newline; if the readme exists, add a blank line and the readme's text.
- **`wheel()`**: raise `ValueError` (with the problems joined by `"; "`) if there are problems; otherwise return `(filename, bytes)` for a `py3-none-any` wheel named from the normalised name with underscores. It contains every file under `src/` (with `src/` removed from the path) except anything in a `__pycache__` folder or ending in `.pyc`, then the `.dist-info` folder with `METADATA`, `WHEEL` (as in Day 3, `Generator: atlas`), `entry_points.txt` if there are scripts, and `RECORD` last.
- **`sdist()`**: the same check, then `(filename, bytes)` for a `.tar.gz` named `{stem}.tar.gz`, where `stem` is the wheel's `name-version`, containing every project file under `{stem}/`, plus `{stem}/PKG-INFO` holding the metadata text.

import base64, csv, hashlib, io, re, tarfile, tomllib, zipfile


def record_hash(data):
    digest = hashlib.sha256(data).digest()
    return "sha256=" + base64.urlsafe_b64encode(digest).rstrip(b"=").decode()


def normalise(name):
    return re.sub(r"[-_.]+", "-", name).lower()


class Project:
    def __init__(self, files):
        self.files = dict(files)

    def metadata(self):
        if "pyproject.toml" not in self.files:
            raise ValueError("no pyproject.toml")
        project = tomllib.loads(self.files["pyproject.toml"]).get("project", {})
        return {
            "name": project.get("name"),
            "version": project.get("version"),
            "summary": project.get("description"),
            "requires_python": project.get("requires-python"),
            "dependencies": list(project.get("dependencies", [])),
            "extras": {k: list(v) for k, v in project.get("optional-dependencies", {}).items()},
            "scripts": dict(project.get("scripts", {})),
            "readme": project.get("readme"),
        }

    def problems(self):
        if "pyproject.toml" not in self.files:
            return ["no pyproject.toml"]
        meta = self.metadata()
        found = []
        if not meta["name"]:
            found.append("no name")
        if not meta["version"]:
            found.append("no version")
        if meta["readme"] and meta["readme"] not in self.files:
            found.append(f"readme not found: {meta['readme']}")
        tops = sorted({p.split("/")[1] for p in self.files if p.startswith("src/") and p.count("/") >= 2})
        if not tops:
            found.append("no package under src/")
        for top in tops:
            if f"src/{top}/__init__.py" not in self.files:
                found.append(f"package {top} has no __init__.py")
        for command, target in meta["scripts"].items():
            module, sep, func = target.partition(":")
            if not sep or not all(p.isidentifier() for p in module.split(".")) or not all(p.isidentifier() for p in func.split(".")):
                found.append(f"bad script: {command}")
        return found

    def package_files(self):
        out = {}
        for path in sorted(self.files):
            if path.startswith("src/") and path.count("/") >= 2:
                if "__pycache__" in path.split("/") or path.endswith(".pyc"):
                    continue
                out[path[4:]] = self.files[path]
        return out

    def _requires_dist(self, meta):
        lines = [f"Requires-Dist: {d}" for d in meta["dependencies"]]
        for extra, deps in meta["extras"].items():
            lines.append(f"Provides-Extra: {extra}")
            for d in deps:
                spec, _, marker = d.partition(";")
                if marker.strip():
                    lines.append(f'Requires-Dist: {spec.strip()}; ({marker.strip()}) and extra == "{extra}"')
                else:
                    lines.append(f'Requires-Dist: {spec.strip()}; extra == "{extra}"')
        return lines

    def metadata_text(self):
        meta = self.metadata()
        lines = ["Metadata-Version: 2.3", f"Name: {meta['name']}", f"Version: {meta['version']}"]
        if meta["summary"]:
            lines.append(f"Summary: {meta['summary']}")
        if meta["requires_python"]:
            lines.append(f"Requires-Python: {meta['requires_python']}")
        lines += self._requires_dist(meta)
        text = "\n".join(lines) + "\n"
        if meta["readme"] and meta["readme"] in self.files:
            text += "\n" + self.files[meta["readme"]]
        return text

    def _stem(self):
        meta = self.metadata()
        return normalise(meta["name"]).replace("-", "_") + "-" + meta["version"]

    def wheel(self):
        problems = self.problems()
        if problems:
            raise ValueError("; ".join(problems))
        meta = self.metadata()
        stem = self._stem()
        info = f"{stem}.dist-info/"
        contents = {p: t.encode() for p, t in self.package_files().items()}
        contents[info + "METADATA"] = self.metadata_text().encode()
        contents[info + "WHEEL"] = b"Wheel-Version: 1.0\nGenerator: atlas\nRoot-Is-Purelib: true\nTag: py3-none-any\n"
        if meta["scripts"]:
            lines = ["[console_scripts]"] + [f"{k} = {v}" for k, v in meta["scripts"].items()]
            contents[info + "entry_points.txt"] = ("\n".join(lines) + "\n").encode()
        record = io.StringIO()
        writer = csv.writer(record, lineterminator="\n")
        for path, data in contents.items():
            writer.writerow([path, record_hash(data), len(data)])
        writer.writerow([info + "RECORD", "", ""])
        contents[info + "RECORD"] = record.getvalue().encode()
        buffer = io.BytesIO()
        with zipfile.ZipFile(buffer, "w", zipfile.ZIP_DEFLATED) as z:
            for path, data in contents.items():
                z.writestr(path, data)
        return f"{stem}-py3-none-any.whl", buffer.getvalue()

    def sdist(self):
        problems = self.problems()
        if problems:
            raise ValueError("; ".join(problems))
        stem = self._stem()
        members = dict(sorted(self.files.items()))
        members["PKG-INFO"] = self.metadata_text()
        buffer = io.BytesIO()
        with tarfile.open(fileobj=buffer, mode="w:gz") as tar:
            for path, text in members.items():
                data = text.encode()
                info = tarfile.TarInfo(f"{stem}/{path}")
                info.size = len(data)
                tar.addfile(info, io.BytesIO(data))
        return f"{stem}.tar.gz", buffer.getvalue()

:::think title="What a backend owes its users"
Notice how much of today was **checking**, not building. A wheel that builds from a broken project (a missing `__init__.py`, a script pointing at nothing, a README that isn't there) fails later, on someone else's machine, where it's hardest to diagnose. Real backends refuse early with a clear message for the same reason. And notice that the wheel contains nothing from `tests/`, while the sdist contains everything: the wheel is what users run, the sdist is what anyone needs to rebuild, test and audit it.

import io, tomllib, zipfile
from email.parser import Parser

def compare(files, wheel_bytes):
    source = {}
    for path, text in files.items():
        if path.startswith("src/") and path.count("/") >= 2:
            if "__pycache__" in path.split("/") or path.endswith(".pyc"):
                continue
            source[path[4:]] = text.encode()
    z = zipfile.ZipFile(io.BytesIO(wheel_bytes))
    built = {n: z.read(n) for n in z.namelist() if ".dist-info/" not in n}
    return {
        "missing": sorted(set(source) - set(built)),
        "extra": sorted(set(built) - set(source)),
        "changed": sorted(p for p in set(source) & set(built) if source[p] != built[p]),
    }

def drift(pyproject, wheel_bytes):
    project = tomllib.loads(pyproject)["project"]
    z = zipfile.ZipFile(io.BytesIO(wheel_bytes))
    meta_name = next(n for n in z.namelist() if n.endswith(".dist-info/METADATA"))
    meta = Parser().parsestr(z.read(meta_name).decode())
    messages = []
    for field, key in (("name", "Name"), ("version", "Version")):
        if project.get(field) != meta[key]:
            messages.append(f"{field}: {project.get(field)} in pyproject, {meta[key]} in wheel")
    ours = list(project.get("dependencies", []))
    theirs = [r for r in (meta.get_all("Requires-Dist") or []) if "extra" not in r.partition(";")[2]]
    messages += [f"not in wheel: {d}" for d in ours if d not in theirs]
    messages += [f"only in wheel: {d}" for d in theirs if d not in ours]
    return messages
:::
