import numpy as np

def fit_scaler(X_train):
    means = X_train.mean(axis=0)
    stds = X_train.std(axis=0)
    return means, np.where(stds == 0, 1.0, stds)

def transform(X, scaler):
    means, stds = scaler
    return (X - means) / stds

import numpy as np

def logistic_fit(X, y, rate=0.5, steps=500):
    w, b = np.zeros(X.shape[1]), 0.0
    losses = []
    for _ in range(steps):
        p = 1 / (1 + np.exp(-(X @ w + b)))
        pc = np.clip(p, 1e-12, 1 - 1e-12)
        losses.append(float(-(y * np.log(pc) + (1 - y) * np.log(1 - pc)).mean()))
        w = w - rate * X.T @ (p - y) / len(y)
        b = b - rate * float((p - y).mean())
    return w, b, losses

def logistic_predict(X, w, b):
    return ((1 / (1 + np.exp(-(X @ w + b)))) >= 0.5).astype(int)

## Forge: a spam filter

A model needs numbers, and a message is words. The standard first step is the **bag of words**: split each message into **tokens** (here, runs of lower-case letters, digits and `£`), build a **vocabulary** of the tokens seen in the training messages, and represent each message by **how many times** it contains each vocabulary word. Word order is thrown away, which sounds crude and works surprisingly well. scikit-learn's `CountVectorizer` does all of this; you'll write it yourself first, then use it.

**Multinomial Naive Bayes** is the classic model for word counts. For each class, it learns how likely each word is (`P(word | spam)`), and scores a message by adding up the logarithms of those probabilities for its words, plus the log of how common the class is. It's "naive" because it treats words as independent of each other, which they aren't; it's fast, needs little data, and is easy to explain, because the ratio `log P(word | spam) − log P(word | ham)` says how much each word pushes a message towards spam.

Write these functions:

- `load_messages(path)`: the messages, with a `spam` column (1 for spam, 0 for ham).
- `tokenise(text)`: the list of tokens, `re.findall(r"[a-z0-9£]+", text.lower())`.
- `vocabulary(texts, min_df=1)`: the sorted list of tokens appearing in at least `min_df` of the messages (count each message once per token).
- `vectorise(texts, vocab)`: a NumPy array of shape `(messages, len(vocab))` of whole-number counts; tokens not in `vocab` are ignored.
- `spam_model()`: an unfitted pipeline of `CountVectorizer(tokenizer=tokenise, token_pattern=None, lowercase=False)` and `MultinomialNB()`.
- `evaluate(df, seed=0)`: split 75/25, stratified by `spam`, with `random_state=seed`; fit a `spam_model()` on the training texts; return `(model, report)` where the report has `accuracy` and `baseline` (always predicting the training majority class) on the test set, `confusion` (`tp`, `fp`, `fn`, `tn`, spam positive), `precision` and `recall` (all 3 places), and the test texts it got wrong: `missed` (spam let through) and `false_alarms` (ham flagged), in test order.
- `spammiest(model, n)`: the `n` vocabulary words with the largest `log P(word | spam) − log P(word | ham)` (from the fitted Naive Bayes step's `feature_log_prob_`), as `(word, value)` pairs with values rounded to 3 places; ties keep alphabetical order.
- `classify(model, texts)`: a list of `"spam"` or `"ham"`.

import re
import numpy as np
import pandas as pd
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import MultinomialNB
from sklearn.pipeline import make_pipeline

def load_messages(path):
    df = pd.read_csv(path)
    df["spam"] = (df["label"] == "spam").astype(int)
    return df

def tokenise(text):
    return re.findall(r"[a-z0-9£]+", text.lower())

def vocabulary(texts, min_df=1):
    seen = {}
    for text in texts:
        for token in set(tokenise(text)):
            seen[token] = seen.get(token, 0) + 1
    return sorted(t for t, n in seen.items() if n >= min_df)

def vectorise(texts, vocab):
    index = {w: i for i, w in enumerate(vocab)}
    counts = np.zeros((len(texts), len(vocab)), dtype=int)
    for row, text in enumerate(texts):
        for token in tokenise(text):
            if token in index:
                counts[row, index[token]] += 1
    return counts

def spam_model():
    return make_pipeline(CountVectorizer(tokenizer=tokenise, token_pattern=None, lowercase=False), MultinomialNB())

def evaluate(df, seed=0):
    train, test = train_test_split(df, test_size=0.25, random_state=seed, stratify=df["spam"])
    model = spam_model().fit(train["text"], train["spam"])
    pred = model.predict(test["text"])
    y = test["spam"].to_numpy()
    c = {"tp": int(((y == 1) & (pred == 1)).sum()), "fp": int(((y == 0) & (pred == 1)).sum()),
         "fn": int(((y == 1) & (pred == 0)).sum()), "tn": int(((y == 0) & (pred == 0)).sum())}
    majority = int(train["spam"].mode()[0])
    texts = test["text"].tolist()
    report = {
        "accuracy": round(float((pred == y).mean()), 3),
        "baseline": round(float((y == majority).mean()), 3),
        "confusion": c,
        "precision": round(c["tp"] / (c["tp"] + c["fp"]), 3) if c["tp"] + c["fp"] else 0.0,
        "recall": round(c["tp"] / (c["tp"] + c["fn"]), 3) if c["tp"] + c["fn"] else 0.0,
        "missed": [t for t, a, p in zip(texts, y, pred) if a == 1 and p == 0],
        "false_alarms": [t for t, a, p in zip(texts, y, pred) if a == 0 and p == 1],
    }
    return model, report

def spammiest(model, n):
    vectoriser, nb = model.steps[0][1], model.steps[-1][1]
    ratio = nb.feature_log_prob_[1] - nb.feature_log_prob_[0]
    words = vectoriser.get_feature_names_out()
    order = np.argsort(-ratio, kind="stable")[:n]
    return [(str(words[i]), round(float(ratio[i]), 3)) for i in order]

def classify(model, texts):
    return ["spam" if p == 1 else "ham" for p in model.predict(list(texts))]

## Forge bonus: Naive Bayes from scratch

Multinomial Naive Bayes is only a few lines of NumPy. For each class `c`, count how often each word appears across that class's training messages, add a **smoothing** constant `α` to every count (so a word never seen in spam doesn't get probability 0, which would make one word veto a whole message), and divide by the class's total: that's `P(word | c)`. Score a message by `log P(c) + Σ count(word) × log P(word | c)` and pick the highest-scoring class. Turning scores into probabilities needs the **log-sum-exp** trick: subtract the largest score before exponentiating, so nothing overflows.

@@solution
import numpy as np

class NaiveBayes:
    def __init__(self, alpha=1.0):
        self.alpha = alpha

    def fit(self, X, y):
        X, y = np.asarray(X, dtype=float), np.asarray(y)
        self.classes_ = np.unique(y)
        counts = np.array([X[y == c].sum(axis=0) for c in self.classes_]) + self.alpha
        self.feature_log_prob_ = np.log(counts / counts.sum(axis=1, keepdims=True))
        self.class_log_prior_ = np.log(np.array([(y == c).mean() for c in self.classes_]))
        return self

    def predict_log_scores(self, X):
        return np.asarray(X, dtype=float) @ self.feature_log_prob_.T + self.class_log_prior_

    def predict(self, X):
        return self.classes_[self.predict_log_scores(X).argmax(axis=1)]

    def predict_proba(self, X):
        scores = self.predict_log_scores(X)
        scores = scores - scores.max(axis=1, keepdims=True)
        e = np.exp(scores)
        return e / e.sum(axis=1, keepdims=True)

def word_evidence(nb, vocab, x, n):
    ratio = nb.feature_log_prob_[1] - nb.feature_log_prob_[0]
    present = np.flatnonzero(np.asarray(x))
    contribution = np.asarray(x)[present] * ratio[present]
    order = present[np.argsort(-contribution, kind="stable")][:n]
    return [(vocab[i], round(float(x[i] * ratio[i]), 3)) for i in order]
