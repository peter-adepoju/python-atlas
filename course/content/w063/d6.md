import weakref

def track_instances(cls):
    live = weakref.WeakSet()
    original_init = cls.__init__

    def __init__(self, *args, **kwargs):
        original_init(self, *args, **kwargs)
        live.add(self)

    cls.__init__ = __init__
    cls.live_count = classmethod(lambda c: sum(1 for obj in live if isinstance(obj, c)))
    return cls

@@solution
from contextlib import contextmanager

_ABSENT = object()

@contextmanager
def overridden(obj, **values):
    saved = {name: obj.__dict__.get(name, _ABSENT) for name in values}
    for name, value in values.items():
        setattr(obj, name, value)
    try:
        yield obj
    finally:
        for name, old in saved.items():
            if old is _ABSENT:
                if name in obj.__dict__:
                    delattr(obj, name)
            else:
                setattr(obj, name, old)

## Forge: an object mapper

An **object mapper** lets you describe stored records as classes and work with them as objects. Write the four parts:

- **`Column(kind, *, default=REQUIRED, primary=False)`**, a data descriptor storing values in the instance's `__dict__`: setting a value that isn't a `kind` raises `TypeError`; changing a primary key that's already set to a **different** value raises `AttributeError`; and every set adds the column's name to the instance's set of changed columns, `obj.__dict__["_dirty"]`.
- **`Model`**, a base class using `__init_subclass__(cls, table=None, abstract=False)`: it collects every `Column` name (base classes first, in definition order) into `cls._columns`, a tuple. Unless the class is `abstract`, it must have **exactly one** primary column (`TypeError` otherwise), stored as `cls._primary`; its table name is `table` or the lower-cased class name plus `"s"`, stored as `cls._table`, and the class is registered in the dict `Model.tables`. `__init__(**values)` sets every column from the keywords or its default (`TypeError` for a missing required column or an unknown keyword). `dirty()` returns the sorted changed columns, `to_dict()` the column values and `repr` looks like `Reader(id=1, name='Ada', city='Lagos')`.
- **`unique(*names)`**, a class decorator storing the names in `cls._unique` (a `TypeError` for a name that isn't a column).
- **`Database()`**: `save(obj)` stores `obj.to_dict()` under its table and primary key, refusing (`ValueError`) a value in a unique column that another row (with a different key) already has; it clears the object's changed columns and returns the key. `get(cls, pk)` returns the **same object** for the same key while it stays remembered (an identity map), building a new instance from the stored row otherwise (with no changed columns), and raises `KeyError` for unknown keys; `forget()` clears the remembered objects. **`db.<table>`** (through `__getattr__`) returns a view of a registered table with `all()` (objects sorted by key), `count()` and `find_by_<column>(value)` (also through `__getattr__`), with `AttributeError` for unknown tables and columns.

@@solution
REQUIRED = object()

class Column:
    def __init__(self, kind, *, default=REQUIRED, primary=False):
        self.kind, self.default, self.primary = kind, default, primary

    def __set_name__(self, owner, name):
        self.name = name

    def __get__(self, obj, objtype=None):
        if obj is None:
            return self
        try:
            return obj.__dict__[self.name]
        except KeyError:
            raise AttributeError(self.name) from None

    def __set__(self, obj, value):
        if not isinstance(value, self.kind):
            raise TypeError(f"{self.name} must be {self.kind.__name__}")
        if self.primary and self.name in obj.__dict__ and obj.__dict__[self.name] != value:
            raise AttributeError("primary key can't change")
        obj.__dict__[self.name] = value
        obj.__dict__.setdefault("_dirty", set()).add(self.name)

class Model:
    tables = {}

    def __init_subclass__(cls, table=None, abstract=False, **kwargs):
        super().__init_subclass__(**kwargs)
        columns = []
        for klass in reversed(cls.__mro__):
            for name, value in vars(klass).items():
                if isinstance(value, Column) and name not in columns:
                    columns.append(name)
        cls._columns = tuple(columns)
        if abstract:
            return
        primaries = [c for c in columns if getattr(cls, c).primary]
        if len(primaries) != 1:
            raise TypeError(f"{cls.__name__} needs exactly one primary key")
        cls._primary = primaries[0]
        cls._table = table or cls.__name__.lower() + "s"
        Model.tables[cls._table] = cls

    def __init__(self, **values):
        for key in values:
            if key not in self._columns:
                raise TypeError(f"unknown column: {key}")
        for name in self._columns:
            if name in values:
                setattr(self, name, values[name])
            else:
                default = getattr(type(self), name).default
                if default is REQUIRED:
                    raise TypeError(f"missing column: {name}")
                setattr(self, name, default)

    def dirty(self):
        return sorted(self.__dict__.get("_dirty", ()))

    def to_dict(self):
        return {c: getattr(self, c) for c in self._columns}

    def __repr__(self):
        inner = ", ".join(f"{k}={v!r}" for k, v in self.to_dict().items())
        return f"{type(self).__name__}({inner})"

def unique(*names):
    def decorate(cls):
        for name in names:
            if name not in cls._columns:
                raise TypeError(f"{cls.__name__} has no column {name!r}")
        cls._unique = tuple(names)
        return cls
    return decorate

class TableView:
    def __init__(self, db, cls):
        self._db, self._cls = db, cls

    def all(self):
        rows = self._db._data.get(self._cls._table, {})
        return [self._db.get(self._cls, pk) for pk in sorted(rows)]

    def count(self):
        return len(self._db._data.get(self._cls._table, {}))

    def __getattr__(self, name):
        if name.startswith("find_by_"):
            column = name[len("find_by_"):]
            if column in self._cls._columns:
                def finder(value):
                    return [obj for obj in self.all() if getattr(obj, column) == value]
                return finder
        raise AttributeError(name)

class Database:
    def __init__(self):
        self._data = {}
        self._loaded = {}

    def save(self, obj):
        cls = type(obj)
        rows = self._data.setdefault(cls._table, {})
        pk = getattr(obj, cls._primary)
        for name in getattr(cls, "_unique", ()):
            value = getattr(obj, name)
            for other_pk, row in rows.items():
                if other_pk != pk and row[name] == value:
                    raise ValueError(f"duplicate {name}: {value!r}")
        rows[pk] = obj.to_dict()
        self._loaded[(cls._table, pk)] = obj
        obj.__dict__["_dirty"] = set()
        return pk

    def get(self, cls, pk):
        key = (cls._table, pk)
        if key in self._loaded:
            return self._loaded[key]
        row = self._data.get(cls._table, {}).get(pk)
        if row is None:
            raise KeyError(pk)
        obj = cls(**row)
        obj.__dict__["_dirty"] = set()
        self._loaded[key] = obj
        return obj

    def forget(self):
        self._loaded.clear()

    def __getattr__(self, name):
        if name.startswith("_"):
            raise AttributeError(name)
        cls = Model.tables.get(name)
        if cls is None:
            raise AttributeError(f"no table {name!r}")
        return TableView(self, cls)
:::

:::think title="Four tools, four jobs"
Each tool in today's mapper does the job it's best at. Descriptors own one attribute each: type checks, the primary-key rule and change tracking live in one place and serve every model. __init_subclass__ runs for every model automatically, because registration and column collection must never be forgotten. The class decorator is opt-in and visible, right for a constraint only some models want. And __getattr__ handles a family of names (find_by_anything) that can't be listed in advance. No metaclass was needed: when the simpler tools fit, they keep the code readable and combinable with other libraries.

## Forge bonus: observable models

:::exercise id=forge2 points=30 title="Forge bonus: observable models"
User interfaces and caches often need to know when an object changes. Write `Watched(default=None)`, a data descriptor (values in the instance's `__dict__`, reading the default when unset) that reports every **real** change (new value not equal to the old one) to its instance, and `Observable`, a base class with:

- `watched`, a tuple of the class's `Watched` names (base classes first), collected by `__init_subclass__`;
- `subscribe(callback, *names)`: `callback(obj, name, old, new)` is called for changes to the named attributes (all watched attributes when no names are given; `AttributeError` for a name that isn't watched). It returns a function that unsubscribes (safe to call twice). Listeners belong to each instance;
- `batch()`: a context manager (reentrant, yielding the object) during which notifications are held back; when the **outermost** batch ends, even after an error, each attribute that changed gets **one** notification with its value before the batch and its final value, in the order the attributes first changed, and none if it ended up back where it started.

Set up per-instance state lazily (the first time it's needed), since subclasses' `__init__` methods might not call yours.

Give `Observable` a method `_changed(name, old, new)` for the descriptor to call. Outside a batch it notifies at once; inside one it records `(old, new)` in a pending dict, keeping the **first** old value if the name is already there. A plain dict keeps first-insertion order for you.
@@solution
from contextlib import contextmanager

class Watched:
    def __init__(self, default=None):
        self.default = default

    def __set_name__(self, owner, name):
        self.name = name

    def __get__(self, obj, objtype=None):
        if obj is None:
            return self
        return obj.__dict__.get(self.name, self.default)

    def __set__(self, obj, value):
        old = self.__get__(obj)
        obj.__dict__[self.name] = value
        if old != value:
            obj._changed(self.name, old, value)

class Observable:
    def __init_subclass__(cls, **kwargs):
        super().__init_subclass__(**kwargs)
        names = []
        for klass in reversed(cls.__mro__):
            for name, value in vars(klass).items():
                if isinstance(value, Watched) and name not in names:
                    names.append(name)
        cls.watched = tuple(names)

    def _state(self):
        if "_listeners" not in self.__dict__:
            self.__dict__["_listeners"] = []
            self.__dict__["_batch_depth"] = 0
            self.__dict__["_pending"] = {}
        return self.__dict__

    def subscribe(self, callback, *names):
        for name in names:
            if name not in self.watched:
                raise AttributeError(f"{name!r} is not watched")
        entry = (callback, frozenset(names))
        self._state()["_listeners"].append(entry)

        def unsubscribe():
            listeners = self._state()["_listeners"]
            if entry in listeners:
                listeners.remove(entry)
        return unsubscribe

    def _notify(self, name, old, new):
        for callback, names in list(self._state()["_listeners"]):
            if not names or name in names:
                callback(self, name, old, new)

    def _changed(self, name, old, new):
        state = self._state()
        if state["_batch_depth"]:
            pending = state["_pending"]
            if name in pending:
                pending[name] = (pending[name][0], new)
            else:
                pending[name] = (old, new)
        else:
            self._notify(name, old, new)

    @contextmanager
    def batch(self):
        state = self._state()
        state["_batch_depth"] += 1
        try:
            yield self
        finally:
            state["_batch_depth"] -= 1
            if state["_batch_depth"] == 0:
                pending, state["_pending"] = state["_pending"], {}
                for name, (old, new) in pending.items():
                    if old != new:
                        self._notify(name, old, new)
