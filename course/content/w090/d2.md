import heapq


class UnknownPlace(LookupError):
    pass


class NoRoute(Exception):
    pass


def naismith_minutes(km, climb_m):
    return round(km / 5 * 60 + climb_m / 600 * 60)


class Route:
    def __init__(self, stops, km, climb_m):
        self.stops, self.km, self.climb_m = list(stops), km, climb_m

    @property
    def minutes(self):
        return naismith_minutes(self.km, self.climb_m)

    def __repr__(self):
        return f"Route({self.stops}, {self.km} km, {self.climb_m} m)"


class PathNetwork:
    def __init__(self, heights):
        self.heights = dict(heights)
        self.paths = {name: {} for name in self.heights}

    def _check(self, *names):
        for name in names:
            if name not in self.heights:
                raise UnknownPlace(name)

    def add_path(self, a, b, km):
        self._check(a, b)
        if km <= 0:
            raise ValueError("km must be positive")
        self.paths[a][b] = km
        self.paths[b][a] = km

    def climb(self, stops):
        return sum(max(0, self.heights[b] - self.heights[a]) for a, b in zip(stops, stops[1:]))

    def shortest(self, start, end):
        self._check(start, end)
        dist, prev, done = {start: 0}, {}, set()
        heap = [(0, start)]
        while heap:
            d, node = heapq.heappop(heap)
            if node in done:
                continue
            done.add(node)
            if node == end:
                break
            for nxt, km in sorted(self.paths[node].items()):
                if d + km < dist.get(nxt, float("inf")):
                    dist[nxt] = d + km
                    prev[nxt] = node
                    heapq.heappush(heap, (d + km, nxt))
        if end not in done:
            raise NoRoute(f"no route from {start} to {end}")
        stops = [end]
        while stops[-1] != start:
            stops.append(prev[stops[-1]])
        stops.reverse()
        return Route(stops, round(dist[end], 2), self.climb(stops))

class GeoNetwork(PathNetwork):
    """A path network whose junctions also have positions, in kilometres east and north."""

    def __init__(self, heights, positions):
        super().__init__(heights)
        self.positions = dict(positions)

    def straight_km(self, a, b):
        return math.dist(self.positions[a], self.positions[b])


def sample_geo():
    network = GeoNetwork(HEIGHTS, POSITIONS)
    for a, b, km in PATHS:
        network.add_path(a, b, km)
    return network


def grid_network(size):
    """A size-by-size grid of junctions named 'r-c', 1 km apart, with slightly longer paths to their neighbours."""
    names = {(r, c): f"{r}-{c}" for r in range(size) for c in range(size)}
    heights = {name: ((r * 13 + c * 7) % 9) * 20 for (r, c), name in names.items()}
    positions = {name: (float(c), float(r)) for (r, c), name in names.items()}
    network = GeoNetwork(heights, positions)
    for (r, c), name in names.items():
        if c + 1 < size:
            network.add_path(name, names[(r, c + 1)], 1.0 + ((r * 7 + c * 3) % 5) / 10)
        if r + 1 < size:
            network.add_path(name, names[(r + 1, c)], 1.0 + ((r * 3 + c * 5) % 5) / 10)
    return network

Ten random bytes (80 bits) make a 16-character code from a 32-letter alphabet. To guess one valid code out of a million in use, an attacker would need around `2**80 / 10**6` attempts, more than a billion billion: the code is the key, so it must come from `secrets`, never from `random`.

## Codes nobody can guess

A share link like `waymark.example/s/mfrggzdfmztwq2lk` is a **bearer** credential: whoever has it can open the route. So the code must be:

- **random from a secure source**: `secrets.token_bytes(10)` (Week 79), never `random`, whose sequence can be predicted from a few outputs, and never a counter or a hash of the route name;
- **long enough**: entropy is `length × log2(alphabet size)` bits; 80 bits is plenty for links that expire in days;
- **easy to type and read**: base32 in lower case avoids confusable characters and case problems in URLs.

For testing, inject the source of random bytes (as with clocks): production passes `secrets.token_bytes`, tests pass something predictable.

## Storing shares safely

Store each share as the **SHA-256 hash** of its code (like the reset tokens of Week 85), with the route, the time it was created, its expiry and whether it's been revoked. A leaked database then gives an attacker no working links. Looking a code up means hashing it and finding the hash.

Answer every failure the same way: unknown, expired and revoked codes all get the same `404` with the same message. Different answers would let an attacker learn which codes once existed. The owner, who is signed in, can list their shares and revoke them; viewers can't.

## GeoJSON out and in

**GeoJSON** (an open standard, RFC 7946) represents shapes as JSON. A route is a `Feature` whose `geometry` is a `LineString`: a list of `[x, y]` positions (in real GeoJSON, `[longitude, latitude]`, **in that order**, a common source of maps drawn in the sea), and whose `properties` hold anything else (name, distance, climb, time). Waymark's positions are kilometres on a local grid rather than longitude and latitude, so its export is GeoJSON in shape for the sake of the exercise; a real export would convert them.

Coming **in** (a walker importing a route someone sent), GeoJSON is untrusted input: check the `type`s, that `coordinates` is a list of at least two pairs of numbers, and then that each point is actually at a junction on the map (within a small tolerance), before treating it as a route.

import hashlib

class ShareStore:
    def __init__(self, clock):
        self.clock = clock
        self.shares = {}

    @staticmethod
    def _key(code):
        return hashlib.sha256(code.encode()).hexdigest()

    def create(self, route, days, code):
        expires = self.clock() + days * 86400
        self.shares[self._key(code)] = {"route": route, "expires": expires, "revoked": False, "views": 0}
        return expires

    def open(self, code):
        share = self.shares.get(self._key(code))
        if share is None or share["revoked"] or self.clock() >= share["expires"]:
            return None
        share["views"] += 1
        return share["route"]

    def revoke(self, code):
        share = self.shares.get(self._key(code))
        if share is None or share["revoked"]:
            return False
        share["revoked"] = True
        return True

    def views(self, code):
        share = self.shares.get(self._key(code))
        return None if share is None else share["views"]

import json
import math

def _number(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool)

def read_line(text):
    try:
        data = json.loads(text)
    except ValueError:
        raise ValueError("not JSON")
    if not isinstance(data, dict) or data.get("type") != "Feature":
        raise ValueError("expected a Feature")
    geometry = data.get("geometry") or {}
    if not isinstance(geometry, dict) or geometry.get("type") != "LineString":
        raise ValueError("expected a LineString")
    coords = geometry.get("coordinates")
    if not isinstance(coords, list) or len(coords) < 2:
        raise ValueError("need at least two points")
    points = []
    for i, point in enumerate(coords):
        if not (isinstance(point, list) and len(point) == 2 and all(_number(v) for v in point)):
            raise ValueError(f"point {i} is not two numbers")
        points.append((point[0], point[1]))
    return points

def junctions_at(positions, points, tolerance=0.05):
    names = []
    for i, point in enumerate(points):
        name = min(positions, key=lambda n: (math.dist(positions[n], point), n))
        if math.dist(positions[name], point) > tolerance:
            raise ValueError(f"point {i} is not at a junction")
        names.append(name)
    return names

import base64
import hashlib
import json
from waymark_domain import Route
from waymark_geo import sample_geo

def make_code(random_bytes, nbytes=10):
    return base64.b32encode(random_bytes(nbytes)).decode().rstrip("=").lower()

class ShareStore:
    def __init__(self, clock):
        self.clock, self.shares = clock, {}
    @staticmethod
    def _key(code):
        return hashlib.sha256(code.encode()).hexdigest()
    def create(self, route, days, code):
        expires = self.clock() + days * 86400
        self.shares[self._key(code)] = {"route": route, "expires": expires, "revoked": False, "views": 0}
        return expires
    def open(self, code):
        share = self.shares.get(self._key(code))
        if share is None or share["revoked"] or self.clock() >= share["expires"]:
            return None
        share["views"] += 1
        return share["route"]
    def revoke(self, code):
        share = self.shares.get(self._key(code))
        if share is None or share["revoked"] or self.clock() >= share["expires"]:
            return False
        share["revoked"] = True
        return True
    def views(self, code):
        share = self.shares.get(self._key(code))
        return None if share is None else share["views"]

def to_geojson(network, name, stops):
    km = sum(network.paths[a][b] for a, b in zip(stops, stops[1:]))
    route = Route(stops, km, network.climb(stops))
    return {"type": "Feature", "geometry": {"type": "LineString", "coordinates": [list(network.positions[s]) for s in stops]},
            "properties": {"name": name, "stops": list(stops), "km": round(km, 1), "climb_m": route.climb_m, "minutes": route.minutes}}

class SharingAPI:
    def __init__(self, clock, random_bytes, routes, network):
        self.store = ShareStore(clock)
        self.random_bytes, self.routes, self.network = random_bytes, routes, network

    def handle(self, method, path, body):
        parts = path.strip("/").split("/")
        if method == "POST" and parts == ["shares"]:
            return self._create(body)
        if method == "GET" and len(parts) == 2 and parts[0] == "s":
            return self._open(parts[1])
        if method == "DELETE" and len(parts) == 2 and parts[0] == "shares":
            return (204, None) if self.store.revoke(parts[1]) else (404, {"error": "no such share"})
        return 404, {"error": "not found"}

    def _create(self, body):
        try:
            data = json.loads(body)
        except ValueError:
            return 400, {"error": "bad request"}
        days = data.get("days") if isinstance(data, dict) else None
        if not (isinstance(days, int) and not isinstance(days, bool) and 1 <= days <= 30):
            return 400, {"error": "days must be 1 to 30"}
        if data.get("route") not in self.routes:
            return 404, {"error": "no such route"}
        code = make_code(self.random_bytes)
        expires = self.store.create(data["route"], days, code)
        return 201, {"code": code, "url": "/s/" + code, "expires": expires}

    def _open(self, code):
        name = self.store.open(code)
        if name is None:
            return 404, {"error": "this link doesn't work"}
        feature = to_geojson(self.network, name, self.routes[name])
        feature["properties"]["views"] = self.store.views(code)
        return 200, feature
