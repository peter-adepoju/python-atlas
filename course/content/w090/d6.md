    walks = [{"date": "2026-01-04", "km": 6.0}, {"date": "2026-03-01", "km": 3.5}, {"date": "2026-01-20", "km": 2.25},

def km_by_month(walks):
    if not walks:
        return {}
    totals = {}
    for walk in walks:
        month = walk["date"][:7]
        totals[month] = totals.get(month, 0) + walk["km"]
    year, month = map(int, min(totals).split("-"))
    last = max(totals)
    result = {}
    while True:
        key = f"{year:04d}-{month:02d}"
        result[key] = round(totals.get(key, 0), 1)
        if key == last:
            return result
        month += 1
        if month == 13:
            year, month = year + 1, 1

import base64
import binascii
import json

def encode_cursor(after):
    raw = json.dumps({"after": after}).encode()
    return base64.urlsafe_b64encode(raw).decode().rstrip("=")

def decode_cursor(text):
    try:
        raw = base64.urlsafe_b64decode(text + "=" * (-len(text) % 4))
        data = json.loads(raw)
    except (binascii.Error, ValueError, UnicodeDecodeError):
        raise ValueError("bad cursor") from None
    if not isinstance(data, dict) or set(data) != {"after"}:
        raise ValueError("bad cursor")
    after = data["after"]
    if isinstance(after, bool) or not isinstance(after, int) or after < 0:
        raise ValueError("bad cursor")
    return after

## What a freeze means

A **feature freeze** is a promise: from today, changes fix, harden or document what exists, and nothing new goes in. It's how a project stops moving long enough to be made reliable. Before freezing, check every story:

- **Done and verified**: it has passing tests, no failing ones, and a demo step that passes. Only these count as finished.
- **Done but unverified**: someone says it's done, but a test fails, there are no tests, or the demo doesn't show it. These block the freeze, because "done" without evidence is a guess.
- **Open**: a **could** that's open simply moves to "later" (a future version). An open **must** or **should** blocks the freeze: either finish it, or cut it openly and tell everyone.
- **Cut**: listed, so nobody wonders where it went.

:::exercise id=forge points=50 title="A feature-complete report"
Write `freeze_report(stories, tests, demo)`. `stories` is a list of dicts with `id` (like `"S11"`), `priority` (`"must"`, `"should"` or `"could"`) and `status` (`"done"`, `"open"` or `"cut"`); `tests` maps a story id to `(passed, failed)`; `demo` is a list of lines like `"PASS S12: share a route"` or `"FAIL S11: profile"` (the first word is the result, the second the story id followed by a colon).

Return `(lines, verdict)`, with one line per story in **numeric** id order:

- a cut story: `"S10 cut"`;
- an open story: `"S13 open (could)"` (with its priority);
- a done story: `"S11 done: TESTS, demo DEMO"`, where TESTS is `"no tests"` (none recorded, or 0 passed and 0 failed), `"N tests"` when none fail, or `"F of T tests failing"`; and DEMO is `"ok"` if a PASS line names it (and no FAIL line does), `"failing"` if a FAIL line names it, else `"missing"`. Add `" UNVERIFIED"` at the end unless it has tests, none failing, and its demo is ok.

The **blocking** stories are the unverified done stories and the open musts and shoulds. `verdict` is `"FREEZE"` if there are none, otherwise `"NOT READY: "` followed by the blocking ids in numeric order, joined by `", "`.

STORIES = [
    {"id": "S12", "priority": "should", "status": "done"}, {"id": "S1", "priority": "must", "status": "done"},

]
TESTS = {"S1": (31, 0), "S9": (12, 0), "S11": (14, 0), "S12": (22, 0)}
DEMO = ["PASS S1: waymark plan Mill Ridge", "PASS S9: avoiding steep paths", "PASS S11: the statistics page",
        "PASS S12: share Mill to Ridge and open it"]

def test_frozen():
    "everything promised is verified"
    lines, verdict = freeze_report(STORIES, TESTS, DEMO)
    assert lines == ["S1 done: 31 tests, demo ok", "S9 done: 12 tests, demo ok", "S10 cut", "S11 done: 14 tests, demo ok",
                     "S12 done: 22 tests, demo ok", "S13 open (could)"]
    assert verdict == "FREEZE"

def test_not_ready():
    "unverified and open stories block"
    tests = {**TESTS, "S12": (21, 1), "S9": (0, 0)}
    demo = [line for line in DEMO if "S11" not in line] + ["FAIL S1: waymark plan Mill Moon"]
    stories = STORIES + [{"id": "S14", "priority": "should", "status": "open"}, {"id": "S2", "priority": "must", "status": "done"}]
    lines, verdict = freeze_report(stories, tests, demo)
    assert lines == ["S1 done: 31 tests, demo failing UNVERIFIED", "S2 done: no tests, demo missing UNVERIFIED",
                     "S9 done: no tests, demo ok UNVERIFIED", "S10 cut", "S11 done: 14 tests, demo missing UNVERIFIED",
                     "S12 done: 1 of 22 tests failing, demo ok UNVERIFIED", "S13 open (could)", "S14 open (should)"]
    assert verdict == "NOT READY: S1, S2, S9, S11, S12, S14"
    assert freeze_report([], {}, []) == ([], "FREEZE")
@@solution
def _num(story_id):
    return int(story_id[1:])

def _demo_results(demo):
    results = {}
    for line in demo:
        words = line.split()
        if len(words) < 2:
            continue
        result, story = words[0], words[1].rstrip(":")
        if result == "FAIL":
            results[story] = "failing"
        elif result == "PASS" and results.get(story) != "failing":
            results[story] = "ok"
    return results

def freeze_report(stories, tests, demo):
    demos = _demo_results(demo)
    lines, blocking = [], []
    for story in sorted(stories, key=lambda s: _num(s["id"])):
        sid, status = story["id"], story["status"]
        if status == "cut":
            lines.append(f"{sid} cut")
            continue
        if status == "open":
            lines.append(f"{sid} open ({story['priority']})")
            if story["priority"] in ("must", "should"):
                blocking.append(sid)
            continue
        passed, failed = tests.get(sid, (0, 0))
        if passed + failed == 0:
            tested, ok = "no tests", False
        elif failed == 0:
            tested, ok = f"{passed} tests", True
        else:
            tested, ok = f"{failed} of {passed + failed} tests failing", False
        shown = demos.get(sid, "missing")
        line = f"{sid} done: {tested}, demo {shown}"
        if not (ok and shown == "ok"):
            line += " UNVERIFIED"
            blocking.append(sid)
        lines.append(line)
    verdict = "FREEZE" if not blocking else "NOT READY: " + ", ".join(blocking)
    return lines, verdict

## Bonus: the changelog

A **changelog** tells users what changed, version by version, in their words rather than the code's. Keep a Changelog (keepachangelog.com, by Olivier Lacan) suggests the format most projects use: newest first, grouped under headings. Waymark's is written from the same story records as the freeze report, so it can't disagree with them.

    "the changelog can't disagree with the stories"
    for finished, message in [({k: v for k, v in FINISHED.items() if k not in ("S3", "S11")}, "no sprint for S3"),
                              ({**FINISHED, "S10": 2}, "S10 is not done")]:

            changelog(STORIES, finished)
        except ValueError as e:

        else:

@@solution
def _num(story_id):
    return int(story_id[1:])

def changelog(stories, finished):
    by_id = {s["id"]: s for s in stories}
    ordered = sorted(stories, key=lambda s: _num(s["id"]))
    for story in ordered:
        if story["status"] == "done" and story["id"] not in finished:
            raise ValueError(f"no sprint for {story['id']}")
    for sid in sorted(finished, key=_num):
        if by_id.get(sid, {}).get("status") != "done":
            raise ValueError(f"{sid} is not done")
    lines = ["# Waymark changelog"]
    for sprint in sorted(set(finished.values()), reverse=True):
        lines += ["", f"## Sprint {sprint}"]
        for story in ordered:
            if finished.get(story["id"]) == sprint:
                lines.append(f"- {story['id']}: {story['title']}")
    rest = [s for s in ordered if s["status"] != "done"]
    if rest:
        lines += ["", "## Not done"]
        for story in rest:
            lines.append(f"- {story['id']}: {story['title']} ({'cut' if story['status'] == 'cut' else 'later'})")
    return "\n".join(lines) + "\n"
