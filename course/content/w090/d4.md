import base64
import json
import re

def check(data, schema, where="body"):
    kind = schema["type"]
    ok = {"object": lambda v: isinstance(v, dict), "string": lambda v: isinstance(v, str),
          "integer": lambda v: isinstance(v, int) and not isinstance(v, bool),
          "number": lambda v: isinstance(v, (int, float)) and not isinstance(v, bool)}[kind]
    if not ok(data):
        return [f"{where}: expected {kind}"]
    problems = []
    if kind == "object":
        for name, sub in schema.get("properties", {}).items():
            if name in data:
                problems += check(data[name], sub, f"{where}.{name}")
            elif name in schema.get("required", []):
                problems.append(f"{where}.{name}: is required")
        if schema.get("additional", True) is False:
            problems += [f"{where}.{n}: is not allowed" for n in data if n not in schema.get("properties", {})]
    elif kind in ("integer", "number"):
        if "minimum" in schema and data < schema["minimum"]:
            problems.append(f"{where}: must be at least {schema['minimum']}")
        if "maximum" in schema and data > schema["maximum"]:
            problems.append(f"{where}: must be at most {schema['maximum']}")
    return problems

WALK_SCHEMA = {"type": "object", "additional": False, "required": ["day", "km", "minutes"],
               "properties": {"day": {"type": "string"}, "km": {"type": "number", "minimum": 0.1},
                              "minutes": {"type": "integer", "minimum": 1, "maximum": 1440}}}

def encode_cursor(last_id):
    return base64.urlsafe_b64encode(json.dumps({"after": last_id}).encode()).decode().rstrip("=")

def decode_cursor(text):
    try:
        data = json.loads(base64.urlsafe_b64decode(text + "=" * (-len(text) % 4)))
    except (ValueError, TypeError):
        raise ValueError("bad cursor")
    after = data.get("after") if isinstance(data, dict) else None
    if not isinstance(after, int) or isinstance(after, bool):
        raise ValueError("bad cursor")
    return after

def page(rows, cursor=None, limit=3):
    after = decode_cursor(cursor) if cursor else None
    remaining = [r for r in rows if after is None or r["id"] > after]
    items = remaining[:limit]
    return {"items": items, "next": encode_cursor(items[-1]["id"]) if len(remaining) > limit else None}

def api_error(status, code, message, details=None):
    body = {"code": code, "message": message}
    if details is not None:
        body["details"] = details
    return status, {"error": body}

def split_version(path, current="v1", deprecated=("v0",)):
    m = re.match(r"/(v\d+)(/.*)?$", path)
    if not m:
        raise LookupError("no version in path")
    version, rest = m.group(1), m.group(2) or "/"
    if version == current:
        return version, rest, {}
    if version in deprecated:
        return version, rest, {"Deprecation": "true", "Sunset": "2027-06-30"}
    raise LookupError(f"unsupported version {version}")

class WalksAPI:
    def __init__(self):
        self.walks = []

    def handle(self, method, path, body=None, query=None):
        try:
            _, rest, headers = split_version(path)
        except LookupError as e:
            return (*api_error(404, "not_found", str(e)), {})
        status, data = self._route(method, rest, body, query or {})
        return status, data, headers

    def _route(self, method, rest, body, query):
        parts = rest.strip("/").split("/")
        if parts[0] != "walks" or len(parts) > 2:
            return api_error(404, "not_found", "no such route")
        if len(parts) == 2:
            if method != "GET":
                return api_error(405, "method_not_allowed", "method not allowed")
            walk = next((w for w in self.walks if str(w["id"]) == parts[1]), None)
            return (200, walk) if walk else api_error(404, "not_found", "no such walk")
        if method == "GET":
            try:
                limit = int(query.get("limit", 10))
            except ValueError:
                limit = 0
            if not 1 <= limit <= 50:
                return api_error(400, "invalid_request", "limit must be 1 to 50")
            try:
                return 200, page(self.walks, query.get("cursor"), limit)
            except ValueError as e:
                return api_error(400, "invalid_request", str(e))
        if method == "POST":
            problems = check(body, WALK_SCHEMA)
            if problems:
                return api_error(400, "invalid_request", "The request body has problems.", problems)
            walk = {"id": len(self.walks) + 1, **body}
            self.walks.append(walk)
            return 201, walk
        return api_error(405, "method_not_allowed", "method not allowed")

The second call uses a value from the first: the id the API assigned. That's a **scenario**, and it's how real clients use an API. Open `walks_api.py` above: it's yesterday's challenge, now the code under test.

## Contracts

A **contract** says what a response looks like: for `GET /v1/walks/N` with `200`, an object with an integer `id`, a string `day`, a number `km` and an integer `minutes`; for any `4xx`, the error envelope. Yesterday's schema checker checks requests; the same checker, pointed at **responses**, turns the contract into a test. Contract tests catch the changes that break clients without failing any other test: a field renamed in a refactor, an id that became a string, an error returned in a different shape.

## Coverage that matters

Line coverage (Week 72) says which lines ran. For an API, a more useful number is **route and status coverage**: of every documented (method, path, status) combination, which has a test actually produced? `DELETE /shares/CODE` returning `404` is a promise in the docs; if no test produces it, nobody knows whether it's kept. Record each response the tests see, compare with the documented list, and list what's missing.

## Scenarios

An **end-to-end scenario** is a list of calls, each with an expected status, where later steps can use values from earlier responses: `POST /v1/walks` then `GET /v1/walks/{id}`. Write scenarios as data (method, path template, body, expected status, values to capture) and run them with one function, so adding one is a few lines, and they read like the acceptance criteria they come from.

## Independence

API tests touch state, so they're the easiest to make depend on each other: a test that assumes the next id is 1 passes alone and fails after another test has created a walk. Each test (and scenario) builds its **own** API instance or database (a fixture), uses values it **captured** rather than assumed, and never relies on the order tests run in.

:::key
Contract tests check every response against a schema of its shape, so renamed fields and changed types fail a test before they reach clients. Measure route-and-status coverage against the documented list, not just lines. Write end-to-end scenarios as data with captured values (`{id}`) and run them with one function. Keep API tests independent: a fresh instance per test, captured values instead of assumptions, no reliance on order.
:::

import re

def check(data, schema, where="body"):
    kind = schema["type"]
    ok = {"object": lambda v: isinstance(v, dict), "string": lambda v: isinstance(v, str), "array": lambda v: isinstance(v, list),
          "integer": lambda v: isinstance(v, int) and not isinstance(v, bool),
          "number": lambda v: isinstance(v, (int, float)) and not isinstance(v, bool)}[kind]
    if not ok(data):
        return [f"{where}: expected {kind}"]
    problems = []
    if kind == "object":
        for name, sub in schema.get("properties", {}).items():
            if name in data:
                problems += check(data[name], sub, f"{where}.{name}")
            elif name in schema.get("required", []):
                problems.append(f"{where}.{name}: is required")
    elif kind == "array" and "items" in schema:
        for i, item in enumerate(data):
            problems += check(item, schema["items"], f"{where}[{i}]")
    return problems

def _matches(route, path):
    pattern = "^" + re.sub(r"\{[^}]+\}", r"[^/]+", route) + "$"
    return re.match(pattern, path) is not None

def _find(contracts, method, path, status):
    candidates = [(m, r, s) for (m, r, s) in contracts if m == method and _matches(r, path)]
    for key in candidates:
        if key[2] == status:
            return contracts[key]
    for key in candidates:
        if key[2] == f"{status // 100}xx":
            return contracts[key]
    return None

def contract_failures(responses, contracts):
    failures = []
    for method, path, status, data in responses:
        schema = _find(contracts, method, path, status)
        label = f"{method} {path} {status}"
        if schema is None:
            failures.append(f"{label}: no contract")
            continue
        failures += [f"{label}: {p}" for p in check(data, schema, "response")]
    return failures

import re

def run_scenario(api, steps):
    values, report = {}, []
    for n, step in enumerate(steps, start=1):
        names = re.findall(r"\{(\w+)\}", step["path"])
        unknown = [name for name in names if name not in values]
        if unknown:
            report.append(f"FAIL {n}: {step['method']} {step['path']}: unknown value {unknown[0]}")
            break
        path = re.sub(r"\{(\w+)\}", lambda m: str(values[m.group(1)]), step["path"])
        status, data, *_ = api.handle(step["method"], path, step.get("body"))
        if status != step["status"]:
            report.append(f"FAIL {n}: {step['method']} {path}: expected {step['status']}, got {status}")
            break
        for name, key in step.get("capture", {}).items():
            values[name] = data[key]
        report.append(f"PASS {n}: {step['method']} {path}")
    return report

def test_pages_cover_everything_once():

Run each scenario (a dict with a `name` and its `steps`) on a **fresh** API from `make_api()`, recording every response as `(method, path, status, data)` (wrap the API so recording happens automatically). Then check all recorded responses against the contracts and compute status coverage. Return text exactly like this:

```
FAILED S5 bad cursor: FAIL 2: GET /v1/walks?cursor=!!: expected 200, got 400
Contracts: 0 failures

```

Query strings in a path (after `?`) are split off and passed to the API as `query`, and paths are recorded without them.

import re
from urllib.parse import parse_qsl

def check(data, schema, where="body"):
    kind = schema["type"]
    ok = {"object": lambda v: isinstance(v, dict), "string": lambda v: isinstance(v, str), "array": lambda v: isinstance(v, list),
          "integer": lambda v: isinstance(v, int) and not isinstance(v, bool),
          "number": lambda v: isinstance(v, (int, float)) and not isinstance(v, bool)}[kind]
    if not ok(data):
        return [f"{where}: expected {kind}"]
    problems = []
    if kind == "object":
        for name, sub in schema.get("properties", {}).items():
            if name in data:
                problems += check(data[name], sub, f"{where}.{name}")
            elif name in schema.get("required", []):
                problems.append(f"{where}.{name}: is required")
    elif kind == "array" and "items" in schema:
        for i, item in enumerate(data):
            problems += check(item, schema["items"], f"{where}[{i}]")
    return problems

def _matches(route, path):
    return re.match("^" + re.sub(r"\{[^}]+\}", r"[^/]+", route) + "$", path) is not None

def contract_failures(responses, contracts):
    failures = []
    for method, path, status, data in responses:
        keys = [k for k in contracts if k[0] == method and _matches(k[1], path)]
        key = next((k for k in keys if k[2] == status), None) or next((k for k in keys if k[2] == f"{status // 100}xx"), None)
        if key is None:
            failures.append(f"{method} {path} {status}: no contract")
        else:
            failures += [f"{method} {path} {status}: {p}" for p in check(data, contracts[key], "response")]
    return failures

def status_coverage(expected, seen):
    missing = [(m, r, s) for m, r, s in expected if not any(sm == m and ss == s and _matches(r, sp) for sm, sp, ss in seen)]
    return (missing, round(100 * (len(expected) - len(missing)) / len(expected))) if expected else ([], 100)

def run_scenario(api, steps):
    values, report = {}, []
    for n, step in enumerate(steps, start=1):
        names = re.findall(r"\{(\w+)\}", step["path"])
        missing = [x for x in names if x not in values]
        if missing:
            report.append(f"FAIL {n}: {step['method']} {step['path']}: unknown value {missing[0]}")

            break
        for name, key in step.get("capture", {}).items():
            values[name] = data[key]
        report.append(f"PASS {n}: {step['method']} {path}")

class Recorder:
    def __init__(self, api, log):
        self.api, self.log = api, log

    def handle(self, method, path, body=None):
        bare, _, qs = path.partition("?")
        status, data, headers = self.api.handle(method, bare, body, dict(parse_qsl(qs)))
        self.log.append((method, bare, status, data))
        return status, data, headers

def api_report(make_api, scenarios, contracts, expected):
    log, failed = [], []
    for scenario in scenarios:
        report = run_scenario(Recorder(make_api(), log), scenario["steps"])
        if report and report[-1].startswith("FAIL"):
            failed.append(f"FAILED {scenario['name']}: {report[-1]}")
    contract = contract_failures(log, contracts)
    missing, percent = status_coverage(expected, [(m, p, s) for m, p, s, _ in log])
    lines = [f"Scenarios: {len(scenarios) - len(failed)} passed, {len(failed)} failed"] + failed
    lines.append(f"Contracts: {len(contract)} failures")
    lines += [f"- {c}" for c in contract]
    if missing:
        lines.append(f"Coverage: {percent}% (missing: " + ", ".join(f"{m} {r} {s}" for m, r, s in missing) + ")")
    else:
        lines.append(f"Coverage: {percent}%")
    return "\n".join(lines)
