import re

VERSION = re.compile(r"v?(0|[1-9]\d*)\.(0|[1-9]\d*)\.(0|[1-9]\d*)")

def sort_tags(tags):
    valid = [t for t in tags if VERSION.fullmatch(t)]
    return sorted(valid, key=lambda t: tuple(int(p) for p in VERSION.fullmatch(t).groups()))

import re

STAMP = re.compile(r"(\d+):(\d+):(\d+),(\d+)")

def _seconds(match):
    h, m, s, ms = map(int, match.groups())
    return h * 3600 + m * 60 + s + ms / 1000

def caption_span(srt):
    stamps = [_seconds(m) for m in STAMP.finditer(srt)]
    return round(stamps[-1] - stamps[0], 3) if stamps else 0.0

## Looking back usefully

A retrospective works when it's **safe** (it examines the process, not people: Norm Kerth's "prime directive" asks everyone to assume each person did the best they could with what they knew at the time), **specific** (evidence, not impressions), and **short on actions** (two or three that will actually happen, each with an owner).

A common format collects notes in four columns: what went **well**, what went **badly**, what we **learned**, and what we'll **try** next. The numbers come from the sprint records: **velocity** (points done per sprint) and its trend, how many stories were **carried over**, and how far **estimates** were from the actual effort (the ratio of actual to estimate: above 1 means it took longer than thought). For a solo capstone, you're the whole team; write the notes anyway.

:::exercise id=forge points=50 title="The retrospective report"
Write `retrospective(sprints, notes)`. Each sprint is a dict with `name`, `planned` and `done` (points), `carried` (a list of story ids not finished) and `estimates` (a list of `(story, estimate, actual)`). Each note is `(column, text)` with columns `"well"`, `"badly"`, `"learned"` and `"try"`. Return a list of lines:

1. per sprint: `"NAME: D of P points (R%)"` (R rounded), then, if any, `" carried: S5, S9"` added to the same line;
2. `"velocity: mean M, TREND"`: M the mean of `done` to 1 decimal place; TREND `"up"` if the last sprint's `done` is more than 10% above the first's, `"down"` if more than 10% below, otherwise `"flat"`;
3. `"estimates: N stories, mean ratio X, most underestimated STORY (Y)"`: ratios are actual over estimate, X their mean and Y the largest ratio, both to 2 decimal places (first story wins ties); or `"estimates: none"`;
4. `"notes: W well, B badly, L learned, T try"`;
5. `"actions:"` and then one line `"  - TEXT"` per `try` note, in order, or `"actions: none"` if there are none.

SPRINTS = [

def test_waymark():

    assert retrospective(SPRINTS, NOTES) == [
        "Sprint 1: 13 of 18 points (72%) carried: S5",
        "Sprint 2: 15 of 16 points (94%)",
        "Sprint 3: 15 of 15 points (100%)",
        "velocity: mean 14.3, up",
        "estimates: 8 stories, mean ratio 1.08, most underestimated S3 (1.60)",
        "notes: 2 well, 2 badly, 1 learned, 2 try",
        "actions:",
        "  - test with users in sprint 1",
        "  - split any story over 5 points",
    ]

    "one sprint, no estimates, no actions"
    sprint = {"name": "Only", "planned": 10, "done": 8, "carried": ["S1", "S2"], "estimates": []}
    assert retrospective([sprint], [("well", "fine")]) == [
        "Only: 8 of 10 points (80%) carried: S1, S2", "velocity: mean 8.0, flat", "estimates: none",
        "notes: 1 well, 0 badly, 0 learned, 0 try", "actions: none"]
    falling = [dict(sprint, done=10, carried=[]), dict(sprint, done=8, carried=[])]
    assert retrospective(falling, [])[2] == "velocity: mean 9.0, down"
@@solution
COLUMNS = ["well", "badly", "learned", "try"]

def retrospective(sprints, notes):
    lines = []
    for sprint in sprints:
        line = f"{sprint['name']}: {sprint['done']} of {sprint['planned']} points ({round(100 * sprint['done'] / sprint['planned'])}%)"
        if sprint["carried"]:
            line += " carried: " + ", ".join(sprint["carried"])
        lines.append(line)
    done = [s["done"] for s in sprints]
    first, last = done[0], done[-1]
    trend = "up" if last > first * 1.1 else "down" if last < first * 0.9 else "flat"
    lines.append(f"velocity: mean {sum(done) / len(done):.1f}, {trend}")
    ratios = [(story, actual / estimate) for s in sprints for story, estimate, actual in s["estimates"]]
    if ratios:
        mean = sum(r for _, r in ratios) / len(ratios)
        worst = max(ratios, key=lambda pair: pair[1])
        lines.append(f"estimates: {len(ratios)} stories, mean ratio {mean:.2f}, most underestimated {worst[0]} ({worst[1]:.2f})")
    else:
        lines.append("estimates: none")
    counts = {c: sum(1 for column, _ in notes if column == c) for c in COLUMNS}
    lines.append(f"notes: {counts['well']} well, {counts['badly']} badly, {counts['learned']} learned, {counts['try']} try")
    actions = [text for column, text in notes if column == "try"]
    if actions:
        lines.append("actions:")
        lines += [f"  - {text}" for text in actions]
    else:
        lines.append("actions: none")
    return lines
:::

## Bonus: following through

Retrospectives fail quietly when their actions are never done, and the same problems come back sprint after sprint. Tracking both takes a few lines.

:::exercise id=forge2 points=30 title="Bonus: actions and recurring problems"

ACTIONS = [{"action": "split stories over 5 points", "owner": "ada", "done": True},
           {"action": "test with users earlier", "owner": "ada", "done": False},
           {"action": "add a benchmark to CI", "owner": "bo", "done": True},
           {"action": "write ADRs as we go", "owner": "bo", "done": False}]

          ["Estimates were low.", "CI was slow", "Tests written late"]]

def test_follow_through():
    "were the actions done?"
    assert follow_through(ACTIONS, RETROS) == [
        "actions: 2 of 4 done (50%)",
        "  open: ada: test with users earlier",
        "  open: bo: write ADRs as we go",
        "recurring in 2 retros: estimates were low",
        "recurring in 3 retros: tests written late",
        "recurring in 2 retros: ci was slow",
    ]

    "a first retrospective"
    assert follow_through([], [["one problem", "one problem"]]) == ["actions: none"]
@@solution
import re

def _normal(text):
    return " ".join(re.sub(r"[^a-z ]", "", text.lower()).split())

def follow_through(actions, retros):
    if actions:
        done = sum(1 for a in actions if a["done"])
        lines = [f"actions: {done} of {len(actions)} done ({round(100 * done / len(actions))}%)"]
        lines += [f"  open: {a['owner']}: {a['action']}" for a in actions if not a["done"]]
    else:
        lines = ["actions: none"]
    seen, order = {}, []
    for retro in retros:
        for problem in {_normal(p) for p in retro}:
            seen[problem] = seen.get(problem, 0) + 1
        for p in retro:
            n = _normal(p)
            if n not in order:
                order.append(n)
    lines += [f"recurring in {seen[p]} retros: {p}" for p in order if seen[p] >= 2]
    return lines
