        self.edges = {}

        self.edges.setdefault(a, {})[b] = cost

        return sorted(self.edges.get(node, {}).items())

from pathkit.graph import Graph


def shortest(graph, start, end):
    heap, seen = [(0, start, [start])], set()
    while heap:
        cost, node, path = heapq.heappop(heap)
        if node == end:
            return cost, path
        if node in seen:
            continue
        seen.add(node)
        for nxt, step in graph.neighbours(node):
            heapq.heappush(heap, (cost + step, nxt, path + [nxt]))
    return None


def reachable(graph, start):
    found, todo = {start}, [start]
    while todo:
        for nxt, _ in graph.neighbours(todo.pop()):
            if nxt not in found:
                found.add(nxt)
                todo.append(nxt)
    return found

`minirepo.py` holds a whole small project, `pathkit`, as a dictionary from file path to file contents. A hundred lines is small enough to read in full, which is the point: today you'll build tools on something you can check by eye, and then use the real versions (`git grep`, your editor's "go to definition") on projects of a hundred thousand lines.

## Strategies for reading

Start from the **outside**: the README (what is it for?), how it's run (a command, a library's public names in `__init__.py`), and the **tests**, which are examples of use written by the people who know the code best. Then go **inside** with a question, not a plan to read everything: "where does a route get its cost?" Follow that one thread, writing down what you learn (a page of notes beats a perfect memory). Expect to understand a small part well and the rest vaguely. That's normal, and enough to start contributing.

## Search and definitions

**Search** is the most used tool: `git grep -n "neighbours("` lists every matching line with its file and line number, much faster than opening files. Search for a name to see where it's used; search for an error message to find where it's raised.

A **definitions index** answers "where is this defined?" without guessing from file names. The syntax tree knows: every `def` and `class` at the top of a module, and every method inside a class, with its line number. Editors build exactly this index for "go to definition".

## Imports and reading order

Imports show the **shape** of a project: which modules depend on which. Draw the graph (module to the modules it imports, ignoring the standard library), and three things stand out. Modules many others import are the **core** (in `pathkit`, `graph`): read them first. Modules nothing imports are **entry points** (the command line, the package's `__init__`) or dead code. And a **reading order** that puts every module after the ones it depends on (a topological order, with ties broken alphabetically to make it repeatable) lets you read without constantly jumping ahead. If there's a cycle, there's no such order: two modules import each other, and that's worth knowing too.

## Calls and tests

"Who calls this?" tells you how a function is really used, and so what a change to it might break. The syntax tree finds every call by name, both `from_pairs(...)` and `graph.neighbours(...)`. Search by name can't know types, so treat the results as candidates: in `pathkit`, the callers of `add` include two functions that call `set.add`, nothing to do with `Graph.add`. That's true of every tool this week.

Two more questions are worth asking on the first day: which modules no test imports (changes there are risky), and which functions nothing calls (perhaps called from outside, perhaps dead).

:::key
Read from the outside in: README, entry points, tests, then one question at a time, with notes. Search with `git grep -n`. Index definitions (functions, classes, methods) with the syntax tree. Map internal imports: much-imported modules are the core, unimported ones are entry points or dead code, and a topological order (alphabetical ties) gives a reading order; a cycle means none exists. Find callers by name, as candidates. Note untested modules and uncalled functions.
:::

import ast

def definitions(repo):
    index = {}
    for path in sorted(repo):
        if not path.endswith(".py"):
            continue
        for node in ast.parse(repo[path]).body:
            if isinstance(node, (ast.FunctionDef, ast.ClassDef)):
                index.setdefault(node.name, []).append(f"{path}:{node.lineno}")
                if isinstance(node, ast.ClassDef):
                    for sub in node.body:
                        if isinstance(sub, ast.FunctionDef):
                            index.setdefault(f"{node.name}.{sub.name}", []).append(f"{path}:{sub.lineno}")
    return index

import heapq

def reading_order(graph):
    waiting = {m: set(deps) for m, deps in graph.items()}
    ready = [m for m, deps in waiting.items() if not deps]
    heapq.heapify(ready)
    order = []
    while ready:
        module = heapq.heappop(ready)
        order.append(module)
        for other, deps in waiting.items():
            if module in deps:
                deps.remove(module)
                if not deps:
                    heapq.heappush(ready, other)
    if len(order) < len(graph):
        raise ValueError("import cycle among: " + ", ".join(sorted(set(graph) - set(order))))
    return order

import ast

def callers(repo, name):
    found = []
    for path in sorted(repo):
        if not path.endswith(".py"):
            continue
        funcs = []
        for node in ast.parse(repo[path]).body:
            if isinstance(node, ast.FunctionDef):
                funcs.append((node.name, node))
            elif isinstance(node, ast.ClassDef):
                funcs += [(f"{node.name}.{s.name}", s) for s in node.body if isinstance(s, ast.FunctionDef)]
        for label, func in funcs:
            for n in ast.walk(func):
                if isinstance(n, ast.Call):
                    target = n.func
                    if (isinstance(target, ast.Name) and target.id == name) or (isinstance(target, ast.Attribute) and target.attr == name):
                        found.append(f"{path}:{label}")
                        break
    return found

        for other, deps in waiting.items():
            if module in deps:
                deps.remove(module)
                if not deps:
                    heapq.heappush(ready, other)
    return order


def _called_names(source):
    names = set()
    for n in ast.walk(ast.parse(source)):
        if isinstance(n, ast.Call):
            if isinstance(n.func, ast.Name):
                names.add(n.func.id)
            elif isinstance(n.func, ast.Attribute):
                names.add(n.func.attr)
    return names


def orientation(repo):
    py = sorted(p for p in repo if p.endswith(".py"))
    modules = {module_name(p): p for p in py}
    graph = {m: sorted(imports(repo[p]) & set(modules)) for m, p in modules.items()}
    total = sum(len(src.splitlines()) for src in repo.values())
    lines = [f"files: {len(repo)}, {len(py)} Python, {total} lines"]
    sizes = sorted(((len(repo[p].splitlines()), p) for p in py), key=lambda t: (-t[0], t[1]))[:3]
    lines.append("largest: " + ", ".join(f"{p} ({n})" for n, p in sizes))
    counts = {m: sum(1 for deps in graph.values() if m in deps) for m in graph}
    top = sorted((m for m in counts if counts[m]), key=lambda m: (-counts[m], m))[:3]
    lines.append("most imported: " + (", ".join(f"{m} ({counts[m]})" for m in top) or "none"))
    source = sorted(m for m in graph if not _is_test(m))
    entry = [m for m in source if counts[m] == 0]
    tested = {d for m, deps in graph.items() if _is_test(m) for d in deps}
    untested = [m for m in source if m not in tested]
    called = set().union(*(_called_names(repo[p]) for p in py)) if py else set()
    defined = sorted(node.name for m in source for node in ast.parse(repo[modules[m]]).body
                     if isinstance(node, ast.FunctionDef) and not node.name.startswith("_"))
    uncalled = [name for name in defined if name not in called]
    lines.append("entry points: " + (", ".join(entry) or "none"))
    lines.append("untested: " + (", ".join(untested) or "none"))
    lines.append("not called in the repo: " + (", ".join(uncalled) or "none"))
    lines.append("start with: " + (", ".join(_order(graph)[:3]) or "none"))
    return lines
