"pathkit/graph.py": '''class Graph:
    """An undirected graph with a cost on each edge."""

    def __init__(self):
        self.edges = {}

    def add(self, a, b, cost):
        self.edges.setdefault(a, {})[b] = cost
        self.edges.setdefault(b, {})[a] = cost

    def neighbours(self, node):
        return sorted(self.edges.get(node, {}).items())


def from_pairs(pairs):
    graph = Graph()
    for a, b, cost in pairs:
        graph.add(a, b, cost)
    return graph
''',
"pathkit/search.py": '''import heapq

from pathkit.graph import Graph


def shortest(graph, start, end):
    heap, seen = [(0, start, [start])], set()
    while heap:
        cost, node, path = heapq.heappop(heap)
        if node == end:
            return cost, path
        if node in seen:
            continue
        seen.add(node)
        for nxt, step in graph.neighbours(node):
            heapq.heappush(heap, (cost + step, nxt, path + [nxt]))
    return None


def reachable(graph, start):
    found, todo = {start}, [start]
    while todo:
        for nxt, _ in graph.neighbours(todo.pop()):
            if nxt not in found:
                found.add(nxt)
                todo.append(nxt)
    return found
''',

import re
from datetime import datetime

PATTERN = re.compile(r"Misc/NEWS\.d/next/([\w-]+)/(\d{4}-\d\d-\d\d-\d\d-\d\d-\d\d)\.gh-issue-(\d+)\.(\w+)\.rst")

def parse_news_path(path):
    match = PATTERN.fullmatch(path)
    if not match:
        raise ValueError(f"not a NEWS entry: {path}")
    section, stamp, issue, nonce = match.groups()
    return {"section": section, "when": datetime.strptime(stamp, "%Y-%m-%d-%H-%M-%S"), "issue": int(issue), "nonce": nonce}

## From issue to plan

A first contribution goes better with a plan written before any code: where the code is (search and definitions), what else uses it (callers, so you know what might break), which tests cover it (the test modules that import it, directly or not), what kind of change it is (bug, docs, feature), and so where the NEWS entry goes and which branches it belongs on. Each of those is a tool from this week. Put them together and you have the first comment of a good pull request description.

import ast
import re

TARGETS = {"feature": set(), "bug": {"bugfix"}, "docs": {"bugfix"}, "security": {"bugfix", "security"}}

def _module_name(path):
    parts = path[:-3].split("/")
    if parts[-1] == "__init__":
        parts = parts[:-1]
    return ".".join(parts)

def _imports(source):
    found = set()
    for node in ast.walk(ast.parse(source)):
        if isinstance(node, ast.Import):
            found |= {a.name for a in node.names}
        elif isinstance(node, ast.ImportFrom) and node.module and node.level == 0:
            found.add(node.module)
    return found

def _definitions(repo):
    index = {}
    for path in sorted(p for p in repo if p.endswith(".py")):
        for node in ast.parse(repo[path]).body:
            if isinstance(node, (ast.FunctionDef, ast.ClassDef)):
                index.setdefault(node.name, []).append(f"{path}:{node.lineno}")
                if isinstance(node, ast.ClassDef):
                    for sub in node.body:
                        if isinstance(sub, ast.FunctionDef):
                            index.setdefault(f"{node.name}.{sub.name}", []).append(f"{path}:{sub.lineno}")
    return index

def _callers(repo, name):
    found = []
    for path in sorted(p for p in repo if p.endswith(".py")):
        funcs = []
        for node in ast.parse(repo[path]).body:
            if isinstance(node, ast.FunctionDef):
                funcs.append((node.name, node))
            elif isinstance(node, ast.ClassDef):
                funcs += [(f"{node.name}.{s.name}", s) for s in node.body if isinstance(s, ast.FunctionDef)]
        for label, func in funcs:
            for n in ast.walk(func):
                if isinstance(n, ast.Call):
                    t = n.func
                    if (isinstance(t, ast.Name) and t.id == name) or (isinstance(t, ast.Attribute) and t.attr == name):
                        found.append(f"{path}:{label}")
                        break
    return found

def _is_test(module):
    return module.startswith("tests.") or module.split(".")[-1].startswith("test_")

def _affected(graph, module):
    importers = {}
    for m, deps in graph.items():
        for d in deps:
            importers.setdefault(d, set()).add(m)
    seen, todo = set(), [module]
    while todo:
        for m in importers.get(todo.pop(), ()):
            if m not in seen:
                seen.add(m)
                todo.append(m)
    return {m for m in seen if _is_test(m)}

def _unique(items):
    out = []
    for item in items:
        if item not in out:
            out.append(item)
    return out

def _join(items):
    return ", ".join(items) if items else "none"

def contribution_plan(repo, issue, branches):
    index = _definitions(repo)
    names = []
    for word in re.findall(r"[A-Za-z_][A-Za-z0-9_.]*", issue["title"]):
        word = word.rstrip(".")
        key = word if word in index else word.split(".")[-1] if word.split(".")[-1] in index else None
        if key:
            names.append(key)
    names = _unique(names)
    places = _unique(p for n in names for p in index[n])
    called = _unique(c for n in names for c in _callers(repo, n.split(".")[-1]))
    modules = {_module_name(p): p for p in repo if p.endswith(".py")}
    graph = {m: sorted(_imports(repo[p]) & set(modules)) for m, p in modules.items()}
    tests = sorted(set().union(*(_affected(graph, _module_name(p.split(":")[0])) for p in places)) if places else set())
    labels = set(issue["labels"])
    kind = "docs" if "docs" in labels else "feature" if "type-feature" in labels else "security" if "type-security" in labels else "bug"
    section = {"docs": "Documentation", "security": "Security"}.get(kind, "Library")
    versions = sorted((v for v, s in branches.items() if s in TARGETS[kind]), key=lambda v: tuple(map(int, v.split("."))), reverse=True)
    return [f"names: {_join(names)}", f"defined in: {_join(places)}", f"called by: {_join(called)}", f"tests to run: {_join(tests)}",
            f"kind: {kind}", f"news section: {section}", "backports: " + _join([f"needs backport to {v}" for v in versions])]

from datetime import date

def timeline(events):
    days = [(date.fromisoformat(d), kind) for d, kind in events]
    def first(kind, after=None):
        return next((d for d, k in days if k == kind and (after is None or d >= after)), None)
    opened = first("issue opened")
    response = next((d for d, k in days if k in ("comment", "review") and opened and d >= opened), None)
    pr, merged = first("pr opened"), first("merged")
    first_review = first("review")
    pushes = sum(1 for d, k in days if k == "changes pushed" and first_review and d >= first_review)
    return [
        f"days to first response: {(response - opened).days if opened and response else '-'}",
        f"days from pull request to merge: {(merged - pr).days if pr and merged else '-'}",
        f"review rounds: {sum(1 for _, k in days if k == 'review')}",
        f"pushes after review: {pushes}",
        f"ci failures: {sum(1 for _, k in days if k == 'ci failed')}",
        f"total days: {(days[-1][0] - days[0][0]).days if days else 0}",
    ]
