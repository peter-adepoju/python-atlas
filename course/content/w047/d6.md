import heapq

def _smash(weights):
    heap = [-w for w in weights]
    heapq.heapify(heap)
    n = 0
    while len(heap) > 1:
        a, b = -heapq.heappop(heap), -heapq.heappop(heap)
        n += 1
        if a != b:
            heapq.heappush(heap, -(a - b))
    return (-heap[0] if heap else 0), n

def last_stone(weights):
    return _smash(weights)[0]

def turns(weights):
    return _smash(weights)[1]

import heapq

def join_order(lengths):
    heap = list(lengths)
    heapq.heapify(heap)
    order = []
    while len(heap) > 1:
        a, b = heapq.heappop(heap), heapq.heappop(heap)
        order.append((a, b))
        heapq.heappush(heap, a + b)
    return order

def cheapest_join(lengths):
    return sum(a + b for a, b in join_order(lengths))

## Forge: a cluster simulator

A small computing cluster has `machines` identical machines, numbered from 0. Jobs arrive over time, each with a duration and a priority (smaller is more urgent). Build an **event-driven simulation**: rather than ticking a clock minute by minute, keep a heap of future events and jump from one to the next.

Write `Cluster(machines)` (`ValueError` if fewer than 1) with:

- `submit(time, name, duration, priority)`: a job named `name` will arrive at `time`. Raise `ValueError` for a name already submitted, a negative time or a duration that isn't positive.
- `cancel(time, name)`: at `time`, if the job is **waiting** (arrived and not started), it's removed and never runs; otherwise (not yet arrived, running, finished, or unknown) nothing happens.
- `run()`: simulate everything and return the **log**, a list of `(time, what, name, machine)` tuples, where `what` is `"start"`, `"finish"` or `"cancel"` (with `machine` `None` for a cancel).

The rules, which decide every tie:

1. Events at the same time are handled in this order: all **finishes** (in machine-number order), then all **cancellations**, then all **arrivals** (cancellations and arrivals each in the order `submit` and `cancel` were called). So a job that finishes at 5 frees its machine for a job that arrives at 5, and a cancellation at 5 can't touch a job arriving at 5.
2. After all the events at one time have been handled, **dispatch**: while a machine is free and a job is waiting, start the most urgent waiting job (smallest priority, then earliest arrival, then name) on the **lowest-numbered** free machine. It finishes at the current time plus its duration.

After `run()`:

- `waits()`: a dict from each job that started to how long it waited (start minus arrival);
- `busiest(k)`: the k machines with the most total running time, as `(machine, busy_time)` pairs, busiest first, ties to the lower number;
- `longest_waits(k)`: the k longest waits as `(name, wait)` pairs, longest first, ties alphabetical.

Use heaps for everything that needs a "next": the events, the waiting jobs (with lazy deletion for cancellations) and the free machines. The checks compare you with a simple minute-by-minute simulation on random workloads, and run 20,000 jobs on 8 machines.

import random
def reference(machines, jobs, cancels):
    t, log = 0, []
    waiting, running = [], {}
    starts = {}
    free = list(range(machines))
    end_time = max([a for a, *_ in jobs] + [c for c, _ in cancels] + [0]) + sum(d for _, _, d, _ in jobs) + 1
    for t in range(end_time + 1):
        for m in sorted(running):
            name, finish = running[m]
            if finish == t:
                log.append((t, "finish", name, m))
                del running[m]
                free.append(m)
        for ct, name in cancels:
            if ct == t and any(w[2] == name for w in waiting):
                waiting = [w for w in waiting if w[2] != name]
                log.append((t, "cancel", name, None))
        for a, name, d, p in jobs:
            if a == t:
                waiting.append((p, a, name))
        while free and waiting:
            waiting.sort()
            p, a, name = waiting.pop(0)
            m = min(free)
            free.remove(m)
            d = next(j[2] for j in jobs if j[1] == name)
            running[m] = (name, t + d)
            starts[name] = t
            log.append((t, "start", name, m))
    log.sort(key=lambda e: (e[0], {"finish": 0, "cancel": 1, "start": 2}[e[1]]))

Make every event a tuple that sorts correctly on its own: `(time, kind, tie, data)` with kind 0 for finish, 1 for cancel and 2 for arrival, and `tie` the machine number for finishes and a counter value for the others. In `run`, pop every event with the same time as the first, handle each, then dispatch. Waiting jobs go in a heap as lists `[priority, arrival, name]`, with a dict from name to entry, so a cancellation can overwrite the name with `None`.
@@solution
import heapq, itertools

FINISH, CANCEL, ARRIVE = 0, 1, 2

class Cluster:
    def __init__(self, machines):
        if machines < 1:
            raise ValueError("need at least one machine")
        self._machines = machines
        self._events = []
        self._seq = itertools.count()
        self._jobs = {}            # name -> [arrival, duration, priority]
        self._log = None
        self._starts = {}
        self._busy = [0] * machines

    def submit(self, time, name, duration, priority):
        if name in self._jobs:
            raise ValueError(f"duplicate job {name!r}")
        if duration <= 0 or time < 0:
            raise ValueError("bad time or duration")
        self._jobs[name] = [time, duration, priority]
        heapq.heappush(self._events, (time, ARRIVE, next(self._seq), name))

    def cancel(self, time, name):
        heapq.heappush(self._events, (time, CANCEL, next(self._seq), name))

    def run(self):
        log, ready, waiting = [], [], {}
        free = list(range(self._machines))
        heapq.heapify(free)
        events = self._events
        while events:
            now = events[0][0]
            while events and events[0][0] == now:
                _, kind, _, data = heapq.heappop(events)
                if kind == FINISH:
                    name, machine = data
                    log.append((now, "finish", name, machine))
                    heapq.heappush(free, machine)
                elif kind == CANCEL:
                    if data in waiting:
                        waiting[data][-1] = None
                        del waiting[data]
                        log.append((now, "cancel", data, None))
                else:
                    arrival, duration, priority = self._jobs[data]
                    entry = [priority, arrival, data]
                    waiting[data] = entry
                    heapq.heappush(ready, entry)
            while free and ready:
                priority, arrival, name = heapq.heappop(ready)
                if name is None:
                    continue
                del waiting[name]
                machine = heapq.heappop(free)
                duration = self._jobs[name][1]
                self._starts[name] = now
                self._busy[machine] += duration
                log.append((now, "start", name, machine))
                heapq.heappush(events, (now + duration, FINISH, machine, (name, machine)))
        self._log = log
        return log

    def waits(self):
        return {name: start - self._jobs[name][0] for name, start in self._starts.items()}

    def busiest(self, k):
        return heapq.nsmallest(k, enumerate(self._busy), key=lambda p: (-p[1], p[0]))

    def longest_waits(self, k):
        return heapq.nsmallest(k, self.waits().items(), key=lambda p: (-p[1], p[0]))

:::think title="Rules for ties"
Most of the specification is about **ties**: what happens when a job finishes at the same moment another arrives, which of two equally urgent jobs goes first, which free machine gets the job. A simulation without these rules still runs, but two correct-looking programs give different answers, and nobody can say which is right. Writing the rules down, and encoding them in the heap entries themselves (kind before counter, priority before arrival before name), is what makes the simulation **deterministic**: the same input always gives the same log, which is what lets the checks compare you with a completely different simulation. The same thinking applies whenever you use a heap: decide what decides a tie, and put it in the tuple.
:::

import heapq, re
from collections import Counter

LEVELS = {"DEBUG", "INFO", "WARNING", "ERROR"}
STAMP = re.compile(r"\d{4}-\d\d-\d\dT\d\d:\d\d:\d\d")

class LogReader:
    """Good lines from one sorted log file, as (stamp, level, message); counts bad ones."""
    def __init__(self, path):
        self._file = open(path)
        self.bad = 0
        self._last = ""

    def __iter__(self):
        return self

    def __next__(self):
        for line in self._file:
            parts = line.rstrip("\n").split(" ", 2)
            if (len(parts) == 3 and STAMP.fullmatch(parts[0]) and parts[1] in LEVELS
                    and parts[0] >= self._last):
                self._last = parts[0]
                return parts[0], parts[1], parts[2]
            if line.strip():
                self.bad += 1
        self._file.close()
        raise StopIteration

def merge_logs(paths, dst):
    readers = [LogReader(p) for p in paths]
    names = [p.rsplit("/", 1)[-1].removesuffix(".log") for p in paths]
    heap = []
    for i, r in enumerate(readers):
        for entry in r:
            heapq.heappush(heap, (entry[0], i, entry))
            break
    with open(dst, "w") as out:
        while heap:
            stamp, i, (_, level, message) = heapq.heappop(heap)
            out.write(f"{stamp} {names[i]} {level} {message}\n")
            for entry in readers[i]:
                heapq.heappush(heap, (entry[0], i, entry))
                break
    return sum(r.bad for r in readers)

def top_errors(path, k):
    counts = Counter()
    with open(path) as f:
        for line in f:
            parts = line.rstrip("\n").split(" ", 3)
            if len(parts) == 4 and parts[2] == "ERROR":
                counts[parts[3]] += 1
    return heapq.nsmallest(k, counts.items(), key=lambda p: (-p[1], p[0]))

def busiest_minute(path):
    counts = Counter()
    with open(path) as f:
        for line in f:
            if line.strip():
                counts[line[:16]] += 1
    if not counts:
        return None
    return min(counts.items(), key=lambda p: (-p[1], p[0]))[0]
