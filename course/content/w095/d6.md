import re


def slugify(title, max_length=40):
    """Turn a title into a lower-case slug of letters, digits and single hyphens."""
    slug = re.sub(r"[^a-z0-9]+", "-", title.lower()).strip("-")
    if len(slug) > max_length:
        slug = slug[:max_length].rstrip("-")
    return slug or "untitled"

    assert hunk_header(1, 1, 1, 1) == "@@ -1 +1 @@"
    assert hunk_header(0, 0, 1, 2) == "@@ -0,0 +1,2 @@"
@@solution
def _side(start, count):
    return f"{start}" if count == 1 else f"{start},{count}"

def hunk_header(old_start, old_count, new_start, new_count):
    return f"@@ -{_side(old_start, old_count)} +{_side(new_start, new_count)} @@"

def news_by_section(paths):
    counts = {}
    prefix = "Misc/NEWS.d/next/"
    for path in paths:
        if path.startswith(prefix):
            section = path[len(prefix):].split("/")[0]
            counts[section] = counts.get(section, 0) + 1
    return dict(sorted(counts.items()))

## A second bug

The report: "`slugify('Mill to Ridge: a long walk over the hill and back', 21)` gives `mill-to-ridge-a-long-`, with a hyphen at the end." Reproduced on `main`; minimal case: any title whose slug, cut at `max_length`, ends at a hyphen. Cause: the slug was cut after stripping hyphens from its ends, so the cut could make a new one. The fix (in `slugs.py`) strips again after cutting: `.rstrip("-")`. What's left is the part reviewers check first: tests that fail without the fix, and that would catch the other mistakes someone might make in this function.

:::exercise id=forge points=50 title="Tests that kill every mutant"
Write a `unittest` suite for `slugify` in `slugs.py` that passes on it and catches all six broken versions: the original bug (no strip after cutting), no lower-casing, hyphen runs kept, ends not stripped, cut one character too short, and no `"untitled"` fallback. Include a regression test that mentions the bug in a comment, and use `subTest` for at least one group of cases.
@@starter
import unittest
from slugs import slugify

class TestSlugify(unittest.TestCase):
    def test_simple(self):
        self.assertEqual(slugify("walk"), "walk")
@@tests
GOOD = open("slugs.py").read()
BROKEN = {
    "the original bug": GOOD.replace('slug = slug[:max_length].rstrip("-")', "slug = slug[:max_length]"),
    "no lower-casing": GOOD.replace("title.lower()", "title"),
    "hyphen runs kept": GOOD.replace('r"[^a-z0-9]+"', 'r"[^a-z0-9]"'),
    "ends not stripped": GOOD.replace('title.lower()).strip("-")', "title.lower())"),
    "cut too short": GOOD.replace("slug[:max_length]", "slug[:max_length - 1]"),
    "no fallback": GOOD.replace('return slug or "untitled"', "return slug"),
}

def test_suite():
    "all six caught"
    for name, code in BROKEN.items():
        assert code != GOOD, name
    assert "subTest" in source, "use subTest for a group of cases"
    assert "#" in source, "say in a comment what the regression test is for"
    r = pytest_run(source)
    assert r["failed"] == [] and r["passed"], r["output"][-600:]
    results = against(source, "slugs", {"correct": GOOD, **BROKEN})
    write_file("slugs.py", GOOD)
    missed = [k for k in BROKEN if not results[k]]
    assert results["correct"] == [] and not missed, f"not caught: {missed}"
@@hint
The regression case is in the report. Then: a title with capitals, one with several spaces or punctuation in a row, one starting and ending with punctuation, a slug that's exactly `max_length` long after cutting, and a title with no letters or digits at all.
@@solution
import unittest
from slugs import slugify

class TestSlugify(unittest.TestCase):
    def test_cut_does_not_end_with_hyphen(self):
        # regression: cutting at max_length left a trailing hyphen
        self.assertEqual(slugify("Mill to Ridge: a long walk over the hill and back", 21), "mill-to-ridge-a-long")

    def test_shapes(self):
        cases = [("Mill to Ridge", "mill-to-ridge"), ("Mill -- to  Ridge", "mill-to-ridge"), ("  ...Lake loop!  ", "lake-loop")]
        for title, expected in cases:
            with self.subTest(title=title):
                self.assertEqual(slugify(title), expected)

    def test_exact_length_kept(self):
        self.assertEqual(slugify("abcde fghij", 11), "abcde-fghij")
        self.assertEqual(slugify("abcde fghij", 10), "abcde-fghi")

    def test_nothing_left(self):
        self.assertEqual(slugify("???"), "untitled")

:::exercise id=forge2 points=30 title="Bonus: resolving conflict blocks"
Write `resolve(text, choose)`: `text` may contain conflict blocks, each a line `<<<<<<< ...`, our lines, a line `=======`, their lines, and a line `>>>>>>> ...`. Replace each block with the lines returned by `choose(ours, theirs)` (two lists of lines). Return `(resolved_text, blocks)`, keeping a final newline if `text` had one. Raise `ValueError("unfinished conflict")` if a block isn't closed, and `ValueError("stray conflict marker")` for a `=======` or `>>>>>>>` line outside a block.
@@starter
def resolve(text, choose):
    ___
@@tests
TEXT = """def shorten(text, width, placeholder=" [...]"):
<<<<<<< ours
    if length + extra + len(placeholder) > width:
=======
    if length + extra + len(placeholder) >= max_width:
>>>>>>> main
        break
<<<<<<< ours
    return " ".join(kept) + placeholder
=======
    return " ".join(kept) + placeholder  # done
>>>>>>> main
"""

def test_resolve():
    "keep both intentions"
    keep_ours = lambda ours, theirs: ours
    resolved, blocks = resolve(TEXT, keep_ours)
    assert blocks == 2 and "<<<<<<<" not in resolved and resolved.endswith("placeholder\n")
    by_hand = lambda ours, theirs: [o.replace("width", "max_width") if ">" in o else t for o, t in zip(ours, theirs)]
    resolved, _ = resolve(TEXT, by_hand)
    assert resolved.splitlines()[1] == "    if length + extra + len(placeholder) > max_width:"
    assert resolve("no conflicts", keep_ours) == ("no conflicts", 0)

def test_malformed():
    "markers that don't match"
    for bad, message in [("<<<<<<< a\nx\n=======\ny\n", "unfinished conflict"), ("x\n=======\ny\n", "stray conflict marker"),
                         ("x\n>>>>>>> main\n", "stray conflict marker")]:
        try:
            resolve(bad, lambda o, t: o)
        except ValueError as e:
            assert str(e) == message
        else:
            raise AssertionError(message)
@@solution
def resolve(text, choose):
    out, blocks = [], 0
    state, ours, theirs = None, [], []
    for line in text.splitlines():
        if line.startswith("<<<<<<<") and state is None:
            state, ours, theirs = "ours", [], []
        elif line.startswith("=======") and state is None or line.startswith(">>>>>>>") and state is None:
            raise ValueError("stray conflict marker")
        elif line.startswith("=======") and state == "ours":
            state = "theirs"
        elif line.startswith(">>>>>>>") and state == "theirs":
            out += choose(ours, theirs)
            blocks += 1
            state = None
        elif state == "ours":
            ours.append(line)
        elif state == "theirs":
            theirs.append(line)
        else:
            out.append(line)
    if state is not None:
        raise ValueError("unfinished conflict")
    return "\n".join(out) + ("\n" if text.endswith("\n") else ""), blocks
