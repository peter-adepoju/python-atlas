Push `(timestamp, server index, position)` for each server's first line; pop the smallest, then push that server's next line.
@@solution
import heapq

def merge_logs(logs):
    heap = [(log[0][0], i, 0) for i, log in enumerate(logs) if log]
    heapq.heapify(heap)
    out = []
    while heap:
        _, i, j = heapq.heappop(heap)
        out.append(logs[i][j])
        if j + 1 < len(logs[i]):
            heapq.heappush(heap, (logs[i][j + 1][0], i, j + 1))
    return out

@@tests
def test_fog():
    "BFS from every source at once"
    assert fog_minutes(["F..", "..#", "#.."]) == 4 and fog_minutes(["F#.", "##."]) == -1
    assert fog_minutes(["FF", "FF"]) == 0 and fog_minutes(["#"]) == 0 and fog_minutes(["F...F"]) == 2

@@hint
Put every `"F"` cell in the queue at minute 0, count the clear cells, and run one breadth-first search; the answer is the last minute reached, if every clear cell was.
@@solution
from collections import deque

def fog_minutes(grid):
    rows, cols = len(grid), len(grid[0]) if grid else 0
    queue = deque((r, c, 0) for r in range(rows) for c in range(cols) if grid[r][c] == "F")
    clear = sum(row.count(".") for row in grid)
    seen = {(r, c) for r, c, _ in queue}
    minutes = 0
    while queue:
        r, c, t = queue.popleft()
        minutes = max(minutes, t)
        for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
            if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] == "." and (nr, nc) not in seen:
                seen.add((nr, nc))
                clear -= 1
                queue.append((nr, nc, t + 1))
    return minutes if clear == 0 else -1

def index_size(docs, words_per_doc, distinct_share, bytes_per_posting, bytes_per_term, vocabulary):
    postings = docs * words_per_doc * distinct_share
    posting_bytes = postings * bytes_per_posting
    dictionary_bytes = vocabulary * bytes_per_term
    return {"postings": round(postings), "posting_gb": round(posting_bytes / 1e9, 1),
            "dictionary_mb": round(dictionary_bytes / 1e6, 1), "total_gb": round((posting_bytes + dictionary_bytes) / 1e9, 1)}

def chunks(items, n):
    if n < 1:
        raise ValueError("n must be at least 1")
    return [items[i:i + n] for i in range(0, len(items), n)]

@@tests
def test_versions():
    "two methods, six comparisons"
    v = [Version(s) for s in ["1.10.0", "1.9.9", "2.0.0", "1.9.10"]]
    assert [str(x.parts) for x in sorted(v)] == ["(1, 9, 9)", "(1, 9, 10)", "(1, 10, 0)", "(2, 0, 0)"]
    assert Version("1.10.0") > Version("1.9.0") and Version("1.2.3") >= Version("1.2.3") and Version("1.2.3") != Version("1.2.4")
    assert len({Version("1.0.0"), Version("1.0.0")}) == 1 and repr(Version("1.10.0")) == "Version('1.10.0')"
    assert (Version("1.0.0") == "1.0.0") is False
    for bad in ["1.0", "a.b.c", "1.0.0.0"]:

        except ValueError:
            pass
        else:
            raise AssertionError(bad)
    assert "__gt__" not in Version.__dict__ or Version.__gt__ is not None
@@solution
import functools

@functools.total_ordering
class Version:
    def __init__(self, text):
        parts = text.split(".")
        if len(parts) != 3 or not all(p.isdigit() for p in parts):
            raise ValueError(f"not a version: {text}")
        self.parts = tuple(int(p) for p in parts)
        self.text = text

    def __eq__(self, other):
        if not isinstance(other, Version):
            return NotImplemented
        return self.parts == other.parts

    def __lt__(self, other):
        if not isinstance(other, Version):
            return NotImplemented
        return self.parts < other.parts

    def __hash__(self):
        return hash(self.parts)

    def __repr__(self):
        return f"Version({self.text!r})"
