:::exercise id=mix_path points=10 title="The easiest way across"

@@starter

    ___
@@tests
def test_effort():
    "DP with one row"
    assert least_effort([[1, 3, 1], [1, 5, 1], [4, 2, 1]]) == 7 and least_effort([[1, 2, 3], [4, 5, 6]]) == 12 and least_effort([[5]]) == 5
@@solution
def least_effort(grid):
    row = []
    for r, line in enumerate(grid):
        new = []
        for c, cost in enumerate(line):
            if r == 0 and c == 0:
                new.append(cost)
            elif r == 0:
                new.append(new[-1] + cost)
            elif c == 0:
                new.append(row[0] + cost)
            else:
                new.append(min(row[c], new[-1]) + cost)
        row = new
    return row[-1]
:::

@@tests

    "one search per part"
    assert parts(5, [(0, 1), (1, 2), (3, 4)]) == 2 and parts(3, []) == 3 and parts(0, []) == 0 and parts(2, [(0, 1), (1, 0)]) == 1
@@solution
from collections import deque

def parts(n, paths):
    links = [[] for _ in range(n)]
    for a, b in paths:
        links[a].append(b)
        links[b].append(a)
    seen, count = set(), 0
    for start in range(n):
        if start in seen:
            continue
        count += 1
        seen.add(start)
        queue = deque([start])
        while queue:
            for nxt in links[queue.popleft()]:
                if nxt not in seen:
                    seen.add(nxt)
                    queue.append(nxt)
    return count

## What ready means

Ready isn't perfect. A sensible bar: your **recent loops** average at least 3 out of 4 with no round far below; you solve **most** of your recent practice problems (say 70%) without the answer; and you have at least **six stories** prepared and timed. If you're below the bar in one place, that's what next week's spare hours are for. And remember that interviews are partly luck: a "no" from one company says little about the next. Keep the log going after the course; it will keep telling you the truth.

:::exercise id=forge points=50 title="The readiness report"
Write `readiness(loops, log, stories)`. `loops` is a list of scorecards (round name to criteria to ratings) in date order; `log` is a list of practice problems (dicts with `solved`, in date order); `stories` is the number of prepared, timed stories. A loop's **overall** score is the mean of its rounds' means. Return `(lines, verdict)`:

1. `"loops: N, overall F to L"` (first and last overall, 2 decimal places);
2. `"trend: rising"` if the last overall is more than 0.25 above the first, `"falling"` if more than 0.25 below, else `"steady"`;
3. `"weak rounds in the last loop: A, B"` (rounds with a mean below 2.5, in order) or `"weak rounds in the last loop: none"`;
4. `"solve rate: P% of N (last 10: Q%)"` (whole percentages);
5. `"stories: S of 6"`.

`verdict` is `"ready to interview"` if the last overall is at least 3, no round is weak, the last-10 solve rate is at least 70% and there are at least 6 stories; otherwise `"keep practising: "` and the failing ones of `loop scores`, `a weak round`, `solve rate` and `stories`, joined by `", "`.

@@tests
LOOPS = [
    {"coding": {"a": 3, "b": 2}, "design": {"a": 2, "b": 3}, "behavioural": {"a": 3}},
    {"coding": {"a": 3, "b": 3}, "design": {"a": 3, "b": 3}, "debugging": {"a": 2, "b": 3}, "behavioural": {"a": 3}},
    {"coding": {"a": 4, "b": 3}, "design": {"a": 3, "b": 3}, "debugging": {"a": 3, "b": 3}, "deep dive": {"a": 3}, "behavioural": {"a": 4}},
]
LOG = [{"solved": s} for s in [False, True, False, True, True, True, False, True, True, True, True, True, True, True]]

def test_ready():
    "the bar, met"
    lines, verdict = readiness(LOOPS, LOG, 7)
    assert lines == ["loops: 3, overall 2.67 to 3.30", "trend: rising", "weak rounds in the last loop: none",
                     "solve rate: 79% of 14 (last 10: 90%)", "stories: 7 of 6"]
    assert verdict == "ready to interview"

def test_not_yet():
    "two things to work on"
    weak = LOOPS[:2] + [{"coding": {"a": 3, "b": 3}, "design": {"a": 2, "b": 2}}]
    lines, verdict = readiness(weak, LOG[:8], 4)
    assert lines[1] == "trend: steady" and lines[2] == "weak rounds in the last loop: design" and lines[3] == "solve rate: 62% of 8 (last 10: 62%)"
    assert verdict == "keep practising: loop scores, a weak round, solve rate, stories"
@@solution
def _mean(values):
    return sum(values) / len(values)

def _overall(loop):
    return _mean([_mean(list(scores.values())) for scores in loop.values()])

def readiness(loops, log, stories):
    first, last = _overall(loops[0]), _overall(loops[-1])
    change = last - first
    trend = "rising" if change > 0.25 else "falling" if change < -0.25 else "steady"
    weak = [name for name, scores in loops[-1].items() if _mean(list(scores.values())) < 2.5]
    solved = [p["solved"] for p in log]
    rate = round(100 * sum(solved) / len(solved)) if solved else 0
    recent = solved[-10:]
    recent_rate = round(100 * sum(recent) / len(recent)) if recent else 0
    lines = [f"loops: {len(loops)}, overall {first:.2f} to {last:.2f}", f"trend: {trend}",
             "weak rounds in the last loop: " + (", ".join(weak) if weak else "none"),
             f"solve rate: {rate}% of {len(solved)} (last 10: {recent_rate}%)", f"stories: {stories} of 6"]
    reasons = []
    if last < 3:
        reasons.append("loop scores")
    if weak:
        reasons.append("a weak round")
    if recent_rate < 70:
        reasons.append("solve rate")
    if stories < 6:
        reasons.append("stories")
    return lines, "ready to interview" if not reasons else "keep practising: " + ", ".join(reasons)
:::

def interview_kit(patterns, stories, questions):
    lines = ["# Interview kit", "", "## Method", "", "Clarify, examples, simple, better, code, test, complexity.", "", "## Patterns", ""]
    lines += [f"- {clue}: {technique}" for clue, technique in sorted(patterns.items())]
    lines += ["", "## Stories", ""] + [f"- {sid}: {', '.join(comps)}" for sid, comps in stories]
    lines += ["", "## Questions to ask", ""] + [f"{n}. {q}" for n, q in enumerate(questions, start=1)]
    return "\n".join(lines)
