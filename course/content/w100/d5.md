For walking along a line, the point that minimises the total distance is the **median**, not the mean: moving towards the side with more people always helps until there are as many on each side. On a grid with paths along the axes, take the median separately in each direction: today's second coding problem.

import heapq

def kth_smallest(nums, k):
    heap = []
    for x in nums:
        if len(heap) < k:
            heapq.heappush(heap, -x)
        elif -x > heap[0]:
            heapq.heapreplace(heap, -x)
    return -heap[0]

@@starter

@@tests
def test_meeting():
    "medians, one axis at a time"
    assert meeting_point([(0, 0), (0, 4), (2, 2)]) == ((0, 2), 6)
    assert meeting_point([(1, 1)]) == ((1, 1), 0) and meeting_point([(0, 0), (0, 10)]) == ((0, 0), 10)
    homes = [(r, c) for r in range(0, 30, 3) for c in range(0, 40, 4)]

    assert total == brute
@@solution
def meeting_point(homes):
    rows = sorted(r for r, _ in homes)
    cols = sorted(c for _, c in homes)
    r, c = rows[(len(rows) - 1) // 2], cols[(len(cols) - 1) // 2]
    return (r, c), sum(abs(a - r) + abs(b - c) for a, b in homes)

@@tests

    "storage grows; the hot set doesn't"

def chat_plan(groups, messages_per_group_per_day, bytes_per_message, years, recent=50, copies=3):
    per_day = groups * messages_per_group_per_day
    stored = per_day * 365 * years * bytes_per_message * copies / 1e12
    cache = groups * recent * bytes_per_message / 1e9
    return {"messages_per_s": round(per_day / 86_400, 1), "stored_tb": round(stored, 1), "cache_gb": round(cache, 1)}

:::exercise id=d2 points=10 title="Debugging: durations that didn't parse"
`parse_duration(text)` should turn `"1h05m"`, `"45m"`, `"2h"` or `"1h 5m"` into minutes, but it only handles one shape. Rewrite it with a regular expression that allows optional hours and optional minutes (at least one), with optional spaces; raise `ValueError("bad duration: TEXT")` otherwise.
@@starter
def parse_duration(text):
    hours, minutes = text.split("h")
    return int(hours) * 60 + int(minutes.rstrip("m"))
@@tests
def test_durations():
    "every shape"
    assert parse_duration("1h05m") == 65 and parse_duration("45m") == 45 and parse_duration("2h") == 120 and parse_duration("1h 5m") == 65
    for bad in ["", "h", "5", "1x", "m"]:
        try:
            parse_duration(bad)
        except ValueError as e:
            assert str(e) == f"bad duration: {bad}"
        else:
            raise AssertionError(bad)
@@solution
import re

def parse_duration(text):
    match = re.fullmatch(r"\s*(?:(\d+)\s*h)?\s*(?:(\d+)\s*m)?\s*", text)
    if not match or not any(match.groups()):
        raise ValueError(f"bad duration: {text}")
    hours, minutes = match.groups()
    return int(hours or 0) * 60 + int(minutes or 0)
:::

It returns the same class.
@@starter

@@tests
def test_repr():
    "generated from the signature"
    @auto_repr
    class Walk:
        def __init__(self, start, end, km=0.0):
            self.start, self.end, self.km = start, end, km
    assert repr(Walk("Mill", "Ridge", 6.0)) == "Walk(start='Mill', end='Ridge', km=6.0)"
    assert repr(Walk("Mill", "Weir")) == "Walk(start='Mill', end='Weir', km=0.0)"
    @auto_repr
    class Empty:
        pass
    assert repr(Empty()) == "Empty()" and Walk.__name__ == "Walk"
@@hint
`list(inspect.signature(cls.__init__).parameters)[1:]` gives the names after `self`. For a class without its own `__init__`, `object.__init__`'s signature has no useful names: treat that as none.
@@solution
import inspect

def auto_repr(cls):
    if cls.__init__ is object.__init__:
        names = []
    else:
        names = list(inspect.signature(cls.__init__).parameters)[1:]

    def __repr__(self):
        shown = ", ".join(f"{n}={getattr(self, n)!r}" for n in names)
        return f"{type(self).__name__}({shown})"

    cls.__repr__ = __repr__
    return cls
:::
