## Warm-up

:::predict
from collections import deque

grid = ["..#", "...", "#.."]
start, goal = (0, 0), (2, 2)
dist, queue = {start: 0}, deque([start])
while queue:
    r, c = queue.popleft()
    for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
        if 0 <= nr < 3 and 0 <= nc < 3 and grid[nr][nc] == "." and (nr, nc) not in dist:
            dist[(nr, nc)] = dist[(r, c)] + 1
            queue.append((nr, nc))
print(dist[goal], len(dist))
:::

Breadth-first search reaches every cell in order of distance, so the first time it reaches the goal is by a shortest route. Mark cells as seen when you **add** them to the queue, not when you take them off, or the same cell is queued many times.

## Standard moves

- **Fewest steps** (unweighted): BFS with a queue and a `dist` dictionary.
- **Cycle in a directed graph** (circular dependencies): depth-first search with three colours: white (unvisited), grey (on the current path), black (finished). Reaching a grey node is a cycle.
- **Two sides** (teams, schedules, "no two neighbours alike"): colour by BFS, alternating; an edge between equal colours means it's impossible.
- **Shortest weighted route**: Dijkstra with a heap, remembering each node's **previous** node, then walking back from the end.
- **Cheapest network joining everything**: Kruskal's algorithm (Joseph Kruskal, 1956): sort edges by cost and add each one that joins two different groups (union-find from last week).
- **Longest chain of dependent tasks**: the critical path: process tasks in topological order, each finishing at its duration plus the latest finish among its dependencies.

from collections import deque

def fewest_steps(grid, start, end):
    rows, cols = len(grid), len(grid[0])
    dist, queue = {start: 0}, deque([start])
    while queue:
        cell = queue.popleft()
        if cell == end:
            return dist[cell]
        r, c = cell
        for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
            if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] == "." and (nr, nc) not in dist:
                dist[(nr, nc)] = dist[cell] + 1
                queue.append((nr, nc))
    return -1

Keep `colour` per node. For each white node push `(node, iterator over its neighbours)`; mark it grey; step the iterator: a grey neighbour means a cycle, a white one is pushed; when the iterator is exhausted, mark the node black and pop it.
@@solution
def has_cycle(n, edges):
    after = [[] for _ in range(n)]
    for a, b in edges:
        after[a].append(b)
    colour = [0] * n
    for root in range(n):
        if colour[root]:
            continue
        colour[root] = 1
        stack = [(root, iter(after[root]))]
        while stack:
            node, neighbours = stack[-1]
            for nxt in neighbours:
                if colour[nxt] == 1:
                    return True
                if colour[nxt] == 0:
                    colour[nxt] = 1
                    stack.append((nxt, iter(after[nxt])))
                    break
            else:
                colour[node] = 2
                stack.pop()
    return False

from collections import deque

def two_groups(n, edges):
    links = [[] for _ in range(n)]
    for a, b in edges:
        links[a].append(b)
        links[b].append(a)
    group = [None] * n
    for root in range(n):
        if group[root] is not None:
            continue
        group[root] = 0
        queue = deque([root])
        while queue:
            node = queue.popleft()
            for nxt in links[node]:
                if group[nxt] is None:
                    group[nxt] = 1 - group[node]
                    queue.append(nxt)
                elif group[nxt] == group[node]:
                    return None
    return group

import heapq

def route(paths, start, end):
    links = {}
    for a, b, km in paths:
        links.setdefault(a, []).append((b, km))
        links.setdefault(b, []).append((a, km))
    best, previous, heap, done = {start: 0}, {}, [(0, start)], set()
    while heap:
        d, node = heapq.heappop(heap)
        if node in done:
            continue
        done.add(node)
        if node == end:
            stops = [end]
            while stops[-1] != start:
                stops.append(previous[stops[-1]])
            return round(d, 2), stops[::-1]
        for nxt, km in links.get(node, ()):
            if d + km < best.get(nxt, float("inf")):
                best[nxt], previous[nxt] = d + km, node
                heapq.heappush(heap, (d + km, nxt))
    return None

         ("Weir", "Farm", 2.5), ("Farm", "Ridge", 2.0), ("Farm", "Wood", 1.5), ("Wood", "Lake", 2.0), ("Lake", "Mill", 3.5)]
NAMES = ["Mill", "Weir", "Church", "Ridge", "Farm", "Wood", "Lake"]

    "join groups, never close a loop"
    total, chosen = cheapest_network(NAMES, PATHS)
    assert total == 11.5 and len(chosen) == 6
    assert chosen[:3] == [("Farm", "Wood", 1.5), ("Mill", "Weir", 1.5), ("Farm", "Ridge", 2.0)]

@@solution
def cheapest_network(junctions, paths):
    parent = {j: j for j in junctions}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    chosen, total = [], 0
    for a, b, km in sorted(paths, key=lambda p: (p[2], p[0], p[1])):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra
            chosen.append((a, b, km))
            total += km
    if len(chosen) != len(junctions) - 1:
        return None
    return round(total, 2), chosen

@@tests

         "pack kit": (25, ["check weather"]), "brief walkers": (15, ["book minibus", "pack kit"]), "print maps": (10, ["plan route"])}

    "the longest chain of dependencies"
    assert critical_path(TASKS) == (65, ["plan route", "book minibus", "brief walkers"])
    assert critical_path({"a": (5, [])}) == (5, ["a"])
    try:

@@solution
def critical_path(tasks):
    finish, via, state = {}, {}, {}

    def visit(name):
        if state.get(name) == 2:
            return finish[name]
        if state.get(name) == 1:
            raise ValueError("cycle")
        state[name] = 1
        minutes, needs = tasks[name]
        best, best_dep = 0, None
        for dep in sorted(needs):
            f = visit(dep)
            if f > best:
                best, best_dep = f, dep
        finish[name], via[name], state[name] = best + minutes, best_dep, 2
        return finish[name]

    for name in sorted(tasks):
        visit(name)
    last = min(finish, key=lambda n: (-finish[n], n))
    chain = [last]
    while via[chain[-1]] is not None:
        chain.append(via[chain[-1]])
    return finish[last], chain[::-1]

@@tests

    "a limit on changes"
    assert cheapest_with_stops(BUSES, "Mill", "Ridge", 0) == 500
    assert cheapest_with_stops(BUSES, "Mill", "Ridge", 1) == 200
    assert cheapest_with_stops(BUSES, "Mill", "Ridge", 2) == 170

@@hint

@@solution
def cheapest_with_stops(buses, start, end, max_stops):
    fare = {start: 0}
    for _ in range(max_stops + 1):
        updated = dict(fare)
        for a, b, cost in buses:
            if a in fare and fare[a] + cost < updated.get(b, float("inf")):
                updated[b] = fare[a] + cost
        fare = updated
    return fare.get(end, -1)
