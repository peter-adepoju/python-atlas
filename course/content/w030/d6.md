class ParseError(ValueError):
    def __init__(self, line_no, text):
        super().__init__(f"line {line_no}: bad number {text!r}")
        self.line_no = line_no
        self.text = text

def parse_numbers(lines):
    numbers = []
    for line_no, line in enumerate(lines, start=1):
        text = line.strip()
        if not text:
            continue
        try:
            numbers.append(int(text))
        except ValueError:
            raise ParseError(line_no, text)
    return numbers
:::

class SplitWords:
    def words(self, text):
        return text.split()

class CleanWords:
    def words(self, text):
        result = []
        for word in text.split():
            word = word.lower().strip(".,!?;:\"'")
            if word:
                result.append(word)
        return result

class WordCounter:
    def __init__(self, tokeniser):
        self.tokeniser = tokeniser

    def count(self, text):
        counts = {}
        for word in self.tokeniser.words(text):
            counts[word] = counts.get(word, 0) + 1
        return counts

## Forge: a bike-hire scheme

A city runs a hire scheme with ordinary bikes, electric bikes and scooters. Riders hire a vehicle by its code and pay when they return it, by the minute. Members pay less, and on some days a price cap applies. Here's the specification. Amounts are whole numbers of pence.

**Exceptions**

- `HireError(Exception)`: the family.
- `UnknownVehicle(HireError)`, made as `UnknownVehicle(code)`, with the message `no vehicle X9` and the attribute `code`.
- `NotAvailable(HireError)`, made as `NotAvailable(code)`, with the message `E1 is not available` and the attribute `code`.

**Vehicles**

- `Vehicle(code)` with class attributes `UNLOCK = 100` and `PER_MIN = 10`, and an attribute `out` (False when made: whether it's on hire). `cost(minutes)` returns `UNLOCK + PER_MIN * minutes`; `minutes` must be an `int` of 0 or more (not a `bool`), otherwise `ValueError`. `can_hire()` returns True when it isn't out. `repr` gives `Bike('B1')`, with the real class name.
- `Bike(Vehicle)`: `PER_MIN = 5`.
- `BatteryMixin`: its `__init__(self, code, battery=100)` passes `code` on with `super().__init__(code)` and stores `battery` (an `int` from 0 to 100, otherwise `ValueError`). It extends `can_hire()` with `super()`: also needs a battery of at least 20. `use(minutes)` takes `minutes` off the battery, stopping at 0.
- `EBike(BatteryMixin, Bike)`: `PER_MIN = 12`.
- `Scooter(BatteryMixin, Vehicle)`: `UNLOCK = 50`, `PER_MIN = 15`.

No vehicle class rewrites `cost`; the battery classes don't rewrite `__repr__`.

**Pricing rules** (each with `price(vehicle, minutes)`)

- `Standard`: the vehicle's own `cost(minutes)`.
- `Member`: the vehicle's cost without the unlock fee.
- `DayCap(inner, cap)`: wraps another rule; its price, but never more than `cap`.

**`Scheme(pricing=None)`**: holds its vehicles, a `pricing` rule (a `Standard()` if none is given; it can be changed at any time), and a list `trips`.

- `add(vehicle)`: a second vehicle with the same code raises `ValueError`.
- `available()`: the codes of every vehicle that `can_hire()`, sorted.
- `hire(code)`: raises `UnknownVehicle` for an unknown code and `NotAvailable` if the vehicle can't be hired; otherwise marks it out.
- `return_vehicle(code, minutes)`: raises `UnknownVehicle` for an unknown code and a plain `HireError` (message `B1 is not out`) if it isn't out. Otherwise works out the price with the pricing rule, then, if the vehicle has a `use` method, calls it with the minutes, marks the vehicle back in, records `(code, minutes, price)` in `trips`, and returns the price. Bad `minutes` raise `ValueError` and change nothing.
- `takings()`: the total of all trip prices.

@@hint

@@solution
class HireError(Exception):
    """Something went wrong with a hire."""

class UnknownVehicle(HireError):
    def __init__(self, code):
        super().__init__(f"no vehicle {code}")
        self.code = code

class NotAvailable(HireError):
    def __init__(self, code):
        super().__init__(f"{code} is not available")
        self.code = code

class Vehicle:
    UNLOCK = 100
    PER_MIN = 10

    def __init__(self, code):
        self.code = code
        self.out = False

    def cost(self, minutes):
        if not isinstance(minutes, int) or isinstance(minutes, bool) or minutes < 0:
            raise ValueError(f"minutes must be a whole number of 0 or more, not {minutes!r}")
        return self.UNLOCK + self.PER_MIN * minutes

    def can_hire(self):
        return not self.out

    def __repr__(self):
        return f"{type(self).__name__}({self.code!r})"

class Bike(Vehicle):
    PER_MIN = 5

class BatteryMixin:
    def __init__(self, code, battery=100):
        if not isinstance(battery, int) or isinstance(battery, bool) or not 0 <= battery <= 100:
            raise ValueError(f"battery must be 0 to 100, not {battery!r}")
        super().__init__(code)
        self.battery = battery

    def can_hire(self):
        return super().can_hire() and self.battery >= 20

    def use(self, minutes):
        self.battery = max(0, self.battery - minutes)

class EBike(BatteryMixin, Bike):
    PER_MIN = 12

class Scooter(BatteryMixin, Vehicle):
    UNLOCK = 50
    PER_MIN = 15

class Standard:
    def price(self, vehicle, minutes):
        return vehicle.cost(minutes)

class Member:
    def price(self, vehicle, minutes):
        return vehicle.cost(minutes) - vehicle.UNLOCK

class DayCap:
    def __init__(self, inner, cap):
        self.inner = inner
        self.cap = cap

    def price(self, vehicle, minutes):
        return min(self.inner.price(vehicle, minutes), self.cap)

class Scheme:
    def __init__(self, pricing=None):
        self.pricing = pricing if pricing is not None else Standard()
        self._vehicles = {}
        self.trips = []

    def add(self, vehicle):
        if vehicle.code in self._vehicles:
            raise ValueError(f"already have {vehicle.code}")
        self._vehicles[vehicle.code] = vehicle

    def _find(self, code):
        if code not in self._vehicles:
            raise UnknownVehicle(code)
        return self._vehicles[code]

    def available(self):
        return sorted(code for code, v in self._vehicles.items() if v.can_hire())

    def hire(self, code):
        vehicle = self._find(code)
        if not vehicle.can_hire():
            raise NotAvailable(code)
        vehicle.out = True

    def return_vehicle(self, code, minutes):
        vehicle = self._find(code)
        if not vehicle.out:
            raise HireError(f"{code} is not out")
        price = self.pricing.price(vehicle, minutes)
        use = getattr(vehicle, "use", None)
        if use is not None:
            use(minutes)
        vehicle.out = False
        self.trips.append((code, minutes, price))
        return price

    def takings(self):
        return sum(price for _, _, price in self.trips)

:::think title="Where each idea earned its place"
Look back at which tool did which job. **Inheritance** gave the vehicles their shared `cost`, `repr` and `can_hire`, with children changing only two numbers. The **mixin** added batteries to two unrelated branches of the family without copying a line, and `super()` threaded its `can_hire` through the MRO. **Composition** let the scheme's pricing change while it was running, which no class family could do. **Duck typing** let `return_vehicle` drain any vehicle that has a battery without asking what kind it was. And the **exception family** let callers catch one problem precisely or every hire problem at once. Which of them would you reach for first next time, and which would you avoid?
:::

## Forge bonus: a pluggable log

Programs record what they do in logs, and different messages go to different places: everything to a file, only problems to the screen. Python's own `logging` module works on this design (a logger that passes records to handlers, each with its own level); here's a small version of it.

:::exercise id=forge2 points=30 title="Forge bonus: a pluggable log"
Write a small logging system:

- `Handler(level="DEBUG")`, a base class. The levels, lowest to highest, are `DEBUG`, `INFO`, `WARNING` and `ERROR`, kept in a class attribute `LEVELS` (a list in that order). An unknown level raises `ValueError`. `handle(level, text)` does nothing if `level` is lower than the handler's level; otherwise it calls the hook `self.emit(line)` with `line` in the form `WARNING: disk nearly full`, and returns True (False if it did nothing). `emit` in `Handler` raises `NotImplementedError`.
- `ListHandler(level="DEBUG")`, a `Handler` whose `emit` appends the line to a list `lines`.
- `StreamHandler(out, level="DEBUG")`, a `Handler` whose `emit` writes the line and a newline to `out`, which can be any object with `write` (a file, `io.StringIO`).
- `CountMixin`, which extends `emit` with `super()` to keep a count of lines emitted in `count` (starting at 0, made the first time it's needed, or in a cooperative `__init__`).
- `CountingListHandler`, using `CountMixin` and `ListHandler`.
- `Log(handlers)`, which keeps its own copy of the handlers. `log(level, text)` passes the message to every handler and returns how many handled it. `debug`, `info`, `warning` and `error` are shortcuts, each taking just the text. An unknown level passed to `log` raises `ValueError`.

@@hint
`CountMixin.emit` is `super().emit(line)` then `self.count += 1`. For `count` to start at 0, give the mixin an `__init__(self, *args, **kwargs)` that calls `super().__init__(*args, **kwargs)` and sets `self.count = 0`.
@@solution
class Handler:
    LEVELS = ["DEBUG", "INFO", "WARNING", "ERROR"]

    def __init__(self, level="DEBUG"):
        if level not in self.LEVELS:
            raise ValueError(f"unknown level {level!r}")
        self.level = level

    def handle(self, level, text):
        if self.LEVELS.index(level) < self.LEVELS.index(self.level):
            return False
        self.emit(f"{level}: {text}")
        return True

    def emit(self, line):
        raise NotImplementedError("handlers must define emit()")

class ListHandler(Handler):
    def __init__(self, level="DEBUG"):
        super().__init__(level)
        self.lines = []

    def emit(self, line):
        self.lines.append(line)

class StreamHandler(Handler):
    def __init__(self, out, level="DEBUG"):
        super().__init__(level)
        self.out = out

    def emit(self, line):
        self.out.write(line + "\n")

class CountMixin:
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.count = 0

    def emit(self, line):
        super().emit(line)
        self.count += 1

class CountingListHandler(CountMixin, ListHandler):
    pass

class Log:
    def __init__(self, handlers):
        self.handlers = list(handlers)

    def log(self, level, text):
        if level not in Handler.LEVELS:
            raise ValueError(f"unknown level {level!r}")
        return sum(h.handle(level, text) for h in self.handlers)

    def debug(self, text):
        return self.log("DEBUG", text)

    def info(self, text):
        return self.log("INFO", text)

    def warning(self, text):
        return self.log("WARNING", text)

    def error(self, text):
        return self.log("ERROR", text)
:::
