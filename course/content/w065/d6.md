def test_nested():
    "finding possible O(n²) code"
    src = (
        "def pairs(xs):\n    out = []\n    for a in xs:\n        for b in xs:\n            out.append((a, b))\n    return out\n"
        "def total(xs):\n    t = 0\n    for x in xs:\n        t += x\n    return t\n"
        "def hidden(xs):\n    for x in xs:\n        ys = [y for y in xs if y > x]\n    return ys\n"

@@solution
import ast

def nested_loops(source):
    found = set()
    loops = (ast.For, ast.While, ast.comprehension)
    for func in ast.walk(ast.parse(source)):
        if isinstance(func, (ast.FunctionDef, ast.AsyncFunctionDef)):
            for outer in ast.walk(func):
                if isinstance(outer, (ast.For, ast.While)):
                    for inner in ast.walk(outer):
                        if inner is not outer and isinstance(inner, loops):
                            found.add(func.name)
    return sorted(found)

@@solution
def chunked(iterable, size):
    chunk = []
    for item in iterable:
        chunk.append(item)
        if len(chunk) == size:
            yield chunk
            chunk = []
    if chunk:
        yield chunk

def chunk_sums(iterable, size):
    for chunk in chunked(iterable, size):
        yield sum(chunk)

## Forge: a faster log analyser

A website's access log is a list of entries like `{"time": 1712, "ip": "10.0.0.7", "page": "/shop"}`, in time order. Here's the current analyser. It's correct, but each function is slow:

```
def unique_visitors(log):                 # an 'in' scan for every entry
    seen = []
    for e in log:
        if e["ip"] not in seen:
            seen.append(e["ip"])
    return seen

def top_pages(log, k):                    # counting by scanning, then sorting everything
    pages = [e["page"] for e in log]
    counts = [(p, pages.count(p)) for p in set(pages)]
    return sorted(counts, key=lambda pc: (-pc[1], pc[0]))[:k]

def visits_by_country(log, lookup):       # calls a slow lookup for every entry
    totals = {}
    for e in log:
        country = lookup(e["ip"])
        totals[country] = totals.get(country, 0) + 1
    return dict(sorted(totals.items()))

def busiest_window(times, width):         # compares every pair of times
    best, start = 0, None
    for a in sorted(times):
        count = sum(1 for b in times if a <= b < a + width)
        if count > best:
            best, start = count, a
    return best, start

def sessions(log, gap):                   # scans the whole log for every visitor
    result = {}
    for ip in unique_visitors(log):
        times = sorted(e["time"] for e in log if e["ip"] == ip)
        result[ip] = 1 + sum(1 for a, b in zip(times, times[1:]) if b - a > gap)
    return dict(sorted(result.items()))
```

@@hint
For the window: sort the times once, then move a `right` index along them, and a `left` index forward while the left time is at least `width` before the right time; the window between them is `right - left + 1` events, starting at `ordered[left]`. For sessions, group the times by address in one pass with a `defaultdict(list)` (the log is already in time order, but sorting each small list is cheap and safe).
@@solution
import bisect
import heapq
from collections import Counter, defaultdict

def unique_visitors(log):
    return list(dict.fromkeys(entry["ip"] for entry in log))

def top_pages(log, k):
    counts = Counter(entry["page"] for entry in log)
    return heapq.nsmallest(k, counts.items(), key=lambda pc: (-pc[1], pc[0]))

def visits_by_country(log, lookup):
    countries = {}
    totals = Counter()
    for entry in log:
        ip = entry["ip"]
        if ip not in countries:
            countries[ip] = lookup(ip)
        totals[countries[ip]] += 1
    return dict(sorted(totals.items()))

def busiest_window(times, width):
    ordered = sorted(times)
    best, best_start, left = 0, None, 0
    for right, t in enumerate(ordered):
        while not ordered[left] > t - width:
            left += 1
        count = right - left + 1
        if count > best:
            best, best_start = count, ordered[left]
    return best, best_start

def sessions(log, gap):
    by_user = defaultdict(list)
    for entry in log:
        by_user[entry["ip"]].append(entry["time"])
    result = {}
    for ip, times in by_user.items():
        times.sort()
        count = 1
        for a, b in zip(times, times[1:]):
            if b - a > gap:
                count += 1
        result[ip] = count
    return dict(sorted(result.items()))

:::think title="Where the speed came from"
None of today's fixes were clever tricks with Python's internals. Each replaced a repeated scan with a structure built once: a dict keeps order and finds duplicates, a Counter counts in one pass, a dict of lookup results stops repeated calls, sorting once makes a sweep possible, and grouping once replaces a scan per visitor. That's the usual shape of real performance work: the profile points at a function, and the fix is a better algorithm or data structure, checked by tests that prove the results haven't changed. Micro-tuning comes last, if at all.
:::

For the moving average, put a 0 in front of the values before taking `np.cumsum`; then the sum of `values[i:i + w]` is `sums[i + w] - sums[i]`, which for all `i` at once is `sums[w:] - sums[:-w]`.
@@solution
import numpy as np

def hourly_counts(times):
    hours = (np.asarray(times, dtype=np.int64) // 3600) % 24
    return np.bincount(hours, minlength=24).tolist()

def moving_average(values, window):
    v = np.asarray(values, dtype=float)
    if len(v) < window:
        return []
    sums = np.cumsum(np.concatenate(([0.0], v)))
    return np.round((sums[window:] - sums[:-window]) / window, 3).tolist()

def outliers(values, z=2.0):
    v = np.asarray(values, dtype=float)
    std = v.std()
    if std == 0:
        return []
    return np.nonzero(np.abs(v - v.mean()) / std > z)[0].tolist()

def response_summary(ms):
    v = np.asarray(ms, dtype=float)
    p50, p95 = np.percentile(v, [50, 95])
    return {"median": round(float(p50), 1), "p95": round(float(p95), 1),
            "slow_share": round(float((v > 1000).mean()), 3)}
