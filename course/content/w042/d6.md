    def __init__(self, value, next=None):
        self.value, self.next = value, next

def remove_from_end(head, n):
    if n < 1:
        raise IndexError("n must be at least 1")
    dummy = Node(None, head)
    lead = trail = dummy
    for _ in range(n):
        lead = lead.next
        if lead is None:
            raise IndexError("the list is too short")
    while lead.next is not None:
        lead, trail = lead.next, trail.next
    trail.next = trail.next.next
    return dummy.next

## Forge: an editor buffer

A text editor keeps its text and a **cursor** (the point between two characters where typing goes). Almost every keystroke works at the cursor: type a character, delete the one before, move left or right. With a Python string, every insertion in the middle copies the whole text: O(n) per keystroke, which is painful for a long document. With a doubly linked list of characters and a cursor that is a **node reference**, each keystroke costs O(1).

Build `Editor`, using a circular doubly linked list with a sentinel (one node per character). The cursor is a reference to the node just **left** of the cursor position, which is the sentinel when the cursor is at the very start.

- `Editor(text="")` holds the text, with the cursor at the **end**.
- `insert(s)` inserts the characters of `s` at the cursor; the cursor ends up after them (as when typing).
- `backspace(k)` deletes up to `k` characters **before** the cursor, and returns how many it deleted.
- `left(k)` and `right(k)` move the cursor up to `k` places (stopping at the ends). Each returns the (up to) 10 characters just left of the cursor afterwards, as a string: what an editor would show beside the cursor.
- `cut(k)` removes up to `k` characters **after** the cursor (the cursor doesn't move) and keeps them as the clipboard, replacing any earlier one; it returns the cut text. `paste()` inserts the clipboard at the cursor (as `insert` does) and returns `len` of what it pasted.
- `text()` returns the whole text; `len(editor)` is its length, in O(1); `cursor_position()` is the number of characters before the cursor, kept up to date as you go (not counted by walking).
- `reverse_word()`: reverse, in place, the run of letters immediately **before** the cursor (stopping at a non-letter or the start), by relinking those nodes. The cursor stays after the run. Return the reversed run.

Every operation except `text()` must take time in proportion to the number of characters it touches (`k`, the length of the string, the run or the clipboard), never the length of the whole document: the checks run tens of thousands of operations on a long text.

Keep `self._sentinel`, `self._cursor` (the node left of the cursor), `self._size`, `self._pos` and `self._clipboard` (a string). Insert a character with "link a new node after the cursor node, then move the cursor to it". Backspace unlinks the cursor node and moves the cursor to its `prev`. Moving left follows `prev` from the cursor; the view walks back up to 10 nodes from the cursor. For `reverse_word`, collect the run's nodes by walking back from the cursor, then relink them in the opposite order between the node before the run and the node after the cursor, and set the cursor to the run's new last node.
@@solution
class Node:
    def __init__(self, char=None):
        self.char = char
        self.prev = self.next = self

class Editor:
    def __init__(self, text=""):
        self._sentinel = Node()
        self._cursor = self._sentinel
        self._size = 0
        self._pos = 0
        self._clipboard = ""
        self.insert(text)

    def _link_after(self, place, node):
        node.prev, node.next = place, place.next
        place.next.prev = node
        place.next = node

    def _unlink(self, node):
        node.prev.next = node.next
        node.next.prev = node.prev

    def __len__(self):
        return self._size

    def cursor_position(self):
        return self._pos

    def text(self):
        out, node = [], self._sentinel.next
        while node is not self._sentinel:
            out.append(node.char)
            node = node.next
        return "".join(out)

    def insert(self, s):
        for ch in s:
            node = Node(ch)
            self._link_after(self._cursor, node)
            self._cursor = node
        self._size += len(s)
        self._pos += len(s)

    def backspace(self, k):
        deleted = 0
        while deleted < k and self._cursor is not self._sentinel:
            node = self._cursor
            self._cursor = node.prev
            self._unlink(node)
            deleted += 1
        self._size -= deleted
        self._pos -= deleted
        return deleted

    def _view(self):
        out, node = [], self._cursor
        while node is not self._sentinel and len(out) < 10:
            out.append(node.char)
            node = node.prev
        return "".join(reversed(out))

    def left(self, k):
        moved = 0
        while moved < k and self._cursor is not self._sentinel:
            self._cursor = self._cursor.prev
            moved += 1
        self._pos -= moved
        return self._view()

    def right(self, k):
        moved = 0
        while moved < k and self._cursor.next is not self._sentinel:
            self._cursor = self._cursor.next
            moved += 1
        self._pos += moved
        return self._view()

    def cut(self, k):
        out = []
        while len(out) < k and self._cursor.next is not self._sentinel:
            node = self._cursor.next
            self._unlink(node)
            out.append(node.char)
        self._size -= len(out)
        self._clipboard = "".join(out)
        return self._clipboard

    def paste(self):
        self.insert(self._clipboard)
        return len(self._clipboard)

    def reverse_word(self):
        run, node = [], self._cursor
        while node is not self._sentinel and node.char.isalpha():
            run.append(node)
            node = node.prev
        if not run:
            return ""
        before, after = node, self._cursor.next
        place = before
        for n in run:                        # run is already in reverse order
            place.next, n.prev = n, place
            place = n
        place.next, after.prev = after, place
        self._cursor = place
        return "".join(n.char for n in run)
:::

:::think title="Why not a string, or a list?"
A Python string can't change at all, so every insertion in the middle builds a whole new string: O(n) per keystroke. A list of characters is mutable, but inserting at position i shifts everything after it (Week 41): still O(n). The linked buffer pays for its O(1) keystrokes with memory (a node object per character) and with slow jumps: going to character 50,000 means walking there. Real editors often use a **gap buffer** (an array with a movable gap at the cursor) or a **rope** (a balanced tree of string pieces), which combine cheap edits with fast jumps; both are built on ideas from this phase. The linked version shows the core idea most plainly: keep the place you're working **as a reference**, and changes there cost nothing to reach.
:::

For `sort_list`, walk `slow` and `fast` with `fast = head.next` to start, so `slow` ends at the end of the first half; cut there; sort both halves recursively (the recursion is only about log n deep); and merge, taking from the first half on ties (`<=`) so equal values keep their order.

class Node:
    def __init__(self, value, next=None):
        self.value, self.next = value, next

def josephus(n, k):
    first = Node(1)
    last = first
    for i in range(2, n + 1):
        last.next = Node(i)
        last = last.next
    last.next = first
    before = last
    out = []
    for _ in range(n):
        for _ in range(k - 1):
            before = before.next
        gone = before.next
        out.append(gone.value)
        before.next = gone.next
    return out

def _merge(a, b):
    dummy = tail = Node(None)
    while a is not None and b is not None:
        if a.value <= b.value:
            tail.next, a = a, a.next
        else:
            tail.next, b = b, b.next
        tail = tail.next
    tail.next = a if a is not None else b
    return dummy.next

def sort_list(head):
    if head is None or head.next is None:
        return head
    slow, fast = head, head.next
    while fast is not None and fast.next is not None:
        slow, fast = slow.next, fast.next.next
    second, slow.next = slow.next, None
    return _merge(sort_list(head), sort_list(second))
