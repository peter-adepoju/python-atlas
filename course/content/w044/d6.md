from collections import Counter

def pairs_with_difference(nums, k):
    counts = Counter(nums)
    if k == 0:
        return sum(1 for c in counts.values() if c >= 2)
    return sum(1 for x in counts if x + k in counts)

## Forge: a search engine

Search engines don't read every document for every query. They build an **inverted index** in advance: for each word, which documents contain it, and how often. A query then only looks up its own words. Build `SearchIndex` on a hash map of your own.

**Your hash map.** Write `HashMap`, with chaining (Day 2): `m[key]` (`KeyError` if missing), `m[key] = value`, `del m[key]`, `in`, `len`, `get(key, default=None)`, `keys()` and `items()` (lists, in any order), doubling when the load factor passes 0.75. **No dict or set anywhere in the Forge** (no `{}` displays, `dict()`, `set()` or `Counter`): every map below is your `HashMap`.

**Words.** `words(text)`: the lower-cased runs of letters and digits in `text`, in order (anything else separates words): `"Don't stop, 2 go!"` gives `["don", "t", "stop", "2", "go"]`.

**The index.** `SearchIndex()` keeps two `HashMap`s: `postings` from each word to a `HashMap` of document id to the number of times the word appears in that document; and `documents` from each document id to a `HashMap` of its words and their counts.

- `add(doc_id, text)`: index a document. Adding an id that's already there raises `ValueError`.
- `remove(doc_id)`: remove it from both maps (a word left with no documents is removed from `postings` entirely); a missing id raises `KeyError`.
- `search(query)`: the ids of documents containing **every** word of the query, ranked by the total number of times the query's words appear in them (most first), then by id. An empty query gives `[]`. Look up the query's **rarest** word first and check only its documents.
- `top_words(k)`: the `k` words that appear most often across all documents, as `(word, total)` pairs, most first, then alphabetically.
- `similar(doc_id)`: the other document sharing the most **distinct** words with it, or `None` if no other document shares any; ties go to the smallest id. Use the index: only documents in some of its words' postings can be candidates.

Sort the results with the key `(-total, doc_id)`.
@@solution
class HashMap:
    def __init__(self):
        self._buckets = [[] for _ in range(8)]
        self._size = 0

    def _find(self, key):
        chain = self._buckets[hash(key) % len(self._buckets)]
        for i, (k, _) in enumerate(chain):
            if k == key:
                return chain, i
        return chain, -1

    def __setitem__(self, key, value):
        chain, i = self._find(key)
        if i >= 0:
            chain[i][1] = value
            return
        chain.append([key, value])
        self._size += 1
        if self._size / len(self._buckets) > 0.75:
            old = self._buckets
            self._buckets = [[] for _ in range(2 * len(old))]
            for c in old:
                for pair in c:
                    self._buckets[hash(pair[0]) % len(self._buckets)].append(pair)

    def __getitem__(self, key):
        chain, i = self._find(key)
        if i < 0:
            raise KeyError(key)
        return chain[i][1]

    def __delitem__(self, key):
        chain, i = self._find(key)
        if i < 0:
            raise KeyError(key)
        chain.pop(i)
        self._size -= 1

    def __contains__(self, key):
        return self._find(key)[1] >= 0

    def __len__(self):
        return self._size

    def get(self, key, default=None):
        chain, i = self._find(key)
        return chain[i][1] if i >= 0 else default

    def items(self):
        return [(k, v) for chain in self._buckets for k, v in chain]

    def keys(self):
        return [k for k, _ in self.items()]


def words(text):
    out, current = [], ""
    for ch in text.lower():
        if ch.isalnum():
            current += ch
        elif current:
            out.append(current)
            current = ""
    if current:
        out.append(current)
    return out


class SearchIndex:
    def __init__(self):
        self.postings = HashMap()
        self.documents = HashMap()

    def add(self, doc_id, text):
        if doc_id in self.documents:
            raise ValueError(f"document {doc_id!r} is already indexed")
        counts = HashMap()
        for w in words(text):
            counts[w] = counts.get(w, 0) + 1
        self.documents[doc_id] = counts
        for w, n in counts.items():
            if w not in self.postings:
                self.postings[w] = HashMap()
            self.postings[w][doc_id] = n

    def remove(self, doc_id):
        counts = self.documents[doc_id]
        for w in counts.keys():
            docs = self.postings[w]
            del docs[doc_id]
            if len(docs) == 0:
                del self.postings[w]
        del self.documents[doc_id]

    def search(self, query):
        terms = words(query)
        if not terms:
            return []
        lists = []
        for t in terms:
            if t not in self.postings:
                return []
            lists.append(self.postings[t])
        rarest = min(lists, key=len)
        results = []
        for doc_id in rarest.keys():
            total = 0
            for docs in lists:
                n = docs.get(doc_id, 0)
                if n == 0:
                    break
                total += n
            else:
                results.append((-total, doc_id))
        return [doc_id for _, doc_id in sorted(results)]

    def top_words(self, k):
        totals = [(-sum(n for _, n in docs.items()), w) for w, docs in self.postings.items()]
        return [(w, -t) for t, w in sorted(totals)[:k]]

    def similar(self, doc_id):
        shared = HashMap()
        for w in self.documents[doc_id].keys():
            for other in self.postings[w].keys():
                if other != doc_id:
                    shared[other] = shared.get(other, 0) + 1
        if len(shared) == 0:
            return None
        return min(shared.items(), key=lambda pair: (-pair[1], pair[0]))[0]

:::think title="Paying at write time"
Every structure this week made the same trade as Week 41's prefix sums: do work when data **arrives**, so questions are cheap later. The index does real work in `add` (counting every word, updating two maps) so that `search` touches only the documents that contain the query's rarest word, not every document. Real search engines go much further (compressed postings, phrase positions, scoring by how rare a word is across the collection, and indexes spread over thousands of machines), but the inverted index at their heart is this one. Notice also how `remove` has to undo exactly what `add` did, in both maps: the price of keeping two views of the same data is keeping them consistent.
:::

## Forge bonus: a Bloom filter

:::exercise id=forge2 points=30 title="Forge bonus: a Bloom filter"
A **Bloom filter** (Burton Bloom, 1970) answers "have I seen this?" using far less memory than a set, by allowing one kind of mistake. It's a list of `m` bits, all 0, and `k` different hash functions. To **add** an item, set the `k` bits its hashes point to. To **check** an item, look at its `k` bits: if any is 0, the item was **definitely never added**; if all are 1, it was **probably** added (or other items happen to have set those bits: a false positive). There are never false negatives. Browsers and databases use them to skip expensive look-ups for things that certainly aren't there.

Write `BloomFilter(m, k)` storing its bits in a `bytearray` of `m` bytes (one per bit, for simplicity), with the hash functions given in the starter (`_hashes`, which uses polynomial hashes with `k` different bases, so it's the same every run). Give it `add(item)` and `might_contain(item)` (items are strings), and `fill()`: the fraction of bits set, rounded to 3 places. Then write `false_positive_rate(m, k, n, trials)`: add the strings `"in0"` to `"in{n-1}"`, check `"out0"` to `"out{trials-1}"` (none of which were added), and return the fraction reported as present, rounded to 3 places. Finally set `estimate` to the textbook approximation `(1 - e^(-k n / m)) ** k` for `m = 10_000`, `k = 7`, `n = 1000`, rounded to 4 places.

import math

class BloomFilter:
    BASES = [31, 37, 41, 43, 47, 53, 59, 61, 67, 71]

    def __init__(self, m, k):
        if not 1 <= k <= len(self.BASES):
            raise ValueError(f"k must be from 1 to {len(self.BASES)}")
        self.m, self.k = m, k
        self.bits = bytearray(m)

    def _hashes(self, item):
        out = []
        for base in self.BASES[:self.k]:
            h = 0
            for ch in item:
                h = (h * base + ord(ch)) % 2_147_483_647
            out.append(h % self.m)
        return out

    def add(self, item):
        for i in self._hashes(item):
            self.bits[i] = 1

    def might_contain(self, item):
        return all(self.bits[i] for i in self._hashes(item))

    def fill(self):
        return round(sum(self.bits) / self.m, 3)

def false_positive_rate(m, k, n, trials):
    b = BloomFilter(m, k)
    for i in range(n):
        b.add(f"in{i}")
    wrong = sum(1 for i in range(trials) if b.might_contain(f"out{i}"))
    return round(wrong / trials, 3)

estimate = round((1 - math.exp(-7 * 1000 / 10_000)) ** 7, 4)
