Write a generator `interleave(*iterables)` that yields one item from each iterable in turn, skipping any that are used up, until all are (`interleave("abc", [1], "xy")` gives `a`, `1`, `x`, `b`, `y`, `c`). @@tests
def test_interleave():
    "round robin"
    import itertools
    assert list(interleave("abc", [1], "xy")) == ["a", 1, "x", "b", "y", "c"]
    assert list(interleave()) == [] and list(interleave([], [])) == []
    assert list(itertools.islice(interleave(itertools.count(), "ab"), 6)) == [0, "a", 1, "b", 2, 3]
    pulled = []
    def src():
        for i in range(10):

            yield i
    g = interleave(src(), "pq")

@@solution
from collections import deque

def interleave(*iterables):
    active = deque(iter(it) for it in iterables)
    while active:
        it = active.popleft()
        try:
            item = next(it)
        except StopIteration:
            continue
        yield item
        active.append(it)

import functools

def once(func):
    state = {}

    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        if "result" not in state:
            state["result"] = func(*args, **kwargs)
        return state["result"]

    wrapper.reset = state.clear
    return wrapper

:::exercise id=b4 points=15 title="B4: Nested timers" exam=yes
Write a class `Timings(clock)`, where `clock()` returns the current time. Its method `section(name)` is a context manager: it times its block with `clock` and adds the elapsed time to `self.totals`, a dict keyed by the section's **path**: its name joined with the names of the sections it's nested inside, using `/` (so `section("parse")` inside `section("load")` is `"load/parse"`). Repeated sections add up. Time is recorded, and the nesting restored, even when the block raises (the exception propagates). `report()` returns the `(path, total)` pairs sorted by path.
@@starter
from contextlib import contextmanager

class Timings:
    ___
@@tests
def test_timings():
    "nested context managers"
    now = [0]
    t = Timings(lambda: now[0])
    with t.section("load"):
        now[0] = 1
        with t.section("parse"):
            now[0] = 4
        with t.section("parse"):
            now[0] = 6
    with t.section("save"):
        now[0] = 7
    assert t.totals == {"load/parse": 5, "load": 6, "save": 1}
    try:
        with t.section("load"):
            now[0] = 9
            raise ValueError
    except ValueError:

    with t.section("end"):
        now[0] = 10

@@solution
from contextlib import contextmanager

class Timings:
    def __init__(self, clock):
        self.clock = clock
        self.totals = {}
        self._stack = []

    @contextmanager
    def section(self, name):
        self._stack.append(name)
        path = "/".join(self._stack)
        start = self.clock()
        try:
            yield
        finally:
            self.totals[path] = self.totals.get(path, 0) + self.clock() - start
            self._stack.pop()

    def report(self):
        return sorted(self.totals.items())

    assert aliases([[1], [1]]) == (3, 0) and aliases(5) == (0, 0)

def aliases(obj):
    seen = set()
    repeated = 0
    stack = [obj]
    while stack:
        item = stack.pop()
        if not isinstance(item, (list, tuple, dict)):
            continue
        if id(item) in seen:
            repeated += 1
            continue
        seen.add(id(item))
        stack.extend(item.values() if isinstance(item, dict) else item)
    return len(seen), repeated

    mine = base.override(size=12)
    assert (mine.colour, mine.size, base.size) == ("blue", 12, 10)

@@solution
class Settings:
    def __init__(self, parent=None, **values):
        object.__setattr__(self, "_parent", parent)
        object.__setattr__(self, "_values", dict(values))

    def __getattr__(self, name):
        if name.startswith("_"):
            raise AttributeError(name)
        if name in self._values:
            return self._values[name]
        if self._parent is not None:
            return getattr(self._parent, name)
        raise AttributeError(f"no setting {name!r}")

    def __setattr__(self, name, value):
        raise AttributeError("settings are read-only")

    def __delattr__(self, name):
        raise AttributeError("settings are read-only")

    def override(self, **values):
        return Settings(self, **values)

    def to_dict(self):
        merged = self._parent.to_dict() if self._parent is not None else {}
        merged.update(self._values)
        return merged

import sys
import importlib.abc
import importlib.util

class SourceFinder(importlib.abc.MetaPathFinder, importlib.abc.Loader):
    def __init__(self, sources):
        self.sources = dict(sources)
        self.loaded = []

    def find_spec(self, name, path=None, target=None):
        if name in self.sources:
            return importlib.util.spec_from_loader(name, self)
        return None

    def exec_module(self, module):
        self.loaded.append(module.__name__)
        exec(self.sources[module.__name__], module.__dict__)

def install(sources):
    finder = SourceFinder(sources)
    sys.meta_path.insert(0, finder)
    return finder

def uninstall(finder):
    if finder in sys.meta_path:
        sys.meta_path.remove(finder)
    for name in finder.loaded:
        sys.modules.pop(name, None)

@@tests

    "a semaphore, simulated"
    def job(waits, result):
        for w in waits:
            yield w
        return result
    def jobs():
        return [job([3], "a"), job([1, 1], "b"), job([1], "c")]
    assert gather_limited(jobs(), 2) == (["a", "b", "c"], 3)
    assert gather_limited(jobs(), 1) == (["a", "b", "c"], 6)
    assert gather_limited(jobs(), 5) == (["a", "b", "c"], 3)
    assert gather_limited([], 2) == ([], 0)
    order = []
    def noisy(name, waits):
        order.append(("start", name))
        for w in waits:
            yield w
        order.append(("end", name))
        return name

@@solution
import heapq

def gather_limited(coros, limit):
    results = [None] * len(coros)
    queue, seq, next_index, total = [], 0, 0, 0

    def start(now):
        nonlocal seq, next_index
        heapq.heappush(queue, (now, seq, next_index))
        seq += 1
        next_index += 1

    while next_index < min(limit, len(coros)):
        start(0)
    while queue:
        now, _, i = heapq.heappop(queue)
        try:
            delay = next(coros[i])
        except StopIteration as stop:
            results[i] = stop.value
            total = max(total, now)
            if next_index < len(coros):
                start(now)
            continue
        heapq.heappush(queue, (now + delay, seq, i))
        seq += 1
    return results, total

:::exercise id=c1 points=50 title="C: A build tool" exam=yes
Write a class `Builder` for describing and running build steps.

- **`rule(name, deps=(), cost=1)`** is a decorator that registers the decorated function as the rule `name`, which depends on the rules named in `deps` and takes `cost` seconds; it returns the function unchanged. Registering a name twice raises `ValueError`.
- **`order(target)`** returns the names of the rules `target` needs, itself included, each once, with every rule after its dependencies: visit each rule's `deps` in the order listed, depth first, adding a rule after its dependencies. An unknown rule raises `KeyError`; a cycle raises `ValueError` whose message starts `"cycle: "` and lists the path, such as `"cycle: x -> y -> x"`.
- **`build(target)`** runs the rules in `order(target)` that don't have a cached result, each called with the results of its `deps` as positional arguments, in order; it caches every result and returns `target`'s.
- **`invalidate(name)`** forgets the cached results of `name` and of every rule that depends on it, directly or indirectly.
- **`dry_run()`** is a context manager yielding a list: inside the block, `build` calls nothing and caches nothing, appends to that list the names it **would** run, and returns `None`. Normal building resumes when the block ends, even after an error.
- **`timeline(target, workers)`** simulates building everything in `order(target)` (ignoring the cache) on `workers` workers: a rule can start once all its dependencies have finished; whenever a worker is free it takes the ready rule that comes **earliest** in `order(target)`. Return `(starts, makespan)`: a dict of each rule's start time, and the time the last one finished.
@@starter
import heapq
from contextlib import contextmanager

    ___
@@tests
def make():
    b = Builder()
    calls = []
    @b.rule("config")
    def config():
        calls.append("config")
        return {"debug": False}
    @b.rule("lib", deps=["config"], cost=3)
    def lib(cfg):
        calls.append("lib")
        return "lib-dbg" if cfg["debug"] else "lib"
    @b.rule("docs", deps=["config"], cost=2)
    def docs(cfg):
        calls.append("docs")
        return "docs"
    @b.rule("app", deps=["lib", "config"], cost=2)
    def app(l, cfg):
        calls.append("app")
        return f"app({l})"
    @b.rule("all", deps=["app", "docs"])
    def everything(a, d):
        calls.append("all")
        return [a, d]
    return b, calls, everything

def test_rules_and_order():
    "registering and ordering"
    b, calls, everything = make()
    assert everything.__name__ == "everything" and calls == []
    assert b.order("all") == ["config", "lib", "app", "docs", "all"] and b.order("lib") == ["config", "lib"]
    try:
        b.rule("lib")(lambda: 0)
        assert False
    except ValueError:
        pass
    c = Builder()
    c.rule("x", deps=["y"])(lambda y: y)
    c.rule("y", deps=["x"])(lambda x: x)
    c.rule("z", deps=["nowhere"])(lambda n: n)

    except ValueError as e:

import heapq
from contextlib import contextmanager

class Builder:
    def __init__(self):
        self.rules = {}
        self.cache = {}
        self._planned = None

    def rule(self, name, deps=(), cost=1):
        def decorate(func):
            if name in self.rules:
                raise ValueError(f"rule {name!r} already exists")
            self.rules[name] = (func, tuple(deps), cost)
            return func
        return decorate

    def order(self, target):
        result, state = [], {}

        def visit(name, path):
            if name not in self.rules:
                raise KeyError(name)
            if state.get(name) == "done":
                return
            if state.get(name) == "visiting":
                raise ValueError("cycle: " + " -> ".join(path + [name]))
            state[name] = "visiting"
            for dep in self.rules[name][1]:
                visit(dep, path + [name])
            state[name] = "done"
            result.append(name)

        visit(target, [])
        return result

    def build(self, target):
        names = self.order(target)
        if self._planned is not None:
            self._planned.extend(n for n in names if n not in self.cache)
            return None
        for name in names:
            if name not in self.cache:
                func, deps, _ = self.rules[name]
                self.cache[name] = func(*[self.cache[d] for d in deps])
        return self.cache[target]

    def invalidate(self, name):
        stale = {name}
        changed = True
        while changed:
            changed = False
            for rule, (_, deps, _) in self.rules.items():
                if rule not in stale and any(d in stale for d in deps):
                    stale.add(rule)
                    changed = True
        for rule in stale:
            self.cache.pop(rule, None)

    @contextmanager
    def dry_run(self):
        planned, previous = [], self._planned
        self._planned = planned
        try:
            yield planned
        finally:
            self._planned = previous

    def timeline(self, target, workers):
        names = self.order(target)
        rank = {n: i for i, n in enumerate(names)}
        waiting = {n: set(self.rules[n][1]) for n in names}
        ready = [rank[n] for n in names if not waiting[n]]
        heapq.heapify(ready)
        running, starts, now = [], {}, 0
        while ready or running:
            while ready and len(running) < workers:
                name = names[heapq.heappop(ready)]
                starts[name] = now
                heapq.heappush(running, (now + self.rules[name][2], rank[name], name))
            end, _, name = heapq.heappop(running)
            now, finished = end, [name]
            while running and running[0][0] == end:
                finished.append(heapq.heappop(running)[2])
            for done in finished:
                for n in names:
                    if done in waiting[n]:
                        waiting[n].discard(done)
                        if not waiting[n]:
                            heapq.heappush(ready, rank[n])
        return starts, now
