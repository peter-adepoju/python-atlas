## Part B: five problems

:::exercise id=p1 points=10 title="P1: Chunks"
Write a generator `chunked(iterable, size)` yielding lists of `size` consecutive items (the last may be shorter, and no empty list is yielded). It must be lazy, working on infinite iterators, and must raise `ValueError` for a `size` below 1 **when called**, not when first iterated (so wrap a generator in an ordinary function).
@@starter
def chunked(iterable, size):
    ___
@@tests
def test_chunked():
    "lazy chunks"
    import itertools
    assert list(chunked(range(7), 3)) == [[0, 1, 2], [3, 4, 5], [6]] and list(chunked([], 2)) == []
    assert list(itertools.islice(chunked(itertools.count(), 2), 2)) == [[0, 1], [2, 3]]
    try:
        chunked([1], 0)
        assert False
    except ValueError:
        pass
@@hint
Check `size` in the outer function, then return an inner generator. Inside it, collect items with `itertools.islice(it, size)` until a chunk comes back empty.
@@solution
import itertools

def chunked(iterable, size):
    if size < 1:
        raise ValueError("size must be at least 1")

    def chunks():
        it = iter(iterable)
        while True:
            chunk = list(itertools.islice(it, size))
            if not chunk:
                return
            yield chunk

    return chunks()

:::exercise id=p2 points=15 title="P2: A rate limiter"
Write `limit_rate(calls, per, clock)`: a decorator factory. The decorated function may run at most `calls` times in any window of `per` seconds, measured with `clock()` (a function returning the current time, passed in so it can be tested). A call that would exceed the limit doesn't run, returns `None`, and adds one to the wrapper's `dropped` attribute. Keep the function's name. (A call at time `t` counts against later calls made before `t + per`.)
@@starter
import functools
from collections import deque

    ___
@@tests
def test_limit():
    "a sliding window"
    now = [0]
    @limit_rate(2, 10, lambda: now[0])
    def ping(x):
        return x * 2
    results = []
    for t in [0, 1, 2, 9, 10, 11, 12]:
        now[0] = t
        results.append(ping(t))
    assert results == [0, 2, None, None, 20, 22, None] and ping.dropped == 3 and ping.__name__ == "ping"
@@hint
Keep a `deque` of the times of calls that ran. Before each call, pop times from the left while they're `<= now - per`; then run only if fewer than `calls` remain.
@@solution
import functools
from collections import deque

def limit_rate(calls, per, clock):
    def decorate(func):
        recent = deque()

        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            now = clock()
            while recent and recent[0] <= now - per:
                recent.popleft()
            if len(recent) >= calls:
                wrapper.dropped += 1
                return None
            recent.append(now)
            return func(*args, **kwargs)

        wrapper.dropped = 0
        return wrapper
    return decorate
:::

@@tests
def test_atomic():
    "a transaction"
    class Cart:
        def __init__(self):
            self.items, self.total = ["tea"], 3
    c = Cart()
    with atomic(c) as same:
        assert same is c
        c.items.append("jam")
        c.total += 2
    assert c.items == ["tea", "jam"] and c.total == 5
    try:
        with atomic(c):
            c.items.append("cake")
            c.total = 99
            c.note = "temp"
            raise RuntimeError
    except RuntimeError:
        pass

@@solution
import copy
from contextlib import contextmanager

@contextmanager
def atomic(obj):
    saved = copy.deepcopy(obj.__dict__)
    try:
        yield obj
    except BaseException:
        obj.__dict__.clear()
        obj.__dict__.update(saved)
        raise

import functools

class Scaled:
    def __init__(self, factor):
        self.factor = factor

    def __get__(self, obj, owner=None):
        if obj is None:
            return self
        return obj.metres / self.factor

    def __set__(self, obj, value):
        obj.metres = value * self.factor

@functools.total_ordering
class Distance:
    km = Scaled(1000)
    cm = Scaled(0.01)
    miles = Scaled(1609.344)

    def __init__(self, metres):
        self.metres = metres

    def __eq__(self, other):
        if not isinstance(other, Distance):
            return NotImplemented
        return self.metres == other.metres

    def __lt__(self, other):
        if not isinstance(other, Distance):
            return NotImplemented
        return self.metres < other.metres

    __hash__ = None

    def __repr__(self):
        return f"Distance({round(self.metres)}m)"

@@tests
def test_pool():
    "first free worker"
    plan, end = schedule([4, 2, 3, 1], 2)
    assert plan == [(0, 0, 4), (1, 0, 2), (1, 2, 5), (0, 4, 5)] and end == 5
    assert schedule([], 3) == ([], 0) and schedule([5], 3) == ([(0, 0, 5)], 5)

@@hint
Keep a heap of `(free_at, worker)`; pop the first free one, record the job, push it back with its new free time. `best_order` is `schedule(sorted(durations, reverse=True), workers)[1]`.
@@solution
import heapq

def schedule(durations, workers):
    free = [(0, w) for w in range(workers)]
    heapq.heapify(free)
    plan = []
    for d in durations:
        start, w = heapq.heappop(free)
        plan.append((w, start, start + d))
        heapq.heappush(free, (start + d, w))
    makespan = max((end for _, _, end in plan), default=0)
    return plan, makespan

def best_order(durations, workers):
    return schedule(sorted(durations, reverse=True), workers)[1]

## Part C: a larger program

:::exercise id=c1 points=30 title="C: A lazy query builder"
Write a class `Query(source)` for building lazy pipelines over any iterable. Each of these returns a **new** `Query` and leaves the original unchanged: `where(pred)` keeps items for which `pred(item)` is true; `select(func)` maps; `limit(n)` keeps at most the first `n` items and **stops reading** the source after that; `batch(n)` groups items into lists of `n` (the last may be shorter). Nothing is read from the source until the query is iterated. Iterating a `Query` (`__iter__`) runs the pipeline afresh each time (so a query over a list can be iterated many times), and records in the query's `read` attribute how many source items that run took. Also `count()` (the number of results), `first(default=None)` (the first result, reading as little as possible) and `to_list()`.
@@starter

@@tests
def test_query_basics():
    "chaining"
    q = Query(range(10))
    evens = q.where(lambda x: x % 2 == 0)
    squares = evens.select(lambda x: x * x)
    assert squares.to_list() == [0, 4, 16, 36, 64] and squares.to_list() == [0, 4, 16, 36, 64]
    assert q.to_list() == list(range(10)) and evens.count() == 5
    assert Query("abcde").batch(2).to_list() == [["a", "b"], ["c", "d"], ["e"]]
    assert Query([]).first("none") == "none" and Query([3, 4]).first() == 3

def test_query_lazy():
    "laziness and reading as little as possible"
    import itertools
    pulled = []
    def source():
        for i in itertools.count():
            pulled.append(i)
            yield i
    q = Query(source()).where(lambda x: x % 3 == 0).limit(3)
    assert pulled == []
    assert q.to_list() == [0, 3, 6] and q.read == 7 and pulled == list(range(7))
    words = Query(["kiwi", "fig", "banana", "apple", "plum"])
    first_long = words.where(lambda w: len(w) > 4)
    assert first_long.first() == "banana" and first_long.read == 3
    assert words.select(len).limit(2).batch(5).to_list() == [[4, 3]]
@@hint
Store the source and a tuple of stages (functions that take an iterator and return one). Each builder returns `Query(self.source, self.stages + (stage,))`. In `__iter__`, wrap the source in a small counting generator that updates `self.read`, pass it through each stage in turn, then `yield from` the result. `limit` is `lambda it: itertools.islice(it, n)`.
@@solution
import itertools

class Query:
    def __init__(self, source, stages=()):
        self.source = source
        self.stages = tuple(stages)
        self.read = 0

    def _add(self, stage):
        return Query(self.source, self.stages + (stage,))

    def where(self, pred):
        return self._add(lambda it: (x for x in it if pred(x)))

    def select(self, func):
        return self._add(lambda it: (func(x) for x in it))

    def limit(self, n):
        return self._add(lambda it: itertools.islice(it, n))

    def batch(self, n):
        def batches(it):
            while True:
                chunk = list(itertools.islice(it, n))
                if not chunk:
                    return
                yield chunk
        return self._add(batches)

    def __iter__(self):
        self.read = 0

        def counted():
            for item in self.source:
                self.read += 1
                yield item

        stream = counted()
        for stage in self.stages:
            stream = stage(stream)
        yield from stream

    def to_list(self):
        return list(self)

    def count(self):
        return sum(1 for _ in self)

    def first(self, default=None):
        return next(iter(self), default)
