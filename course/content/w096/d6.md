@@tests
def test_parse():
    "back into parts"
    assert parse_comment("Lib/wraptext.py:34: issue (blocking): a bare except") == {
        "path": "Lib/wraptext.py", "line": 34, "label": "issue", "decoration": "blocking", "subject": "a bare except"}
    assert parse_comment("praise: thanks!") == {"path": None, "line": None, "label": "praise", "decoration": None, "subject": "thanks!"}
    try:
        parse_comment("This is wrong.")
    except ValueError as e:
        assert str(e) == "not a conventional comment"
    else:
        raise AssertionError("plain text")
@@solution
import re

PATTERN = re.compile(r"(?:(?P<path>[^\s:]+):(?P<line>\d+): )?(?P<label>praise|nitpick|suggestion|issue|question|thought)"
                     r"(?: \((?P<decoration>[\w-]+)\))?: (?P<subject>.+)", re.S)

def parse_comment(text):
    m = PATTERN.fullmatch(text)
    if not m:
        raise ValueError("not a conventional comment")
    return {"path": m["path"], "line": int(m["line"]) if m["line"] else None, "label": m["label"],
            "decoration": m["decoration"], "subject": m["subject"]}
:::

import itertools

def exhaustive(fn, oracle, alphabet, max_len):
    found = []
    for n in range(max_len + 1):
        for chars in itertools.product(alphabet, repeat=n):
            line = "".join(chars)
            if fn(line) != oracle(line):
                found.append(line)
    return found

## Several at once

Open source runs on other people's time, so a contributor usually has several things in flight: a pull request waiting for review, another with a CI failure, a third with a question to answer, a review promised to someone. The trick is a short daily routine with a clear order: first anything **broken** (a failing CI run, a conflict with `main`), then anything **waiting on you** (a reviewer's comment), then anything that's been **waiting on others** for a long time (a single polite note after two weeks or so: "Is there anything else needed here?"), then new work. Closed pull requests get a line in the log and, if the reason is useful, in the write-up.

:::exercise id=forge points=50 title="The sprint board"
Write `sprint_board(entries, today)`. Each entry has `pr`, `title`, `status` (`"open"`, `"merged"` or `"closed"`), `ci` (`"passed"`, `"failed"` or `"pending"`), `waiting_on` (`"me"`, `"reviewer"` or `None`), `conflict` (a bool) and `last_activity` (`"YYYY-MM-DD"`). For **open** entries, actions in priority order:

1. `"fix CI: #PR TITLE"` if CI failed;
2. `"resolve conflict: #PR TITLE"` if there's a conflict;
3. `"reply: #PR TITLE"` if it's waiting on you;
4. `"polite ping: #PR TITLE (D days)"` if it's waiting on the reviewer and the last activity was at least 14 days before today.

def e(pr, title, status="open", ci="passed", waiting_on=None, conflict=False, last="2026-12-01"):

ENTRIES = [
    e(140210, "Clarify json.dumps sort_keys", status="merged"),
    e(140188, "Keep a word that exactly fills the width", waiting_on="reviewer", last="2026-11-20"),
    e(2815, "Fix a broken example", ci="failed", waiting_on="me", last="2026-12-08"),
    e(140240, "Test calendar.monthrange", conflict=True, waiting_on="reviewer", last="2026-12-02"),
    e(140199, "Clarify difflib cutoff", status="closed"),
    e(140250, "Document shutil.copy2 times", waiting_on="me", last="2026-12-09"),
]

    "this morning's list"
    lines, next_step = sprint_board(ENTRIES, "2026-12-10")
    assert lines == [
        "fix CI: #2815 Fix a broken example",
        "resolve conflict: #140240 Test calendar.monthrange",
        "reply: #2815 Fix a broken example",
        "reply: #140250 Document shutil.copy2 times",
        "polite ping: #140188 Keep a word that exactly fills the width (20 days)",
        "open: 4, merged: 1, closed: 1",
    ]
    assert next_step == "next: fix CI: #2815 Fix a broken example"

def test_quiet():
    "nothing waiting on you"
    lines, next_step = sprint_board([e(1, "A", waiting_on="reviewer", last="2026-12-01")], "2026-12-10")
    assert lines == ["open: 1, merged: 0, closed: 0"] and next_step == "next: start something new"
    assert sprint_board([e(1, "A", waiting_on="reviewer", last="2026-11-26")], "2026-12-10")[0][0] == "polite ping: #1 A (14 days)"
@@solution
from datetime import date

def sprint_board(entries, today):
    now = date.fromisoformat(today)
    open_ = sorted((x for x in entries if x["status"] == "open"), key=lambda x: x["pr"])
    actions = []
    for x in open_:
        if x["ci"] == "failed":
            actions.append(f"fix CI: #{x['pr']} {x['title']}")
    for x in open_:
        if x["conflict"]:
            actions.append(f"resolve conflict: #{x['pr']} {x['title']}")
    for x in open_:
        if x["waiting_on"] == "me":
            actions.append(f"reply: #{x['pr']} {x['title']}")
    for x in open_:
        days = (now - date.fromisoformat(x["last_activity"])).days
        if x["waiting_on"] == "reviewer" and days >= 14:
            actions.append(f"polite ping: #{x['pr']} {x['title']} ({days} days)")
    count = lambda s: sum(1 for x in entries if x["status"] == s)
    lines = actions + [f"open: {count('open')}, merged: {count('merged')}, closed: {count('closed')}"]
    return lines, "next: " + (actions[0] if actions else "start something new")

## Bonus: what to review next

:::exercise id=forge2 points=30 title="Bonus: choosing a pull request to review"

def p(pr, title, author="someone", reviews=0, has_test=True, lines=30, labels=(), opened="2026-12-01"):
    return {"pr": pr, "author": author, "title": title, "reviews": reviews, "has_test": has_test, "lines": lines,
            "labels": list(labels), "opened": opened}

PRS = [
    p(140260, "Fix difflib cutoff=0"),
    p(140261, "Rewrite the tokenizer", lines=4200, has_test=False),
    p(140262, "Clarify json docs", labels=["docs"], has_test=False, lines=8),
    p(140263, "My own fix", author="ada"),
    p(140264, "Add calendar tests", labels=["tests"], reviews=2, lines=120),
    p(139001, "Old small fix", opened="2026-08-01", lines=12),
    p(140265, "Experiment", labels=["do-not-merge"]),
]

def test_queue():
    "where a review helps most"
    assert review_queue(PRS, "2026-12-10", "ada") == ["#140260 (6): Fix difflib cutoff=0", "#140262 (6): Clarify json docs",
                                                      "#139001 (5): Old small fix"]
    assert len(review_queue(PRS, "2026-12-10", "ada", n=10)) == 5
@@solution
from datetime import date

def review_queue(prs, today, me, n=3):
    now = date.fromisoformat(today)
    scored = []
    for x in prs:
        if x["author"] == me or "do-not-merge" in x["labels"]:
            continue
        score = 3 * (x["reviews"] == 0) + x["has_test"] + (2 if x["lines"] < 50 else 1 if x["lines"] < 200 else 0)
        score += bool({"docs", "tests"} & set(x["labels"]))
        score -= (now - date.fromisoformat(x["opened"])).days > 60
        scored.append((score, x))
    scored.sort(key=lambda pair: (-pair[0], pair[1]["pr"]))
    return [f"#{x['pr']} ({s}): {x['title']}" for s, x in scored[:n]]
