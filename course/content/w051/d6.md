def _mst(vertices, edges, skip=None):
    parent = {v: v for v in vertices}
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    total, used = 0, []
    for i in sorted(range(len(edges)), key=lambda i: edges[i][2]):
        if i == skip:
            continue
        u, v, w = edges[i]
        if find(u) != find(v):
            parent[find(u)] = find(v)
            total += w
            used.append(i)
    if len(used) != len(parent) - 1:
        return None, used
    return total, used

def second_mst_total(vertices, edges):
    total, used = _mst(vertices, edges)
    if total is None:
        return None
    options = [_mst(vertices, edges, skip=i)[0] for i in used]
    options = [t for t in options if t is not None]
    return min(options) if options else None

from collections import deque

def escape_steps(grid):
    rows, cols = len(grid), len(grid[0])
    start = ((0, 0), 0)
    dist = {start: 0}
    queue = deque([start])
    while queue:
        (r, c), broken = queue.popleft()
        if (r, c) == (rows - 1, cols - 1):
            return dist[((r, c), broken)]
        for nr, nc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            if 0 <= nr < rows and 0 <= nc < cols:
                nb = broken + (grid[nr][nc] == "#")
                if nb <= 1 and ((nr, nc), nb) not in dist:
                    dist[((nr, nc), nb)] = dist[((r, c), broken)] + 1
                    queue.append(((nr, nc), nb))
    return None

## Forge: a delivery planner

A district's roads take a number of minutes to drive; most are two-way, some one-way. A delivery firm wants a planner.

Write `RoadMap(roads, one_way=())`, where both are lists of `(a, b, minutes)`:

- Store the graph as `graph`, a dict of dicts (if two roads join the same pair, keep the quicker). Raise `ValueError` for a road from a place to itself or with minutes that aren't positive. Unknown places given to any method raise `KeyError`.
- `places()`: sorted.
- `travel_time(a, b)` and `route(a, b)`: the quickest time and route (`None` if unreachable), with Dijkstra's algorithm; explore neighbours in alphabetical order so routes are predictable.
- `all_times(start)`: a dict of the quickest time to every reachable place.
- `close(a, b)` and `reopen(a, b)`: close or reopen the road between a and b, **in both directions**; closed roads are ignored by every search until reopened.
- `nearest_depot(place, depots)`: `(depot, minutes)` for the depot that can reach `place` soonest (alphabetically first on a tie), or `None`.
- `delivery_round(depot, stops)`: a greedy round: from the depot, repeatedly drive to the **nearest** unvisited stop (by travel time, ties alphabetical), then back to the depot. Return `(order, total_minutes)`; raise `ValueError` if a stop or the depot can't be reached. (Finding the truly best round is the travelling salesperson problem, which no known algorithm solves quickly; greedy is a common, decent heuristic.)
- `cable_plan()`: fibre is laid along **two-way** roads only, and must connect every place. Return `(total_minutes_of_road_used, roads_used)` from Kruskal's algorithm, considering roads by length (ties by their names); raise `ValueError` if the two-way roads don't connect everything.

One private `_dijkstra(sources)` that skips closed roads and returns `(dist, parent)` serves `travel_time`, `route`, `all_times`, `nearest_depot` (run it from each depot) and `delivery_round` (run it from wherever the van is). Keep the two-way roads in their own list for `cable_plan`.
@@solution
import heapq

class RoadMap:
    def __init__(self, roads, one_way=()):
        self.graph = {}
        self.two_way = []
        self.closed = set()
        for a, b, minutes in roads:
            self._check(a, b, minutes)
            self._add(a, b, minutes)
            self._add(b, a, minutes)
            self.two_way.append((a, b, minutes))
        for a, b, minutes in one_way:
            self._check(a, b, minutes)
            self._add(a, b, minutes)
            self.graph.setdefault(b, {})

    def _check(self, a, b, minutes):
        if a == b or minutes <= 0:
            raise ValueError(f"bad road {a!r} to {b!r} ({minutes} minutes)")

    def _add(self, a, b, minutes):
        self.graph.setdefault(a, {})
        self.graph[a][b] = min(minutes, self.graph[a].get(b, float("inf")))

    def places(self):
        return sorted(self.graph)

    def _known(self, *places):
        for p in places:
            if p not in self.graph:
                raise KeyError(p)

    def close(self, a, b):
        self._known(a, b)
        self.closed.add((a, b))
        self.closed.add((b, a))

    def reopen(self, a, b):
        self.closed.discard((a, b))
        self.closed.discard((b, a))

    def _dijkstra(self, sources):
        dist = {s: 0 for s in sources}
        parent = {s: None for s in sources}
        heap = [(0, s) for s in sorted(sources)]
        heapq.heapify(heap)
        done = set()
        while heap:
            d, u = heapq.heappop(heap)
            if u in done:
                continue
            done.add(u)
            for v, w in sorted(self.graph[u].items()):
                if (u, v) in self.closed:
                    continue
                if d + w < dist.get(v, float("inf")):
                    dist[v], parent[v] = d + w, u
                    heapq.heappush(heap, (d + w, v))
        return dist, parent

    def all_times(self, start):
        self._known(start)
        return self._dijkstra([start])[0]

    def travel_time(self, a, b):
        self._known(a, b)
        return self._dijkstra([a])[0].get(b)

    def route(self, a, b):
        self._known(a, b)
        dist, parent = self._dijkstra([a])
        if b not in dist:
            return None
        path = []
        while b is not None:
            path.append(b)
            b = parent[b]
        return path[::-1]

    def nearest_depot(self, place, depots):
        self._known(place, *depots)
        best = None
        for d in sorted(depots):
            t = self._dijkstra([d])[0].get(place)
            if t is not None and (best is None or t < best[1]):
                best = (d, t)
        return best

    def delivery_round(self, depot, stops):
        self._known(depot, *stops)
        todo = set(stops) - {depot}
        here, order, total = depot, [], 0
        while todo:
            dist = self._dijkstra([here])[0]
            options = [(dist[s], s) for s in todo if s in dist]
            if not options:
                raise ValueError("some stops can't be reached")
            t, nxt = min(options)
            order.append(nxt)
            total += t
            todo.discard(nxt)
            here = nxt
        back = self._dijkstra([here])[0].get(depot)
        if back is None:
            raise ValueError("can't get back to the depot")
        return order, total + back

    def cable_plan(self):
        parent = {p: p for p in self.graph}
        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x
        total, used = 0, []
        for a, b, m in sorted(self.two_way, key=lambda r: (r[2], r[0], r[1])):
            ra, rb = find(a), find(b)
            if ra != rb:
                parent[ra] = rb
                total += m
                used.append((a, b, m))
        if len({find(p) for p in self.graph}) > 1:
            raise ValueError("the two-way roads don't connect every place")
        return total, used

:::think title="Which algorithm, and why"
Every method in the Forge uses one of this week's algorithms, chosen by the shape of its question. Travel times have non-negative weights, so Dijkstra's algorithm. The cable plan wants everything connected cheaply, not short routes, so a minimum spanning tree, and only over two-way roads. The delivery round is a much harder question (visit everything, then return) that no efficient algorithm answers exactly, so the planner uses a reasonable greedy rule built from shortest paths, and says so. Recognising which kind of question you're facing, and when a problem is hard enough that a good approximation is the honest answer, matters as much as knowing the algorithms.
:::

import math

class ExchangeDesk:
    def __init__(self, rates, fee=0.0):
        if not 0 <= fee < 1:
            raise ValueError("the fee must be from 0 up to (not including) 1")
        self.rates = {}
        for (a, b), r in rates.items():
            if r <= 0 or a == b:
                raise ValueError(f"bad rate {a} to {b}: {r}")
            self.rates[(a, b)] = r * (1 - fee)
        self.currencies = sorted({c for pair in self.rates for c in pair})

    def _weights(self):
        return [(a, b, -math.log(r)) for (a, b), r in self.rates.items()]

    def loop(self):
        dist = {c: 0.0 for c in self.currencies}
        parent = {c: None for c in self.currencies}
        last = None
        for _ in range(len(self.currencies)):
            last = None
            for a, b, w in self._weights():
                if dist[a] + w < dist[b] - 1e-12:
                    dist[b], parent[b] = dist[a] + w, a
                    last = b
        if last is None:
            return None
        for _ in range(len(self.currencies)):
            last = parent[last]
        cycle, v = [last], parent[last]
        while v != last:
            cycle.append(v)
            v = parent[v]
        return cycle[::-1]

    def best(self, source, target):
        if source not in self.currencies or target not in self.currencies:
            raise KeyError(source if source not in self.currencies else target)
        if self.loop() is not None:
            raise ValueError("the rates contain a money-making loop")
        dist = {c: float("inf") for c in self.currencies}
        parent = {c: None for c in self.currencies}
        dist[source] = 0.0
        for _ in range(len(self.currencies) - 1):
            changed = False
            for a, b, w in self._weights():
                if dist[a] + w < dist[b] - 1e-12:
                    dist[b], parent[b] = dist[a] + w, a
                    changed = True
            if not changed:
                break
        if dist[target] == float("inf"):
            return None
        path, c = [], target
        while c is not None:
            path.append(c)
            c = parent[c]
        return math.exp(-dist[target]), path[::-1]

    def convert(self, amount, path):
        for a, b in zip(path, path[1:]):
            amount *= self.rates[(a, b)]
        return amount
