import heapq


class UnknownPlace(LookupError):
    pass


class NoRoute(Exception):
    pass


def naismith_minutes(km, climb_m):
    return round(km / 5 * 60 + climb_m / 600 * 60)


class Route:
    def __init__(self, stops, km, climb_m):
        self.stops, self.km, self.climb_m = list(stops), km, climb_m

    @property
    def minutes(self):
        return naismith_minutes(self.km, self.climb_m)

    def __repr__(self):
        return f"Route({self.stops}, {self.km} km, {self.climb_m} m)"


class PathNetwork:
    def __init__(self, heights):
        self.heights = dict(heights)
        self.paths = {name: {} for name in self.heights}

    def _check(self, *names):
        for name in names:
            if name not in self.heights:
                raise UnknownPlace(name)

    def add_path(self, a, b, km):
        self._check(a, b)
        if km <= 0:
            raise ValueError("km must be positive")
        self.paths[a][b] = km
        self.paths[b][a] = km

    def climb(self, stops):
        return sum(max(0, self.heights[b] - self.heights[a]) for a, b in zip(stops, stops[1:]))

    def shortest(self, start, end):
        self._check(start, end)
        dist, prev, done = {start: 0}, {}, set()
        heap = [(0, start)]
        while heap:
            d, node = heapq.heappop(heap)
            if node in done:
                continue
            done.add(node)
            if node == end:
                break
            for nxt, km in sorted(self.paths[node].items()):
                if d + km < dist.get(nxt, float("inf")):
                    dist[nxt] = d + km
                    prev[nxt] = node
                    heapq.heappush(heap, (d + km, nxt))
        if end not in done:
            raise NoRoute(f"no route from {start} to {end}")
        stops = [end]
        while stops[-1] != start:
            stops.append(prev[stops[-1]])
        stops.reverse()
        return Route(stops, round(dist[end], 2), self.climb(stops))

## Migrations

Last week's skeleton created its tables when it started. That stops working the day there's real data: you can't drop and recreate a database with someone's walks in it, and every copy of the database (yours, your tests', a user's) may be at a different stage. A **migration** is one numbered change to the schema (`1: create junction and path`, `2: create saved_route and saved_stop`), kept in the code forever. The database records which migrations it has had in a `schema_version` table, and on start-up the program applies the missing ones **in order**.

Rules that keep migrations safe:

- **Never edit a migration that has been applied** anywhere: write a new one. (The challenge detects edits by storing a hash of each.)
- **Each migration is all or nothing**: run it inside a transaction, so a failure halfway leaves the database as it was. SQLite can roll back schema changes too. Note that `executescript` commits as it goes, so run the statements one by one inside `BEGIN` and `COMMIT` instead.
- **Test migrations** on a copy of real data before they meet real data.

:::example
import sqlite3

MIGRATIONS = [
    "CREATE TABLE junction (name TEXT PRIMARY KEY, height_m INTEGER NOT NULL)",
    "CREATE TABLE saved_route (id INTEGER PRIMARY KEY, name TEXT NOT NULL UNIQUE COLLATE NOCASE)",
]

def migrate(conn):
    conn.execute("CREATE TABLE IF NOT EXISTS schema_version (version INTEGER NOT NULL)")
    done = conn.execute("SELECT COALESCE(MAX(version), 0) FROM schema_version").fetchone()[0]
    for number, sql in enumerate(MIGRATIONS, start=1):
        if number > done:
            with conn:
                conn.execute(sql)
                conn.execute("INSERT INTO schema_version VALUES (?)", (number,))
            print("applied", number)

conn = sqlite3.connect(":memory:")
migrate(conn)
migrate(conn)
print([r[0] for r in conn.execute("SELECT name FROM sqlite_master WHERE type = 'table' ORDER BY name")])
:::

The second call applies nothing: the database knows it's up to date.

## The repository

A saved route is a name and an **ordered** list of stops. Two tables model it: `saved_route (id, name UNIQUE COLLATE NOCASE)` and `saved_stop (route_id, position, junction)`, with the primary key `(route_id, position)` so the order is stored, not assumed. The repository (Week 36) hides the SQL behind methods in the domain's terms: `add(name, stops)`, `get(name)`, `names()`, `delete(name)`. It turns the database's `IntegrityError` for a duplicate name into the domain's own `DuplicateName`, so nothing above it needs to know which database it uses.

## Service and API

Two kinds of checking, in two places:

- **At the edge** (the API handler or the service's first lines): is the input well formed? A name of 1 to 40 characters, letters, digits, spaces, hyphens and apostrophes, with extra spaces tidied; stops as a list of at least two strings.
- **In the domain and service**: does it make sense? Every stop must exist, and each consecutive pair must be joined by a path. The service asks the domain (`network.paths`), then saves inside one transaction, and returns the route's distance, climb and time so the interface can show them.

The API follows Day 3's table: `POST /saved-routes` (201, or 400 for bad input, 404 for an unknown place or missing path, 409 for a taken name), `GET /saved-routes`, `GET /saved-routes/NAME` and `DELETE /saved-routes/NAME` (204 or 404), with the name URL-decoded (`urllib.parse.unquote`).

:::key
Change live schemas with numbered migrations recorded in a `schema_version` table and applied in order on start-up; never edit an applied migration; run each in a transaction (statement by statement, not `executescript`) so it's all or nothing. Model ordered children with a position column in the primary key. Use `COLLATE NOCASE` for names compared without capitals. Repositories speak the domain's language and translate database errors into domain exceptions. Validate form at the edge and meaning in the domain and service; return plain data to interfaces, which map exceptions to status codes (400, 404, 409).

import sqlite3

def current_version(conn):
    conn.execute("CREATE TABLE IF NOT EXISTS schema_version (version INTEGER NOT NULL)")
    return conn.execute("SELECT COALESCE(MAX(version), 0) FROM schema_version").fetchone()[0]

def migrate(conn, migrations):
    applied = []
    start = current_version(conn)
    for number, sql in enumerate(migrations, start=1):
        if number <= start:
            continue
        with conn:
            conn.execute(sql)
            conn.execute("INSERT INTO schema_version VALUES (?)", (number,))
        applied.append(number)
    return applied

import sqlite3

class DuplicateName(Exception):
    pass

class RouteRepository:
    def __init__(self, conn):
        self.conn = conn
        conn.execute("CREATE TABLE IF NOT EXISTS saved_route (id INTEGER PRIMARY KEY, name TEXT NOT NULL UNIQUE COLLATE NOCASE)")
        conn.execute("CREATE TABLE IF NOT EXISTS saved_stop (route_id INTEGER NOT NULL, position INTEGER NOT NULL, "
                     "junction TEXT NOT NULL, PRIMARY KEY (route_id, position))")

    def add(self, name, stops):
        try:
            with self.conn:
                cur = self.conn.execute("INSERT INTO saved_route (name) VALUES (?)", (name,))
                self.conn.executemany("INSERT INTO saved_stop VALUES (?, ?, ?)",
                                      [(cur.lastrowid, i, stop) for i, stop in enumerate(stops)])
        except sqlite3.IntegrityError:
            raise DuplicateName(name)
        return cur.lastrowid

    def get(self, name):
        row = self.conn.execute("SELECT id, name FROM saved_route WHERE name = ?", (name,)).fetchone()
        if row is None:
            return None
        stops = [r[0] for r in self.conn.execute("SELECT junction FROM saved_stop WHERE route_id = ? ORDER BY position", (row[0],))]
        return row[1], stops

    def names(self):
        return [r[0] for r in self.conn.execute("SELECT name FROM saved_route ORDER BY name COLLATE NOCASE")]

    def delete(self, name):
        row = self.conn.execute("SELECT id FROM saved_route WHERE name = ?", (name,)).fetchone()
        if row is None:
            return False
        with self.conn:
            self.conn.execute("DELETE FROM saved_stop WHERE route_id = ?", (row[0],))
            self.conn.execute("DELETE FROM saved_route WHERE id = ?", (row[0],))
        return True

    name = clean_name(name)
    if len(stops) < 2:
        raise ValueError("a route needs at least two stops")
    for stop in stops:
        if stop not in network.heights:
            raise UnknownPlace(stop)
    km = 0.0
    for a, b in zip(stops, stops[1:]):

class DuplicateName(Exception):
    pass

class RouteRepository:
    def __init__(self, conn):
        self.conn = conn
        conn.execute("CREATE TABLE IF NOT EXISTS saved_route (id INTEGER PRIMARY KEY, name TEXT NOT NULL UNIQUE COLLATE NOCASE)")
        conn.execute("CREATE TABLE IF NOT EXISTS saved_stop (route_id INTEGER, position INTEGER, junction TEXT, PRIMARY KEY (route_id, position))")
    def add(self, name, stops):
        try:
            with self.conn:
                cur = self.conn.execute("INSERT INTO saved_route (name) VALUES (?)", (name,))
                self.conn.executemany("INSERT INTO saved_stop VALUES (?, ?, ?)", [(cur.lastrowid, i, s) for i, s in enumerate(stops)])
        except sqlite3.IntegrityError:
            raise DuplicateName(name)
    def get(self, name):
        row = self.conn.execute("SELECT id, name FROM saved_route WHERE name = ?", (name,)).fetchone()
        if row is None:
            return None
        return row[1], [r[0] for r in self.conn.execute("SELECT junction FROM saved_stop WHERE route_id = ? ORDER BY position", (row[0],))]
    def names(self):
        return [r[0] for r in self.conn.execute("SELECT name FROM saved_route ORDER BY name COLLATE NOCASE")]
    def delete(self, name):
        row = self.conn.execute("SELECT id FROM saved_route WHERE name = ?", (name,)).fetchone()
        if row is None:
            return False
        with self.conn:
            self.conn.execute("DELETE FROM saved_stop WHERE route_id = ?", (row[0],))
            self.conn.execute("DELETE FROM saved_route WHERE id = ?", (row[0],))
        return True

def save_route(repo, network, name, stops):
    name = " ".join(name.split())
    if not name or len(stops) < 2:
        raise ValueError("a route needs a name and at least two stops")
    for stop in stops:
        if stop not in network.heights:
            raise UnknownPlace(stop)
    for a, b in zip(stops, stops[1:]):
        if b not in network.paths[a]:
            raise NoRoute(f"no path from {a} to {b}")
    repo.add(name, stops)
    return {"name": name, "stops": list(stops)}

NOT_FOUND = (404, {"error": "not found"})

def handle(method, path, body, repo, network):
    parts = path.strip("/").split("/")
    if parts[0] != "saved-routes" or len(parts) > 2:
        return NOT_FOUND
    if len(parts) == 1:
        if method == "GET":
            return 200, {"names": repo.names()}
        if method != "POST":
            return NOT_FOUND
        try:
            data = json.loads(body)
            name, stops = data["name"], data["stops"]
            if not isinstance(name, str) or not isinstance(stops, list) or not all(isinstance(s, str) for s in stops):
                raise ValueError("bad fields")
            return 201, save_route(repo, network, name, stops)
        except (ValueError, KeyError, TypeError) as e:
            return 400, {"error": "bad request"}
        except (UnknownPlace, NoRoute) as e:
            return 404, {"error": e.args[0]}
        except DuplicateName:
            return 409, {"error": "name taken"}
    name = unquote(parts[1])
    if method == "GET":
        found = repo.get(name)
        return (200, {"name": found[0], "stops": found[1]}) if found else NOT_FOUND
    if method == "DELETE":
        return (204, None) if repo.delete(name) else NOT_FOUND
    return NOT_FOUND

import sqlite3

MIGRATIONS = ["CREATE TABLE junction (name TEXT PRIMARY KEY)"]

def start(conn):
    conn.execute("CREATE TABLE IF NOT EXISTS schema_version (version INTEGER NOT NULL)")
    done = conn.execute("SELECT COALESCE(MAX(version), 0) FROM schema_version").fetchone()[0]
    for number, sql in enumerate(MIGRATIONS, start=1):
        if number > done:
            with conn:
                conn.execute(sql)
                conn.execute("INSERT INTO schema_version VALUES (?)", (number,))

import hashlib
import sqlite3

class MigrationError(Exception):
    pass

def _digest(sql):
    return hashlib.sha256(sql.encode()).hexdigest()

def migrate_safely(conn, migrations):
    conn.execute("CREATE TABLE IF NOT EXISTS schema_version (version INTEGER PRIMARY KEY, checksum TEXT NOT NULL)")
    applied = dict(conn.execute("SELECT version, checksum FROM schema_version"))
    if applied and max(applied) > len(migrations):
        raise MigrationError(f"database is at version {max(applied)} but only {len(migrations)} migrations exist")
    for number in sorted(applied):
        if applied[number] != _digest(migrations[number - 1]):
            raise MigrationError(f"migration {number} was changed after it was applied")
    done = []
    for number, sql in enumerate(migrations, start=1):
        if number in applied:
            continue
        conn.execute("BEGIN")
        try:
            for statement in sql.split(";"):
                if statement.strip():
                    conn.execute(statement)
            conn.execute("INSERT INTO schema_version VALUES (?, ?)", (number, _digest(sql)))
            conn.commit()
        except sqlite3.Error as e:
            conn.rollback()
            raise MigrationError(f"migration {number} failed: {e}")
        done.append(number)
    return done
