import heapq


class UnknownPlace(LookupError):
    pass


class NoRoute(Exception):
    pass


def naismith_minutes(km, climb_m):
    return round(km / 5 * 60 + climb_m / 600 * 60)


class Route:
    def __init__(self, stops, km, climb_m):
        self.stops, self.km, self.climb_m = list(stops), km, climb_m

    @property
    def minutes(self):
        return naismith_minutes(self.km, self.climb_m)

    def __repr__(self):
        return f"Route({self.stops}, {self.km} km, {self.climb_m} m)"


class PathNetwork:
    def __init__(self, heights):
        self.heights = dict(heights)
        self.paths = {name: {} for name in self.heights}

    def _check(self, *names):
        for name in names:
            if name not in self.heights:
                raise UnknownPlace(name)

    def add_path(self, a, b, km):
        self._check(a, b)
        if km <= 0:
            raise ValueError("km must be positive")
        self.paths[a][b] = km
        self.paths[b][a] = km

    def climb(self, stops):
        return sum(max(0, self.heights[b] - self.heights[a]) for a, b in zip(stops, stops[1:]))

    def shortest(self, start, end):
        self._check(start, end)
        dist, prev, done = {start: 0}, {}, set()
        heap = [(0, start)]
        while heap:
            d, node = heapq.heappop(heap)
            if node in done:
                continue
            done.add(node)
            if node == end:
                break
            for nxt, km in sorted(self.paths[node].items()):
                if d + km < dist.get(nxt, float("inf")):
                    dist[nxt] = d + km
                    prev[nxt] = node
                    heapq.heappush(heap, (d + km, nxt))
        if end not in done:
            raise NoRoute(f"no route from {start} to {end}")
        stops = [end]
        while stops[-1] != start:
            stops.append(prev[stops[-1]])
        stops.reverse()
        return Route(stops, round(dist[end], 2), self.climb(stops))

def pace_factor(walks, minimum=3):
    if len(walks) < minimum:
        return 1.0
    return round(statistics.median(w["minutes"] / naismith_minutes(w["km"], w["climb_m"]) for w in walks), 2)

    return round(naismith_minutes(km, climb_m) * pace_factor(walks))

import csv
import io
from collections import deque

def read_junctions(text):
    reader = csv.DictReader(io.StringIO(text))
    if reader.fieldnames != ["name", "height_m"]:
        return {}, ["line 1: expected columns name,height_m"]
    heights, problems = {}, []
    for line, row in enumerate(reader, start=2):
        name = (row["name"] or "").strip()
        try:
            height = int((row["height_m"] or "").strip())
        except ValueError:
            problems.append(f"line {line}: height must be a whole number")
            continue
        if not name:
            problems.append(f"line {line}: missing name")
        elif name in heights:
            problems.append(f"line {line}: duplicate junction {name}")
        else:
            heights[name] = height
    return heights, problems

def read_paths(text, heights):
    reader = csv.DictReader(io.StringIO(text))
    if reader.fieldnames != ["from", "to", "km"]:
        return [], ["line 1: expected columns from,to,km"]
    paths, problems = [], []
    for line, row in enumerate(reader, start=2):
        a, b = (row["from"] or "").strip(), (row["to"] or "").strip()
        unknown = next((n for n in (a, b) if n not in heights), None)
        if not a or not b:
            problems.append(f"line {line}: missing junction")
            continue
        try:
            km = float(row["km"])
        except (TypeError, ValueError):
            km = 0
        if unknown:
            problems.append(f"line {line}: unknown junction {unknown}")
        elif a == b:
            problems.append(f"line {line}: path from {a} to itself")
        elif km <= 0:
            problems.append(f"line {line}: km must be a positive number")
        else:
            paths.append((a, b, km))
    return paths, problems

def outside_largest(names, paths):
    neighbours = {n: set() for n in names}
    for a, b, _ in paths:
        neighbours[a].add(b)
        neighbours[b].add(a)
    seen, parts = set(), []
    for start in sorted(neighbours):
        if start not in seen:
            seen.add(start)
            queue, part = deque([start]), []
            while queue:
                node = queue.popleft()
                part.append(node)
                for nxt in neighbours[node] - seen:
                    seen.add(nxt)
                    queue.append(nxt)
            parts.append(sorted(part))
    parts.sort(key=lambda p: (-len(p), p[0]))
    return sorted(n for p in parts[1:] for n in p)

def check_map(junctions_text, paths_text, steep_limit=250):
    heights, junction_errors = read_junctions(junctions_text)
    paths, path_errors = read_paths(paths_text, heights)
    warnings = []
    steep = []
    for a, b, km in paths:
        rate = abs(heights[b] - heights[a]) / km
        if rate > steep_limit:
            steep.append((a, b, round(rate)))
    warnings += [f"steep path {a} to {b} ({rate} m per km)" for a, b, rate in sorted(steep, key=lambda s: -s[2])]
    counts = {}
    for a, b, _ in paths:
        key = tuple(sorted((a, b)))
        counts[key] = counts.get(key, 0) + 1
    warnings += [f"duplicate path between {a} and {b}" for (a, b), n in sorted(counts.items()) if n > 1]
    apart = outside_largest(heights, paths)
    if apart:
        warnings.append("not connected to the rest: " + ", ".join(apart))
    errors = junction_errors + path_errors
    return {"heights": heights, "paths": paths, "errors": errors, "warnings": warnings, "ok": not errors,
            "summary": f"{len(heights)} junctions, {len(paths)} paths, {len(errors)} errors, {len(warnings)} warnings"}

Joining each new junction to one earlier one always gives a **connected** network (a random tree), and the seed makes it the same every run. Generators like this let one test check thousands of networks, which is how the bugs nobody thought of get found.

## Where the gaps are

Start the tests day with evidence, not a feeling:

- **Trace stories to tests** (Week 87's `trace`): any must without a test is the first gap.
- **Look at the unhappy paths**: every `raise` in the code and every error status in the API should have a test that makes it happen.
- **Look at what's hard to test** (dates, randomness, the database, long outputs) and make it easy: inject it, build it, or compare it with a file.
- **Ask what a bug would look like**: the mutation question from Week 72. If changing `<=` to `<` in the pace code wouldn't fail a test, there's a gap.

## Builders

Tests that build their data by hand repeat the same twelve fields, and hide the one that matters. A **builder** has a sensible default for everything and a method for each thing a test might want to change, so a test reads like its scenario: `WalkBuilder().on("2026-01-04").km(15.5).build()`. Each method returns the builder (so calls chain) and `build()` returns a fresh dict each time. When a field is added to walks, only the builder changes.

## Oracles and properties

Dijkstra's algorithm is easy to get subtly wrong (Week 87's bug hunt finished nodes too early) and hard to test by hand on more than a few graphs. But there's a slow, obviously correct way to find the shortest route on a small network: try **every** simple path and take the shortest. That's an **oracle**. A property test generates many random small networks, asks both, and fails with the first network where they disagree.

Other useful properties for Waymark need no oracle at all: a route starts and ends where it should; its distance equals the sum of its paths; reversing it gives the same distance; the climb there plus the climb back equals the sum of all the height changes. Hypothesis (Week 72) automates this and shrinks failures; the idea works with a seeded `random.Random` too.

:::example
import itertools
import random
from waymark_domain import PathNetwork

def all_simple_routes(network, start, end, path=None):
    path = path or [start]
    if path[-1] == end:
        yield path
        return
    for nxt in network.paths[path[-1]]:
        if nxt not in path:
            yield from all_simple_routes(network, start, end, path + [nxt])

network = PathNetwork({"A": 0, "B": 0, "C": 0, "D": 0})

## Golden files and fuzzing

Some outputs are long: the year report, a CLI's help text, an exported map. Asserting them line by line is tedious; instead, save a known-good copy as a **golden file**, and have the test compare the current output with it, showing a **diff** when they differ. When the output changes on purpose, regenerate the file (with a flag) and review the diff in the pull request like any other change.

**Fuzzing** turns property testing on input parsers: generate valid input, damage it at random (cut a line short, empty a field, put a word where a number goes, repeat a line), and check that the parser never crashes and that its report stays consistent. Parsers are where untrusted data enters a program, so they're where fuzzing pays most.

Whatever the technique, tests must be **independent**: no shared state between tests (each builds its own data), no dependence on the real clock (inject it) or on unseeded randomness (seed it), and no dependence on the order they run in. A test that sometimes fails is worse than none, because people learn to ignore it.

:::key
Find gaps from evidence: stories without tests, unhappy paths never triggered, hard-to-test parts, and mutations no test would catch. Use builders with defaults so each test states only what matters. Check algorithms against brute-force oracles on many seeded random inputs, and test properties that need no oracle. Compare long outputs with golden files and show diffs; regenerate deliberately. Fuzz parsers with damaged input and check they never crash and their reports stay consistent. Keep tests independent of each other, the clock and chance.
:::

from datetime import date, timedelta

class WalkBuilder:
    def __init__(self):
        self._data = {"day": "2026-01-01", "km": 5.0, "climb_m": 0, "minutes": 60, "route": None}

    def _set(self, key, value):
        self._data[key] = value
        return self

    def on(self, day):
        return self._set("day", day)

    def km(self, value):
        return self._set("km", value)

    def climb(self, metres):
        return self._set("climb_m", metres)

    def taking(self, minutes):
        return self._set("minutes", minutes)

    def along(self, route):
        return self._set("route", route)

    def build(self):
        return dict(self._data)

def walks(n, start="2026-01-01", every_days=7, **changes):
    first = date.fromisoformat(start)
    result = []
    for i in range(n):
        walk = WalkBuilder().on((first + timedelta(days=i * every_days)).isoformat()).build()
        walk.update(changes)
        result.append(walk)
    return result

import random
from waymark_domain import PathNetwork

def random_network(rng, n, extra):
    names = [f"J{i}" for i in range(n)]
    network = PathNetwork({name: rng.randint(0, 300) for name in names})
    for i in range(1, n):
        network.add_path(names[i], names[rng.randrange(i)], round(rng.uniform(0.5, 3), 1))
    for _ in range(extra):
        a, b = rng.sample(names, 2)
        km = round(rng.uniform(0.5, 3), 1)
        if b not in network.paths[a]:
            network.add_path(a, b, km)
    return network

def brute_force_km(network, start, end):
    best = float("inf")
    def walk(path, km):
        nonlocal best
        if path[-1] == end:
            best = min(best, km)
            return
        for nxt, d in network.paths[path[-1]].items():
            if nxt not in path:
                walk(path + [nxt], km + d)
    walk([start], 0)
    return round(best, 2)

def disagreement(rng, trials, n=6, extra=4):
    for _ in range(trials):
        network = random_network(rng, n, extra)
        for start in network.heights:
            for end in network.heights:
                if start != end and round(network.shortest(start, end).km, 2) != brute_force_km(network, start, end):
                    return network, start, end
    return None

import difflib
import os

def check_golden(output, path, update=False):
    if update or not os.path.exists(path):
        with open(path, "w") as f:
            f.write(output)
        return "written"
    with open(path) as f:
        expected = f.read()
    if expected == output:
        return "same"
    return "".join(difflib.unified_diff(expected.splitlines(keepends=True), output.splitlines(keepends=True),
                                        fromfile="golden", tofile="actual"))

import random
import re

def _damage(lines, rng):
    out = [lines[0]]
    for line in lines[1:]:
        if rng.random() < 0.3:
            fields = line.split(",")
            kind = rng.choice(["first", "empty", "word", "twice"])
            if kind == "first":
                out.append(fields[0])
            elif kind == "empty":
                out.append(",".join(fields[:-1] + [""]))
            elif kind == "word":
                out.append(",".join(fields[:-1] + ["x"]))
            else:
                out += [line, line]
        else:
            out.append(line)
    return "\n".join(out) + "\n"

def fuzz(check, rng, trials=300):
    for _ in range(trials):
        names = [f"J{i}" for i in range(rng.randint(2, 6))]
        junction_lines = ["name,height_m"] + [f"{n},{rng.randint(0, 300)}" for n in names]
        path_lines = ["from,to,km"]
        for _ in range(rng.randint(1, 6)):
            a, b = rng.sample(names, 2)
            path_lines.append(f"{a},{b},{round(rng.uniform(0.1, 3), 1)}")
        jt, pt = _damage(junction_lines, rng), _damage(path_lines, rng)
        try:
            report = check(jt, pt)
        except Exception as e:
            return jt, pt, f"crashed: {type(e).__name__}"
        if report["ok"] != (not report["errors"]):
            return jt, pt, "ok is wrong"
        longest = max(jt.count("\n"), pt.count("\n"))
        for error in report["errors"]:
            m = re.match(r"line (\d+): ", error)
            if not m or not 2 <= int(m.group(1)) <= longest:
                return jt, pt, "bad line number"
    return None
