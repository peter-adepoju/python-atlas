import heapq


class UnknownPlace(LookupError):
    pass


class NoRoute(Exception):
    pass


def naismith_minutes(km, climb_m):
    return round(km / 5 * 60 + climb_m / 600 * 60)


class Route:
    def __init__(self, stops, km, climb_m):
        self.stops, self.km, self.climb_m = list(stops), km, climb_m

    @property
    def minutes(self):
        return naismith_minutes(self.km, self.climb_m)

    def __repr__(self):
        return f"Route({self.stops}, {self.km} km, {self.climb_m} m)"


class PathNetwork:
    def __init__(self, heights):
        self.heights = dict(heights)
        self.paths = {name: {} for name in self.heights}

    def _check(self, *names):
        for name in names:
            if name not in self.heights:
                raise UnknownPlace(name)

    def add_path(self, a, b, km):
        self._check(a, b)
        if km <= 0:
            raise ValueError("km must be positive")
        self.paths[a][b] = km
        self.paths[b][a] = km

    def climb(self, stops):
        return sum(max(0, self.heights[b] - self.heights[a]) for a, b in zip(stops, stops[1:]))

    def shortest(self, start, end):
        self._check(start, end)
        dist, prev, done = {start: 0}, {}, set()
        heap = [(0, start)]
        while heap:
            d, node = heapq.heappop(heap)
            if node in done:
                continue
            done.add(node)
            if node == end:
                break
            for nxt, km in sorted(self.paths[node].items()):
                if d + km < dist.get(nxt, float("inf")):
                    dist[nxt] = d + km
                    prev[nxt] = node
                    heapq.heappush(heap, (d + km, nxt))
        if end not in done:
            raise NoRoute(f"no route from {start} to {end}")
        stops = [end]
        while stops[-1] != start:
            stops.append(prev[stops[-1]])
        stops.reverse()
        return Route(stops, round(dist[end], 2), self.climb(stops))

import argparse
import contextlib
import io
from waymark_domain import NoRoute, UnknownPlace
from waymark_map import sample_network


def main(argv, out):
    parser = argparse.ArgumentParser(prog="waymark")
    sub = parser.add_subparsers(dest="command", required=True)
    p = sub.add_parser("plan")
    p.add_argument("start")
    p.add_argument("end")
    try:
        with contextlib.redirect_stderr(io.StringIO()):
            args = parser.parse_args(argv)
    except SystemExit:
        out.write("usage: waymark plan START END\n")
        return 2
    try:
        route = sample_network().shortest(args.start, args.end)
    except UnknownPlace as e:
        out.write(f"unknown place: {e.args[0]}\n")
        return 2
    except NoRoute as e:
        out.write(f"{e}\n")
        return 1
    hours, minutes = divmod(route.minutes, 60)
    out.write(f"{' -> '.join(route.stops)}: {round(route.km, 1)} km, {hours} h {minutes:02d} min\n")
    return 0

## The review

A sprint review has three parts, about an hour in all (shorter alone, but don't skip it):

1. **The demo.** Show each finished story **working**, from the user's side: the command, the request, the screen. Only finished stories (Week 87's definition of done) appear; "nearly done" is carried over, not demonstrated. The safest demo is one you've **run as a test**: a script of commands and their expected output, checked automatically, so the demo can't drift from the program. That's the bonus.
2. **The numbers.** Points finished (this sprint's **velocity**), what's carried over and why, the burndown's shape (a late cliff means stories were too big to finish one at a time), and the forecast for the rest of the backlog as a **range** from your slowest to your fastest sprint. With two sprints left in the capstone's build phase, the forecast tells you now whether the musts will fit, while there's still time to cut.
3. **The retrospective** (Week 87): what went well, what didn't, one to three actions for the next sprint, and whether last sprint's actions happened.

Then **release** what's done: a version from the commits (Week 87, Day 4), a changelog entry, and a tag. A release at the end of every sprint keeps the project always shippable.

import re

    recent = history[-window:]
    if not recent:
        return 0.0, 0, 0
    return round(sum(recent) / len(recent), 1), min(recent), max(recent)

import math
import re

def velocity(history, window=3):
    recent = history[-window:]
    if not recent:
        return 0.0, 0, 0
    return round(sum(recent) / len(recent), 1), min(recent), max(recent)

def forecast(points, history, window=3):
    average, low, high = velocity(history, window)
    if average == 0:
        raise ValueError("no velocity yet")
    return {"expected": math.ceil(points / average), "best": math.ceil(points / high),
            "worst": math.ceil(points / low) if low else None}

HEADER = re.compile(r"(?P<type>[a-z]+)(?:\([^)]+\))?(?P<bang>!)?: \S")

def next_version(current, messages):
    major, minor, patch = (int(p) for p in current.split("."))
    kinds = [HEADER.match(m) for m in messages]
    if any(k and (k.group("bang") or "BREAKING CHANGE:" in m) for k, m in zip(kinds, messages)):
        return f"{major + 1}.0.0"
    if any(k and k.group("type") == "feat" for k in kinds):
        return f"{major}.{minor + 1}.0"
    if any(k and k.group("type") in ("fix", "perf") for k in kinds):
        return f"{major}.{minor}.{patch + 1}"
    return current

def _list(ids, planned):
    points = sum(planned[s] for s in ids)
    return f"{', '.join(ids) if ids else 'none'} ({points} points)"

def sprint_report(name, planned, done_on, days, history, backlog_points, version, messages, date):
    done = [s for s in planned if s in done_on]
    carried = [s for s in planned if s not in done_on]
    points = sum(planned[s] for s in done)
    new_history = history + [points]
    remaining = [sum(planned.values()) - sum(planned[s] for s, d in done_on.items() if d <= day) for day in range(days + 1)]
    left = backlog_points + sum(planned[s] for s in carried)
    outlook = forecast(left, new_history)
    new_version = next_version(version, messages)
    release = new_version if new_version != version else None
    average = velocity(new_history)[0]
    text = "\n".join([
        f"# {name} review",
        f"Done: {_list(done, planned)}",
        f"Carried over: {_list(carried, planned)}",
        "Burndown: " + " ".join(str(r) for r in remaining),
        f"Velocity: {points} (average of last 3: {average})",
        f"Forecast for {left} points: {outlook['expected']} sprints (best {outlook['best']}, worst {outlook['worst'] or 'unknown'})",
        f"Release: {release}" if release else "Release: nothing to release",
    ]) + "\n"
    return {"velocity": points, "history": new_history, "carried": carried, "remaining": remaining,
            "forecast": outlook, "release": release, "text": text}

## Bonus: a demo that checks itself

A demo that's typed live goes wrong in front of people; one that's described ("and then it would show...") proves nothing. The fix is a **demo script**: the commands you'll show and exactly what each prints, kept in the repository and **run as a test** in CI, like a golden file for the whole user experience. If the program changes, the test fails, and the demo is updated deliberately. It doubles as documentation: the README's usage section can be the same file. (Python's `doctest` does the same for examples in docstrings.)

import io

def parse_demo(text):
    steps, command, output, code = [], None, [], 0
    for line in text.splitlines():
        if line.startswith("$ "):
            if command is not None:
                steps.append((command, "".join(o + "\n" for o in output), code))
            command, output, code = line[2:], [], 0
        elif line.startswith("[exit ") and line.endswith("]"):
            code = int(line[6:-1])
        elif command is not None:
            output.append(line)
    if command is not None:
        steps.append((command, "".join(o + "\n" for o in output), code))
    return steps

def run_demo(steps, main):
    report = []
    for n, (command, expected, code) in enumerate(steps, start=1):
        out = io.StringIO()
        got = main(command.split()[1:], out)
        if got != code:
            report.append(f"FAIL {n}: {command}: expected exit {code}, got {got}")
        elif out.getvalue() != expected:
            report.append(f"FAIL {n}: {command}: output differs")
        else:
            report.append(f"PASS {n}: {command}")
    return report

def demo_ok(report):
    return all(line.startswith("PASS") for line in report)
