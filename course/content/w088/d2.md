import heapq


class UnknownPlace(LookupError):
    pass


class NoRoute(Exception):
    pass


def naismith_minutes(km, climb_m):
    return round(km / 5 * 60 + climb_m / 600 * 60)


class Route:
    def __init__(self, stops, km, climb_m):
        self.stops, self.km, self.climb_m = list(stops), km, climb_m

    @property
    def minutes(self):
        return naismith_minutes(self.km, self.climb_m)

    def __repr__(self):
        return f"Route({self.stops}, {self.km} km, {self.climb_m} m)"


class PathNetwork:
    def __init__(self, heights):
        self.heights = dict(heights)
        self.paths = {name: {} for name in self.heights}

    def _check(self, *names):
        for name in names:
            if name not in self.heights:
                raise UnknownPlace(name)

    def add_path(self, a, b, km):
        self._check(a, b)
        if km <= 0:
            raise ValueError("km must be positive")
        self.paths[a][b] = km
        self.paths[b][a] = km

    def climb(self, stops):
        return sum(max(0, self.heights[b] - self.heights[a]) for a, b in zip(stops, stops[1:]))

    def shortest(self, start, end):
        self._check(start, end)
        dist, prev, done = {start: 0}, {}, set()
        heap = [(0, start)]
        while heap:
            d, node = heapq.heappop(heap)
            if node in done:
                continue
            done.add(node)
            if node == end:
                break
            for nxt, km in sorted(self.paths[node].items()):
                if d + km < dist.get(nxt, float("inf")):
                    dist[nxt] = d + km
                    prev[nxt] = node
                    heapq.heappush(heap, (d + km, nxt))
        if end not in done:
            raise NoRoute(f"no route from {start} to {end}")
        stops = [end]
        while stops[-1] != start:
            stops.append(prev[stops[-1]])
        stops.reverse()
        return Route(stops, round(dist[end], 2), self.climb(stops))

Dates that are always written `YYYY-MM-DD` with zeros happen to sort correctly as text, which is why databases store them that way; but anything typed by a person must be **parsed** into a `date` first, and rejected if it can't be.

## Validating a walk

A walk record has a day, a distance, a climb, the minutes it took, and optionally the saved route it followed. Validation should:

- **collect every problem**, not stop at the first, so the walker can fix them all at once: return a list of messages, empty if all is well;
- **parse, then check**: turn the day into a `date` (`date.fromisoformat`), and the numbers into numbers, before checking ranges;
- **check against an injected `today`**, never `date.today()` inside the function, so tests can say what today is and the "no walks in the future" rule can be tested on any day;
- **normalise** what it accepts (a `date` back to a `YYYY-MM-DD` string, distances rounded), so everything stored looks the same.

## Storing and summarising

Walks go in a `walk` table, with `CHECK` constraints repeating the most important rules (Day 3's second line of defence). Summaries are SQL's job: `strftime('%Y-%m', day)` gives each walk's month, and `GROUP BY` totals them. SQL only returns months that **have** walks, so a year's summary fills the missing months with zeros in Python: a chart with a gap where February should be is misleading.

:::example
import sqlite3
from walks import WALKS

conn = sqlite3.connect(":memory:")
conn.execute("CREATE TABLE walk (day TEXT NOT NULL, km REAL NOT NULL CHECK (km > 0), climb_m INTEGER NOT NULL, minutes INTEGER NOT NULL)")
conn.executemany("INSERT INTO walk VALUES (:day, :km, :climb_m, :minutes)", WALKS)
rows = conn.execute('''SELECT strftime('%m', day) AS month, COUNT(*), SUM(km) FROM walk
                       WHERE strftime('%Y', day) = ? GROUP BY month ORDER BY month''', ("2026",)).fetchall()
print(rows)
by_month = {int(m): (n, km) for m, n, km in rows}
print([by_month.get(m, (0, 0.0))[1] for m in range(1, 13)])

## A personal pace

Naismith's rule is for an average walker. Waymark knows better: every logged walk has an estimate (from its distance and climb) and an actual time. The ratio `actual / estimate` says how this walker compares: 1.1 means 10% slower than the rule. Use the **median** of the ratios, not the mean, so one walk with a long lunch stop doesn't change every future estimate. With too few walks (fewer than three, say), there isn't enough evidence: fall back to the rule.

:::example
import statistics
from waymark_domain import naismith_minutes
from walks import WALKS

ratios = [w["minutes"] / naismith_minutes(w["km"], w["climb_m"]) for w in WALKS]
print([round(r, 2) for r in ratios])
print("mean", round(statistics.mean(ratios), 3), "median", round(statistics.median(ratios), 3))
print("Ridge walk for this walker:", round(naismith_minutes(6, 270) * statistics.median(ratios)), "minutes, not", naismith_minutes(6, 270))
:::

## Streaks and weeks

"Walked every week for six weeks" is motivating, and easy to get wrong. A week is identified by its **ISO year and week number**, `date.isocalendar()`, because the last days of December can belong to week 1 of the next year and the first days of January to week 52 or 53 of the last. The safest way to test "consecutive weeks" avoids week numbers altogether: two walks are in consecutive weeks if the **Mondays** of their weeks are exactly 7 days apart (`day - timedelta(days=day.weekday())` is that Monday).

:::key
Validate records by collecting every problem; parse text into real types (`date.fromisoformat`) before checking; inject "today" rather than calling `date.today()` inside logic; normalise what you store. Summarise in SQL with `strftime` and `GROUP BY`, and fill empty periods in Python. Personalise estimates with the median of a person's own `actual / estimate` ratios, falling back to the general rule when there's too little history. Identify weeks by their Monday (or ISO year and week), never by week number alone, so streaks survive the new year.
:::

from datetime import date

def _number(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool)

def _whole(x):
    return isinstance(x, int) and not isinstance(x, bool)

def check_walk(data, today):
    problems = []
    try:
        day = date.fromisoformat(data.get("day", ""))
    except (TypeError, ValueError):
        day = None
        problems.append("day must be a date like 2026-11-14")
    if day is not None and day > today:
        problems.append("day is in the future")
    km, climb, minutes, route = data.get("km"), data.get("climb_m"), data.get("minutes"), data.get("route")
    if not (_number(km) and km > 0):
        problems.append("km must be a positive number")
    if not (_whole(climb) and climb >= 0):
        problems.append("climb_m must be a whole number of metres, 0 or more")
    if not (_whole(minutes) and 1 <= minutes <= 1440):
        problems.append("minutes must be between 1 and 1440")
    if route is not None and not isinstance(route, str):
        problems.append("route must be text")
    if problems:
        return None, problems
    return {"day": day.isoformat(), "km": round(km, 2), "climb_m": climb, "minutes": minutes, "route": route}, []

import sqlite3

class WalkLog:
    def __init__(self, conn):
        self.conn = conn
        conn.execute('''CREATE TABLE IF NOT EXISTS walk (id INTEGER PRIMARY KEY, day TEXT NOT NULL,
                        km REAL NOT NULL CHECK (km > 0), climb_m INTEGER NOT NULL CHECK (climb_m >= 0),
                        minutes INTEGER NOT NULL CHECK (minutes > 0), route TEXT)''')

    def add(self, walk):
        with self.conn:
            cur = self.conn.execute("INSERT INTO walk (day, km, climb_m, minutes, route) VALUES (?, ?, ?, ?, ?)",
                                    (walk["day"], walk["km"], walk["climb_m"], walk["minutes"], walk.get("route")))
        return cur.lastrowid

    def months(self, year):
        rows = self.conn.execute('''SELECT CAST(strftime('%m', day) AS INTEGER), COUNT(*), SUM(km), SUM(climb_m), SUM(minutes)
                                    FROM walk WHERE strftime('%Y', day) = ? GROUP BY 1''', (str(year),)).fetchall()
        found = {m: (n, round(km, 1), climb, mins) for m, n, km, climb, mins in rows}
        return [(m, *found.get(m, (0, 0.0, 0, 0))) for m in range(1, 13)]

    def total_km(self, year):
        return round(sum(m[2] for m in self.months(year)), 1)

from datetime import date, timedelta

def week_start(day):
    return day - timedelta(days=day.weekday())

def _runs(days):
    weeks = sorted({week_start(day) for day in days})
    runs, length = [], 0
    for i, week in enumerate(weeks):
        length = length + 1 if i and week - weeks[i - 1] == timedelta(days=7) else 1
        runs.append((week, length))
    return runs

def longest_streak(days):
    return max((length for _, length in _runs(days)), default=0)

def current_streak(days, today):
    runs = _runs(days)
    if not runs:
        return 0
    last_week, length = runs[-1]
    this_week = week_start(today)
    return length if this_week - last_week <= timedelta(days=7) else 0

import calendar
import statistics
from datetime import date, timedelta
from waymark_domain import naismith_minutes

def pace_factor(walks, minimum=3):
    if len(walks) < minimum:
        return 1.0
    return round(statistics.median(w["minutes"] / naismith_minutes(w["km"], w["climb_m"]) for w in walks), 2)

def week_start(day):
    return day - timedelta(days=day.weekday())

def streaks(days, today):
    weeks = sorted({week_start(d) for d in days})
    best = run = 0
    for i, w in enumerate(weeks):
        run = run + 1 if i and w - weeks[i - 1] == timedelta(days=7) else 1
        best = max(best, run)
    current = run if weeks and week_start(today) - weeks[-1] <= timedelta(days=7) else 0
    return best, current

def year_report(walks, year, today):
    mine = [w for w in walks if w["day"].startswith(f"{year}-")]
    lines = [f"Waymark: your {year} in walks"]
    if not mine:
        return "\n".join(lines + ["No walks logged."]) + "\n"
    km = round(sum(w["km"] for w in mine), 1)
    climb = sum(w["climb_m"] for w in mine)
    hours, minutes = divmod(sum(w["minutes"] for w in mine), 60)
    longest = max(mine, key=lambda w: (w["km"], [-ord(c) for c in w["day"]]))
    by_month = {}
    for w in mine:
        month = int(w["day"][5:7])
        by_month[month] = by_month.get(month, 0) + w["km"]
    busiest = min(by_month, key=lambda m: (-by_month[m], m))
    factor = pace_factor(mine)
    percent = round(abs(factor - 1) * 100)
    pace = "exactly as Naismith's rule" if percent == 0 else f"{percent}% {'slower' if factor > 1 else 'faster'} than Naismith's rule"
    best, current = streaks([date.fromisoformat(w["day"]) for w in mine], today)
    lines += [f"Walks: {len(mine)}", f"Distance: {km} km, climb {climb} m", f"Time on foot: {hours} h {minutes} min",
              f"Longest walk: {longest['km']} km on {longest['day']} ({longest['route'] or 'no route'})",
              f"Busiest month: {calendar.month_name[busiest]} ({round(by_month[busiest], 1)} km)",
              f"Pace: {pace}", f"Weekly streaks: longest {best}, current {current}"]
    return "\n".join(lines) + "\n"
