:::exercise id=mix_trace points=10 title="Tracing recursion"
Write a decorator `traced(func)` that prints each call and its result, indented by two spaces per level of recursion, using a closure variable for the current depth (with `nonlocal`, restoring it even if the call raises). Each call prints `"{indent}{name}({args})"` before and `"{indent}-> {result}"` after, with arguments shown by `repr` and joined by `", "`.
@@starter

    ___
@@tests
def test_traced():
    "nonlocal depth"
    @traced
    def fact(n):
        return 1 if n <= 1 else n * fact(n - 1)
    import io, contextlib
    buf = io.StringIO()
    with contextlib.redirect_stdout(buf):
        fact(3)
    lines = buf.getvalue().strip().splitlines()
    assert lines == ["fact(3)", "  fact(2)", "    fact(1)", "    -> 1", "  -> 2", "-> 6"], lines
    assert fact.__name__ == "fact"
@@solution
import functools

def traced(func):
    depth = 0
    @functools.wraps(func)
    def wrapper(*args):
        nonlocal depth
        indent = "  " * depth
        print(f"{indent}{func.__name__}({', '.join(map(repr, args))})")
        depth += 1
        try:
            result = func(*args)
        finally:
            depth -= 1
        print(f"{indent}-> {result!r}")
        return result
    return wrapper
:::

:::exercise id=mix_lazy points=10 title="A generator with a memory"
Write `make_numberer(start=1)`: returns a **generator function** `numbered(items)` that yields `(n, item)` pairs, where the numbering **continues** from where the previous call left off (a closure with `nonlocal` holding the next number, shared by every call of `numbered`).

@@tests
import types
def test_numberer():
    "closures and generators"
    numbered = make_numberer()
    assert list(numbered("ab")) == [(1, "a"), (2, "b")] and list(numbered(["c"])) == [(3, "c")]
    other = make_numberer(10)
    assert list(other("x")) == [(10, "x")] and list(numbered("d")) == [(4, "d")]
    assert isinstance(numbered("z"), types.GeneratorType)
@@solution
def make_numberer(start=1):
    next_number = start
    def numbered(items):
        nonlocal next_number
        for item in items:
            yield next_number, item
            next_number += 1
    return numbered

## Forge: a plugin host

Write a class `PluginHost()` and two decorator factories:

- `@host.plugin` or `@host.plugin(name=..., priority=...)` registers a function (under its own name unless `name` is given; default priority 0) and returns it **unchanged**. Registering a name twice raises `ValueError`.
- `@host.hook("before")` and `@host.hook("after")` register hook functions; any other kind raises `ValueError` when `hook` is called. Before-hooks are called as `hook(name, args)` before each plugin run, after-hooks as `hook(name, result)` after it, in the order they were registered.
- `host.plugins()`: sorted names. `host.run(name, *args, **kwargs)`: run one plugin with its hooks and return its result (`KeyError` for an unknown name, `RuntimeError` if it's disabled). `host.run_all(*args, **kwargs)`: run every enabled plugin, highest priority first and then by name, returning a dict from name to result in that order. `host.disable(name)` and `host.enable(name)` (`KeyError` if unknown). `host.stats()`: a dict from every plugin's name to how many times it has run **successfully**.
- `timed(clock)`: a decorator factory recording the duration of the last call (by the given clock function) as `wrapper.last` (`None` before any call), even when the call raises.
- `expects(*types)`: a decorator factory checking positional arguments against the types in order, raising `TypeError(f"{name}: argument {i} must be {type_name}")`.

Both factories keep the function's name with `functools.wraps`.

`plugin` is a method that works with or without brackets: `def plugin(self, func=None, *, name=None, priority=0)`, returning `lambda f: self.plugin(f, name=name, priority=priority)` when `func` is `None`. `hook` is a factory method returning a registering decorator. Keep plugins in a dict from name to `(priority, function)` and a separate dict of run counts.
@@solution
import functools

class PluginHost:
    def __init__(self):
        self._plugins = {}         # name -> (priority, wrapped function)
        self._disabled = set()
        self._hooks = {"before": [], "after": []}
        self._runs = {}

    def plugin(self, func=None, *, name=None, priority=0):
        if func is None:
            return lambda f: self.plugin(f, name=name, priority=priority)
        key = name or func.__name__
        if key in self._plugins:
            raise ValueError(f"plugin {key!r} is already registered")
        self._plugins[key] = (priority, func)
        self._runs[key] = 0
        return func

    def hook(self, when):
        if when not in self._hooks:
            raise ValueError(f"no such hook: {when!r}")
        def decorator(func):
            self._hooks[when].append(func)
            return func
        return decorator

    def plugins(self):
        return sorted(self._plugins)

    def disable(self, name):
        self._known(name)
        self._disabled.add(name)

    def enable(self, name):
        self._known(name)
        self._disabled.discard(name)

    def _known(self, name):
        if name not in self._plugins:
            raise KeyError(name)

    def run(self, name, *args, **kwargs):
        self._known(name)
        if name in self._disabled:
            raise RuntimeError(f"plugin {name!r} is disabled")
        for h in self._hooks["before"]:
            h(name, args)
        result = self._plugins[name][1](*args, **kwargs)
        self._runs[name] += 1
        for h in self._hooks["after"]:
            h(name, result)
        return result

    def run_all(self, *args, **kwargs):
        order = sorted((n for n in self._plugins if n not in self._disabled),
                       key=lambda n: (-self._plugins[n][0], n))
        return {n: self.run(n, *args, **kwargs) for n in order}

    def stats(self):
        return dict(self._runs)

def timed(clock):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            start = clock()
            try:
                return func(*args, **kwargs)
            finally:
                wrapper.last = clock() - start
        wrapper.last = None
        return wrapper
    return decorator

def expects(*types):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for i, (value, kind) in enumerate(zip(args, types)):
                if not isinstance(value, kind):
                    raise TypeError(f"{func.__name__}: argument {i} must be {kind.__name__}")
            return func(*args, **kwargs)
        return wrapper
    return decorator
:::

:::think title="Decorators as an interface"
The plugin authors never call `host.register(...)`; they write `@host.plugin` above an ordinary function. That's why frameworks lean on decorators: registration sits right next to the code it registers, plugins stay plain functions you can call and test directly (the decorator returns them unchanged), and settings read naturally (`@host.plugin(priority=5)`). Flask's `@app.route("/")`, pytest's `@pytest.fixture` and click's `@click.command()` all follow this shape, and all of them are closures and factories like the ones you wrote this week.
:::

## Forge bonus: cleaning recipes

:::exercise id=forge2 points=30 title="Forge bonus: cleaning recipes"
A data team writes text-cleaning recipes as strings like `"strip|lower|squash|replace:tea:coffee|cut:12"`. Build the machinery:

- `steps = {}` and a registering decorator factory `step(name)` that stores a function under that name and returns it unchanged. Register `strip`, `lower`, `squash` (collapse runs of whitespace to single spaces and trim the ends), `replace(text, old, new)` and `cut(text, n)` (the first n characters; `n` may arrive as a string).
- `compose(*funcs)`: a function applying them left to right, with `functools.reduce` (no functions: the identity).
- `build(spec)`: split the recipe on `"|"`; each part is a step name followed by `:`-separated arguments, which are passed to the step after the text, as strings, by name (use `functools.partial` with the step's parameter names after the first, read from `func.__code__`). Unknown steps raise `KeyError`. Return the composed function.
- `describe(spec)`: a readable version: `"strip then replace(a, b) then cut(3)"`.

New steps registered later with `@step(...)` must work in `build` straight away.

def test_steps():
    "registered steps"
    assert sorted(steps) == ["cut", "lower", "replace", "squash", "strip"]
    assert steps["squash"]("  a   b ") == "a b" and steps["replace"]("aXa", "X", "-") == "a-a" and steps["cut"]("abcdef", "3") == "abc"

    "partial and reduce"
    clean = build("strip|lower|squash|replace:tea:coffee|cut:12")
    assert clean("   I like   TEA and   tea  ") == "i like coffe"
    assert build("lower")("ABC") == "abc" and build("replace:a:b|replace:b:c")("aab") == "ccc"
    try:
        build("strip|explode")
        assert False
    except KeyError:
        pass
    assert describe("strip|replace:a:b|cut:3") == "strip then replace(a, b) then cut(3)"

def test_extend():
    "registering a new step from outside"
    @step("title")
    def title(text):
        return text.title()
    assert build("squash|title")(" the  long walk ") == "The Long Walk"
    del steps["title"]
@@solution
from functools import partial, reduce

steps = {}

def step(name):
    def decorator(func):
        steps[name] = func
        return func
    return decorator

@step("strip")
def strip(text):
    return text.strip()

@step("lower")
def lower(text):
    return text.lower()

@step("squash")
def squash(text):
    return " ".join(text.split())

@step("replace")
def replace_text(text, old, new):
    return text.replace(old, new)

@step("cut")
def cut(text, n):
    return text[:int(n)]

def compose(*funcs):
    return lambda text: reduce(lambda acc, f: f(acc), funcs, text)

def build(spec):
    funcs = []
    for part in spec.split("|"):
        name, *args = part.split(":")
        if name not in steps:
            raise KeyError(name)
        funcs.append(partial(steps[name], **dict(zip(_params(steps[name]), args))) if args else steps[name])
    return compose(*funcs)

def _params(func):
    code = func.__code__
    return code.co_varnames[1:code.co_argcount]

def describe(spec):
    out = []
    for part in spec.split("|"):
        name, *args = part.split(":")
        out.append(f"{name}({', '.join(args)})" if args else name)
    return " then ".join(out)
