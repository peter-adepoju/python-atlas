## Step 21: testing the tokeniser

Each broken version below is a mistake that's easy to make and easy to miss, and each one changes how some Tiny programs behave. A good test suite is one that would catch them.

class TinyError(Exception):
    """Base class for every error the Tiny interpreter reports."""

    def __init__(self, message, line=None, column=None):
        super().__init__(message)
        self.message, self.line, self.column = message, line, column

    def __str__(self):
        if self.line is None:
            return self.message
        if self.column is None:
            return f"line {self.line}: {self.message}"
        return f"line {self.line}, column {self.column}: {self.message}"

    line, column, text = cursor.line, cursor.column, ""

def read_string(cursor):
    line, column = cursor.line, cursor.column
    text = cursor.advance()
    value = ""
    while True:
        if cursor.at_end() or cursor.peek() == "\n":
            raise LexError("unterminated string", line, column)
        ch = cursor.advance()
        text += ch
        if ch == '"':
            return Token("STRING", text, value, line, column)
        if ch == "\\":
            if cursor.at_end():
                raise LexError("unterminated string", line, column)
            esc_line, esc_col = cursor.line, cursor.column - 1
            code = cursor.advance()
            text += code
            if code not in ESCAPES:
                raise LexError(f"unknown escape \\{code}", esc_line, esc_col)
            value += ESCAPES[code]
        else:
            value += ch

def read_operator(cursor):
    line, column = cursor.line, cursor.column
    pair = cursor.peek() + cursor.peek(1)
    if pair in TWO_CHAR:
        cursor.advance()
        cursor.advance()
        return Token("OP", pair, pair, line, column)
    ch = cursor.peek()
    if ch in ONE_CHAR:
        cursor.advance()
        return Token("OP", ch, ch, line, column)
    raise LexError(f"unexpected character {ch!r}", line, column)


def skip_space_and_comments(cursor):
    while not cursor.at_end():
        ch = cursor.peek()
        if ch in " \t\r\n":
            cursor.advance()
        elif ch == "#":
            while not cursor.at_end() and cursor.peek() != "\n":
                cursor.advance()
        else:
            return

    cursor = Cursor(source)

    while True:
        skip_space_and_comments(cursor)
        if cursor.at_end():
            tokens.append(Token("EOF", "", None, cursor.line, cursor.column))

        if ch.isdigit():

        elif ch == '"':
            tokens.append(read_string(cursor))
        elif ch.isalpha() or ch == "_":
            tokens.append(read_name(cursor))
        else:
            tokens.append(read_operator(cursor))

import pytest
from tiny_lex import tokenize, LexError

def kinds(source):
    return [t.kind for t in tokenize(source)]

def test_two_character_operators():
    assert [t.text for t in tokenize("a <= b == c")][:5] == ["a", "<=", "b", "==", "c"]

def test_a_dot_needs_a_digit():
    assert tokenize("3.5")[0].value == 3.5
    with pytest.raises(LexError):
        tokenize("3.")

def test_keyword_values():
    values = [t.value for t in tokenize("true false nil")[:3]]
    assert values == [True, False, None]

def test_positions():
    toks = tokenize("let x = 1;\n  print x;")
    assert (toks[1].line, toks[1].column) == (1, 5)
    assert (toks[5].line, toks[5].column) == (2, 3)

def test_comments_end_at_the_line():
    assert kinds("# a comment\nprint 1; # more\nprint 2;") == ["PRINT", "NUMBER", "OP", "PRINT", "NUMBER", "OP", "EOF"]

def test_escapes():
    assert tokenize(r'"a\nb\t\"c\""')[0].value == 'a\nb\t"c"'

class Parser:
    def __init__(self, tokens):
        self.tokens, self.pos = tokens, 0

    def peek(self):
        return self.tokens[self.pos]

    def at(self, kind, text=None):
        tok = self.peek()
        return tok.kind == kind and (text is None or tok.text == text)

    def match(self, kind, text=None):
        if self.at(kind, text):
            tok = self.peek()
            self.pos += 1
            return tok
        return None

    def expect(self, kind, text=None, what=None):
        tok = self.match(kind, text)
        if tok is None:
            got = self.peek()
            shown = "end of input" if got.kind == "EOF" else repr(got.text)
            raise ParseError(f"expected {what or text or kind.lower()}, got {shown}", got.line, got.column)
        return tok

    # --- expressions, lowest precedence first ---
    def expression(self):
        return self.logic_or()

    def logic_or(self):
        node = self.logic_and()
        while self.at("OR"):
            line = self.match("OR").line
            node = Logical("or", node, self.logic_and(), line)
        return node

    def logic_and(self):
        node = self.equality()
        while self.at("AND"):
            line = self.match("AND").line
            node = Logical("and", node, self.equality(), line)
        return node

    def binary(self, operators, operand):
        node = operand()
        while self.peek().kind == "OP" and self.peek().text in operators:
            tok = self.match("OP")
            node = Binary(tok.text, node, operand(), tok.line)
        return node

    def equality(self):
        return self.binary({"==", "!="}, self.comparison)

    def comparison(self):
        return self.binary({"<", "<=", ">", ">="}, self.term)

    def term(self):
        return self.binary({"+", "-"}, self.factor)

    def factor(self):
        return self.binary({"*", "/", "%"}, self.unary)

    def unary(self):
        if self.at("OP", "-"):
            line = self.match("OP").line
            return Unary("-", self.unary(), line)
        if self.at("NOT"):
            line = self.match("NOT").line
            return Unary("not", self.unary(), line)
        return self.call()

    def call(self):
        node = self.primary()
        while self.at("OP", "("):
            line = self.match("OP").line
            args = []
            if not self.at("OP", ")"):
                args.append(self.expression())
                while self.match("OP", ","):
                    args.append(self.expression())
            self.expect("OP", ")")
            node = Call(node, args, line)
        return node

    def primary(self):
        tok = self.peek()
        if self.match("NUMBER"):
            return Number(tok.value, tok.line)
        if self.match("STRING"):
            return String(tok.value, tok.line)
        if tok.kind in ("TRUE", "FALSE", "NIL"):
            self.pos += 1
            return Literal(tok.value, tok.line)
        if self.match("NAME"):
            return Name(tok.text, tok.line)
        if self.match("OP", "("):
            node = self.expression()
            self.expect("OP", ")")
            return node
        shown = "end of input" if tok.kind == "EOF" else repr(tok.text)
        raise ParseError(f"expected an expression, got {shown}", tok.line, tok.column)

    # --- statements ---
    def statement(self):
        tok = self.peek()
        if self.match("LET"):
            name = self.expect("NAME", what="a name").text
            self.expect("OP", "=")
            value = self.expression()
            self.expect("OP", ";")
            return Let(name, value, tok.line)
        if self.match("PRINT"):
            value = self.expression()
            self.expect("OP", ";")
            return Print(value, tok.line)
        if self.at("OP", "{"):
            return Block(self.block(), tok.line)
        if self.match("IF"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            then = self.statement()
            otherwise = self.statement() if self.match("ELSE") else None
            return If(test, then, otherwise, tok.line)
        if self.match("WHILE"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            return While(test, self.statement(), tok.line)
        if self.match("FN"):
            name = self.expect("NAME", what="a function name").text
            self.expect("OP", "(")
            params = []
            if not self.at("OP", ")"):
                params.append(self.expect("NAME", what="a parameter name").text)
                while self.match("OP", ","):
                    params.append(self.expect("NAME", what="a parameter name").text)
            self.expect("OP", ")")
            return Function(name, params, self.block(), tok.line)
        if self.match("RETURN"):
            value = None if self.at("OP", ";") else self.expression()
            self.expect("OP", ";")
            return Return(value, tok.line)
        if tok.kind == "NAME" and self.tokens[self.pos + 1].kind == "OP" and self.tokens[self.pos + 1].text == "=":
            self.pos += 2
            value = self.expression()
            self.expect("OP", ";")
            return Assign(tok.text, value, tok.line)
        value = self.expression()
        self.expect("OP", ";")
        return ExprStmt(value, tok.line)

    def block(self):
        self.expect("OP", "{")
        body = []
        while not self.at("OP", "}"):
            if self.at("EOF"):
                tok = self.peek()
                raise ParseError("expected '}', got end of input", tok.line, tok.column)
            body.append(self.statement())
        self.expect("OP", "}")
        return body

    def program(self):
        body = []
        while not self.at("EOF"):
            body.append(self.statement())
        return body


def parse(source):

class Environment:
    def __init__(self, parent=None):
        self.values, self.parent = {}, parent

    def define(self, name, value):
        self.values[name] = value

    def find(self, name):
        env = self
        while env is not None:
            if name in env.values:
                return env
            env = env.parent
        return None

    def get(self, name, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"undefined name {name!r}", line)
        return env.values[name]

    def assign(self, name, value, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"cannot assign to undefined name {name!r}", line)
        env.values[name] = value

def truthy(value):

def show(value):
    if value is None:
        return "nil"
    if value is True:
        return "true"
    if value is False:
        return "false"
    if isinstance(value, float) and value.is_integer():
        return str(int(value)) if abs(value) < 1e16 else str(value)
    return str(value)


def type_name(value):
    if value is None:
        return "nil"
    if isinstance(value, bool):
        return "bool"
    if isinstance(value, (int, float)):
        return "number"
    if isinstance(value, str):
        return "string"
    return "function"


def is_number(value):
    return isinstance(value, (int, float)) and not isinstance(value, bool)


class Interpreter:
    def __init__(self, output=None, max_depth=100):
        self.output = [] if output is None else output
        self.globals = Environment()
        self.max_depth, self.depth = max_depth, 0
        self.steps = 0
        for name, arity, func in [
            ("len", 1, self._len), ("str", 1, lambda v: show(v)), ("type", 1, type_name),
        ]:
            self.globals.define(name, Builtin(name, arity, func))

    @staticmethod
    def _len(value):
        if not isinstance(value, str):
            raise TinyRuntimeError(f"len() needs a string, not {type_name(value)}")
        return len(value)

    # --- expressions ---
    def evaluate(self, node, env):
        self.steps += 1
        if isinstance(node, (Number, String, Literal)):
            return node.value
        if isinstance(node, Name):
            return env.get(node.name, node.line)
        if isinstance(node, Unary):
            value = self.evaluate(node.operand, env)
            if node.op == "-":
                if not is_number(value):
                    raise TinyRuntimeError(f"cannot negate {type_name(value)}", node.line)
                return -value
            return not truthy(value)
        if isinstance(node, Logical):
            left = self.evaluate(node.left, env)
            if node.op == "or":
                return left if truthy(left) else self.evaluate(node.right, env)
            return left if not truthy(left) else self.evaluate(node.right, env)
        if isinstance(node, Binary):
            return self.binary(node, self.evaluate(node.left, env), self.evaluate(node.right, env))
        if isinstance(node, Call):
            return self.call(node, env)
        raise TinyRuntimeError(f"cannot evaluate {type(node).__name__}", getattr(node, "line", None))

    def binary(self, node, left, right):
        op = node.op
        if op == "==":
            return type_name(left) == type_name(right) and left == right
        if op == "!=":
            return not (type_name(left) == type_name(right) and left == right)
        if op == "+" and isinstance(left, str) and isinstance(right, str):
            return left + right
        if not (is_number(left) and is_number(right)):
            raise TinyRuntimeError(
                f"cannot use {op} with {type_name(left)} and {type_name(right)}", node.line)
        if op in ("/", "%") and right == 0:
            raise TinyRuntimeError("division by zero", node.line)
        if op == "/":
            result = left / right
            return int(result) if isinstance(left, int) and isinstance(right, int) and result.is_integer() else result
        return {"+": lambda: left + right, "-": lambda: left - right, "*": lambda: left * right,
                "%": lambda: left % right, "<": lambda: left < right, "<=": lambda: left <= right,
                ">": lambda: left > right, ">=": lambda: left >= right}[op]()

    def call(self, node, env):
        callee = self.evaluate(node.callee, env)
        args = [self.evaluate(a, env) for a in node.args]
        if isinstance(callee, Builtin):
            if len(args) != callee.arity:
                raise TinyRuntimeError(f"{callee.name}() takes {callee.arity} argument(s), got {len(args)}", node.line)
            try:
                return callee.func(*args)
            except TinyRuntimeError as e:
                raise TinyRuntimeError(e.message, node.line) from None
        if not isinstance(callee, TinyFunction):
            raise TinyRuntimeError(f"cannot call {type_name(callee)}", node.line)
        decl = callee.decl
        if len(args) != len(decl.params):
            raise TinyRuntimeError(f"{decl.name}() takes {len(decl.params)} argument(s), got {len(args)}", node.line)
        if self.depth >= self.max_depth:
            raise TinyRuntimeError("too much recursion", node.line)
        local = Environment(callee.closure)
        for name, value in zip(decl.params, args):
            local.define(name, value)
        self.depth += 1
        try:
            self.run_block(decl.body, local)
        except ReturnSignal as signal:
            return signal.value
        finally:
            self.depth -= 1
        return None

    # --- statements ---
    def execute(self, node, env):
        self.steps += 1
        if isinstance(node, Let):
            env.define(node.name, self.evaluate(node.value, env))
        elif isinstance(node, Assign):
            env.assign(node.name, self.evaluate(node.value, env), node.line)
        elif isinstance(node, Print):
            self.output.append(show(self.evaluate(node.value, env)))
        elif isinstance(node, ExprStmt):
            self.evaluate(node.value, env)
        elif isinstance(node, Block):
            self.run_block(node.body, Environment(env))
        elif isinstance(node, If):
            if truthy(self.evaluate(node.test, env)):
                self.execute(node.then, env)
            elif node.otherwise is not None:
                self.execute(node.otherwise, env)
        elif isinstance(node, While):
            while truthy(self.evaluate(node.test, env)):
                self.execute(node.body, env)
        elif isinstance(node, Function):
            env.define(node.name, TinyFunction(node, env))
        elif isinstance(node, Return):
            raise ReturnSignal(None if node.value is None else self.evaluate(node.value, env))
        else:
            raise TinyRuntimeError(f"cannot execute {type(node).__name__}", getattr(node, "line", None))

    def run_block(self, statements, env):
        for stmt in statements:
            self.execute(stmt, env)

    def run(self, program):
        try:
            self.run_block(program, self.globals)
        except ReturnSignal:
            raise TinyRuntimeError("return outside a function") from None
        return self.output

class TinyError(Exception):
    """Base class for every error the Tiny interpreter reports."""

    def __init__(self, message, line=None, column=None):
        super().__init__(message)
        self.message, self.line, self.column = message, line, column

    def __str__(self):
        if self.line is None:
            return self.message
        if self.column is None:
            return f"line {self.line}: {self.message}"
        return f"line {self.line}, column {self.column}: {self.message}"

class Cursor:
    def __init__(self, source):
        self.source, self.pos, self.line, self.column = source, 0, 1, 1

    def at_end(self):
        return self.pos >= len(self.source)

    def peek(self, ahead=0):
        i = self.pos + ahead
        return self.source[i] if i < len(self.source) else ""

    def advance(self):
        ch = self.source[self.pos]
        self.pos += 1
        if ch == "\n":
            self.line, self.column = self.line + 1, 1
        else:
            self.column += 1
        return ch


def read_number(cursor):
    line, column, text = cursor.line, cursor.column, ""
    while cursor.peek().isdigit():
        text += cursor.advance()
    if cursor.peek() == "." and cursor.peek(1).isdigit():
        text += cursor.advance()
        while cursor.peek().isdigit():
            text += cursor.advance()
        return Token("NUMBER", text, float(text), line, column)
    return Token("NUMBER", text, int(text), line, column)


ESCAPES = {"n": "\n", "t": "\t", '"': '"', "\\": "\\"}


def read_string(cursor):
    line, column = cursor.line, cursor.column
    text = cursor.advance()
    value = ""
    while True:
        if cursor.at_end() or cursor.peek() == "\n":
            raise LexError("unterminated string", line, column)
        ch = cursor.advance()
        text += ch
        if ch == '"':
            return Token("STRING", text, value, line, column)
        if ch == "\\":
            if cursor.at_end():
                raise LexError("unterminated string", line, column)
            esc_line, esc_col = cursor.line, cursor.column - 1
            code = cursor.advance()
            text += code
            if code not in ESCAPES:
                raise LexError(f"unknown escape \\{code}", esc_line, esc_col)
            value += ESCAPES[code]
        else:
            value += ch


KEYWORDS = {"let", "fn", "return", "if", "else", "while", "print", "true", "false", "nil", "and", "or", "not"}


TWO_CHAR = {"==", "!=", "<=", ">="}


ONE_CHAR = set("+-*/%<>=(){},;")


def read_name(cursor):
    line, column, text = cursor.line, cursor.column, ""
    while cursor.peek().isalnum() or cursor.peek() == "_":
        text += cursor.advance()
    if text in KEYWORDS:
        value = {"true": True, "false": False, "nil": None}.get(text)
        return Token(text.upper(), text, value, line, column)
    return Token("NAME", text, text, line, column)


def read_operator(cursor):
    line, column = cursor.line, cursor.column
    pair = cursor.peek() + cursor.peek(1)
    if pair in TWO_CHAR:
        cursor.advance()
        cursor.advance()
        return Token("OP", pair, pair, line, column)
    ch = cursor.peek()
    if ch in ONE_CHAR:
        cursor.advance()
        return Token("OP", ch, ch, line, column)
    raise LexError(f"unexpected character {ch!r}", line, column)


def skip_space_and_comments(cursor):
    while not cursor.at_end():
        ch = cursor.peek()
        if ch in " \t\r\n":
            cursor.advance()
        elif ch == "#":
            while not cursor.at_end() and cursor.peek() != "\n":
                cursor.advance()
        else:
            return


def tokenize(source):
    cursor = Cursor(source)
    tokens = []
    while True:
        skip_space_and_comments(cursor)
        if cursor.at_end():
            tokens.append(Token("EOF", "", None, cursor.line, cursor.column))
            return tokens
        ch = cursor.peek()
        if ch.isdigit():
            tokens.append(read_number(cursor))
        elif ch == '"':
            tokens.append(read_string(cursor))
        elif ch.isalpha() or ch == "_":
            tokens.append(read_name(cursor))
        else:
            tokens.append(read_operator(cursor))

class Parser:
    def __init__(self, tokens):
        self.tokens, self.pos = tokens, 0

    def peek(self):
        return self.tokens[self.pos]

    def at(self, kind, text=None):
        tok = self.peek()
        return tok.kind == kind and (text is None or tok.text == text)

    def match(self, kind, text=None):
        if self.at(kind, text):
            tok = self.peek()
            self.pos += 1
            return tok
        return None

    def expect(self, kind, text=None, what=None):
        tok = self.match(kind, text)
        if tok is None:
            got = self.peek()
            shown = "end of input" if got.kind == "EOF" else repr(got.text)
            raise ParseError(f"expected {what or text or kind.lower()}, got {shown}", got.line, got.column)
        return tok

    # --- expressions, lowest precedence first ---
    def expression(self):
        return self.logic_or()

    def logic_or(self):
        node = self.logic_and()
        while self.at("OR"):
            line = self.match("OR").line
            node = Logical("or", node, self.logic_and(), line)
        return node

    def logic_and(self):
        node = self.equality()
        while self.at("AND"):
            line = self.match("AND").line
            node = Logical("and", node, self.equality(), line)
        return node

    def binary(self, operators, operand):
        node = operand()
        while self.peek().kind == "OP" and self.peek().text in operators:
            tok = self.match("OP")
            node = Binary(tok.text, node, operand(), tok.line)
        return node

    def equality(self):
        return self.binary({"==", "!="}, self.comparison)

    def comparison(self):
        return self.binary({"<", "<=", ">", ">="}, self.term)

    def term(self):
        return self.binary({"+", "-"}, self.factor)

    def factor(self):
        return self.binary({"*", "/", "%"}, self.unary)

    def unary(self):
        if self.at("OP", "-"):
            line = self.match("OP").line
            return Unary("-", self.unary(), line)
        if self.at("NOT"):
            line = self.match("NOT").line
            return Unary("not", self.unary(), line)
        return self.call()

    def call(self):
        node = self.primary()
        while self.at("OP", "("):
            line = self.match("OP").line
            args = []
            if not self.at("OP", ")"):
                args.append(self.expression())
                while self.match("OP", ","):
                    args.append(self.expression())
            self.expect("OP", ")")
            node = Call(node, args, line)
        return node

    def primary(self):
        tok = self.peek()
        if self.match("NUMBER"):
            return Number(tok.value, tok.line)
        if self.match("STRING"):
            return String(tok.value, tok.line)
        if tok.kind in ("TRUE", "FALSE", "NIL"):
            self.pos += 1
            return Literal(tok.value, tok.line)
        if self.match("NAME"):
            return Name(tok.text, tok.line)
        if self.match("OP", "("):
            node = self.expression()
            self.expect("OP", ")")
            return node
        shown = "end of input" if tok.kind == "EOF" else repr(tok.text)
        raise ParseError(f"expected an expression, got {shown}", tok.line, tok.column)

    # --- statements ---
    def statement(self):
        tok = self.peek()
        if self.match("LET"):
            name = self.expect("NAME", what="a name").text
            self.expect("OP", "=")
            value = self.expression()
            self.expect("OP", ";")
            return Let(name, value, tok.line)
        if self.match("PRINT"):
            value = self.expression()
            self.expect("OP", ";")
            return Print(value, tok.line)
        if self.at("OP", "{"):
            return Block(self.block(), tok.line)
        if self.match("IF"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            then = self.statement()
            otherwise = self.statement() if self.match("ELSE") else None
            return If(test, then, otherwise, tok.line)
        if self.match("WHILE"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            return While(test, self.statement(), tok.line)
        if self.match("FN"):
            name = self.expect("NAME", what="a function name").text
            self.expect("OP", "(")
            params = []
            if not self.at("OP", ")"):
                params.append(self.expect("NAME", what="a parameter name").text)
                while self.match("OP", ","):
                    params.append(self.expect("NAME", what="a parameter name").text)
            self.expect("OP", ")")
            return Function(name, params, self.block(), tok.line)
        if self.match("RETURN"):
            value = None if self.at("OP", ";") else self.expression()
            self.expect("OP", ";")
            return Return(value, tok.line)
        if tok.kind == "NAME" and self.tokens[self.pos + 1].kind == "OP" and self.tokens[self.pos + 1].text == "=":
            self.pos += 2
            value = self.expression()
            self.expect("OP", ";")
            return Assign(tok.text, value, tok.line)
        value = self.expression()
        self.expect("OP", ";")
        return ExprStmt(value, tok.line)

    def block(self):
        self.expect("OP", "{")
        body = []
        while not self.at("OP", "}"):
            if self.at("EOF"):
                tok = self.peek()
                raise ParseError("expected '}', got end of input", tok.line, tok.column)
            body.append(self.statement())
        self.expect("OP", "}")
        return body

    def program(self):
        body = []
        while not self.at("EOF"):
            body.append(self.statement())
        return body


def parse(source):
    return Parser(tokenize(source)).program()

class Environment:
    def __init__(self, parent=None):
        self.values, self.parent = {}, parent

    def define(self, name, value):
        self.values[name] = value

    def find(self, name):
        env = self
        while env is not None:
            if name in env.values:
                return env
            env = env.parent
        return None

    def get(self, name, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"undefined name {name!r}", line)
        return env.values[name]

    def assign(self, name, value, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"cannot assign to undefined name {name!r}", line)
        env.values[name] = value

def truthy(value):
    return value is not None and value is not False


def show(value):
    if value is None:
        return "nil"
    if value is True:
        return "true"
    if value is False:
        return "false"
    if isinstance(value, float) and value.is_integer():
        return str(int(value)) if abs(value) < 1e16 else str(value)
    return str(value)


def type_name(value):
    if value is None:
        return "nil"
    if isinstance(value, bool):
        return "bool"
    if isinstance(value, (int, float)):
        return "number"
    if isinstance(value, str):
        return "string"
    return "function"


def is_number(value):
    return isinstance(value, (int, float)) and not isinstance(value, bool)


class Interpreter:
    def __init__(self, output=None, max_depth=100):
        self.output = [] if output is None else output
        self.globals = Environment()
        self.max_depth, self.depth = max_depth, 0
        self.steps = 0
        for name, arity, func in [
            ("len", 1, self._len), ("str", 1, lambda v: show(v)), ("type", 1, type_name),
        ]:
            self.globals.define(name, Builtin(name, arity, func))

    @staticmethod
    def _len(value):
        if not isinstance(value, str):
            raise TinyRuntimeError(f"len() needs a string, not {type_name(value)}")
        return len(value)

    # --- expressions ---
    def evaluate(self, node, env):
        self.steps += 1
        if isinstance(node, (Number, String, Literal)):
            return node.value
        if isinstance(node, Name):
            return env.get(node.name, node.line)
        if isinstance(node, Unary):
            value = self.evaluate(node.operand, env)
            if node.op == "-":
                if not is_number(value):
                    raise TinyRuntimeError(f"cannot negate {type_name(value)}", node.line)
                return -value
            return not truthy(value)
        if isinstance(node, Logical):
            left = self.evaluate(node.left, env)
            if node.op == "or":
                return left if truthy(left) else self.evaluate(node.right, env)
            return left if not truthy(left) else self.evaluate(node.right, env)
        if isinstance(node, Binary):
            return self.binary(node, self.evaluate(node.left, env), self.evaluate(node.right, env))
        if isinstance(node, Call):
            return self.call(node, env)
        raise TinyRuntimeError(f"cannot evaluate {type(node).__name__}", getattr(node, "line", None))

    def binary(self, node, left, right):
        op = node.op
        if op == "==":
            return type_name(left) == type_name(right) and left == right
        if op == "!=":
            return not (type_name(left) == type_name(right) and left == right)
        if op == "+" and isinstance(left, str) and isinstance(right, str):
            return left + right
        if not (is_number(left) and is_number(right)):
            raise TinyRuntimeError(
                f"cannot use {op} with {type_name(left)} and {type_name(right)}", node.line)
        if op in ("/", "%") and right == 0:
            raise TinyRuntimeError("division by zero", node.line)
        if op == "/":
            result = left / right
            return int(result) if isinstance(left, int) and isinstance(right, int) and result.is_integer() else result
        return {"+": lambda: left + right, "-": lambda: left - right, "*": lambda: left * right,
                "%": lambda: left % right, "<": lambda: left < right, "<=": lambda: left <= right,
                ">": lambda: left > right, ">=": lambda: left >= right}[op]()

    def call(self, node, env):
        callee = self.evaluate(node.callee, env)
        args = [self.evaluate(a, env) for a in node.args]
        if isinstance(callee, Builtin):
            if len(args) != callee.arity:
                raise TinyRuntimeError(f"{callee.name}() takes {callee.arity} argument(s), got {len(args)}", node.line)
            try:
                return callee.func(*args)
            except TinyRuntimeError as e:
                raise TinyRuntimeError(e.message, node.line) from None
        if not isinstance(callee, TinyFunction):
            raise TinyRuntimeError(f"cannot call {type_name(callee)}", node.line)
        decl = callee.decl
        if len(args) != len(decl.params):
            raise TinyRuntimeError(f"{decl.name}() takes {len(decl.params)} argument(s), got {len(args)}", node.line)
        if self.depth >= self.max_depth:
            raise TinyRuntimeError("too much recursion", node.line)
        local = Environment(callee.closure)
        for name, value in zip(decl.params, args):
            local.define(name, value)
        self.depth += 1
        try:
            self.run_block(decl.body, local)
        except ReturnSignal as signal:
            return signal.value
        finally:
            self.depth -= 1
        return None

    # --- statements ---
    def execute(self, node, env):
        self.steps += 1
        if isinstance(node, Let):
            env.define(node.name, self.evaluate(node.value, env))
        elif isinstance(node, Assign):
            env.assign(node.name, self.evaluate(node.value, env), node.line)
        elif isinstance(node, Print):
            self.output.append(show(self.evaluate(node.value, env)))
        elif isinstance(node, ExprStmt):
            self.evaluate(node.value, env)
        elif isinstance(node, Block):
            self.run_block(node.body, Environment(env))
        elif isinstance(node, If):
            if truthy(self.evaluate(node.test, env)):
                self.execute(node.then, env)
            elif node.otherwise is not None:
                self.execute(node.otherwise, env)
        elif isinstance(node, While):
            while truthy(self.evaluate(node.test, env)):
                self.execute(node.body, env)
        elif isinstance(node, Function):
            env.define(node.name, TinyFunction(node, env))
        elif isinstance(node, Return):
            raise ReturnSignal(None if node.value is None else self.evaluate(node.value, env))
        else:
            raise TinyRuntimeError(f"cannot execute {type(node).__name__}", getattr(node, "line", None))

    def run_block(self, statements, env):
        for stmt in statements:
            self.execute(stmt, env)

    def run(self, program):
        try:
            self.run_block(program, self.globals)
        except ReturnSignal:
            raise TinyRuntimeError("return outside a function") from None
        return self.output


def run_source(source):
    interpreter = Interpreter()

def read_string(cursor):
    line, column = cursor.line, cursor.column
    text = cursor.advance()
    value = ""
    while True:
        if cursor.at_end() or cursor.peek() == "\n":
            raise LexError("unterminated string", line, column)
        ch = cursor.advance()
        text += ch
        if ch == '"':
            return Token("STRING", text, value, line, column)
        if ch == "\\":
            if cursor.at_end():
                raise LexError("unterminated string", line, column)

            code = cursor.advance()
            text += code

                raise LexError(f"unknown escape \\{code}", esc_line, esc_col)

            value += ch

def read_name(cursor):
    line, column, text = cursor.line, cursor.column, ""
    while cursor.peek().isalnum() or cursor.peek() == "_":
        text += cursor.advance()
    if text in KEYWORDS:
        value = {"true": True, "false": False, "nil": None}.get(text)
        return Token(text.upper(), text, value, line, column)
    return Token("NAME", text, text, line, column)


def read_operator(cursor):
    line, column = cursor.line, cursor.column
    pair = cursor.peek() + cursor.peek(1)
    if pair in TWO_CHAR:
        cursor.advance()
        cursor.advance()
        return Token("OP", pair, pair, line, column)
    ch = cursor.peek()
    if ch in ONE_CHAR:
        cursor.advance()
        return Token("OP", ch, ch, line, column)
    raise LexError(f"unexpected character {ch!r}", line, column)


def skip_space_and_comments(cursor):
    while not cursor.at_end():
        ch = cursor.peek()
        if ch in " \t\r\n":
            cursor.advance()
        elif ch == "#":
            while not cursor.at_end() and cursor.peek() != "\n":
                cursor.advance()
        else:
            return


def tokenize(source):
    cursor = Cursor(source)
    tokens = []
    while True:
        skip_space_and_comments(cursor)
        if cursor.at_end():
            tokens.append(Token("EOF", "", None, cursor.line, cursor.column))
            return tokens
        ch = cursor.peek()
        if ch.isdigit():
            tokens.append(read_number(cursor))
        elif ch == '"':
            tokens.append(read_string(cursor))
        elif ch.isalpha() or ch == "_":
            tokens.append(read_name(cursor))
        else:
            tokens.append(read_operator(cursor))

class Parser:
    def __init__(self, tokens):
        self.tokens, self.pos = tokens, 0

    def peek(self):
        return self.tokens[self.pos]

    def at(self, kind, text=None):
        tok = self.peek()
        return tok.kind == kind and (text is None or tok.text == text)

    def match(self, kind, text=None):
        if self.at(kind, text):
            tok = self.peek()
            self.pos += 1
            return tok
        return None

    def expect(self, kind, text=None, what=None):
        tok = self.match(kind, text)
        if tok is None:
            got = self.peek()
            shown = "end of input" if got.kind == "EOF" else repr(got.text)
            raise ParseError(f"expected {what or text or kind.lower()}, got {shown}", got.line, got.column)
        return tok

    # --- expressions, lowest precedence first ---
    def expression(self):
        return self.logic_or()

    def logic_or(self):
        node = self.logic_and()
        while self.at("OR"):
            line = self.match("OR").line
            node = Logical("or", node, self.logic_and(), line)
        return node

    def logic_and(self):
        node = self.equality()
        while self.at("AND"):
            line = self.match("AND").line
            node = Logical("and", node, self.equality(), line)
        return node

    def binary(self, operators, operand):
        node = operand()
        while self.peek().kind == "OP" and self.peek().text in operators:
            tok = self.match("OP")
            node = Binary(tok.text, node, operand(), tok.line)
        return node

    def equality(self):
        return self.binary({"==", "!="}, self.comparison)

    def comparison(self):
        return self.binary({"<", "<=", ">", ">="}, self.term)

    def term(self):
        return self.binary({"+", "-"}, self.factor)

    def factor(self):
        return self.binary({"*", "/", "%"}, self.unary)

    def unary(self):
        if self.at("OP", "-"):
            line = self.match("OP").line
            return Unary("-", self.unary(), line)
        if self.at("NOT"):
            line = self.match("NOT").line
            return Unary("not", self.unary(), line)
        return self.call()

    def call(self):
        node = self.primary()
        while self.at("OP", "("):
            line = self.match("OP").line
            args = []
            if not self.at("OP", ")"):
                args.append(self.expression())
                while self.match("OP", ","):
                    args.append(self.expression())
            self.expect("OP", ")")
            node = Call(node, args, line)
        return node

    def primary(self):
        tok = self.peek()
        if self.match("NUMBER"):
            return Number(tok.value, tok.line)
        if self.match("STRING"):
            return String(tok.value, tok.line)
        if tok.kind in ("TRUE", "FALSE", "NIL"):
            self.pos += 1
            return Literal(tok.value, tok.line)
        if self.match("NAME"):
            return Name(tok.text, tok.line)
        if self.match("OP", "("):
            node = self.expression()
            self.expect("OP", ")")
            return node
        shown = "end of input" if tok.kind == "EOF" else repr(tok.text)
        raise ParseError(f"expected an expression, got {shown}", tok.line, tok.column)

    # --- statements ---
    def statement(self):
        tok = self.peek()
        if self.match("LET"):
            name = self.expect("NAME", what="a name").text
            self.expect("OP", "=")
            value = self.expression()
            self.expect("OP", ";")
            return Let(name, value, tok.line)
        if self.match("PRINT"):
            value = self.expression()
            self.expect("OP", ";")
            return Print(value, tok.line)
        if self.at("OP", "{"):
            return Block(self.block(), tok.line)
        if self.match("IF"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            then = self.statement()
            otherwise = self.statement() if self.match("ELSE") else None
            return If(test, then, otherwise, tok.line)
        if self.match("WHILE"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            return While(test, self.statement(), tok.line)
        if self.match("FN"):
            name = self.expect("NAME", what="a function name").text
            self.expect("OP", "(")
            params = []
            if not self.at("OP", ")"):
                params.append(self.expect("NAME", what="a parameter name").text)
                while self.match("OP", ","):
                    params.append(self.expect("NAME", what="a parameter name").text)
            self.expect("OP", ")")
            return Function(name, params, self.block(), tok.line)
        if self.match("RETURN"):
            value = None if self.at("OP", ";") else self.expression()
            self.expect("OP", ";")
            return Return(value, tok.line)
        if tok.kind == "NAME" and self.tokens[self.pos + 1].kind == "OP" and self.tokens[self.pos + 1].text == "=":
            self.pos += 2
            value = self.expression()
            self.expect("OP", ";")
            return Assign(tok.text, value, tok.line)
        value = self.expression()
        self.expect("OP", ";")
        return ExprStmt(value, tok.line)

    def block(self):
        self.expect("OP", "{")
        body = []
        while not self.at("OP", "}"):
            if self.at("EOF"):
                tok = self.peek()
                raise ParseError("expected '}', got end of input", tok.line, tok.column)
            body.append(self.statement())
        self.expect("OP", "}")
        return body

    def program(self):
        body = []
        while not self.at("EOF"):
            body.append(self.statement())
        return body


def parse(source):
    return Parser(tokenize(source)).program()

class Environment:
    def __init__(self, parent=None):
        self.values, self.parent = {}, parent

    def define(self, name, value):
        self.values[name] = value

    def find(self, name):
        env = self
        while env is not None:
            if name in env.values:
                return env
            env = env.parent
        return None

    def get(self, name, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"undefined name {name!r}", line)
        return env.values[name]

    def assign(self, name, value, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"cannot assign to undefined name {name!r}", line)
        env.values[name] = value

class Interpreter:
    def __init__(self, output=None, max_depth=100):
        self.output = [] if output is None else output
        self.globals = Environment()
        self.max_depth, self.depth = max_depth, 0
        self.steps = 0
        for name, arity, func in [
            ("len", 1, self._len), ("str", 1, lambda v: show(v)), ("type", 1, type_name),
        ]:
            self.globals.define(name, Builtin(name, arity, func))

    @staticmethod
    def _len(value):
        if not isinstance(value, str):
            raise TinyRuntimeError(f"len() needs a string, not {type_name(value)}")
        return len(value)

    # --- expressions ---
    def evaluate(self, node, env):
        self.steps += 1
        if isinstance(node, (Number, String, Literal)):
            return node.value
        if isinstance(node, Name):
            return env.get(node.name, node.line)
        if isinstance(node, Unary):
            value = self.evaluate(node.operand, env)
            if node.op == "-":
                if not is_number(value):
                    raise TinyRuntimeError(f"cannot negate {type_name(value)}", node.line)
                return -value
            return not truthy(value)
        if isinstance(node, Logical):
            left = self.evaluate(node.left, env)
            if node.op == "or":
                return left if truthy(left) else self.evaluate(node.right, env)
            return left if not truthy(left) else self.evaluate(node.right, env)
        if isinstance(node, Binary):
            return self.binary(node, self.evaluate(node.left, env), self.evaluate(node.right, env))
        if isinstance(node, Call):
            return self.call(node, env)
        raise TinyRuntimeError(f"cannot evaluate {type(node).__name__}", getattr(node, "line", None))

    def binary(self, node, left, right):
        op = node.op
        if op == "==":
            return type_name(left) == type_name(right) and left == right
        if op == "!=":
            return not (type_name(left) == type_name(right) and left == right)
        if op == "+" and isinstance(left, str) and isinstance(right, str):
            return left + right
        if not (is_number(left) and is_number(right)):
            raise TinyRuntimeError(
                f"cannot use {op} with {type_name(left)} and {type_name(right)}", node.line)
        if op in ("/", "%") and right == 0:
            raise TinyRuntimeError("division by zero", node.line)
        if op == "/":
            result = left / right
            return int(result) if isinstance(left, int) and isinstance(right, int) and result.is_integer() else result
        return {"+": lambda: left + right, "-": lambda: left - right, "*": lambda: left * right,
                "%": lambda: left % right, "<": lambda: left < right, "<=": lambda: left <= right,
                ">": lambda: left > right, ">=": lambda: left >= right}[op]()

    def call(self, node, env):
        callee = self.evaluate(node.callee, env)
        args = [self.evaluate(a, env) for a in node.args]
        if isinstance(callee, Builtin):
            if len(args) != callee.arity:
                raise TinyRuntimeError(f"{callee.name}() takes {callee.arity} argument(s), got {len(args)}", node.line)
            try:
                return callee.func(*args)
            except TinyRuntimeError as e:
                raise TinyRuntimeError(e.message, node.line) from None
        if not isinstance(callee, TinyFunction):
            raise TinyRuntimeError(f"cannot call {type_name(callee)}", node.line)
        decl = callee.decl
        if len(args) != len(decl.params):
            raise TinyRuntimeError(f"{decl.name}() takes {len(decl.params)} argument(s), got {len(args)}", node.line)
        if self.depth >= self.max_depth:
            raise TinyRuntimeError("too much recursion", node.line)
        local = Environment(callee.closure)
        for name, value in zip(decl.params, args):
            local.define(name, value)
        self.depth += 1
        try:
            self.run_block(decl.body, local)
        except ReturnSignal as signal:
            return signal.value
        finally:
            self.depth -= 1
        return None

    # --- statements ---
    def execute(self, node, env):
        self.steps += 1
        if isinstance(node, Let):
            env.define(node.name, self.evaluate(node.value, env))
        elif isinstance(node, Assign):
            env.assign(node.name, self.evaluate(node.value, env), node.line)
        elif isinstance(node, Print):
            self.output.append(show(self.evaluate(node.value, env)))
        elif isinstance(node, ExprStmt):
            self.evaluate(node.value, env)
        elif isinstance(node, Block):
            self.run_block(node.body, Environment(env))
        elif isinstance(node, If):
            if truthy(self.evaluate(node.test, env)):
                self.execute(node.then, env)
            elif node.otherwise is not None:
                self.execute(node.otherwise, env)
        elif isinstance(node, While):
            while truthy(self.evaluate(node.test, env)):
                self.execute(node.body, env)
        elif isinstance(node, Function):
            env.define(node.name, TinyFunction(node, env))
        elif isinstance(node, Return):
            raise ReturnSignal(None if node.value is None else self.evaluate(node.value, env))
        else:
            raise TinyRuntimeError(f"cannot execute {type(node).__name__}", getattr(node, "line", None))

    def run_block(self, statements, env):
        for stmt in statements:
            self.execute(stmt, env)

    def run(self, program):
        try:
            self.run_block(program, self.globals)
        except ReturnSignal:
            raise TinyRuntimeError("return outside a function") from None
        return self.output

            value += ESCAPES[code]
        else:
            value += ch


KEYWORDS = {"let", "fn", "return", "if", "else", "while", "print", "true", "false", "nil", "and", "or", "not"}

def read_name(cursor):
    line, column, text = cursor.line, cursor.column, ""
    while cursor.peek().isalnum() or cursor.peek() == "_":
        text += cursor.advance()
    if text in KEYWORDS:
        value = {"true": True, "false": False, "nil": None}.get(text)
        return Token(text.upper(), text, value, line, column)
    return Token("NAME", text, text, line, column)


def read_operator(cursor):
    line, column = cursor.line, cursor.column
    pair = cursor.peek() + cursor.peek(1)
    if pair in TWO_CHAR:
        cursor.advance()
        cursor.advance()
        return Token("OP", pair, pair, line, column)
    ch = cursor.peek()
    if ch in ONE_CHAR:
        cursor.advance()
        return Token("OP", ch, ch, line, column)
    raise LexError(f"unexpected character {ch!r}", line, column)


def skip_space_and_comments(cursor):
    while not cursor.at_end():
        ch = cursor.peek()
        if ch in " \t\r\n":
            cursor.advance()
        elif ch == "#":
            while not cursor.at_end() and cursor.peek() != "\n":
                cursor.advance()
        else:
            return


def tokenize(source):
    cursor = Cursor(source)
    tokens = []
    while True:
        skip_space_and_comments(cursor)
        if cursor.at_end():
            tokens.append(Token("EOF", "", None, cursor.line, cursor.column))
            return tokens
        ch = cursor.peek()
        if ch.isdigit():
            tokens.append(read_number(cursor))
        elif ch == '"':
            tokens.append(read_string(cursor))
        elif ch.isalpha() or ch == "_":
            tokens.append(read_name(cursor))
        else:
            tokens.append(read_operator(cursor))

class Parser:
    def __init__(self, tokens):
        self.tokens, self.pos = tokens, 0

    def peek(self):
        return self.tokens[self.pos]

    def at(self, kind, text=None):
        tok = self.peek()
        return tok.kind == kind and (text is None or tok.text == text)

    def match(self, kind, text=None):
        if self.at(kind, text):
            tok = self.peek()
            self.pos += 1
            return tok
        return None

    def expect(self, kind, text=None, what=None):
        tok = self.match(kind, text)
        if tok is None:
            got = self.peek()
            shown = "end of input" if got.kind == "EOF" else repr(got.text)
            raise ParseError(f"expected {what or text or kind.lower()}, got {shown}", got.line, got.column)
        return tok

    # --- expressions, lowest precedence first ---
    def expression(self):
        return self.logic_or()

    def logic_or(self):
        node = self.logic_and()
        while self.at("OR"):
            line = self.match("OR").line
            node = Logical("or", node, self.logic_and(), line)
        return node

    def logic_and(self):
        node = self.equality()
        while self.at("AND"):
            line = self.match("AND").line
            node = Logical("and", node, self.equality(), line)
        return node

    def binary(self, operators, operand):
        node = operand()
        while self.peek().kind == "OP" and self.peek().text in operators:
            tok = self.match("OP")
            node = Binary(tok.text, node, operand(), tok.line)
        return node

    def equality(self):
        return self.binary({"==", "!="}, self.comparison)

    def comparison(self):
        return self.binary({"<", "<=", ">", ">="}, self.term)

    def term(self):
        return self.binary({"+", "-"}, self.factor)

    def factor(self):
        return self.binary({"*", "/", "%"}, self.unary)

    def unary(self):
        if self.at("OP", "-"):
            line = self.match("OP").line
            return Unary("-", self.unary(), line)
        if self.at("NOT"):
            line = self.match("NOT").line
            return Unary("not", self.unary(), line)
        return self.call()

    def call(self):
        node = self.primary()
        while self.at("OP", "("):
            line = self.match("OP").line
            args = []
            if not self.at("OP", ")"):
                args.append(self.expression())
                while self.match("OP", ","):
                    args.append(self.expression())
            self.expect("OP", ")")
            node = Call(node, args, line)
        return node

    def primary(self):
        tok = self.peek()
        if self.match("NUMBER"):
            return Number(tok.value, tok.line)
        if self.match("STRING"):
            return String(tok.value, tok.line)
        if tok.kind in ("TRUE", "FALSE", "NIL"):
            self.pos += 1
            return Literal(tok.value, tok.line)
        if self.match("NAME"):
            return Name(tok.text, tok.line)
        if self.match("OP", "("):
            node = self.expression()
            self.expect("OP", ")")
            return node
        shown = "end of input" if tok.kind == "EOF" else repr(tok.text)
        raise ParseError(f"expected an expression, got {shown}", tok.line, tok.column)

    # --- statements ---
    def statement(self):
        tok = self.peek()
        if self.match("LET"):
            name = self.expect("NAME", what="a name").text
            self.expect("OP", "=")
            value = self.expression()
            self.expect("OP", ";")
            return Let(name, value, tok.line)
        if self.match("PRINT"):
            value = self.expression()
            self.expect("OP", ";")
            return Print(value, tok.line)
        if self.at("OP", "{"):
            return Block(self.block(), tok.line)
        if self.match("IF"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            then = self.statement()
            otherwise = self.statement() if self.match("ELSE") else None
            return If(test, then, otherwise, tok.line)
        if self.match("WHILE"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            return While(test, self.statement(), tok.line)
        if self.match("FN"):
            name = self.expect("NAME", what="a function name").text
            self.expect("OP", "(")
            params = []
            if not self.at("OP", ")"):
                params.append(self.expect("NAME", what="a parameter name").text)
                while self.match("OP", ","):
                    params.append(self.expect("NAME", what="a parameter name").text)
            self.expect("OP", ")")
            return Function(name, params, self.block(), tok.line)
        if self.match("RETURN"):
            value = None if self.at("OP", ";") else self.expression()
            self.expect("OP", ";")
            return Return(value, tok.line)
        if tok.kind == "NAME" and self.tokens[self.pos + 1].kind == "OP" and self.tokens[self.pos + 1].text == "=":
            self.pos += 2
            value = self.expression()
            self.expect("OP", ";")
            return Assign(tok.text, value, tok.line)
        value = self.expression()
        self.expect("OP", ";")
        return ExprStmt(value, tok.line)

    def block(self):
        self.expect("OP", "{")
        body = []
        while not self.at("OP", "}"):
            if self.at("EOF"):
                tok = self.peek()
                raise ParseError("expected '}', got end of input", tok.line, tok.column)
            body.append(self.statement())
        self.expect("OP", "}")
        return body

    def program(self):
        body = []
        while not self.at("EOF"):
            body.append(self.statement())
        return body


def parse(source):
    return Parser(tokenize(source)).program()

class Environment:
    def __init__(self, parent=None):
        self.values, self.parent = {}, parent

    def define(self, name, value):
        self.values[name] = value

    def find(self, name):
        env = self
        while env is not None:
            if name in env.values:
                return env
            env = env.parent
        return None

    def get(self, name, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"undefined name {name!r}", line)
        return env.values[name]

    def assign(self, name, value, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"cannot assign to undefined name {name!r}", line)
        env.values[name] = value

class Interpreter:
    def __init__(self, output=None, max_depth=100):
        self.output = [] if output is None else output
        self.globals = Environment()
        self.max_depth, self.depth = max_depth, 0
        self.steps = 0
        for name, arity, func in [
            ("len", 1, self._len), ("str", 1, lambda v: show(v)), ("type", 1, type_name),
        ]:
            self.globals.define(name, Builtin(name, arity, func))

    @staticmethod
    def _len(value):
        if not isinstance(value, str):
            raise TinyRuntimeError(f"len() needs a string, not {type_name(value)}")
        return len(value)

    # --- expressions ---
    def evaluate(self, node, env):
        self.steps += 1
        if isinstance(node, (Number, String, Literal)):
            return node.value
        if isinstance(node, Name):
            return env.get(node.name, node.line)
        if isinstance(node, Unary):
            value = self.evaluate(node.operand, env)
            if node.op == "-":
                if not is_number(value):
                    raise TinyRuntimeError(f"cannot negate {type_name(value)}", node.line)
                return -value
            return not truthy(value)
        if isinstance(node, Logical):
            left = self.evaluate(node.left, env)
            if node.op == "or":
                return left if truthy(left) else self.evaluate(node.right, env)
            return left if not truthy(left) else self.evaluate(node.right, env)
        if isinstance(node, Binary):
            return self.binary(node, self.evaluate(node.left, env), self.evaluate(node.right, env))
        if isinstance(node, Call):
            return self.call(node, env)
        raise TinyRuntimeError(f"cannot evaluate {type(node).__name__}", getattr(node, "line", None))

    def binary(self, node, left, right):
        op = node.op
        if op == "==":
            return type_name(left) == type_name(right) and left == right
        if op == "!=":
            return not (type_name(left) == type_name(right) and left == right)
        if op == "+" and isinstance(left, str) and isinstance(right, str):
            return left + right
        if not (is_number(left) and is_number(right)):
            raise TinyRuntimeError(
                f"cannot use {op} with {type_name(left)} and {type_name(right)}", node.line)
        if op in ("/", "%") and right == 0:
            raise TinyRuntimeError("division by zero", node.line)
        if op == "/":
            result = left / right
            return int(result) if isinstance(left, int) and isinstance(right, int) and result.is_integer() else result
        return {"+": lambda: left + right, "-": lambda: left - right, "*": lambda: left * right,
                "%": lambda: left % right, "<": lambda: left < right, "<=": lambda: left <= right,
                ">": lambda: left > right, ">=": lambda: left >= right}[op]()

    def call(self, node, env):
        callee = self.evaluate(node.callee, env)
        args = [self.evaluate(a, env) for a in node.args]
        if isinstance(callee, Builtin):
            if len(args) != callee.arity:
                raise TinyRuntimeError(f"{callee.name}() takes {callee.arity} argument(s), got {len(args)}", node.line)
            try:
                return callee.func(*args)
            except TinyRuntimeError as e:
                raise TinyRuntimeError(e.message, node.line) from None
        if not isinstance(callee, TinyFunction):
            raise TinyRuntimeError(f"cannot call {type_name(callee)}", node.line)
        decl = callee.decl
        if len(args) != len(decl.params):
            raise TinyRuntimeError(f"{decl.name}() takes {len(decl.params)} argument(s), got {len(args)}", node.line)
        if self.depth >= self.max_depth:
            raise TinyRuntimeError("too much recursion", node.line)
        local = Environment(callee.closure)
        for name, value in zip(decl.params, args):
            local.define(name, value)
        self.depth += 1
        try:
            self.run_block(decl.body, local)
        except ReturnSignal as signal:
            return signal.value
        finally:
            self.depth -= 1
        return None

    # --- statements ---
    def execute(self, node, env):
        self.steps += 1
        if isinstance(node, Let):
            env.define(node.name, self.evaluate(node.value, env))
        elif isinstance(node, Assign):
            env.assign(node.name, self.evaluate(node.value, env), node.line)
        elif isinstance(node, Print):
            self.output.append(show(self.evaluate(node.value, env)))
        elif isinstance(node, ExprStmt):
            self.evaluate(node.value, env)
        elif isinstance(node, Block):
            self.run_block(node.body, Environment(env))
        elif isinstance(node, If):
            if truthy(self.evaluate(node.test, env)):
                self.execute(node.then, env)
            elif node.otherwise is not None:
                self.execute(node.otherwise, env)
        elif isinstance(node, While):
            while truthy(self.evaluate(node.test, env)):
                self.execute(node.body, env)
        elif isinstance(node, Function):
            env.define(node.name, TinyFunction(node, env))
        elif isinstance(node, Return):
            raise ReturnSignal(None if node.value is None else self.evaluate(node.value, env))
        else:
            raise TinyRuntimeError(f"cannot execute {type(node).__name__}", getattr(node, "line", None))

    def run_block(self, statements, env):
        for stmt in statements:
            self.execute(stmt, env)

    def run(self, program):
        try:
            self.run_block(program, self.globals)
        except ReturnSignal:
            raise TinyRuntimeError("return outside a function") from None
        return self.output


def run_source(source):
    interpreter = Interpreter()
    interpreter.run(parse(source))
    return interpreter.output

## Step 24: a constant-folding pass

**Constant folding** is one of the oldest optimisations: compute at compile time whatever doesn't depend on the program's input. CPython does it too (Week 64: `2 * 3 + 1` is stored as `7` in the bytecode). The rule that matters most is that the optimised program must behave **exactly** like the original, errors included.

:::exercise id=step24 points=20 title="Step 24: a constant-folding pass"
An optimiser can do some work before the program runs. Write `fold(node)`: return a copy of a tree (or a list of nodes) in which every `Binary` with two `Number` operands and an operator in `FOLDABLE` is replaced by its result (computed with the interpreter's own `binary`, so the rules match exactly; comparisons give a `Literal`), every `-` of a `Number` by a `Number`, and every `+` of two `String`s by a `String`, working **from the leaves up** so `2 * 3 + 1` folds completely. Don't fold `/` or `%` (which might divide by zero at run time, and must still raise then), and never change the original tree (use `dataclasses.replace`). Then `steps_saved(source)`: run the program with and without folding, check the outputs are equal, and return how many fewer interpreter steps the folded one took.

class Parser:
    def __init__(self, tokens):
        self.tokens, self.pos = tokens, 0

    def peek(self):
        return self.tokens[self.pos]

    def at(self, kind, text=None):
        tok = self.peek()
        return tok.kind == kind and (text is None or tok.text == text)

    def match(self, kind, text=None):
        if self.at(kind, text):
            tok = self.peek()
            self.pos += 1
            return tok
        return None

    def expect(self, kind, text=None, what=None):
        tok = self.match(kind, text)
        if tok is None:
            got = self.peek()
            shown = "end of input" if got.kind == "EOF" else repr(got.text)
            raise ParseError(f"expected {what or text or kind.lower()}, got {shown}", got.line, got.column)
        return tok

    # --- expressions, lowest precedence first ---
    def expression(self):
        return self.logic_or()

    def logic_or(self):
        node = self.logic_and()
        while self.at("OR"):
            line = self.match("OR").line
            node = Logical("or", node, self.logic_and(), line)
        return node

    def logic_and(self):
        node = self.equality()
        while self.at("AND"):
            line = self.match("AND").line
            node = Logical("and", node, self.equality(), line)
        return node

    def binary(self, operators, operand):
        node = operand()
        while self.peek().kind == "OP" and self.peek().text in operators:
            tok = self.match("OP")
            node = Binary(tok.text, node, operand(), tok.line)
        return node

    def equality(self):
        return self.binary({"==", "!="}, self.comparison)

    def comparison(self):
        return self.binary({"<", "<=", ">", ">="}, self.term)

    def term(self):
        return self.binary({"+", "-"}, self.factor)

    def factor(self):
        return self.binary({"*", "/", "%"}, self.unary)

    def unary(self):
        if self.at("OP", "-"):
            line = self.match("OP").line
            return Unary("-", self.unary(), line)
        if self.at("NOT"):
            line = self.match("NOT").line
            return Unary("not", self.unary(), line)
        return self.call()

    def call(self):
        node = self.primary()
        while self.at("OP", "("):
            line = self.match("OP").line
            args = []
            if not self.at("OP", ")"):
                args.append(self.expression())
                while self.match("OP", ","):
                    args.append(self.expression())
            self.expect("OP", ")")
            node = Call(node, args, line)
        return node

    def primary(self):
        tok = self.peek()
        if self.match("NUMBER"):
            return Number(tok.value, tok.line)
        if self.match("STRING"):
            return String(tok.value, tok.line)
        if tok.kind in ("TRUE", "FALSE", "NIL"):
            self.pos += 1
            return Literal(tok.value, tok.line)
        if self.match("NAME"):
            return Name(tok.text, tok.line)
        if self.match("OP", "("):
            node = self.expression()
            self.expect("OP", ")")
            return node
        shown = "end of input" if tok.kind == "EOF" else repr(tok.text)
        raise ParseError(f"expected an expression, got {shown}", tok.line, tok.column)

    # --- statements ---
    def statement(self):
        tok = self.peek()
        if self.match("LET"):
            name = self.expect("NAME", what="a name").text
            self.expect("OP", "=")
            value = self.expression()
            self.expect("OP", ";")
            return Let(name, value, tok.line)
        if self.match("PRINT"):
            value = self.expression()
            self.expect("OP", ";")
            return Print(value, tok.line)
        if self.at("OP", "{"):
            return Block(self.block(), tok.line)
        if self.match("IF"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            then = self.statement()
            otherwise = self.statement() if self.match("ELSE") else None
            return If(test, then, otherwise, tok.line)
        if self.match("WHILE"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            return While(test, self.statement(), tok.line)
        if self.match("FN"):
            name = self.expect("NAME", what="a function name").text
            self.expect("OP", "(")
            params = []
            if not self.at("OP", ")"):
                params.append(self.expect("NAME", what="a parameter name").text)
                while self.match("OP", ","):
                    params.append(self.expect("NAME", what="a parameter name").text)
            self.expect("OP", ")")
            return Function(name, params, self.block(), tok.line)
        if self.match("RETURN"):
            value = None if self.at("OP", ";") else self.expression()
            self.expect("OP", ";")
            return Return(value, tok.line)
        if tok.kind == "NAME" and self.tokens[self.pos + 1].kind == "OP" and self.tokens[self.pos + 1].text == "=":
            self.pos += 2
            value = self.expression()
            self.expect("OP", ";")
            return Assign(tok.text, value, tok.line)
        value = self.expression()
        self.expect("OP", ";")
        return ExprStmt(value, tok.line)

    def block(self):
        self.expect("OP", "{")
        body = []
        while not self.at("OP", "}"):
            if self.at("EOF"):
                tok = self.peek()
                raise ParseError("expected '}', got end of input", tok.line, tok.column)
            body.append(self.statement())
        self.expect("OP", "}")
        return body

    def program(self):
        body = []
        while not self.at("EOF"):
            body.append(self.statement())
        return body


def parse(source):
    return Parser(tokenize(source)).program()

class Environment:
    def __init__(self, parent=None):
        self.values, self.parent = {}, parent

    def define(self, name, value):
        self.values[name] = value

    def find(self, name):
        env = self
        while env is not None:
            if name in env.values:
                return env
            env = env.parent
        return None

    def get(self, name, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"undefined name {name!r}", line)
        return env.values[name]

    def assign(self, name, value, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"cannot assign to undefined name {name!r}", line)
        env.values[name] = value

    if value is True:

    return str(value)

    if value is None:
        return "nil"
    if isinstance(value, bool):
        return "bool"
    if isinstance(value, (int, float)):
        return "number"
    if isinstance(value, str):
        return "string"
    return "function"

    return isinstance(value, (int, float)) and not isinstance(value, bool)


class Interpreter:
    def __init__(self, output=None, max_depth=100):
        self.output = [] if output is None else output
        self.globals = Environment()
        self.max_depth, self.depth = max_depth, 0
        self.steps = 0
        for name, arity, func in [
            ("len", 1, self._len), ("str", 1, lambda v: show(v)), ("type", 1, type_name),
        ]:
            self.globals.define(name, Builtin(name, arity, func))

    @staticmethod
    def _len(value):
        if not isinstance(value, str):
            raise TinyRuntimeError(f"len() needs a string, not {type_name(value)}")
        return len(value)

    # --- expressions ---
    def evaluate(self, node, env):
        self.steps += 1
        if isinstance(node, (Number, String, Literal)):
            return node.value
        if isinstance(node, Name):
            return env.get(node.name, node.line)
        if isinstance(node, Unary):
            value = self.evaluate(node.operand, env)
            if node.op == "-":
                if not is_number(value):
                    raise TinyRuntimeError(f"cannot negate {type_name(value)}", node.line)
                return -value
            return not truthy(value)
        if isinstance(node, Logical):
            left = self.evaluate(node.left, env)
            if node.op == "or":
                return left if truthy(left) else self.evaluate(node.right, env)
            return left if not truthy(left) else self.evaluate(node.right, env)
        if isinstance(node, Binary):
            return self.binary(node, self.evaluate(node.left, env), self.evaluate(node.right, env))
        if isinstance(node, Call):
            return self.call(node, env)
        raise TinyRuntimeError(f"cannot evaluate {type(node).__name__}", getattr(node, "line", None))

    def binary(self, node, left, right):
        op = node.op
        if op == "==":
            return type_name(left) == type_name(right) and left == right
        if op == "!=":
            return not (type_name(left) == type_name(right) and left == right)
        if op == "+" and isinstance(left, str) and isinstance(right, str):
            return left + right
        if not (is_number(left) and is_number(right)):
            raise TinyRuntimeError(
                f"cannot use {op} with {type_name(left)} and {type_name(right)}", node.line)
        if op in ("/", "%") and right == 0:
            raise TinyRuntimeError("division by zero", node.line)
        if op == "/":
            result = left / right
            return int(result) if isinstance(left, int) and isinstance(right, int) and result.is_integer() else result
        return {"+": lambda: left + right, "-": lambda: left - right, "*": lambda: left * right,
                "%": lambda: left % right, "<": lambda: left < right, "<=": lambda: left <= right,
                ">": lambda: left > right, ">=": lambda: left >= right}[op]()

    def call(self, node, env):
        callee = self.evaluate(node.callee, env)
        args = [self.evaluate(a, env) for a in node.args]
        if isinstance(callee, Builtin):
            if len(args) != callee.arity:
                raise TinyRuntimeError(f"{callee.name}() takes {callee.arity} argument(s), got {len(args)}", node.line)
            try:
                return callee.func(*args)
            except TinyRuntimeError as e:
                raise TinyRuntimeError(e.message, node.line) from None
        if not isinstance(callee, TinyFunction):
            raise TinyRuntimeError(f"cannot call {type_name(callee)}", node.line)
        decl = callee.decl
        if len(args) != len(decl.params):
            raise TinyRuntimeError(f"{decl.name}() takes {len(decl.params)} argument(s), got {len(args)}", node.line)
        if self.depth >= self.max_depth:
            raise TinyRuntimeError("too much recursion", node.line)
        local = Environment(callee.closure)
        for name, value in zip(decl.params, args):
            local.define(name, value)
        self.depth += 1
        try:
            self.run_block(decl.body, local)
        except ReturnSignal as signal:
            return signal.value
        finally:
            self.depth -= 1
        return None

    # --- statements ---
    def execute(self, node, env):
        self.steps += 1
        if isinstance(node, Let):
            env.define(node.name, self.evaluate(node.value, env))
        elif isinstance(node, Assign):
            env.assign(node.name, self.evaluate(node.value, env), node.line)
        elif isinstance(node, Print):
            self.output.append(show(self.evaluate(node.value, env)))
        elif isinstance(node, ExprStmt):
            self.evaluate(node.value, env)
        elif isinstance(node, Block):
            self.run_block(node.body, Environment(env))
        elif isinstance(node, If):
            if truthy(self.evaluate(node.test, env)):
                self.execute(node.then, env)
            elif node.otherwise is not None:
                self.execute(node.otherwise, env)
        elif isinstance(node, While):
            while truthy(self.evaluate(node.test, env)):
                self.execute(node.body, env)
        elif isinstance(node, Function):
            env.define(node.name, TinyFunction(node, env))
        elif isinstance(node, Return):
            raise ReturnSignal(None if node.value is None else self.evaluate(node.value, env))
        else:
            raise TinyRuntimeError(f"cannot execute {type(node).__name__}", getattr(node, "line", None))

    def run_block(self, statements, env):
        for stmt in statements:
            self.execute(stmt, env)

    def run(self, program):
        try:
            self.run_block(program, self.globals)
        except ReturnSignal:
            raise TinyRuntimeError("return outside a function") from None
        return self.output


def run_source(source):
    interpreter = Interpreter()
    interpreter.run(parse(source))
    return interpreter.output

class Cursor:
    def __init__(self, source):
        self.source, self.pos, self.line, self.column = source, 0, 1, 1

    def at_end(self):
        return self.pos >= len(self.source)

    def peek(self, ahead=0):
        i = self.pos + ahead
        return self.source[i] if i < len(self.source) else ""

    def advance(self):
        ch = self.source[self.pos]
        self.pos += 1
        if ch == "\n":
            self.line, self.column = self.line + 1, 1
        else:
            self.column += 1
        return ch


def read_number(cursor):
    line, column, text = cursor.line, cursor.column, ""
    while cursor.peek().isdigit():
        text += cursor.advance()
    if cursor.peek() == "." and cursor.peek(1).isdigit():
        text += cursor.advance()
        while cursor.peek().isdigit():
            text += cursor.advance()
        return Token("NUMBER", text, float(text), line, column)
    return Token("NUMBER", text, int(text), line, column)


ESCAPES = {"n": "\n", "t": "\t", '"': '"', "\\": "\\"}


def read_string(cursor):
    line, column = cursor.line, cursor.column
    text = cursor.advance()
    value = ""
    while True:
        if cursor.at_end() or cursor.peek() == "\n":
            raise LexError("unterminated string", line, column)
        ch = cursor.advance()
        text += ch
        if ch == '"':
            return Token("STRING", text, value, line, column)
        if ch == "\\":
            if cursor.at_end():
                raise LexError("unterminated string", line, column)
            esc_line, esc_col = cursor.line, cursor.column - 1
            code = cursor.advance()
            text += code
            if code not in ESCAPES:
                raise LexError(f"unknown escape \\{code}", esc_line, esc_col)
            value += ESCAPES[code]
        else:
            value += ch

def read_name(cursor):
    line, column, text = cursor.line, cursor.column, ""
    while cursor.peek().isalnum() or cursor.peek() == "_":
        text += cursor.advance()
    if text in KEYWORDS:
        value = {"true": True, "false": False, "nil": None}.get(text)
        return Token(text.upper(), text, value, line, column)
    return Token("NAME", text, text, line, column)


def read_operator(cursor):
    line, column = cursor.line, cursor.column
    pair = cursor.peek() + cursor.peek(1)
    if pair in TWO_CHAR:
        cursor.advance()
        cursor.advance()
        return Token("OP", pair, pair, line, column)
    ch = cursor.peek()
    if ch in ONE_CHAR:
        cursor.advance()
        return Token("OP", ch, ch, line, column)
    raise LexError(f"unexpected character {ch!r}", line, column)


def skip_space_and_comments(cursor):
    while not cursor.at_end():
        ch = cursor.peek()
        if ch in " \t\r\n":
            cursor.advance()
        elif ch == "#":
            while not cursor.at_end() and cursor.peek() != "\n":
                cursor.advance()
        else:
            return


def tokenize(source):
    cursor = Cursor(source)
    tokens = []
    while True:
        skip_space_and_comments(cursor)
        if cursor.at_end():
            tokens.append(Token("EOF", "", None, cursor.line, cursor.column))
            return tokens
        ch = cursor.peek()
        if ch.isdigit():
            tokens.append(read_number(cursor))
        elif ch == '"':
            tokens.append(read_string(cursor))
        elif ch.isalpha() or ch == "_":
            tokens.append(read_name(cursor))
        else:
            tokens.append(read_operator(cursor))

class Parser:
    def __init__(self, tokens):
        self.tokens, self.pos = tokens, 0

    def peek(self):
        return self.tokens[self.pos]

    def at(self, kind, text=None):
        tok = self.peek()
        return tok.kind == kind and (text is None or tok.text == text)

    def match(self, kind, text=None):
        if self.at(kind, text):
            tok = self.peek()
            self.pos += 1
            return tok
        return None

    def expect(self, kind, text=None, what=None):
        tok = self.match(kind, text)
        if tok is None:
            got = self.peek()
            shown = "end of input" if got.kind == "EOF" else repr(got.text)
            raise ParseError(f"expected {what or text or kind.lower()}, got {shown}", got.line, got.column)
        return tok

    # --- expressions, lowest precedence first ---
    def expression(self):
        return self.logic_or()

    def logic_or(self):
        node = self.logic_and()
        while self.at("OR"):
            line = self.match("OR").line
            node = Logical("or", node, self.logic_and(), line)
        return node

    def logic_and(self):
        node = self.equality()
        while self.at("AND"):
            line = self.match("AND").line
            node = Logical("and", node, self.equality(), line)
        return node

    def binary(self, operators, operand):
        node = operand()
        while self.peek().kind == "OP" and self.peek().text in operators:
            tok = self.match("OP")
            node = Binary(tok.text, node, operand(), tok.line)
        return node

    def equality(self):
        return self.binary({"==", "!="}, self.comparison)

    def comparison(self):
        return self.binary({"<", "<=", ">", ">="}, self.term)

    def term(self):
        return self.binary({"+", "-"}, self.factor)

    def factor(self):
        return self.binary({"*", "/", "%"}, self.unary)

    def unary(self):
        if self.at("OP", "-"):
            line = self.match("OP").line
            return Unary("-", self.unary(), line)
        if self.at("NOT"):
            line = self.match("NOT").line
            return Unary("not", self.unary(), line)
        return self.call()

    def call(self):
        node = self.primary()
        while self.at("OP", "("):
            line = self.match("OP").line
            args = []
            if not self.at("OP", ")"):
                args.append(self.expression())
                while self.match("OP", ","):
                    args.append(self.expression())
            self.expect("OP", ")")
            node = Call(node, args, line)
        return node

    def primary(self):
        tok = self.peek()
        if self.match("NUMBER"):
            return Number(tok.value, tok.line)
        if self.match("STRING"):
            return String(tok.value, tok.line)
        if tok.kind in ("TRUE", "FALSE", "NIL"):
            self.pos += 1
            return Literal(tok.value, tok.line)
        if self.match("NAME"):
            return Name(tok.text, tok.line)
        if self.match("OP", "("):
            node = self.expression()
            self.expect("OP", ")")
            return node
        shown = "end of input" if tok.kind == "EOF" else repr(tok.text)
        raise ParseError(f"expected an expression, got {shown}", tok.line, tok.column)

    # --- statements ---
    def statement(self):
        tok = self.peek()
        if self.match("LET"):
            name = self.expect("NAME", what="a name").text
            self.expect("OP", "=")
            value = self.expression()
            self.expect("OP", ";")
            return Let(name, value, tok.line)
        if self.match("PRINT"):
            value = self.expression()
            self.expect("OP", ";")
            return Print(value, tok.line)
        if self.at("OP", "{"):
            return Block(self.block(), tok.line)
        if self.match("IF"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            then = self.statement()
            otherwise = self.statement() if self.match("ELSE") else None
            return If(test, then, otherwise, tok.line)
        if self.match("WHILE"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            return While(test, self.statement(), tok.line)
        if self.match("FN"):
            name = self.expect("NAME", what="a function name").text
            self.expect("OP", "(")
            params = []
            if not self.at("OP", ")"):
                params.append(self.expect("NAME", what="a parameter name").text)
                while self.match("OP", ","):
                    params.append(self.expect("NAME", what="a parameter name").text)
            self.expect("OP", ")")
            return Function(name, params, self.block(), tok.line)
        if self.match("RETURN"):
            value = None if self.at("OP", ";") else self.expression()
            self.expect("OP", ";")
            return Return(value, tok.line)
        if tok.kind == "NAME" and self.tokens[self.pos + 1].kind == "OP" and self.tokens[self.pos + 1].text == "=":
            self.pos += 2
            value = self.expression()
            self.expect("OP", ";")
            return Assign(tok.text, value, tok.line)
        value = self.expression()
        self.expect("OP", ";")
        return ExprStmt(value, tok.line)

    def block(self):
        self.expect("OP", "{")
        body = []
        while not self.at("OP", "}"):
            if self.at("EOF"):
                tok = self.peek()
                raise ParseError("expected '}', got end of input", tok.line, tok.column)
            body.append(self.statement())
        self.expect("OP", "}")
        return body

    def program(self):
        body = []
        while not self.at("EOF"):
            body.append(self.statement())
        return body


def parse(source):
    return Parser(tokenize(source)).program()

        return "nil"

class Interpreter:
    def __init__(self, output=None, max_depth=100):
        self.output = [] if output is None else output
        self.globals = Environment()
        self.max_depth, self.depth = max_depth, 0
        self.steps = 0
        for name, arity, func in [
            ("len", 1, self._len), ("str", 1, lambda v: show(v)), ("type", 1, type_name),
        ]:
            self.globals.define(name, Builtin(name, arity, func))

    @staticmethod
    def _len(value):
        if not isinstance(value, str):
            raise TinyRuntimeError(f"len() needs a string, not {type_name(value)}")
        return len(value)

    # --- expressions ---
    def evaluate(self, node, env):
        self.steps += 1
        if isinstance(node, (Number, String, Literal)):
            return node.value
        if isinstance(node, Name):
            return env.get(node.name, node.line)
        if isinstance(node, Unary):
            value = self.evaluate(node.operand, env)
            if node.op == "-":
                if not is_number(value):
                    raise TinyRuntimeError(f"cannot negate {type_name(value)}", node.line)
                return -value
            return not truthy(value)
        if isinstance(node, Logical):
            left = self.evaluate(node.left, env)
            if node.op == "or":
                return left if truthy(left) else self.evaluate(node.right, env)
            return left if not truthy(left) else self.evaluate(node.right, env)
        if isinstance(node, Binary):
            return self.binary(node, self.evaluate(node.left, env), self.evaluate(node.right, env))
        if isinstance(node, Call):
            return self.call(node, env)
        raise TinyRuntimeError(f"cannot evaluate {type(node).__name__}", getattr(node, "line", None))

    def binary(self, node, left, right):
        op = node.op
        if op == "==":
            return type_name(left) == type_name(right) and left == right
        if op == "!=":
            return not (type_name(left) == type_name(right) and left == right)
        if op == "+" and isinstance(left, str) and isinstance(right, str):
            return left + right
        if not (is_number(left) and is_number(right)):
            raise TinyRuntimeError(
                f"cannot use {op} with {type_name(left)} and {type_name(right)}", node.line)
        if op in ("/", "%") and right == 0:
            raise TinyRuntimeError("division by zero", node.line)
        if op == "/":
            result = left / right
            return int(result) if isinstance(left, int) and isinstance(right, int) and result.is_integer() else result
        return {"+": lambda: left + right, "-": lambda: left - right, "*": lambda: left * right,
                "%": lambda: left % right, "<": lambda: left < right, "<=": lambda: left <= right,
                ">": lambda: left > right, ">=": lambda: left >= right}[op]()

    def call(self, node, env):
        callee = self.evaluate(node.callee, env)
        args = [self.evaluate(a, env) for a in node.args]
        if isinstance(callee, Builtin):
            if len(args) != callee.arity:
                raise TinyRuntimeError(f"{callee.name}() takes {callee.arity} argument(s), got {len(args)}", node.line)
            try:
                return callee.func(*args)
            except TinyRuntimeError as e:
                raise TinyRuntimeError(e.message, node.line) from None
        if not isinstance(callee, TinyFunction):
            raise TinyRuntimeError(f"cannot call {type_name(callee)}", node.line)
        decl = callee.decl
        if len(args) != len(decl.params):
            raise TinyRuntimeError(f"{decl.name}() takes {len(decl.params)} argument(s), got {len(args)}", node.line)
        if self.depth >= self.max_depth:
            raise TinyRuntimeError("too much recursion", node.line)
        local = Environment(callee.closure)
        for name, value in zip(decl.params, args):
            local.define(name, value)
        self.depth += 1
        try:
            self.run_block(decl.body, local)
        except ReturnSignal as signal:
            return signal.value
        finally:
            self.depth -= 1
        return None

    # --- statements ---
    def execute(self, node, env):
        self.steps += 1
        if isinstance(node, Let):
            env.define(node.name, self.evaluate(node.value, env))
        elif isinstance(node, Assign):
            env.assign(node.name, self.evaluate(node.value, env), node.line)
        elif isinstance(node, Print):
            self.output.append(show(self.evaluate(node.value, env)))
        elif isinstance(node, ExprStmt):
            self.evaluate(node.value, env)
        elif isinstance(node, Block):
            self.run_block(node.body, Environment(env))
        elif isinstance(node, If):
            if truthy(self.evaluate(node.test, env)):
                self.execute(node.then, env)
            elif node.otherwise is not None:
                self.execute(node.otherwise, env)
        elif isinstance(node, While):
            while truthy(self.evaluate(node.test, env)):
                self.execute(node.body, env)
        elif isinstance(node, Function):
            env.define(node.name, TinyFunction(node, env))
        elif isinstance(node, Return):
            raise ReturnSignal(None if node.value is None else self.evaluate(node.value, env))
        else:
            raise TinyRuntimeError(f"cannot execute {type(node).__name__}", getattr(node, "line", None))

    def run_block(self, statements, env):
        for stmt in statements:
            self.execute(stmt, env)

    def run(self, program):
        try:
            self.run_block(program, self.globals)
        except ReturnSignal:
            raise TinyRuntimeError("return outside a function") from None
        return self.output


def run_source(source):
    interpreter = Interpreter()
    interpreter.run(parse(source))
    return interpreter.output

def fold(node):
    if isinstance(node, list):
        return [fold(n) for n in node]
    if not dataclasses.is_dataclass(node):
        return node
    changes = {f.name: fold(getattr(node, f.name)) for f in dataclasses.fields(node) if f.name != "line"}
    node = dataclasses.replace(node, **changes)
    if isinstance(node, Unary) and node.op == "-" and isinstance(node.operand, Number):
        return Number(-node.operand.value, node.line)
    if isinstance(node, Binary) and node.op in FOLDABLE:
        left, right = node.left, node.right
        if isinstance(left, Number) and isinstance(right, Number):
            value = Interpreter().binary(node, left.value, right.value)
            return Literal(value, node.line) if isinstance(value, bool) else Number(value, node.line)
        if node.op == "+" and isinstance(left, String) and isinstance(right, String):
            return String(left.value + right.value, node.line)
    return node

class Cursor:
    def __init__(self, source):
        self.source, self.pos, self.line, self.column = source, 0, 1, 1

    def at_end(self):
        return self.pos >= len(self.source)

    def peek(self, ahead=0):
        i = self.pos + ahead
        return self.source[i] if i < len(self.source) else ""

    def advance(self):
        ch = self.source[self.pos]
        self.pos += 1
        if ch == "\n":
            self.line, self.column = self.line + 1, 1
        else:
            self.column += 1
        return ch


def read_number(cursor):
    line, column, text = cursor.line, cursor.column, ""
    while cursor.peek().isdigit():
        text += cursor.advance()
    if cursor.peek() == "." and cursor.peek(1).isdigit():
        text += cursor.advance()
        while cursor.peek().isdigit():
            text += cursor.advance()
        return Token("NUMBER", text, float(text), line, column)
    return Token("NUMBER", text, int(text), line, column)


ESCAPES = {"n": "\n", "t": "\t", '"': '"', "\\": "\\"}


def read_string(cursor):
    line, column = cursor.line, cursor.column
    text = cursor.advance()
    value = ""
    while True:
        if cursor.at_end() or cursor.peek() == "\n":
            raise LexError("unterminated string", line, column)
        ch = cursor.advance()
        text += ch
        if ch == '"':
            return Token("STRING", text, value, line, column)
        if ch == "\\":
            if cursor.at_end():
                raise LexError("unterminated string", line, column)
            esc_line, esc_col = cursor.line, cursor.column - 1
            code = cursor.advance()
            text += code
            if code not in ESCAPES:
                raise LexError(f"unknown escape \\{code}", esc_line, esc_col)
            value += ESCAPES[code]
        else:
            value += ch


KEYWORDS = {"let", "fn", "return", "if", "else", "while", "print", "true", "false", "nil", "and", "or", "not"}


TWO_CHAR = {"==", "!=", "<=", ">="}


ONE_CHAR = set("+-*/%<>=(){},;")


def read_name(cursor):
    line, column, text = cursor.line, cursor.column, ""
    while cursor.peek().isalnum() or cursor.peek() == "_":
        text += cursor.advance()
    if text in KEYWORDS:
        value = {"true": True, "false": False, "nil": None}.get(text)
        return Token(text.upper(), text, value, line, column)
    return Token("NAME", text, text, line, column)


def read_operator(cursor):
    line, column = cursor.line, cursor.column
    pair = cursor.peek() + cursor.peek(1)
    if pair in TWO_CHAR:
        cursor.advance()
        cursor.advance()
        return Token("OP", pair, pair, line, column)
    ch = cursor.peek()
    if ch in ONE_CHAR:
        cursor.advance()
        return Token("OP", ch, ch, line, column)
    raise LexError(f"unexpected character {ch!r}", line, column)


def skip_space_and_comments(cursor):
    while not cursor.at_end():
        ch = cursor.peek()
        if ch in " \t\r\n":
            cursor.advance()
        elif ch == "#":
            while not cursor.at_end() and cursor.peek() != "\n":
                cursor.advance()
        else:
            return


def tokenize(source):
    cursor = Cursor(source)
    tokens = []
    while True:
        skip_space_and_comments(cursor)
        if cursor.at_end():
            tokens.append(Token("EOF", "", None, cursor.line, cursor.column))
            return tokens
        ch = cursor.peek()
        if ch.isdigit():
            tokens.append(read_number(cursor))
        elif ch == '"':
            tokens.append(read_string(cursor))
        elif ch.isalpha() or ch == "_":
            tokens.append(read_name(cursor))
        else:
            tokens.append(read_operator(cursor))

class Parser:
    def __init__(self, tokens):
        self.tokens, self.pos = tokens, 0

    def peek(self):
        return self.tokens[self.pos]

    def at(self, kind, text=None):
        tok = self.peek()
        return tok.kind == kind and (text is None or tok.text == text)

    def match(self, kind, text=None):
        if self.at(kind, text):
            tok = self.peek()
            self.pos += 1
            return tok
        return None

    def expect(self, kind, text=None, what=None):
        tok = self.match(kind, text)
        if tok is None:
            got = self.peek()
            shown = "end of input" if got.kind == "EOF" else repr(got.text)
            raise ParseError(f"expected {what or text or kind.lower()}, got {shown}", got.line, got.column)
        return tok

    # --- expressions, lowest precedence first ---
    def expression(self):
        return self.logic_or()

    def logic_or(self):
        node = self.logic_and()
        while self.at("OR"):
            line = self.match("OR").line
            node = Logical("or", node, self.logic_and(), line)
        return node

    def logic_and(self):
        node = self.equality()
        while self.at("AND"):
            line = self.match("AND").line
            node = Logical("and", node, self.equality(), line)
        return node

    def binary(self, operators, operand):
        node = operand()
        while self.peek().kind == "OP" and self.peek().text in operators:
            tok = self.match("OP")
            node = Binary(tok.text, node, operand(), tok.line)
        return node

    def equality(self):
        return self.binary({"==", "!="}, self.comparison)

    def comparison(self):
        return self.binary({"<", "<=", ">", ">="}, self.term)

    def term(self):
        return self.binary({"+", "-"}, self.factor)

    def factor(self):
        return self.binary({"*", "/", "%"}, self.unary)

    def unary(self):
        if self.at("OP", "-"):
            line = self.match("OP").line
            return Unary("-", self.unary(), line)
        if self.at("NOT"):
            line = self.match("NOT").line
            return Unary("not", self.unary(), line)
        return self.call()

    def call(self):
        node = self.primary()
        while self.at("OP", "("):
            line = self.match("OP").line
            args = []
            if not self.at("OP", ")"):
                args.append(self.expression())
                while self.match("OP", ","):
                    args.append(self.expression())
            self.expect("OP", ")")
            node = Call(node, args, line)
        return node

    def primary(self):
        tok = self.peek()
        if self.match("NUMBER"):
            return Number(tok.value, tok.line)
        if self.match("STRING"):
            return String(tok.value, tok.line)
        if tok.kind in ("TRUE", "FALSE", "NIL"):
            self.pos += 1
            return Literal(tok.value, tok.line)
        if self.match("NAME"):
            return Name(tok.text, tok.line)
        if self.match("OP", "("):
            node = self.expression()
            self.expect("OP", ")")
            return node
        shown = "end of input" if tok.kind == "EOF" else repr(tok.text)
        raise ParseError(f"expected an expression, got {shown}", tok.line, tok.column)

    # --- statements ---
    def statement(self):
        tok = self.peek()
        if self.match("LET"):
            name = self.expect("NAME", what="a name").text
            self.expect("OP", "=")
            value = self.expression()
            self.expect("OP", ";")
            return Let(name, value, tok.line)
        if self.match("PRINT"):
            value = self.expression()
            self.expect("OP", ";")
            return Print(value, tok.line)
        if self.at("OP", "{"):
            return Block(self.block(), tok.line)
        if self.match("IF"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            then = self.statement()
            otherwise = self.statement() if self.match("ELSE") else None
            return If(test, then, otherwise, tok.line)
        if self.match("WHILE"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            return While(test, self.statement(), tok.line)
        if self.match("FN"):
            name = self.expect("NAME", what="a function name").text
            self.expect("OP", "(")
            params = []
            if not self.at("OP", ")"):
                params.append(self.expect("NAME", what="a parameter name").text)
                while self.match("OP", ","):
                    params.append(self.expect("NAME", what="a parameter name").text)
            self.expect("OP", ")")
            return Function(name, params, self.block(), tok.line)
        if self.match("RETURN"):
            value = None if self.at("OP", ";") else self.expression()
            self.expect("OP", ";")
            return Return(value, tok.line)
        if tok.kind == "NAME" and self.tokens[self.pos + 1].kind == "OP" and self.tokens[self.pos + 1].text == "=":
            self.pos += 2
            value = self.expression()
            self.expect("OP", ";")
            return Assign(tok.text, value, tok.line)
        value = self.expression()
        self.expect("OP", ";")
        return ExprStmt(value, tok.line)

    def block(self):
        self.expect("OP", "{")
        body = []
        while not self.at("OP", "}"):
            if self.at("EOF"):
                tok = self.peek()
                raise ParseError("expected '}', got end of input", tok.line, tok.column)
            body.append(self.statement())
        self.expect("OP", "}")
        return body

    def program(self):
        body = []
        while not self.at("EOF"):
            body.append(self.statement())
        return body


def parse(source):

    if value is True:
        return "true"

        return "false"
    if isinstance(value, float) and value.is_integer():

    return str(value)

    if isinstance(value, bool):

    if isinstance(value, (int, float)):
        return "number"
    if isinstance(value, str):

    return "function"

class Interpreter:
    def __init__(self, output=None, max_depth=100):
        self.output = [] if output is None else output
        self.globals = Environment()
        self.max_depth, self.depth = max_depth, 0
        self.steps = 0
        for name, arity, func in [
            ("len", 1, self._len), ("str", 1, lambda v: show(v)), ("type", 1, type_name),
        ]:
            self.globals.define(name, Builtin(name, arity, func))

    @staticmethod
    def _len(value):
        if not isinstance(value, str):
            raise TinyRuntimeError(f"len() needs a string, not {type_name(value)}")
        return len(value)

    # --- expressions ---
    def evaluate(self, node, env):
        self.steps += 1
        if isinstance(node, (Number, String, Literal)):
            return node.value
        if isinstance(node, Name):
            return env.get(node.name, node.line)
        if isinstance(node, Unary):
            value = self.evaluate(node.operand, env)
            if node.op == "-":
                if not is_number(value):
                    raise TinyRuntimeError(f"cannot negate {type_name(value)}", node.line)
                return -value
            return not truthy(value)
        if isinstance(node, Logical):
            left = self.evaluate(node.left, env)
            if node.op == "or":
                return left if truthy(left) else self.evaluate(node.right, env)
            return left if not truthy(left) else self.evaluate(node.right, env)
        if isinstance(node, Binary):
            return self.binary(node, self.evaluate(node.left, env), self.evaluate(node.right, env))
        if isinstance(node, Call):
            return self.call(node, env)
        raise TinyRuntimeError(f"cannot evaluate {type(node).__name__}", getattr(node, "line", None))

    def binary(self, node, left, right):
        op = node.op
        if op == "==":
            return type_name(left) == type_name(right) and left == right
        if op == "!=":
            return not (type_name(left) == type_name(right) and left == right)
        if op == "+" and isinstance(left, str) and isinstance(right, str):
            return left + right
        if not (is_number(left) and is_number(right)):
            raise TinyRuntimeError(
                f"cannot use {op} with {type_name(left)} and {type_name(right)}", node.line)
        if op in ("/", "%") and right == 0:
            raise TinyRuntimeError("division by zero", node.line)
        if op == "/":
            result = left / right
            return int(result) if isinstance(left, int) and isinstance(right, int) and result.is_integer() else result
        return {"+": lambda: left + right, "-": lambda: left - right, "*": lambda: left * right,
                "%": lambda: left % right, "<": lambda: left < right, "<=": lambda: left <= right,
                ">": lambda: left > right, ">=": lambda: left >= right}[op]()

    def call(self, node, env):
        callee = self.evaluate(node.callee, env)
        args = [self.evaluate(a, env) for a in node.args]
        if isinstance(callee, Builtin):
            if len(args) != callee.arity:
                raise TinyRuntimeError(f"{callee.name}() takes {callee.arity} argument(s), got {len(args)}", node.line)
            try:
                return callee.func(*args)
            except TinyRuntimeError as e:
                raise TinyRuntimeError(e.message, node.line) from None
        if not isinstance(callee, TinyFunction):
            raise TinyRuntimeError(f"cannot call {type_name(callee)}", node.line)
        decl = callee.decl
        if len(args) != len(decl.params):
            raise TinyRuntimeError(f"{decl.name}() takes {len(decl.params)} argument(s), got {len(args)}", node.line)
        if self.depth >= self.max_depth:
            raise TinyRuntimeError("too much recursion", node.line)
        local = Environment(callee.closure)
        for name, value in zip(decl.params, args):
            local.define(name, value)
        self.depth += 1
        try:
            self.run_block(decl.body, local)
        except ReturnSignal as signal:
            return signal.value
        finally:
            self.depth -= 1
        return None

    # --- statements ---
    def execute(self, node, env):
        self.steps += 1
        if isinstance(node, Let):
            env.define(node.name, self.evaluate(node.value, env))
        elif isinstance(node, Assign):
            env.assign(node.name, self.evaluate(node.value, env), node.line)
        elif isinstance(node, Print):
            self.output.append(show(self.evaluate(node.value, env)))
        elif isinstance(node, ExprStmt):
            self.evaluate(node.value, env)
        elif isinstance(node, Block):
            self.run_block(node.body, Environment(env))
        elif isinstance(node, If):
            if truthy(self.evaluate(node.test, env)):
                self.execute(node.then, env)
            elif node.otherwise is not None:
                self.execute(node.otherwise, env)
        elif isinstance(node, While):
            while truthy(self.evaluate(node.test, env)):
                self.execute(node.body, env)
        elif isinstance(node, Function):
            env.define(node.name, TinyFunction(node, env))
        elif isinstance(node, Return):
            raise ReturnSignal(None if node.value is None else self.evaluate(node.value, env))
        else:
            raise TinyRuntimeError(f"cannot execute {type(node).__name__}", getattr(node, "line", None))

    def run_block(self, statements, env):
        for stmt in statements:
            self.execute(stmt, env)

    def run(self, program):
        try:
            self.run_block(program, self.globals)
        except ReturnSignal:
            raise TinyRuntimeError("return outside a function") from None
        return self.output


def run_source(source):
    interpreter = Interpreter()
    interpreter.run(parse(source))
    return interpreter.output

class TinyError(Exception):
    """Base class for every error the Tiny interpreter reports."""

    def __init__(self, message, line=None, column=None):
        super().__init__(message)
        self.message, self.line, self.column = message, line, column

    def __str__(self):
        if self.line is None:
            return self.message
        if self.column is None:
            return f"line {self.line}: {self.message}"
        return f"line {self.line}, column {self.column}: {self.message}"

class Cursor:
    def __init__(self, source):
        self.source, self.pos, self.line, self.column = source, 0, 1, 1

    def at_end(self):
        return self.pos >= len(self.source)

    def peek(self, ahead=0):
        i = self.pos + ahead
        return self.source[i] if i < len(self.source) else ""

    def advance(self):
        ch = self.source[self.pos]
        self.pos += 1
        if ch == "\n":
            self.line, self.column = self.line + 1, 1
        else:
            self.column += 1
        return ch


def read_number(cursor):
    line, column, text = cursor.line, cursor.column, ""
    while cursor.peek().isdigit():
        text += cursor.advance()
    if cursor.peek() == "." and cursor.peek(1).isdigit():
        text += cursor.advance()
        while cursor.peek().isdigit():
            text += cursor.advance()
        return Token("NUMBER", text, float(text), line, column)
    return Token("NUMBER", text, int(text), line, column)


ESCAPES = {"n": "\n", "t": "\t", '"': '"', "\\": "\\"}


def read_string(cursor):
    line, column = cursor.line, cursor.column
    text = cursor.advance()
    value = ""
    while True:
        if cursor.at_end() or cursor.peek() == "\n":
            raise LexError("unterminated string", line, column)
        ch = cursor.advance()
        text += ch
        if ch == '"':
            return Token("STRING", text, value, line, column)
        if ch == "\\":
            if cursor.at_end():
                raise LexError("unterminated string", line, column)
            esc_line, esc_col = cursor.line, cursor.column - 1
            code = cursor.advance()
            text += code
            if code not in ESCAPES:
                raise LexError(f"unknown escape \\{code}", esc_line, esc_col)
            value += ESCAPES[code]
        else:
            value += ch


KEYWORDS = {"let", "fn", "return", "if", "else", "while", "print", "true", "false", "nil", "and", "or", "not"}


TWO_CHAR = {"==", "!=", "<=", ">="}


ONE_CHAR = set("+-*/%<>=(){},;")


def read_name(cursor):
    line, column, text = cursor.line, cursor.column, ""
    while cursor.peek().isalnum() or cursor.peek() == "_":
        text += cursor.advance()
    if text in KEYWORDS:
        value = {"true": True, "false": False, "nil": None}.get(text)
        return Token(text.upper(), text, value, line, column)
    return Token("NAME", text, text, line, column)


def read_operator(cursor):
    line, column = cursor.line, cursor.column
    pair = cursor.peek() + cursor.peek(1)
    if pair in TWO_CHAR:
        cursor.advance()
        cursor.advance()
        return Token("OP", pair, pair, line, column)
    ch = cursor.peek()
    if ch in ONE_CHAR:
        cursor.advance()
        return Token("OP", ch, ch, line, column)
    raise LexError(f"unexpected character {ch!r}", line, column)


def skip_space_and_comments(cursor):
    while not cursor.at_end():
        ch = cursor.peek()
        if ch in " \t\r\n":
            cursor.advance()
        elif ch == "#":
            while not cursor.at_end() and cursor.peek() != "\n":
                cursor.advance()
        else:
            return


def tokenize(source):
    cursor = Cursor(source)
    tokens = []
    while True:
        skip_space_and_comments(cursor)
        if cursor.at_end():
            tokens.append(Token("EOF", "", None, cursor.line, cursor.column))
            return tokens
        ch = cursor.peek()
        if ch.isdigit():
            tokens.append(read_number(cursor))
        elif ch == '"':
            tokens.append(read_string(cursor))
        elif ch.isalpha() or ch == "_":
            tokens.append(read_name(cursor))
        else:
            tokens.append(read_operator(cursor))

class Parser:
    def __init__(self, tokens):
        self.tokens, self.pos = tokens, 0

    def peek(self):
        return self.tokens[self.pos]

    def at(self, kind, text=None):
        tok = self.peek()
        return tok.kind == kind and (text is None or tok.text == text)

    def match(self, kind, text=None):
        if self.at(kind, text):
            tok = self.peek()
            self.pos += 1
            return tok
        return None

    def expect(self, kind, text=None, what=None):
        tok = self.match(kind, text)
        if tok is None:
            got = self.peek()
            shown = "end of input" if got.kind == "EOF" else repr(got.text)
            raise ParseError(f"expected {what or text or kind.lower()}, got {shown}", got.line, got.column)
        return tok

    # --- expressions, lowest precedence first ---
    def expression(self):
        return self.logic_or()

    def logic_or(self):
        node = self.logic_and()
        while self.at("OR"):
            line = self.match("OR").line
            node = Logical("or", node, self.logic_and(), line)
        return node

    def logic_and(self):
        node = self.equality()
        while self.at("AND"):
            line = self.match("AND").line
            node = Logical("and", node, self.equality(), line)
        return node

    def binary(self, operators, operand):
        node = operand()
        while self.peek().kind == "OP" and self.peek().text in operators:
            tok = self.match("OP")
            node = Binary(tok.text, node, operand(), tok.line)
        return node

    def equality(self):
        return self.binary({"==", "!="}, self.comparison)

    def comparison(self):
        return self.binary({"<", "<=", ">", ">="}, self.term)

    def term(self):
        return self.binary({"+", "-"}, self.factor)

    def factor(self):
        return self.binary({"*", "/", "%"}, self.unary)

    def unary(self):
        if self.at("OP", "-"):
            line = self.match("OP").line
            return Unary("-", self.unary(), line)
        if self.at("NOT"):
            line = self.match("NOT").line
            return Unary("not", self.unary(), line)
        return self.call()

    def call(self):
        node = self.primary()
        while self.at("OP", "("):
            line = self.match("OP").line
            args = []
            if not self.at("OP", ")"):
                args.append(self.expression())
                while self.match("OP", ","):
                    args.append(self.expression())
            self.expect("OP", ")")
            node = Call(node, args, line)
        return node

    def primary(self):
        tok = self.peek()
        if self.match("NUMBER"):
            return Number(tok.value, tok.line)
        if self.match("STRING"):
            return String(tok.value, tok.line)
        if tok.kind in ("TRUE", "FALSE", "NIL"):
            self.pos += 1
            return Literal(tok.value, tok.line)
        if self.match("NAME"):
            return Name(tok.text, tok.line)
        if self.match("OP", "("):
            node = self.expression()
            self.expect("OP", ")")
            return node
        shown = "end of input" if tok.kind == "EOF" else repr(tok.text)
        raise ParseError(f"expected an expression, got {shown}", tok.line, tok.column)

    # --- statements ---
    def statement(self):
        tok = self.peek()
        if self.match("LET"):
            name = self.expect("NAME", what="a name").text
            self.expect("OP", "=")
            value = self.expression()
            self.expect("OP", ";")
            return Let(name, value, tok.line)
        if self.match("PRINT"):
            value = self.expression()
            self.expect("OP", ";")
            return Print(value, tok.line)
        if self.at("OP", "{"):
            return Block(self.block(), tok.line)
        if self.match("IF"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            then = self.statement()
            otherwise = self.statement() if self.match("ELSE") else None
            return If(test, then, otherwise, tok.line)
        if self.match("WHILE"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            return While(test, self.statement(), tok.line)
        if self.match("FN"):
            name = self.expect("NAME", what="a function name").text
            self.expect("OP", "(")
            params = []
            if not self.at("OP", ")"):
                params.append(self.expect("NAME", what="a parameter name").text)
                while self.match("OP", ","):
                    params.append(self.expect("NAME", what="a parameter name").text)
            self.expect("OP", ")")
            return Function(name, params, self.block(), tok.line)
        if self.match("RETURN"):
            value = None if self.at("OP", ";") else self.expression()
            self.expect("OP", ";")
            return Return(value, tok.line)
        if tok.kind == "NAME" and self.tokens[self.pos + 1].kind == "OP" and self.tokens[self.pos + 1].text == "=":
            self.pos += 2
            value = self.expression()
            self.expect("OP", ";")
            return Assign(tok.text, value, tok.line)
        value = self.expression()
        self.expect("OP", ";")
        return ExprStmt(value, tok.line)

    def block(self):
        self.expect("OP", "{")
        body = []
        while not self.at("OP", "}"):
            if self.at("EOF"):
                tok = self.peek()
                raise ParseError("expected '}', got end of input", tok.line, tok.column)
            body.append(self.statement())
        self.expect("OP", "}")
        return body

    def program(self):
        body = []
        while not self.at("EOF"):
            body.append(self.statement())
        return body


def parse(source):
    return Parser(tokenize(source)).program()

class Environment:
    def __init__(self, parent=None):
        self.values, self.parent = {}, parent

    def define(self, name, value):
        self.values[name] = value

    def find(self, name):
        env = self
        while env is not None:
            if name in env.values:
                return env
            env = env.parent
        return None

    def get(self, name, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"undefined name {name!r}", line)
        return env.values[name]

    def assign(self, name, value, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"cannot assign to undefined name {name!r}", line)
        env.values[name] = value

        if len(args) != len(decl.params):
            raise TinyRuntimeError(f"{decl.name}() takes {len(decl.params)} argument(s), got {len(args)}", node.line)
        if self.depth >= self.max_depth:
            raise TinyRuntimeError("too much recursion", node.line)
        local = Environment(callee.closure)
        for name, value in zip(decl.params, args):
            local.define(name, value)
        self.depth += 1
        try:
            self.run_block(decl.body, local)
        except ReturnSignal as signal:
            return signal.value
        finally:
            self.depth -= 1
        return None

    # --- statements ---
    def execute(self, node, env):
        self.steps += 1
        if isinstance(node, Let):
            env.define(node.name, self.evaluate(node.value, env))
        elif isinstance(node, Assign):
            env.assign(node.name, self.evaluate(node.value, env), node.line)
        elif isinstance(node, Print):
            self.output.append(show(self.evaluate(node.value, env)))
        elif isinstance(node, ExprStmt):
            self.evaluate(node.value, env)
        elif isinstance(node, Block):
            self.run_block(node.body, Environment(env))
        elif isinstance(node, If):
            if truthy(self.evaluate(node.test, env)):
                self.execute(node.then, env)
            elif node.otherwise is not None:
                self.execute(node.otherwise, env)
        elif isinstance(node, While):
            while truthy(self.evaluate(node.test, env)):
                self.execute(node.body, env)
        elif isinstance(node, Function):
            env.define(node.name, TinyFunction(node, env))
        elif isinstance(node, Return):
            raise ReturnSignal(None if node.value is None else self.evaluate(node.value, env))
        else:
            raise TinyRuntimeError(f"cannot execute {type(node).__name__}", getattr(node, "line", None))

    def run_block(self, statements, env):
        for stmt in statements:
            self.execute(stmt, env)

    def run(self, program):
        try:
            self.run_block(program, self.globals)
        except ReturnSignal:
            raise TinyRuntimeError("return outside a function") from None
        return self.output

class Cursor:
    def __init__(self, source):
        self.source, self.pos, self.line, self.column = source, 0, 1, 1

    def at_end(self):
        return self.pos >= len(self.source)

    def peek(self, ahead=0):
        i = self.pos + ahead
        return self.source[i] if i < len(self.source) else ""

    def advance(self):
        ch = self.source[self.pos]
        self.pos += 1
        if ch == "\n":
            self.line, self.column = self.line + 1, 1
        else:
            self.column += 1
        return ch


def read_number(cursor):
    line, column, text = cursor.line, cursor.column, ""
    while cursor.peek().isdigit():
        text += cursor.advance()
    if cursor.peek() == "." and cursor.peek(1).isdigit():
        text += cursor.advance()
        while cursor.peek().isdigit():
            text += cursor.advance()
        return Token("NUMBER", text, float(text), line, column)
    return Token("NUMBER", text, int(text), line, column)


ESCAPES = {"n": "\n", "t": "\t", '"': '"', "\\": "\\"}


def read_string(cursor):
    line, column = cursor.line, cursor.column
    text = cursor.advance()
    value = ""
    while True:
        if cursor.at_end() or cursor.peek() == "\n":
            raise LexError("unterminated string", line, column)
        ch = cursor.advance()
        text += ch
        if ch == '"':
            return Token("STRING", text, value, line, column)
        if ch == "\\":
            if cursor.at_end():
                raise LexError("unterminated string", line, column)
            esc_line, esc_col = cursor.line, cursor.column - 1
            code = cursor.advance()
            text += code
            if code not in ESCAPES:
                raise LexError(f"unknown escape \\{code}", esc_line, esc_col)
            value += ESCAPES[code]
        else:
            value += ch


KEYWORDS = {"let", "fn", "return", "if", "else", "while", "print", "true", "false", "nil", "and", "or", "not"}


TWO_CHAR = {"==", "!=", "<=", ">="}


ONE_CHAR = set("+-*/%<>=(){},;")


def read_name(cursor):
    line, column, text = cursor.line, cursor.column, ""
    while cursor.peek().isalnum() or cursor.peek() == "_":
        text += cursor.advance()
    if text in KEYWORDS:
        value = {"true": True, "false": False, "nil": None}.get(text)
        return Token(text.upper(), text, value, line, column)
    return Token("NAME", text, text, line, column)


def read_operator(cursor):
    line, column = cursor.line, cursor.column
    pair = cursor.peek() + cursor.peek(1)
    if pair in TWO_CHAR:
        cursor.advance()
        cursor.advance()
        return Token("OP", pair, pair, line, column)
    ch = cursor.peek()
    if ch in ONE_CHAR:
        cursor.advance()
        return Token("OP", ch, ch, line, column)
    raise LexError(f"unexpected character {ch!r}", line, column)


def skip_space_and_comments(cursor):
    while not cursor.at_end():
        ch = cursor.peek()
        if ch in " \t\r\n":
            cursor.advance()
        elif ch == "#":
            while not cursor.at_end() and cursor.peek() != "\n":
                cursor.advance()
        else:
            return


def tokenize(source):
    cursor = Cursor(source)
    tokens = []
    while True:
        skip_space_and_comments(cursor)
        if cursor.at_end():
            tokens.append(Token("EOF", "", None, cursor.line, cursor.column))
            return tokens
        ch = cursor.peek()
        if ch.isdigit():
            tokens.append(read_number(cursor))
        elif ch == '"':
            tokens.append(read_string(cursor))
        elif ch.isalpha() or ch == "_":
            tokens.append(read_name(cursor))
        else:
            tokens.append(read_operator(cursor))

class Parser:
    def __init__(self, tokens):
        self.tokens, self.pos = tokens, 0

    def peek(self):
        return self.tokens[self.pos]

    def at(self, kind, text=None):
        tok = self.peek()
        return tok.kind == kind and (text is None or tok.text == text)

    def match(self, kind, text=None):
        if self.at(kind, text):
            tok = self.peek()
            self.pos += 1
            return tok
        return None

    def expect(self, kind, text=None, what=None):
        tok = self.match(kind, text)
        if tok is None:
            got = self.peek()
            shown = "end of input" if got.kind == "EOF" else repr(got.text)
            raise ParseError(f"expected {what or text or kind.lower()}, got {shown}", got.line, got.column)
        return tok

    # --- expressions, lowest precedence first ---
    def expression(self):
        return self.logic_or()

    def logic_or(self):
        node = self.logic_and()
        while self.at("OR"):
            line = self.match("OR").line
            node = Logical("or", node, self.logic_and(), line)
        return node

    def logic_and(self):
        node = self.equality()
        while self.at("AND"):
            line = self.match("AND").line
            node = Logical("and", node, self.equality(), line)
        return node

    def binary(self, operators, operand):
        node = operand()
        while self.peek().kind == "OP" and self.peek().text in operators:
            tok = self.match("OP")
            node = Binary(tok.text, node, operand(), tok.line)
        return node

    def equality(self):
        return self.binary({"==", "!="}, self.comparison)

    def comparison(self):
        return self.binary({"<", "<=", ">", ">="}, self.term)

    def term(self):
        return self.binary({"+", "-"}, self.factor)

    def factor(self):
        return self.binary({"*", "/", "%"}, self.unary)

    def unary(self):
        if self.at("OP", "-"):
            line = self.match("OP").line
            return Unary("-", self.unary(), line)
        if self.at("NOT"):
            line = self.match("NOT").line
            return Unary("not", self.unary(), line)
        return self.call()

    def call(self):
        node = self.primary()
        while self.at("OP", "("):
            line = self.match("OP").line
            args = []
            if not self.at("OP", ")"):
                args.append(self.expression())
                while self.match("OP", ","):
                    args.append(self.expression())
            self.expect("OP", ")")
            node = Call(node, args, line)
        return node

    def primary(self):
        tok = self.peek()
        if self.match("NUMBER"):
            return Number(tok.value, tok.line)
        if self.match("STRING"):
            return String(tok.value, tok.line)
        if tok.kind in ("TRUE", "FALSE", "NIL"):
            self.pos += 1
            return Literal(tok.value, tok.line)
        if self.match("NAME"):
            return Name(tok.text, tok.line)
        if self.match("OP", "("):
            node = self.expression()
            self.expect("OP", ")")
            return node
        shown = "end of input" if tok.kind == "EOF" else repr(tok.text)
        raise ParseError(f"expected an expression, got {shown}", tok.line, tok.column)

    # --- statements ---
    def statement(self):
        tok = self.peek()
        if self.match("LET"):
            name = self.expect("NAME", what="a name").text
            self.expect("OP", "=")
            value = self.expression()
            self.expect("OP", ";")
            return Let(name, value, tok.line)
        if self.match("PRINT"):
            value = self.expression()
            self.expect("OP", ";")
            return Print(value, tok.line)
        if self.at("OP", "{"):
            return Block(self.block(), tok.line)
        if self.match("IF"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            then = self.statement()
            otherwise = self.statement() if self.match("ELSE") else None
            return If(test, then, otherwise, tok.line)
        if self.match("WHILE"):
            self.expect("OP", "(")
            test = self.expression()
            self.expect("OP", ")")
            return While(test, self.statement(), tok.line)
        if self.match("FN"):
            name = self.expect("NAME", what="a function name").text
            self.expect("OP", "(")
            params = []
            if not self.at("OP", ")"):
                params.append(self.expect("NAME", what="a parameter name").text)
                while self.match("OP", ","):
                    params.append(self.expect("NAME", what="a parameter name").text)
            self.expect("OP", ")")
            return Function(name, params, self.block(), tok.line)
        if self.match("RETURN"):
            value = None if self.at("OP", ";") else self.expression()
            self.expect("OP", ";")
            return Return(value, tok.line)
        if tok.kind == "NAME" and self.tokens[self.pos + 1].kind == "OP" and self.tokens[self.pos + 1].text == "=":
            self.pos += 2
            value = self.expression()
            self.expect("OP", ";")
            return Assign(tok.text, value, tok.line)
        value = self.expression()
        self.expect("OP", ";")
        return ExprStmt(value, tok.line)

    def block(self):
        self.expect("OP", "{")
        body = []
        while not self.at("OP", "}"):
            if self.at("EOF"):
                tok = self.peek()
                raise ParseError("expected '}', got end of input", tok.line, tok.column)
            body.append(self.statement())
        self.expect("OP", "}")
        return body

    def program(self):
        body = []
        while not self.at("EOF"):
            body.append(self.statement())
        return body


def parse(source):
    return Parser(tokenize(source)).program()

class Environment:
    def __init__(self, parent=None):
        self.values, self.parent = {}, parent

    def define(self, name, value):
        self.values[name] = value

    def find(self, name):
        env = self
        while env is not None:
            if name in env.values:
                return env
            env = env.parent
        return None

    def get(self, name, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"undefined name {name!r}", line)
        return env.values[name]

    def assign(self, name, value, line=None):
        env = self.find(name)
        if env is None:
            raise TinyRuntimeError(f"cannot assign to undefined name {name!r}", line)
        env.values[name] = value

        return "nil"
    if value is True:
        return "true"
    if value is False:
        return "false"
    if isinstance(value, float) and value.is_integer():
        return str(int(value)) if abs(value) < 1e16 else str(value)
    return str(value)


def type_name(value):
    if value is None:
        return "nil"
    if isinstance(value, bool):
        return "bool"
    if isinstance(value, (int, float)):
        return "number"
    if isinstance(value, str):
        return "string"
    return "function"


def is_number(value):
    return isinstance(value, (int, float)) and not isinstance(value, bool)


class Interpreter:
    def __init__(self, output=None, max_depth=100):
        self.output = [] if output is None else output
        self.globals = Environment()
        self.max_depth, self.depth = max_depth, 0
        self.steps = 0
        for name, arity, func in [
            ("len", 1, self._len), ("str", 1, lambda v: show(v)), ("type", 1, type_name),
        ]:
            self.globals.define(name, Builtin(name, arity, func))

    @staticmethod
    def _len(value):
        if not isinstance(value, str):
            raise TinyRuntimeError(f"len() needs a string, not {type_name(value)}")
        return len(value)

    # --- expressions ---
    def evaluate(self, node, env):
        self.steps += 1
        if isinstance(node, (Number, String, Literal)):
            return node.value
        if isinstance(node, Name):
            return env.get(node.name, node.line)
        if isinstance(node, Unary):
            value = self.evaluate(node.operand, env)
            if node.op == "-":
                if not is_number(value):
                    raise TinyRuntimeError(f"cannot negate {type_name(value)}", node.line)
                return -value
            return not truthy(value)
        if isinstance(node, Logical):
            left = self.evaluate(node.left, env)
            if node.op == "or":
                return left if truthy(left) else self.evaluate(node.right, env)
            return left if not truthy(left) else self.evaluate(node.right, env)
        if isinstance(node, Binary):
            return self.binary(node, self.evaluate(node.left, env), self.evaluate(node.right, env))
        if isinstance(node, Call):
            return self.call(node, env)
        raise TinyRuntimeError(f"cannot evaluate {type(node).__name__}", getattr(node, "line", None))

    def binary(self, node, left, right):
        op = node.op
        if op == "==":
            return type_name(left) == type_name(right) and left == right
        if op == "!=":
            return not (type_name(left) == type_name(right) and left == right)
        if op == "+" and isinstance(left, str) and isinstance(right, str):
            return left + right
        if not (is_number(left) and is_number(right)):
            raise TinyRuntimeError(
                f"cannot use {op} with {type_name(left)} and {type_name(right)}", node.line)
        if op in ("/", "%") and right == 0:
            raise TinyRuntimeError("division by zero", node.line)
        if op == "/":
            result = left / right
            return int(result) if isinstance(left, int) and isinstance(right, int) and result.is_integer() else result
        return {"+": lambda: left + right, "-": lambda: left - right, "*": lambda: left * right,
                "%": lambda: left % right, "<": lambda: left < right, "<=": lambda: left <= right,
                ">": lambda: left > right, ">=": lambda: left >= right}[op]()

    def call(self, node, env):
        callee = self.evaluate(node.callee, env)
        args = [self.evaluate(a, env) for a in node.args]
        if isinstance(callee, Builtin):
            if len(args) != callee.arity:
                raise TinyRuntimeError(f"{callee.name}() takes {callee.arity} argument(s), got {len(args)}", node.line)
            try:
                return callee.func(*args)
            except TinyRuntimeError as e:
                raise TinyRuntimeError(e.message, node.line) from None
        if not isinstance(callee, TinyFunction):
            raise TinyRuntimeError(f"cannot call {type_name(callee)}", node.line)
        decl = callee.decl
        if len(args) != len(decl.params):
            raise TinyRuntimeError(f"{decl.name}() takes {len(decl.params)} argument(s), got {len(args)}", node.line)
        if self.depth >= self.max_depth:
            raise TinyRuntimeError("too much recursion", node.line)
        local = Environment(callee.closure)
        for name, value in zip(decl.params, args):
            local.define(name, value)
        self.depth += 1
        try:
            self.run_block(decl.body, local)
        except ReturnSignal as signal:
            return signal.value
        finally:
            self.depth -= 1
        return None

    # --- statements ---
    def execute(self, node, env):
        self.steps += 1
        if isinstance(node, Let):
            env.define(node.name, self.evaluate(node.value, env))
        elif isinstance(node, Assign):
            env.assign(node.name, self.evaluate(node.value, env), node.line)
        elif isinstance(node, Print):
            self.output.append(show(self.evaluate(node.value, env)))
        elif isinstance(node, ExprStmt):
            self.evaluate(node.value, env)
        elif isinstance(node, Block):
            self.run_block(node.body, Environment(env))
        elif isinstance(node, If):
            if truthy(self.evaluate(node.test, env)):
                self.execute(node.then, env)
            elif node.otherwise is not None:
                self.execute(node.otherwise, env)
        elif isinstance(node, While):
            while truthy(self.evaluate(node.test, env)):
                self.execute(node.body, env)
        elif isinstance(node, Function):
            env.define(node.name, TinyFunction(node, env))
        elif isinstance(node, Return):
            raise ReturnSignal(None if node.value is None else self.evaluate(node.value, env))
        else:
            raise TinyRuntimeError(f"cannot execute {type(node).__name__}", getattr(node, "line", None))

    def run_block(self, statements, env):
        for stmt in statements:
            self.execute(stmt, env)

    def run(self, program):
        try:
            self.run_block(program, self.globals)
        except ReturnSignal:
            raise TinyRuntimeError("return outside a function") from None
        return self.output

    if error.line is None:
        return str(error)
    text = source.split("\n")[error.line - 1]

def check(cases):
    failures = []
    for name, src, expected in cases:
        try:
            got = run_source(src)
        except TinyError as e:
            got = "error: " + str(e)
        if got != expected:
            failures.append((name, expected, got))
    return failures


def report(failures):
    if not failures:
        return "all passed"
    lines = []
    for name, expected, got in failures:
        lines.append(f"FAIL {name}: expected {expected!r}, got {got!r}")
    return "\n".join(lines)

class CountingInterpreter(Interpreter):
    def __init__(self):
        super().__init__()
        self.counts = Counter()

    def evaluate(self, node, env):
        self.counts[type(node).__name__] += 1
        return super().evaluate(node, env)

    def execute(self, node, env):
        self.counts[type(node).__name__] += 1
        return super().execute(node, env)


def node_counts(source):
    interp = CountingInterpreter()
    interp.run(parse(source))
    return interp.counts


def hottest(source, n=3):
    counts = node_counts(source)
    return sorted(counts.items(), key=lambda kv: (-kv[1], kv[0]))[:n]


def python_calls(source, method_name):
    profiler = cProfile.Profile()
    program = parse(source)
    interp = Interpreter()
    profiler.enable()
    try:
        interp.run(program)
    finally:
        profiler.disable()
    total = 0
    for (file, line, name), (primitive, calls, *rest) in pstats.Stats(profiler).stats.items():
        if name == method_name:
            total += calls
    return total

def fold(node):
    if isinstance(node, list):
        return [fold(n) for n in node]
    if not dataclasses.is_dataclass(node):
        return node
    changes = {f.name: fold(getattr(node, f.name)) for f in dataclasses.fields(node) if f.name != "line"}
    node = dataclasses.replace(node, **changes)
    if isinstance(node, Unary) and node.op == "-" and isinstance(node.operand, Number):
        return Number(-node.operand.value, node.line)
    if isinstance(node, Binary) and node.op in FOLDABLE:
        left, right = node.left, node.right
        if isinstance(left, Number) and isinstance(right, Number):
            value = Interpreter().binary(node, left.value, right.value)
            return Literal(value, node.line) if isinstance(value, bool) else Number(value, node.line)
        if node.op == "+" and isinstance(left, String) and isinstance(right, String):
            return String(left.value + right.value, node.line)
    return node


def steps_saved(source):
    program = parse(source)
    plain, folded = Interpreter(), Interpreter()
    out_plain = plain.run(program)
    out_folded = folded.run(fold(program))
    assert out_plain == out_folded
    return plain.steps - folded.steps


cases = [("fib", 'fn fib(n) { if (n < 2) return n; return fib(n - 1) + fib(n - 2); } print fib(10);', ["55"]),
         ("strings", 'print "ab" + "cd";', ["abcd"]),
         ("error", "print undefined;", "error: line 1: undefined name 'undefined'")]
print(report(check(cases)))
print(hottest('fn fib(n) { if (n < 2) return n; return fib(n - 1) + fib(n - 2); } print fib(10);', 4))
print("folding saves", steps_saved("let i = 0; while (i < 5) { print 60 * 60 * 24 * i; i = i + 1; }"), "steps")
:::

:::think title="Tests make optimisation safe"
Step 24 changed the program the interpreter runs, and the only reason you can be confident it's correct is that Steps 21 and 22 gave you a way to check. That's the general rule: performance work without tests is guesswork, because an optimisation that changes behaviour in a corner case (folding a division by zero, say, or changing the order of an error) is a bug, however fast it is. Real compilers have test suites of tens of thousands of programs for exactly this reason.
:::
