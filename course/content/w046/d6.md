class TreeNode:
    def __init__(self, val, left=None, right=None):
        self.val, self.left, self.right = val, left, right

def min_gap(root):
    best, prev, stack, node = float("inf"), None, [], root
    while stack or node is not None:
        while node is not None:
            stack.append(node)
            node = node.left
        node = stack.pop()
        if prev is not None:
            best = min(best, node.val - prev)
        prev = node.val
        node = node.right
    return best

class TreeNode:
    def __init__(self, val, left=None, right=None):
        self.val, self.left, self.right = val, left, right

class InOrder:
    def __init__(self, root):
        self._stack = []
        self._push(root)

    def _push(self, node):
        while node is not None:
            self._stack.append(node)
            node = node.left

    def __iter__(self):
        return self

    def __next__(self):
        if not self._stack:
            raise StopIteration
        node = self._stack.pop()
        self._push(node.right)
        return node.val

def same_values(a, b):
    done = object()
    ia, ib = InOrder(a), InOrder(b)
    while True:
        x, y = next(ia, done), next(ib, done)
        if x is done or y is done:
            return x is y
        if x != y:
            return False

## Forge: a booking calendar

A meeting room's calendar holds bookings, each a half-open interval of whole minutes `[start, end)` with a name: a booking from 600 to 660 and another from 660 to 720 don't clash. Bookings never overlap, so ordering them by start time orders them completely, and a balanced BST keyed by start answers every question with a floor or successor search. Use an **AVL tree** (rotations, heights in the provided `Node`), so the calendar stays fast however bookings arrive; keep a single attribute holding the root.

Build `Calendar()` with:

- `book(start, end, name)`: add the booking and return `True`, or return `False` (changing nothing) if it would overlap an existing booking. Raise `ValueError` if `start >= end`. Only the bookings just before and just after `start` can clash: check those two, not every booking.
- `cancel(start)`: remove the booking that starts at `start` and return its name; `KeyError` if none does.
- `at(t)`: the name of the booking covering minute `t`, or `None`.
- `between(t1, t2)`: every booking overlapping `[t1, t2)`, as sorted `(start, end, name)` tuples, in O(h + k): only the booking found by a floor search on `t1` can start before `t1` and still overlap.
- `next_free(t, length)`: the earliest minute `s >= t` such that `[s, s + length)` overlaps no booking.
- `bookings()`, `len(c)` and `height()` (−1 when empty).
- `save()`: JSON text `{"format": "calendar", "version": 1, "bookings": [[start, end, name], ...]}` in start order, and a class method `Calendar.load(text)` that rebuilds a calendar, raising `ValueError` (only) for anything wrong: invalid JSON, a wrong format name or version, bookings that aren't a list of three-item lists, times that aren't whole numbers (`bool` doesn't count), names that aren't strings, `start >= end`, or overlapping bookings.

Write `_floor(t)` (the booking with the largest start ≤ t) and `_first_from(t)` (the smallest start ≥ t) as node-returning loops; nearly everything else is built from them. `book` clashes if the floor booking ends after `start`, or the next booking (smallest start greater than `start`) begins before `end`. import json

class Node:
    def __init__(self, start, end, name):
        self.start, self.end, self.name = start, end, name
        self.left = self.right = None
        self.height = 0

def height(n):
    return n.height if n else -1

def update(n):
    n.height = 1 + max(height(n.left), height(n.right))

def balance(n):
    return height(n.left) - height(n.right)

def rotate_right(y):
    x = y.left
    y.left, x.right = x.right, y
    update(y)
    update(x)
    return x

def rotate_left(x):
    y = x.right
    x.right, y.left = y.left, x
    update(x)
    update(y)
    return y

def rebalance(n):
    update(n)
    if balance(n) > 1:
        if balance(n.left) < 0:
            n.left = rotate_left(n.left)
        return rotate_right(n)
    if balance(n) < -1:
        if balance(n.right) > 0:
            n.right = rotate_right(n.right)
        return rotate_left(n)
    return n

class Calendar:
    def __init__(self):
        self._root, self._size = None, 0

    def __len__(self):
        return self._size

    def height(self):
        return height(self._root)

    def _floor(self, t):
        """The booking with the largest start <= t, or None."""
        best, node = None, self._root
        while node is not None:
            if node.start <= t:
                best, node = node, node.right
            else:
                node = node.left
        return best

    def _after(self, t):
        """The booking with the smallest start > t, or None."""
        best, node = None, self._root
        while node is not None:
            if node.start > t:
                best, node = node, node.left
            else:
                node = node.right
        return best

    def book(self, start, end, name):
        if not start < end:
            raise ValueError("a booking must end after it starts")
        before = self._floor(start)
        if before is not None and before.end > start:
            return False
        after = self._after(start)
        if after is not None and after.start < end:
            return False
        self._root = self._insert(self._root, Node(start, end, name))
        self._size += 1
        return True

    def _insert(self, n, new):
        if n is None:
            return new
        if new.start < n.start:
            n.left = self._insert(n.left, new)
        else:
            n.right = self._insert(n.right, new)
        return rebalance(n)

    def cancel(self, start):
        found = self._floor(start)
        if found is None or found.start != start:
            raise KeyError(start)
        name = found.name
        self._root = self._delete(self._root, start)
        self._size -= 1
        return name

    def _delete(self, n, start):
        if start < n.start:
            n.left = self._delete(n.left, start)
        elif start > n.start:
            n.right = self._delete(n.right, start)
        else:
            if n.left is None:
                return n.right
            if n.right is None:
                return n.left
            succ = n.right
            while succ.left is not None:
                succ = succ.left
            n.start, n.end, n.name = succ.start, succ.end, succ.name
            n.right = self._delete(n.right, succ.start)
        return rebalance(n)

    def at(self, t):
        b = self._floor(t)
        return b.name if b is not None and t < b.end else None

    def between(self, t1, t2):
        out = []
        first = self._floor(t1)
        lo = first.start if first is not None else t1
        stack, node = [], self._root
        while stack or node is not None:
            while node is not None:
                stack.append(node)
                node = node.left if node.start > lo else None
            node = stack.pop()
            if lo <= node.start < t2 and node.end > t1:
                out.append((node.start, node.end, node.name))
            node = node.right if node.start < t2 else None
        return out

    def next_free(self, t, length):
        b = self._floor(t)
        if b is not None and b.end > t:
            t = b.end
        while True:
            nxt = self._first_from(t)
            if nxt is None or nxt.start - t >= length:
                return t
            t = max(t, nxt.end)

    def _first_from(self, t):
        best, node = None, self._root
        while node is not None:
            if node.start >= t:
                best, node = node, node.left
            else:
                node = node.right
        return best

    def bookings(self):
        out, stack, node = [], [], self._root
        while stack or node is not None:
            while node is not None:
                stack.append(node)
                node = node.left
            node = stack.pop()
            out.append((node.start, node.end, node.name))
            node = node.right
        return out

    def save(self):

:::think title="Invariants"
The calendar has two rules that must hold after every operation: **no two bookings overlap**, and **the tree is an AVL tree** (ordered by start, balanced, heights correct). The first is the calendar's own rule; the second is how it stays fast. Both are **invariants**, things that are true before and after every public method, though they may be briefly false in the middle of one (half-way through a rotation, the heights are wrong). The checks today test them directly, by walking your tree. When you write a data structure, write a `check()` method that verifies every invariant, and call it in your tests after each change: it finds bugs at the operation that caused them, not three hundred operations later when an answer comes out wrong. And notice where the overlap rule is enforced: in `book`, and again in `load`, because a file is just another way for data to arrive.

## Forge bonus: a treap

:::exercise id=forge2 points=30 title="Forge bonus: a treap"
A **treap** (Raimund Seidel and Cecilia Aragon, 1989) balances itself by chance instead of by height. Each node gets a random **priority** when it's created, and the tree keeps two orders at once: it's a **BST by key**, and a **heap by priority** (every node's priority is at least as high as its children's, so the highest priority is the root). For given priorities, only one tree satisfies both, and it's the BST you'd get by inserting keys in decreasing priority order: random priorities make it a random BST (height about 3 log₂ n, Day 1) whatever order the keys actually arrive in.

Write `Treap` (the constructor is given: priorities come from `self._priority(key)`, random unless the checks pass their own function):

- `add(key)`: insert as a leaf, as in a plain BST; then, on the way back up, while the new node's priority is higher than its parent's, rotate it up (a right rotation if it's a left child, a left one if it's a right child). With the subtree-returning style, that's one check after each recursive call. No heights to maintain.
- `discard(key)`: find the node and rotate it **down**, each time lifting whichever child has the higher priority (so the heap order holds), until it has at most one child; then remove it as in Day 2.

import random

class TNode:
    def __init__(self, key, priority):
        self.key, self.priority = key, priority
        self.left = self.right = None

def rotate_right(y):
    x = y.left
    y.left, x.right = x.right, y
    return x

def rotate_left(x):
    y = x.right
    x.right, y.left = y.left, x
    return y

class Treap:
    def __init__(self, seed=None, priority=None):
        rng = random.Random(seed)
        self._priority = priority if priority is not None else (lambda key: rng.random())
        self._root, self._size = None, 0

    def __len__(self):
        return self._size

    def __contains__(self, key):
        node = self._root
        while node is not None and node.key != key:
            node = node.left if key < node.key else node.right
        return node is not None

    def add(self, key):
        if key not in self:
            self._root = self._insert(self._root, key)
            self._size += 1

    def _insert(self, n, key):
        if n is None:
            return TNode(key, self._priority(key))
        if key < n.key:
            n.left = self._insert(n.left, key)
            if n.left.priority > n.priority:
                n = rotate_right(n)
        else:
            n.right = self._insert(n.right, key)
            if n.right.priority > n.priority:
                n = rotate_left(n)
        return n

    def discard(self, key):
        if key in self:
            self._root = self._delete(self._root, key)
            self._size -= 1

    def _delete(self, n, key):
        if key < n.key:
            n.left = self._delete(n.left, key)
        elif key > n.key:
            n.right = self._delete(n.right, key)
        else:
            if n.left is None:
                return n.right
            if n.right is None:
                return n.left
            if n.left.priority > n.right.priority:
                n = rotate_right(n)
                n.right = self._delete(n.right, key)
            else:
                n = rotate_left(n)
                n.left = self._delete(n.left, key)
        return n

    def height(self):
        best, stack = -1, [(self._root, 0)] if self._root else []
        while stack:
            n, d = stack.pop()
            best = max(best, d)
            stack.extend((c, d + 1) for c in (n.left, n.right) if c)
        return best

    def to_list(self):
        out, stack, node = [], [], self._root
        while stack or node is not None:
            while node is not None:
                stack.append(node)
                node = node.left
            node = stack.pop()
            out.append(node.key)
            node = node.right
        return out
