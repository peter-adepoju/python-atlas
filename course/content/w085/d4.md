from urllib.parse import parse_qs, urlencode, urlsplit, urlunsplit

def with_params(url, **changes):
    parts = urlsplit(url)
    params = parse_qs(parts.query)
    for name, value in changes.items():
        if value is None:
            params.pop(name, None)
        else:
            params[name] = value if isinstance(value, list) else [value]
    query = urlencode(sorted(params.items()), doseq=True)
    return urlunsplit((parts.scheme, parts.netloc, parts.path, query, parts.fragment))

def get_param(url, name, default=None):
    values = parse_qs(urlsplit(url).query).get(name)
    return values[0] if values else default

import sqlite3
from datetime import date, timedelta

def make_db():
    conn = sqlite3.connect(":memory:")
    conn.executescript('''
    CREATE TABLE books (id INTEGER PRIMARY KEY, title TEXT);
    CREATE TABLE loans (id INTEGER PRIMARY KEY, book_id INTEGER, borrower TEXT, due TEXT, returned TEXT);
    INSERT INTO books VALUES (1, 'Dune'), (2, 'Emma'), (3, 'Ulysses');
    ''')
    return conn

def borrow(conn, book_id, borrower, today):
    with conn:
        open_loan = conn.execute("SELECT 1 FROM loans WHERE book_id = ? AND returned IS NULL", (book_id,)).fetchone()
        if open_loan:
            raise ValueError("on loan")
        due = (date.fromisoformat(today) + timedelta(days=14)).isoformat()
        conn.execute("INSERT INTO loans (book_id, borrower, due) VALUES (?, ?, ?)", (book_id, borrower, due))

def give_back(conn, book_id, today):
    with conn:
        cur = conn.execute("UPDATE loans SET returned = ? WHERE book_id = ? AND returned IS NULL", (today, book_id))
        if cur.rowcount == 0:
            raise ValueError("not on loan")

def overdue(conn, today):
    rows = conn.execute('''
        SELECT b.title, l.borrower, l.due FROM loans l JOIN books b ON b.id = l.book_id
        WHERE l.returned IS NULL AND l.due < ?
    ''', (today,)).fetchall()
    late = [(t, who, (date.fromisoformat(today) - date.fromisoformat(due)).days) for t, who, due in rows]
    return sorted(late, key=lambda r: (-r[2], r[0]))

import pandas as pd

def monthly(df):
    g = (df.assign(month=df["date"].str[:7])
           .groupby(["city", "month"])
           .agg(mean_temp=("temp", "mean"), total_rain=("rain", "sum"))
           .round(2).reset_index())
    return [(c, m, float(t), float(r)) for c, m, t, r in g.itertuples(index=False)]

def filled(df):
    city_mean = df.groupby("city")["temp"].transform("mean").round(2)
    return df.assign(temp=df["temp"].fillna(city_mean), rain=df["rain"].fillna(0))

def wettest(df):
    row = df.loc[df["rain"].idxmax()]
    return row["date"], row["city"]

@@solution
class TokenBucket:
    def __init__(self, rate, capacity, clock):
        self.rate, self.capacity, self.clock = rate, capacity, clock
        self.tokens = float(capacity)
        self.updated = clock()

    def _refill(self):
        now = self.clock()
        self.tokens = min(self.capacity, self.tokens + (now - self.updated) * self.rate)
        self.updated = now

    def allow(self, cost=1):
        self._refill()
        if self.tokens >= cost:
            self.tokens -= cost
            return True
        return False

    def retry_after(self, cost=1):
        self._refill()
        return round(max(0.0, (cost - self.tokens) / self.rate), 2)

class Limiter:
    def __init__(self, rate, capacity, clock):
        self.rate, self.capacity, self.clock = rate, capacity, clock
        self.buckets = {}

    def check(self, client):
        bucket = self.buckets.setdefault(client, TokenBucket(self.rate, self.capacity, self.clock))
        if bucket.allow():
            return 200, None
        return 429, bucket.retry_after()

Base 62: repeatedly `divmod(n, 62)`, collecting digits, then reverse. To decode, `n = n * 62 + ALPHABET.index(ch)` for each character. For the limit, count the user's links with `created > clock() - 60`.
@@solution
import json
import sqlite3

ALPHABET = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ"

def encode(n):
    digits = ""
    while True:
        n, r = divmod(n, 62)
        digits = ALPHABET[r] + digits
        if n == 0:
            return digits

def decode(code):
    n = 0
    for ch in code:
        if ch not in ALPHABET:
            return None
        n = n * 62 + ALPHABET.index(ch)
    return n

class Shortener:
    def __init__(self, clock, per_minute=5):
        self.clock, self.per_minute = clock, per_minute
        self.conn = sqlite3.connect(":memory:")
        self.conn.execute("CREATE TABLE links (id INTEGER PRIMARY KEY, url TEXT, user TEXT, created REAL, clicks INTEGER)")

    def _id(self, code):
        n = decode(code) if code else None
        return None if n is None else n - 10_000

    def handle(self, method, path, body):
        parts = path.strip("/").split("/")
        if method == "POST" and path == "/links":
            return self._create(body)
        if method == "GET" and len(parts) == 2 and parts[0] == "stats":
            return self._stats(parts[1])
        if method == "GET" and len(parts) == 1 and parts[0]:
            return self._follow(parts[0])
        return 404, {"error": "not found"}

    def _create(self, body):
        try:
            data = json.loads(body)
        except ValueError:
            return 400, {"error": "bad request"}
        url, user = (data.get("url"), data.get("user")) if isinstance(data, dict) else (None, None)
        if not (isinstance(url, str) and isinstance(user, str) and url.startswith(("http://", "https://"))):
            return 400, {"error": "bad request"}
        existing = self.conn.execute("SELECT id FROM links WHERE url = ? AND user = ?", (url, user)).fetchone()
        if existing:
            return 200, {"code": encode(existing[0] + 10_000)}
        now = self.clock()
        recent = self.conn.execute("SELECT COUNT(*) FROM links WHERE user = ? AND created > ?", (user, now - 60)).fetchone()[0]
        if recent >= self.per_minute:
            return 429, {"error": "slow down"}
        with self.conn:
            cur = self.conn.execute("INSERT INTO links (url, user, created, clicks) VALUES (?, ?, ?, 0)", (url, user, now))
        return 201, {"code": encode(cur.lastrowid + 10_000)}

    def _follow(self, code):
        link_id = self._id(code)
        row = self.conn.execute("SELECT url FROM links WHERE id = ?", (link_id,)).fetchone() if link_id else None
        if row is None:
            return 404, {"error": "not found"}
        with self.conn:
            self.conn.execute("UPDATE links SET clicks = clicks + 1 WHERE id = ?", (link_id,))
        return 302, {"location": row[0]}

    def _stats(self, code):
        link_id = self._id(code)
        row = self.conn.execute("SELECT url, clicks, user FROM links WHERE id = ?", (link_id,)).fetchone() if link_id else None
        if row is None:
            return 404, {"error": "not found"}
        return 200, {"url": row[0], "clicks": row[1], "user": row[2]}
:::
