## Warm-up

:::predict
import bisect

tails = []
for x in [3, 1, 4, 1, 5, 9, 2, 6]:
    i = bisect.bisect_left(tails, x)
    tails[i:i + 1] = [x]
print(tails, len(tails))
:::

`tails[i]` is the smallest possible last value of an increasing run of length `i + 1` seen so far. Its length is the length of the longest increasing subsequence (here 4, for example 1, 4, 5, 9), found with one binary search per number. The list itself isn't a subsequence of the input; only its length means anything.

## Harder patterns

- "How many at once?" (rooms, servers, overlapping walks): sort by start and keep a **min-heap of end times**; the heap's size at its largest is the answer.
- "All combinations or arrangements": **backtracking**: choose, recurse, un-choose. Avoid duplicates by only choosing candidates from the current index onwards.
- "Longest increasing...": the `tails` trick above, O(n log n), instead of the O(n²) table.
- "Fewest edits from one string to another": a table `d[i][j]` of the edit distance between prefixes, filled from the top left: `d[i][j] = d[i-1][j-1]` if the characters match, otherwise 1 plus the smallest of the three neighbours. This is Levenshtein distance, after Vladimir Levenshtein.
- "Shortest time through a weighted network": **Dijkstra's algorithm** with a heap (Phase 4's Week 49).

@@solution
import heapq

def rooms_needed(meetings):
    ends = []
    for start, end in sorted(meetings):
        if ends and ends[0] <= start:
            heapq.heapreplace(ends, end)
        else:
            heapq.heappush(ends, end)
    return len(ends)

@@solution
def combinations_to(candidates, target):
    candidates = sorted(candidates)
    found = []

    def explore(start, remaining, chosen):
        if remaining == 0:
            found.append(list(chosen))
            return
        for i in range(start, len(candidates)):
            c = candidates[i]
            if c > remaining:
                break
            chosen.append(c)
            explore(i, remaining - c, chosen)
            chosen.pop()

    explore(0, target, [])
    return sorted(found)

def lis_length(nums):
    tails = []
    for x in nums:
        i = bisect.bisect_left(tails, x)

        else:

@@solution
def edit_distance(a, b):
    previous = list(range(len(b) + 1))
    for i, ca in enumerate(a, start=1):
        current = [i]
        for j, cb in enumerate(b, start=1):
            if ca == cb:
                current.append(previous[j - 1])
            else:
                current.append(1 + min(previous[j - 1], previous[j], current[j - 1]))
        previous = current
    return previous[-1]

Adjacency lists, a heap of `(time, node)`, and a `done` set; the answer is the largest finished time, if all `n` nodes finished.
@@solution
import heapq

def network_delay(times, n, start):
    links = {}
    for a, b, d in times:
        links.setdefault(a, []).append((b, d))
    best, heap, done = {}, [(0, start)], set()
    while heap:
        t, node = heapq.heappop(heap)
        if node in done:
            continue
        done.add(node)
        best[node] = t
        for nxt, d in links.get(node, ()):
            if nxt not in done:
                heapq.heappush(heap, (t + d, nxt))
    return max(best.values()) if len(done) == n else -1

import random, time

def test_trapped():
    "water level is the lower of the tallest walls on each side"
    assert trapped([0, 1, 0, 2, 1, 0, 1, 3, 2, 1, 2, 1]) == 6 and trapped([4, 2, 0, 3, 2, 5]) == 9
    assert trapped([]) == 0 and trapped([3]) == 0 and trapped([1, 2, 3]) == 0 and trapped([3, 0, 3]) == 3

def test_fast():
    "200,000 walls"
    rng = random.Random(2)
    h = [rng.randint(0, 1000) for _ in range(200_000)]
    start = time.perf_counter()
    left = right = 0
    expected = 0
    prefix = []
    for x in h:
        left = max(left, x)
        prefix.append(left)
    for i in range(len(h) - 1, -1, -1):
        right = max(right, h[i])
        expected += min(prefix[i], right) - h[i]
    start = time.perf_counter()
    assert trapped(h) == expected and time.perf_counter() - start < 1.5
@@hint
Two pointers from both ends with the tallest wall seen from each side. Move the side whose tallest wall is lower: its water level is decided.
@@solution
def trapped(heights):
    lo, hi = 0, len(heights) - 1
    left_max = right_max = water = 0
    while lo < hi:
        if heights[lo] < heights[hi]:
            left_max = max(left_max, heights[lo])
            water += left_max - heights[lo]
            lo += 1
        else:
            right_max = max(right_max, heights[hi])
            water += right_max - heights[hi]
            hi -= 1
    return water
:::
