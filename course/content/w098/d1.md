class ListNode:
    def __init__(self, value, next=None):
        self.value, self.next = value, next


class TreeNode:
    def __init__(self, value, left=None, right=None):
        self.value, self.left, self.right = value, left, right


def make_list(values):
    head = None
    for v in reversed(values):
        head = ListNode(v, head)
    return head


def to_list(head, limit=10_000):
    out = []
    while head is not None and len(out) < limit:
        out.append(head.value)
        head = head.next
    return out


def make_tree(values):
    """Build a tree from level order, with None for missing children."""
    if not values or values[0] is None:
        return None
    root = TreeNode(values[0])
    queue, i = [root], 1
    while queue and i < len(values):
        node = queue.pop(0)
        for side in ("left", "right"):
            if i < len(values) and values[i] is not None:
                child = TreeNode(values[i])
                setattr(node, side, child)
                queue.append(child)
            i += 1
    return root

`make_tree` reads a tree in **level order**, with `None` for a missing child: 5 at the root, 3 and 8 below it, then 3's children (none on the left, 4 on the right). That's how interview problems usually write trees, so it's worth being able to read it at a glance.

## Pointers and trees

**Linked lists**: draw the boxes and arrows before coding, and name every reference you need (`previous`, `current`, `following`). Most bugs are a lost `next` because it was overwritten before it was saved. Two pointers moving at different speeds (Robert Floyd's "tortoise and hare") find the middle of a list or a cycle in O(1) extra space.

**Trees**: depth-first (recursive, or a stack) for questions about paths and depths; breadth-first with a queue for questions about levels. A **binary search tree** keeps every value in the left subtree smaller and every value in the right larger, which is a property of the whole subtree, not just a node's children: pass the allowed **bounds** down as you go.

Recursion depth: a tree of 100,000 nodes in a line is 100,000 calls deep. Say so in the interview, and use an explicit stack when it matters.

@@solution
from nodes import ListNode

def reverse(head):
    previous, current = None, head
    while current is not None:
        following = current.next
        current.next = previous
        previous, current = current, following
    return previous

def test_cycle():

@@solution
def cycle_start(head):
    slow = fast = head
    while fast is not None and fast.next is not None:
        slow, fast = slow.next, fast.next.next
        if slow is fast:
            slow = head
            while slow is not fast:
                slow, fast = slow.next, fast.next
            return slow
    return None

from collections import deque

def levels(root):
    if root is None:
        return []
    out, queue = [], deque([root])
    while queue:
        level = []
        for _ in range(len(queue)):
            node = queue.popleft()
            level.append(node.value)
            queue.extend(child for child in (node.left, node.right) if child is not None)
        out.append(level)
    return out

Use an explicit stack of `(node, low, high)`: each node's value must lie strictly between its bounds; its left child gets `(low, value)` and its right child `(value, high)`.
@@solution
def is_bst(root):
    stack = [(root, float("-inf"), float("inf"))]
    while stack:
        node, low, high = stack.pop()
        if node is None:
            continue
        if not low < node.value < high:
            return False
        stack.append((node.left, low, node.value))
        stack.append((node.right, node.value, high))
    return True

def common_ancestor(root, a, b):
    node = root
    while node is not None:
        if a < node.value and b < node.value:
            node = node.left
        elif a > node.value and b > node.value:
            node = node.right
        else:
            return node.value
    return None

:::bughunt id=bug2 points=10 title="Checking only the children"
Each node here is larger than its left child and smaller than its right one, but 6 sits in 5's left subtree, so this isn't a search tree; checking only children prints `True`. Check bounds for whole subtrees so it prints `False`.
@@starter
from nodes import make_tree

def is_bst(node):
    if node is None:
        return True
    if node.left and node.left.value >= node.value:
        return False
    if node.right and node.right.value <= node.value:
        return False
    return is_bst(node.left) and is_bst(node.right)

from nodes import make_tree

def is_bst(node, low=float("-inf"), high=float("inf")):

        return True
    if not low < node.value < high:
        return False

from nodes import TreeNode, make_tree

def same(a, b):
    stack = [(a, b)]
    while stack:
        x, y = stack.pop()
        if x is None or y is None:
            if x is not y:
                return False
            continue
        if x.value != y.value:
            return False
        stack += [(x.left, y.left), (x.right, y.right)]
    return True

def test_round_trip():
    "the same tree back"
    for values in ([1, 2, 3, None, None, 4, 5], [], [7], [1, None, 2, None, 3], [-5, 10, -20]):
        tree = make_tree(values)
        text = serialise(tree)
        assert isinstance(text, str) and same(deserialise(text), tree), values

    "a long chain"

        node = node.left

@@hint
Level order works well: write values breadth first with `#` for a missing child, separated by commas; rebuild with a queue, reading two children for each node.
@@solution
from collections import deque
from nodes import TreeNode

def serialise(root):
    out, queue = [], deque([root])
    while queue:
        node = queue.popleft()
        if node is None:
            out.append("#")
        else:
            out.append(str(node.value))
            queue.append(node.left)
            queue.append(node.right)
    return ",".join(out)

def deserialise(text):
    parts = text.split(",")
    if not parts or parts[0] in ("", "#"):
        return None
    root = TreeNode(int(parts[0]))
    queue, i = deque([root]), 1
    while queue and i < len(parts):
        node = queue.popleft()
        for side in ("left", "right"):
            if i < len(parts) and parts[i] != "#":
                child = TreeNode(int(parts[i]))
                setattr(node, side, child)
                queue.append(child)
            i += 1
    return root
