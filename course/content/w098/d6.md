:::exercise id=mix_overlap points=10 title="The busiest moment"
Walkers are on a path from `start` to `end` (times in minutes; someone leaving at the same moment another arrives doesn't overlap). Write `most_at_once(intervals)`: the largest number on the path at any moment, using a sweep over sorted start and end events.
@@starter
def most_at_once(intervals):
    ___
@@tests
def test_overlap():
    "a sweep"
    assert most_at_once([(0, 30), (5, 10), (15, 20)]) == 2 and most_at_once([(1, 5), (5, 8)]) == 1
    assert most_at_once([]) == 0 and most_at_once([(1, 9), (2, 9), (3, 9), (9, 10)]) == 3
@@solution
def most_at_once(intervals):
    events = sorted([(s, 1) for s, _ in intervals] + [(e, -1) for _, e in intervals])
    current = best = 0
    for _, change in events:
        current += change
        best = max(best, current)
    return best
:::

def error_counts(tracebacks):
    counts = {}
    for tb in tracebacks:
        name = tb.strip().splitlines()[-1].split(": ", 1)[0]
        counts[name] = counts.get(name, 0) + 1
    return dict(sorted(counts.items(), key=lambda pair: (-pair[1], pair[0])))

:::exercise id=forge points=50 title="Coding round: autocomplete"
Write a class `Autocomplete` with `add(word, count=1)` (record that the word was searched `count` more times) and `suggest(prefix, k=3)`: up to `k` recorded words starting with `prefix`, most searched first, ties alphabetical. Also write `count(word)` (0 if never added).
@@starter

@@tests
import random, time

def test_suggest():
    "most searched first"
    ac = Autocomplete()
    for word, n in [("mill", 5), ("mile", 2), ("milestone", 2), ("moor", 9), ("mill", 1), ("ridge", 4)]:
        ac.add(word, n)
    assert ac.suggest("mi") == ["mill", "mile", "milestone"] and ac.suggest("m", k=2) == ["moor", "mill"]
    assert ac.suggest("x") == [] and ac.suggest("ridge") == ["ridge"] and ac.count("mill") == 6 and ac.count("mil") == 0
    ac.add("mil")
    assert ac.suggest("mil", k=5) == ["mill", "mile", "milestone", "mil"]

def test_fast():
    "50,000 words"
    rng = random.Random(4)
    letters = "abcdefghij"
    ac = Autocomplete()
    words = ["".join(rng.choice(letters) for _ in range(rng.randint(4, 8))) for _ in range(50_000)]
    for w in words:
        ac.add(w, rng.randint(1, 100))
    start = time.perf_counter()
    for _ in range(2000):
        prefix = "".join(rng.choice(letters) for _ in range(3))
        result = ac.suggest(prefix)
        assert all(w.startswith(prefix) for w in result)
    assert time.perf_counter() - start < 6, "too slow: search only under the prefix"
@@hint
Each node is a dict of children plus the count of searches that end there (store it under a key that can't be a letter, such as `None`). `suggest` walks down to the prefix's node, then collects every word below it with its count, and sorts by `(-count, word)`. (In the design discussion: precompute the top k at each node to make suggestions O(prefix length).)
@@solution
import heapq

class Autocomplete:
    def __init__(self):
        self.root = {}

    def add(self, word, count=1):
        node = self.root
        for ch in word:
            node = node.setdefault(ch, {})
        node[None] = node.get(None, 0) + count

    def _node(self, prefix):
        node = self.root
        for ch in prefix:
            if ch not in node:
                return None
            node = node[ch]
        return node

    def count(self, word):
        node = self._node(word)
        return node.get(None, 0) if node else 0

    def suggest(self, prefix, k=3):
        start = self._node(prefix)
        if start is None:
            return []
        found, stack = [], [(start, prefix)]
        while stack:
            node, text = stack.pop()
            for key, child in node.items():
                if key is None:
                    found.append((-child, text))
                else:
                    stack.append((child, text + key))
        return [word for _, word in heapq.nsmallest(k, found)]

## Bonus: two loops compared

:::exercise id=forge2 points=30 title="Bonus: what improved?"
Write `compare_loops(first, second)`: each loop maps round names to a dict of criteria and ratings (1 to 4).     ___
@@tests
FIRST = {"coding": {"solving": 4, "testing": 3}, "design": {"requirements": 2, "estimates": 3, "trade-offs": 2},
         "behavioural": {"ownership": 3, "learning": 3}}
SECOND = {"coding": {"solving": 4, "testing": 4}, "design": {"requirements": 3, "estimates": 3, "trade-offs": 3},
          "debugging": {"method": 3, "fix": 2}, "behavioural": {"ownership": 3, "learning": 3}}

def test_compare():
    "the point of mock loops"
    assert compare_loops(FIRST, SECOND) == [
        "coding: 3.50 to 4.00 (+0.50)", "design: 2.33 to 3.00 (+0.67)", "debugging: new, 2.50",
        "behavioural: 3.00 to 3.00 (same)", "most improved: design", "focus next: debugging", "weakest criterion: fix (2)"]

def test_worse():
    "nothing rose"
    lines = compare_loops({"coding": {"x": 4}}, {"coding": {"x": 3}})
    assert lines == ["coding: 4.00 to 3.00 (-1.00)", "most improved: none", "focus next: coding", "weakest criterion: x (3)"]
@@solution
def _mean(scores):
    return sum(scores.values()) / len(scores)

def compare_loops(first, second):
    lines, rises = [], []
    for name, scores in second.items():
        b = _mean(scores)
        if name not in first:
            lines.append(f"{name}: new, {b:.2f}")
            continue
        a = _mean(first[name])
        change = round(b - a, 2)
        shown = "same" if change == 0 else f"{change:+.2f}"
        lines.append(f"{name}: {a:.2f} to {b:.2f} ({shown})")
        rises.append((change, name))
    best = max((r for r in rises if r[0] > 0), key=lambda r: r[0], default=None)
    lines.append(f"most improved: {best[1] if best else 'none'}")
    lowest = min(second, key=lambda n: _mean(second[n]))
    lines.append(f"focus next: {lowest}")
    criterion, rating = min(((c, r) for scores in second.values() for c, r in scores.items()), key=lambda cr: cr[1])
    lines.append(f"weakest criterion: {criterion} ({rating})")
    return lines
