def test_compare():

    assert compare({"p": 1}, {"p": 1, "q": 0}) == "equal"
    assert compare({"p": 1}, {"p": 2, "q": 1}) == "before"
    assert compare({"p": 3, "q": 1}, {"p": 2}) == "after"

@@solution
def compare(a, b):
    nodes = set(a) | set(b)
    a_le = all(a.get(n, 0) <= b.get(n, 0) for n in nodes)
    b_le = all(b.get(n, 0) <= a.get(n, 0) for n in nodes)
    if a_le and b_le:
        return "equal"
    if a_le:
        return "before"
    if b_le:
        return "after"
    return "concurrent"

def merge(a, b):
    return {n: max(a.get(n, 0), b.get(n, 0)) for n in set(a) | set(b)}

def budget_report(slo, total, failed, days_elapsed, window_days=30):
    budget = (1 - slo) * total
    spent = failed / budget
    burn = spent / (days_elapsed / window_days)
    if spent >= 1:
        verdict = "freeze"
    elif burn > 1:
        verdict = "slow down"
    else:
        verdict = "ok"
    return {"budget": round(budget), "spent": round(spent, 3), "burn_rate": round(burn, 2), "verdict": verdict}

## Forge: seat booking

A concert goes on sale at 10:00 and 80,000 people try to buy 20,000 seats. The rules a booking system must never break are short: **a seat is sold at most once**, and **someone who paid gets their seats**. Everything else follows from them.

- **Holds.** Choosing seats and paying takes minutes, so the system **holds** the chosen seats for that person (ten minutes, say) and shows them to nobody else. A hold that isn't paid for **expires** on its own, so abandoned checkouts free their seats without anyone cleaning up.
- **All or nothing.** Four friends want four seats together. If any one of them is taken, the hold fails and none are held, otherwise the system fills up with useless fragments of groups.
- **Atomic claims.** Two people clicking the same seat at the same instant must not both get it. In a real system the claim is one atomic step (a conditional write, like Wednesday's rate-limiter store: "set this seat's holder only if it's free"); here, one method call does the whole check-and-claim.
- **Idempotent purchase.** The payment page times out and the browser retries. An **idempotency key** (Week 77) makes the retry return the same order instead of charging twice.

Build `BookingService(clock, hold_ttl=600, max_seats=6)`:

- `add_event(event_id, seats)` with a list of seat ids; a repeated id raises `ValueError("event exists")`.
- `hold(event_id, user, seats)` returns a hold id (`"h1"`, `"h2"`, ...). It raises `ValueError("choose 1 to N seats")` (N being `max_seats`) for an empty list or too many, `ValueError("unknown seat X")` for the first seat (in the order given) not in the event, and `ValueError("seat X is taken")` for the first seat that is sold or held by an **unexpired** hold (a user's own hold counts too). Holds expire `hold_ttl` seconds after they were made (a hold is expired when `now >= expires`). All seats are held or none.
- `release(hold_id)` cancels a hold (an unknown or finished hold is ignored).
- `purchase(hold_id, user, key)`: if `key` was used before, return the earlier order unchanged. Otherwise an unknown hold raises `KeyError("no such hold")`; another user's hold raises `PermissionError("not your hold")`; an expired or released hold raises `ValueError("hold expired")`. On success the seats are sold, the hold is finished, and it returns an order `{"order": "o1", "event": ..., "user": ..., "seats": [sorted seat ids]}` (numbered `"o1"`, `"o2"`, ...).
- `available(event_id)`: the sorted seats neither sold nor held by an unexpired hold. `stats(event_id)`: `{"sold": n, "held": n, "available": n}` (seats, not holds).

class BookingService:
    def __init__(self, clock, hold_ttl=600, max_seats=6):
        self.clock = clock
        self.hold_ttl = hold_ttl
        self.max_seats = max_seats
        self.events = {}
        self.sold = {}
        self.holds = {}
        self.orders = {}
        self.next_hold = 1
        self.next_order = 1

    def add_event(self, event_id, seats):
        if event_id in self.events:
            raise ValueError("event exists")
        self.events[event_id] = list(seats)
        self.sold[event_id] = set()

    def _active_holds(self, event_id):
        now = self.clock()
        return [h for h in self.holds.values()
                if h["event"] == event_id and h["state"] == "held" and now < h["expires"]]

    def _held_seats(self, event_id):
        return {seat for h in self._active_holds(event_id) for seat in h["seats"]}

    def hold(self, event_id, user, seats):
        if not 1 <= len(seats) <= self.max_seats:
            raise ValueError(f"choose 1 to {self.max_seats} seats")
        known = set(self.events[event_id])
        for seat in seats:
            if seat not in known:
                raise ValueError(f"unknown seat {seat}")
        taken = self.sold[event_id] | self._held_seats(event_id)
        for seat in seats:
            if seat in taken:
                raise ValueError(f"seat {seat} is taken")
        hold_id = f"h{self.next_hold}"
        self.next_hold += 1
        self.holds[hold_id] = {"event": event_id, "user": user, "seats": sorted(set(seats)),
                               "expires": self.clock() + self.hold_ttl, "state": "held"}
        return hold_id

    def release(self, hold_id):
        h = self.holds.get(hold_id)
        if h and h["state"] == "held":
            h["state"] = "released"

    def purchase(self, hold_id, user, key):
        if key in self.orders:
            return self.orders[key]
        if hold_id not in self.holds:
            raise KeyError("no such hold")
        h = self.holds[hold_id]
        if h["user"] != user:
            raise PermissionError("not your hold")
        if h["state"] != "held" or self.clock() >= h["expires"]:
            raise ValueError("hold expired")
        self.sold[h["event"]].update(h["seats"])
        h["state"] = "purchased"
        order = {"order": f"o{self.next_order}", "event": h["event"], "user": user, "seats": list(h["seats"])}
        self.next_order += 1
        self.orders[key] = order
        return order

    def available(self, event_id):
        taken = self.sold[event_id] | self._held_seats(event_id)
        return sorted(s for s in self.events[event_id] if s not in taken)

    def stats(self, event_id):
        held = len(self._held_seats(event_id))
        sold = len(self.sold[event_id])
        return {"sold": sold, "held": held, "available": len(self.events[event_id]) - sold - held}

## Forge bonus: a waiting room

At 10:00 the booking service could take perhaps 2,000 buyers at once; 80,000 arrive. Letting them all in makes the site slow for everyone, and the fastest scripts win. A **virtual waiting room** puts arrivals in a first-come queue and **admits** them in batches as active buyers finish, so the service runs at a pace it can sustain and the order is fair.

:::exercise id=forge2 points=30 title="Forge bonus: a waiting room"

@@solution
import math

class WaitingRoom:
    def __init__(self, clock, capacity, session_ttl):
        self.clock = clock
        self.capacity = capacity
        self.session_ttl = session_ttl
        self.queue = []
        self.active = {}

    def _expired(self, user):
        return self.clock() >= self.active[user] + self.session_ttl

    def join(self, user):
        if user in self.active and self._expired(user):
            del self.active[user]
        if user not in self.active and user not in self.queue:
            self.queue.append(user)
        return self.status(user)

    def admit(self):
        for user in [u for u in self.active if self._expired(u)]:
            del self.active[user]
        admitted = []
        while self.queue and len(self.active) < self.capacity:
            user = self.queue.pop(0)
            self.active[user] = self.clock()
            admitted.append(user)
        return admitted

    def leave(self, user):
        self.active.pop(user, None)
        if user in self.queue:
            self.queue.remove(user)

    def status(self, user):
        if user in self.active and not self._expired(user):
            return ("active", None)
        if user in self.queue:
            return ("waiting", self.queue.index(user) + 1)
        return ("absent", None)

    def wait_estimate(self, user, minutes_per_session):
        state, position = self.status(user)
        if state == "active":
            return 0
        if state == "waiting":
            return math.ceil(position / self.capacity) * minutes_per_session
        return None
