def none(pence):
    return pence

def student(pence):
    return pence - pence // 10

def half(pence):
    return pence // 2

DISCOUNTS = {"none": none, "student": student, "half": half}

def charge(pence, name="none"):
    key = name.lower()
    if key not in DISCOUNTS:
        raise ValueError(f"unknown discount {name!r}; choose from {', '.join(sorted(DISCOUNTS))}")
    return DISCOUNTS[key](pence)

class Watched:
    def __init__(self, value):
        self._value = value
        self._handlers = []

    def subscribe(self, handler):
        self._handlers.append(handler)
        return lambda: self._handlers.remove(handler)

    @property
    def value(self):
        return self._value

    @value.setter
    def value(self, new):
        if new == self._value:
            return
        old, self._value = self._value, new
        for h in list(self._handlers):
            h(old, new)

class Alarm:
    def __init__(self, limit):
        self.limit = limit
        self.triggered = []

    def __call__(self, old, new):
        if old <= self.limit < new:
            self.triggered.append(new)

## Forge: a smart-home hub

A home hub controls lamps, heaters and plugs, and lets other parts of the program (a phone app, an energy report, a log) react to every change. Here's the specification, with the pattern each part uses.

**Devices, made by a factory.**

- `Device` is the base class. Subclasses register themselves with `__init_subclass__` and a `kind` keyword (`class Lamp(Device, kind="lamp")`) in `Device.kinds`. Every device has a `name`, a method `state()` returning a **new** dictionary describing it, and `command(action, **settings)`, which raises `ValueError` for an action it doesn't know.
- `Lamp(name)`: state `{"on": False, "brightness": 100}` to start. Actions `"on"`, `"off"`, and `"dim"` with a setting `level` (a whole number from 0 to 100, otherwise `ValueError`), which sets the brightness.
- `Heater(name)`: state `{"on": False, "target": 18}`. Actions `"on"`, `"off"`, and `"set"` with a setting `target` (5 to 30, otherwise `ValueError`).
- `make_device(spec)` makes a device from a dictionary such as `{"kind": "lamp", "name": "hall"}`; an unknown kind, or settings the class doesn't take, raise `ValueError`.

**A plug from another maker, adapted.** `AcmePlug` (given, and not to be changed) has `label`, `switch(on)` and `is_live()`. `PlugAdapter(plug)` makes it work like a device: its `name` is the plug's label, its state is `{"on": ...}` from `is_live()`, and it understands `"on"` and `"off"` (anything else raises `ValueError`). It must not inherit from `AcmePlug`.

**Logging, as a decorator.** `Logged(device, log)` wraps any device, with the same `name`, `state()` and `command(...)`. After a command succeeds, it appends a line to `log` like `hall: dim level=40` (the device's name, the action, then any settings as `key=value` in alphabetical order of key, all separated by single spaces). Failed commands aren't logged.

**The hub, as a subject.** `Hub()` holds devices by name:

- `add(device)` (a duplicate name raises `ValueError`) and `device(name)` (an unknown name raises `KeyError`);
- `subscribe(handler)` returns an unsubscribe function; after every successful command, each handler is called as `handler(name, action, state)` with the device's new state;
- `command(name, action, **settings)` passes the command to the device, then notifies.

**Heating, by strategy.** A heating strategy is a function from an hour (0 to 23) to a target temperature, or `None` meaning "off".

- `constant(temp)` returns a strategy that always gives `temp`.
- `schedule(times)` returns a strategy from a dictionary like `{7: 21, 9: 16, 17: 21, 22: None}`: at a given hour, use the entry for the latest hour at or before it; before the first entry of the day, use the **last** entry (from the night before).
- `Hub.tick(hour, strategy)` applies the strategy to every device whose state has a `"target"` key, in name order, using `command` so observers hear about it: for `None`, send `"off"` if it's on; for a temperature, send `"set"` with that target if it differs, and `"on"` if it's off. It returns the number of commands sent.

class AcmePlug:
    """Another maker's smart plug. Don't change it."""
    def __init__(self, label):
        self.label = label
        self._live = False

    def switch(self, on):
        self._live = bool(on)

    def is_live(self):
        return self._live

class Device:
    kinds = {}

    def __init_subclass__(cls, kind=None, **kwargs):
        super().__init_subclass__(**kwargs)
        if kind is not None:
            Device.kinds[kind] = cls

    def __init__(self, name):
        self.name = name

    def state(self):
        raise NotImplementedError

    def command(self, action, **settings):
        raise ValueError(f"{self.name} doesn't understand {action!r}")

class Lamp(Device, kind="lamp"):
    def __init__(self, name):
        super().__init__(name)
        self._on = False
        self._brightness = 100

    def state(self):
        return {"on": self._on, "brightness": self._brightness}

    def command(self, action, **settings):
        if action in ("on", "off") and not settings:
            self._on = action == "on"
        elif action == "dim":
            level = settings.get("level")
            if not isinstance(level, int) or not 0 <= level <= 100:
                raise ValueError("level must be a whole number from 0 to 100")
            self._brightness = level
        else:
            super().command(action, **settings)

class Heater(Device, kind="heater"):
    def __init__(self, name):
        super().__init__(name)
        self._on = False
        self._target = 18

    def state(self):
        return {"on": self._on, "target": self._target}

    def command(self, action, **settings):
        if action in ("on", "off") and not settings:
            self._on = action == "on"
        elif action == "set":
            target = settings.get("target")
            if not isinstance(target, (int, float)) or not 5 <= target <= 30:
                raise ValueError("target must be from 5 to 30")
            self._target = target
        else:
            super().command(action, **settings)

def make_device(spec):
    settings = dict(spec)
    kind = settings.pop("kind", None)
    if kind not in Device.kinds:
        raise ValueError(f"unknown kind {kind!r}")
    try:
        return Device.kinds[kind](**settings)
    except TypeError as err:
        raise ValueError(f"bad settings for {kind}: {err}") from None

class PlugAdapter:
    def __init__(self, plug):
        self.plug = plug

    @property
    def name(self):
        return self.plug.label

    def state(self):
        return {"on": self.plug.is_live()}

    def command(self, action, **settings):
        if action not in ("on", "off") or settings:
            raise ValueError(f"{self.name} doesn't understand {action!r}")
        self.plug.switch(action == "on")

class Logged:
    def __init__(self, device, log):
        self.inner = device
        self.log = log

    @property
    def name(self):
        return self.inner.name

    def state(self):
        return self.inner.state()

    def command(self, action, **settings):
        self.inner.command(action, **settings)
        parts = [f"{self.name}:", action] + [f"{k}={settings[k]}" for k in sorted(settings)]
        self.log.append(" ".join(parts))

class Hub:
    def __init__(self):
        self._devices = {}
        self._handlers = []

    def add(self, device):
        if device.name in self._devices:
            raise ValueError(f"already have a device called {device.name!r}")
        self._devices[device.name] = device

    def device(self, name):
        return self._devices[name]

    def subscribe(self, handler):
        self._handlers.append(handler)
        return lambda: self._handlers.remove(handler)

    def command(self, name, action, **settings):
        device = self._devices[name]
        device.command(action, **settings)
        state = device.state()
        for h in list(self._handlers):
            h(name, action, dict(state))

    def tick(self, hour, strategy):
        sent = 0
        target = strategy(hour)
        for name in sorted(self._devices):
            state = self._devices[name].state()
            if "target" not in state:
                continue
            if target is None:
                if state["on"]:
                    self.command(name, "off")
                    sent += 1
                continue
            if state["target"] != target:
                self.command(name, "set", target=target)
                sent += 1
            if not state["on"]:
                self.command(name, "on")
                sent += 1
        return sent

def constant(temp):
    return lambda hour: temp

def schedule(times):
    hours = sorted(times)

    def strategy(hour):
        earlier = [h for h in hours if h <= hour]
        return times[earlier[-1] if earlier else hours[-1]]

    return strategy

:::think title="Patterns as a vocabulary"
Count the patterns in the hub: a factory makes devices, an adapter fits the plug, a decorator logs commands, the hub is an observable subject, and heating schedules are strategies. Yet no single part is complicated, and each answers one question: how is it made, how does it fit, what extra does it do, who hears about it, how does it decide? That's the real value of patterns: not clever code, but a shared vocabulary for small, familiar shapes, so that someone reading `Logged(PlugAdapter(AcmePlug("kettle")), log)` understands it at a glance. Notice, too, what isn't here: no pattern was used where an `if` or a plain function would do.

## Forge bonus: newsroom alerts

:::exercise id=forge2 points=30 title="Forge bonus: newsroom alerts"
A newsroom sends stories to readers through channels, each reader choosing what they want. Build it from this week's patterns:

- `Story`, a frozen dataclass: `headline`, `section` and `urgency` (1 to 3, 3 being breaking news).
- **Filters** (strategies) are functions from a `Story` to `bool`: `section_is(name)`, `urgent_at_least(level)`, and `all_of(*filters)`, which is True only when every filter is (and True for no filters).
- **Channels** have `send(text)`. `Inbox` keeps every text in `messages`. Decorators: `Prefixed(inner, prefix)` adds the prefix; `Digest(inner, size)` collects texts and, every time it has `size` of them, sends **one** text joining them with `" | "`; `Capitals(inner)` sends the text in capitals.
- `make_channel(spec)`, a **recursive** factory: `{"type": "inbox"}`, `{"type": "prefixed", "prefix": "[NEWS] ", "inner": {...}}`, `{"type": "digest", "size": 2, "inner": {...}}`, `{"type": "capitals", "inner": {...}}`. An unknown type raises `ValueError`. It must return the `Inbox` it made as well, so the result is a pair `(channel, inbox)`.
- `Desk` (the subject): `subscribe(story_filter, channel)` returns an unsubscribe function; `publish(story)` sends the headline to every subscribed channel whose filter accepts the story, and returns how many channels it was sent to.

@@solution
from dataclasses import dataclass

@dataclass(frozen=True)
class Story:
    headline: str
    section: str
    urgency: int

def section_is(name):
    return lambda story: story.section == name

def urgent_at_least(level):
    return lambda story: story.urgency >= level

def all_of(*filters):
    return lambda story: all(f(story) for f in filters)

class Inbox:
    def __init__(self):
        self.messages = []

    def send(self, text):
        self.messages.append(text)

class Prefixed:
    def __init__(self, inner, prefix):
        self.inner, self.prefix = inner, prefix

    def send(self, text):
        self.inner.send(self.prefix + text)

class Capitals:
    def __init__(self, inner):
        self.inner = inner

    def send(self, text):
        self.inner.send(text.upper())

class Digest:
    def __init__(self, inner, size):
        self.inner, self.size = inner, size
        self.waiting = []

    def send(self, text):
        self.waiting.append(text)
        if len(self.waiting) == self.size:
            self.inner.send(" | ".join(self.waiting))
            self.waiting = []

def make_channel(spec):
    kind = spec.get("type")
    if kind == "inbox":
        box = Inbox()
        return box, box
    if kind not in ("prefixed", "digest", "capitals"):
        raise ValueError(f"unknown channel type {kind!r}")
    if "inner" not in spec:
        raise ValueError(f"{kind} needs an inner channel")
    inner, box = make_channel(spec["inner"])
    if kind == "prefixed":
        return Prefixed(inner, spec["prefix"]), box
    if kind == "digest":
        return Digest(inner, spec["size"]), box
    return Capitals(inner), box

class Desk:
    def __init__(self):
        self._subs = []

    def subscribe(self, story_filter, channel):
        pair = (story_filter, channel)
        self._subs.append(pair)
        return lambda: self._subs.remove(pair)

    def publish(self, story):
        sent = 0
        for story_filter, channel in list(self._subs):
            if story_filter(story):
                channel.send(story.headline)
                sent += 1
        return sent
