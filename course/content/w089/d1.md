import heapq


class UnknownPlace(LookupError):
    pass


class NoRoute(Exception):
    pass


def naismith_minutes(km, climb_m):
    return round(km / 5 * 60 + climb_m / 600 * 60)


class Route:
    def __init__(self, stops, km, climb_m):
        self.stops, self.km, self.climb_m = list(stops), km, climb_m

    @property
    def minutes(self):
        return naismith_minutes(self.km, self.climb_m)

    def __repr__(self):
        return f"Route({self.stops}, {self.km} km, {self.climb_m} m)"


class PathNetwork:
    def __init__(self, heights):
        self.heights = dict(heights)
        self.paths = {name: {} for name in self.heights}

    def _check(self, *names):
        for name in names:
            if name not in self.heights:
                raise UnknownPlace(name)

    def add_path(self, a, b, km):
        self._check(a, b)
        if km <= 0:
            raise ValueError("km must be positive")
        self.paths[a][b] = km
        self.paths[b][a] = km

    def climb(self, stops):
        return sum(max(0, self.heights[b] - self.heights[a]) for a, b in zip(stops, stops[1:]))

    def shortest(self, start, end):
        self._check(start, end)
        dist, prev, done = {start: 0}, {}, set()
        heap = [(0, start)]
        while heap:
            d, node = heapq.heappop(heap)
            if node in done:
                continue
            done.add(node)
            if node == end:
                break
            for nxt, km in sorted(self.paths[node].items()):
                if d + km < dist.get(nxt, float("inf")):
                    dist[nxt] = d + km
                    prev[nxt] = node
                    heapq.heappush(heap, (d + km, nxt))
        if end not in done:
            raise NoRoute(f"no route from {start} to {end}")
        stops = [end]
        while stops[-1] != start:
            stops.append(prev[stops[-1]])
        stops.reverse()
        return Route(stops, round(dist[end], 2), self.climb(stops))

import math
from waymark_domain import PathNetwork
from waymark_map import HEIGHTS, PATHS

POSITIONS = {"Mill": (0.0, 0.0), "Weir": (1.4, 0.4), "Church": (2.8, 1.2), "Ridge": (4.4, 3.4), "Farm": (2.8, 2.4),
             "Wood": (2.0, 3.5), "Lake": (0.6, 2.8), "Island": (-3.0, 5.0)}


class GeoNetwork(PathNetwork):
    """A path network whose junctions also have positions, in kilometres east and north."""

    def __init__(self, heights, positions):
        super().__init__(heights)
        self.positions = dict(positions)

    def straight_km(self, a, b):
        return math.dist(self.positions[a], self.positions[b])


def sample_geo():
    network = GeoNetwork(HEIGHTS, POSITIONS)
    for a, b, km in PATHS:
        network.add_path(a, b, km)
    return network


def grid_network(size):
    """A size-by-size grid of junctions named 'r-c', 1 km apart, with slightly longer paths to their neighbours."""
    names = {(r, c): f"{r}-{c}" for r in range(size) for c in range(size)}
    heights = {name: ((r * 13 + c * 7) % 9) * 20 for (r, c), name in names.items()}
    positions = {name: (float(c), float(r)) for (r, c), name in names.items()}
    network = GeoNetwork(heights, positions)
    for (r, c), name in names.items():
        if c + 1 < size:
            network.add_path(name, names[(r, c + 1)], 1.0 + ((r * 7 + c * 3) % 5) / 10)
        if r + 1 < size:
            network.add_path(name, names[(r + 1, c)], 1.0 + ((r * 3 + c * 5) % 5) / 10)
    return network

Every path is at least as long as the straight line between its ends: you can't walk from Mill to Ridge in less than 5.4 km, whatever the paths do. That simple fact is what makes today's faster search correct.

## Measure first

"Make it faster" starts with a number. Timing is noisy (other programs, caches, a browser tab), so for algorithms count the **work** instead: here, how many junctions the search **finishes** (pops from the heap and expands) before it reaches the destination. Dijkstra's algorithm finishes every junction closer to the start than the destination, in **every direction**: to walk a few kilometres east it explores just as far west. On a 30 by 30 grid, from one corner to a junction nearby, that's 80 junctions to find a route through 13.

Then measure **where** the time goes in the whole request, not just the algorithm: in Waymark, loading the map from the database for each request turns out to cost far more than the search (Week 65's profiling). Fix the biggest cost first.

## A*: searching towards the goal

**A\*** is Dijkstra's algorithm with a sense of direction. Instead of always expanding the junction with the smallest distance **so far** (`g`), it expands the one with the smallest `g + h`, where `h` is an **estimate** of the distance still to go. With `h` the straight-line distance to the destination, junctions heading away from it look worse and are left alone.

A* still finds the shortest route if the estimate is **admissible**: it never **over**estimates the true remaining distance. The straight line is admissible because no path can be shorter than the straight line between its ends. Overestimate (multiply `h` by 2, say) and A* gets faster but can return a longer route. The optimisation therefore depends on the **data**: a path recorded as shorter than the straight line (a typo in the map, or positions in the wrong units) makes `h` inadmissible. Check that assumption when importing.

## Not loading the map every time

The map changes rarely (when someone imports a new one) and is read on every request, which is the perfect case for a **cache** (Week 81): load it once, keep it in memory, and reuse it. The hard part, as always, is **invalidation**: after an import, the cached map is stale. Waymark keeps a `map_version` number in the database that every import increases; the cache remembers which version it holds and reloads only when the number has changed. Checking a number is far cheaper than loading thousands of rows.

## Where am I?

To plan "from here", Waymark needs the junction nearest to a point. Checking every junction works for 8 and is slow for 50,000. A **grid index** divides the map into square cells and records which junctions are in each; to find the nearest to a point, look in the point's own cell, then the ring of cells around it, then the next ring, stopping once no unexplored cell could hold anything closer than the best found so far. It's the same idea as a hash table (Week 44), keyed by position.

:::key

A* expands the junction with the smallest `g + h`; with an admissible heuristic (never an overestimate, like the straight-line distance) it still finds the shortest route while exploring less; an overestimate makes it wrong. Check the data supports the heuristic. Cache rarely changing data and invalidate with a version number that every change increases. Find nearest points with a grid index, searching outward ring by ring and stopping when no closer point is possible.

import heapq
from waymark_domain import NoRoute
from waymark_geo import grid_network, sample_geo

def dijkstra_work(network, start, end):
    dist, done, heap = {start: 0}, set(), [(0, start)]
    while heap:
        d, node = heapq.heappop(heap)
        if node in done:
            continue
        done.add(node)
        if node == end:
            return round(d, 2), len(done)
        for nxt, km in sorted(network.paths[node].items()):
            if d + km < dist.get(nxt, float("inf")):
                dist[nxt] = d + km
                heapq.heappush(heap, (d + km, nxt))
    raise NoRoute(f"no route from {start} to {end}")

import heapq
from waymark_domain import NoRoute, Route, UnknownPlace
from waymark_geo import grid_network, sample_geo

def astar(network, start, end):
    for name in (start, end):
        if name not in network.heights:
            raise UnknownPlace(name)
    g, prev, done = {start: 0}, {}, set()
    heap = [(network.straight_km(start, end), start)]
    while heap:
        _, node = heapq.heappop(heap)
        if node in done:
            continue
        done.add(node)
        if node == end:
            stops = [end]
            while stops[-1] != start:
                stops.append(prev[stops[-1]])
            stops.reverse()
            return Route(stops, round(g[end], 2), network.climb(stops)), len(done)
        for nxt, km in sorted(network.paths[node].items()):
            if g[node] + km < g.get(nxt, float("inf")):
                g[nxt] = g[node] + km
                prev[nxt] = node
                heapq.heappush(heap, (g[nxt] + network.straight_km(nxt, end), nxt))
    raise NoRoute(f"no route from {start} to {end}")

class MapCache:
    def __init__(self, load, version):
        self._load, self._version = load, version
        self._value = None
        self._at = None
        self.loads = self.hits = 0

    def get(self):
        current = self._version()
        if self._value is not None and current == self._at:
            self.hits += 1
            return self._value
        self._value, self._at = self._load(), current
        self.loads += 1
        return self._value

    def clear(self):
        self._value = self._at = None

import math

class GridIndex:
    def __init__(self, positions, cell=1.0):
        self.positions, self.cell = dict(positions), cell
        self.cells = {}
        for name, (x, y) in self.positions.items():
            self.cells.setdefault(self._cell(x, y), []).append(name)
        self.cells_checked = 0

    def _cell(self, x, y):
        return math.floor(x / self.cell), math.floor(y / self.cell)

    def nearest(self, x, y):
        self.cells_checked = 0
        if not self.positions:
            return None
        cx, cy = self._cell(x, y)
        best, best_d = None, float("inf")
        limit = max(abs(a - cx) + abs(b - cy) for a, b in self.cells) + 1
        for r in range(limit + 1):
            for gx in range(cx - r, cx + r + 1):
                for gy in range(cy - r, cy + r + 1):
                    if max(abs(gx - cx), abs(gy - cy)) != r:
                        continue
                    self.cells_checked += 1
                    for name in self.cells.get((gx, gy), ()):
                        d = math.dist(self.positions[name], (x, y))
                        if (d, name) < (best_d, best or ""):
                            best, best_d = name, d
            if best is not None and best_d <= r * self.cell:
                break
        return best

def too_short(network, tolerance=0.01):

class GridIndex:
    def __init__(self, positions, cell=1.0):
        self.positions, self.cell, self.cells = dict(positions), cell, {}
        for name, (x, y) in self.positions.items():
            self.cells.setdefault((math.floor(x / cell), math.floor(y / cell)), []).append(name)
    def nearest(self, x, y):
        if not self.positions:
            return None
        return min(self.positions, key=lambda n: (math.dist(self.positions[n], (x, y)), n))

class Planner:
    def __init__(self, load, version, cell=1.0):
        self.cache = MapCache(load, version)
        self.cell = cell
        self.index, self._indexed = None, None
        self.searches = 0

    def _ready(self):
        network = self.cache.get()
        if network is not self._indexed:
            self.index, self._indexed = GridIndex(network.positions, self.cell), network
        if not network.heights:
            raise ValueError("no map loaded")
        return network

    def plan_from(self, x, y, end):
        network = self._ready()
        start = self.index.nearest(x, y)
        walk = math.dist(network.positions[start], (x, y))
        route, _ = astar(network, start, end)
        self.searches += 1
        total = route.km + walk
        return {"start": start, "walk_to_start_km": round(walk, 2), "stops": route.stops, "km": round(total, 1),
                "minutes": naismith_minutes(total, route.climb_m)}
