import heapq


class UnknownPlace(LookupError):
    pass


class NoRoute(Exception):
    pass


def naismith_minutes(km, climb_m):
    return round(km / 5 * 60 + climb_m / 600 * 60)


class Route:
    def __init__(self, stops, km, climb_m):
        self.stops, self.km, self.climb_m = list(stops), km, climb_m

    @property
    def minutes(self):
        return naismith_minutes(self.km, self.climb_m)

    def __repr__(self):
        return f"Route({self.stops}, {self.km} km, {self.climb_m} m)"


class PathNetwork:
    def __init__(self, heights):
        self.heights = dict(heights)
        self.paths = {name: {} for name in self.heights}

    def _check(self, *names):
        for name in names:
            if name not in self.heights:
                raise UnknownPlace(name)

    def add_path(self, a, b, km):
        self._check(a, b)
        if km <= 0:
            raise ValueError("km must be positive")
        self.paths[a][b] = km
        self.paths[b][a] = km

    def climb(self, stops):
        return sum(max(0, self.heights[b] - self.heights[a]) for a, b in zip(stops, stops[1:]))

    def shortest(self, start, end):
        self._check(start, end)
        dist, prev, done = {start: 0}, {}, set()
        heap = [(0, start)]
        while heap:
            d, node = heapq.heappop(heap)
            if node in done:
                continue
            done.add(node)
            if node == end:
                break
            for nxt, km in sorted(self.paths[node].items()):
                if d + km < dist.get(nxt, float("inf")):
                    dist[nxt] = d + km
                    prev[nxt] = node
                    heapq.heappush(heap, (d + km, nxt))
        if end not in done:
            raise NoRoute(f"no route from {start} to {end}")
        stops = [end]
        while stops[-1] != start:
            stops.append(prev[stops[-1]])
        stops.reverse()
        return Route(stops, round(dist[end], 2), self.climb(stops))

    def __init__(self, heights, positions):

def grid_network(size):
    """A size-by-size grid of junctions named 'r-c', 1 km apart, with slightly longer paths to their neighbours."""
    names = {(r, c): f"{r}-{c}" for r in range(size) for c in range(size)}
    heights = {name: ((r * 13 + c * 7) % 9) * 20 for (r, c), name in names.items()}
    positions = {name: (float(c), float(r)) for (r, c), name in names.items()}
    network = GeoNetwork(heights, positions)
    for (r, c), name in names.items():
        if c + 1 < size:
            network.add_path(name, names[(r, c + 1)], 1.0 + ((r * 7 + c * 3) % 5) / 10)
        if r + 1 < size:
            network.add_path(name, names[(r + 1, c)], 1.0 + ((r * 3 + c * 5) % 5) / 10)

import heapq
from dataclasses import dataclass
from waymark_domain import NoRoute, Route, UnknownPlace
from waymark_kinds import kind_of
from waymark_map import sample_network

@dataclass(frozen=True)
class Preferences:
    avoid_roads: bool = False
    avoid_steep: bool = False
    road_factor: float = 3.0
    steep_factor: float = 2.0
    steep_limit: int = 80

    def __post_init__(self):
        if self.road_factor < 1 or self.steep_factor < 1:
            raise ValueError("factors must be at least 1")
        if not self.steep_limit > 0:
            raise ValueError("steep_limit must be positive")

    @classmethod
    def from_query(cls, query):
        avoid = {a.strip() for a in query.get("avoid", "").split(",") if a.strip()}
        for item in sorted(avoid):
            if item not in ("roads", "steep"):
                raise ValueError(f"cannot avoid {item}")
        options = {"avoid_roads": "roads" in avoid, "avoid_steep": "steep" in avoid}
        if "steep_limit" in query:
            try:
                options["steep_limit"] = int(query["steep_limit"])
            except ValueError:
                raise ValueError("steep_limit must be a whole number")
        return cls(**options)

def step_cost(network, a, b, prefs):
    km = network.paths[a][b]
    cost = km * (prefs.road_factor if prefs.avoid_roads and kind_of(a, b) == "road" else 1)
    climb = max(0, network.heights[b] - network.heights[a])
    return cost * (prefs.steep_factor if prefs.avoid_steep and climb / km > prefs.steep_limit else 1)

def best_route(network, start, end, prefs):
    best, prev, done, heap = {start: 0}, {}, set(), [(0, start)]
    while heap:
        c, node = heapq.heappop(heap)
        if node in done:
            continue
        done.add(node)
        if node == end:
            stops = [end]
            while stops[-1] != start:
                stops.append(prev[stops[-1]])
            stops.reverse()
            km = sum(network.paths[a][b] for a, b in zip(stops, stops[1:]))
            return Route(stops, round(km, 2), network.climb(stops)), round(c, 2)
        for nxt in sorted(network.paths[node]):
            step = step_cost(network, node, nxt, prefs)
            if c + step < best.get(nxt, float("inf")):
                best[nxt], prev[nxt] = c + step, node
                heapq.heappush(heap, (c + step, nxt))
    raise NoRoute(f"no route from {start} to {end}")

def explain(network, stops, shortest_km, steep_limit=80):
    steps = list(zip(stops, stops[1:]))
    out = []
    road = sum(network.paths[a][b] for a, b in steps if kind_of(a, b) == "road")
    if road:
        out.append(f"{round(road, 1)} km on roads")
    steep = [f"{a} to {b}" for a, b in steps if max(0, network.heights[b] - network.heights[a]) / network.paths[a][b] > steep_limit]
    out.append(f"{len(steep)} steep climb{'s' if len(steep) > 1 else ''}: {', '.join(steep)}" if steep else "no steep climbs")
    km = sum(network.paths[a][b] for a, b in steps)
    if km - shortest_km > 0.05:
        out.append(f"{round(km - shortest_km, 1)} km longer than the shortest route")
    return out

CHOICES = [("shortest", Preferences()), ("flattest", Preferences(avoid_steep=True)), ("quietest", Preferences(avoid_roads=True))]

def alternatives(network, start, end):
    found = {}
    for label, prefs in CHOICES:
        route, _ = best_route(network, start, end, prefs)
        key = tuple(route.stops)
        if key in found:
            found[key]["labels"].append(label)
        else:
            found[key] = {"labels": [label], "route": route}
    shortest_km = next(iter(found.values()))["route"].km
    return [{"label": " and ".join(v["labels"]), "stops": v["route"].stops, "km": round(v["route"].km, 1),
             "minutes": v["route"].minutes, "why": explain(network, v["route"].stops, shortest_km)}
            for v in found.values()]

import heapq
from waymark_domain import Route


def shortest_avoiding(network, start, end, banned):
    dist, prev, done, heap = {start: 0}, {}, set(), [(0, start)]
    while heap:
        d, node = heapq.heappop(heap)
        if node in done:
            continue
        done.add(node)
        if node == end:
            stops = [end]
            while stops[-1] != start:
                stops.append(prev[stops[-1]])
            stops.reverse()
            return Route(stops, round(d, 2), network.climb(stops))
        for nxt, km in sorted(network.paths[node].items()):
            if frozenset({node, nxt}) in banned:
                continue
            if d + km < dist.get(nxt, float("inf")):
                dist[nxt], prev[nxt] = d + km, node
                heapq.heappush(heap, (d + km, nxt))
    return None


def paths_of(stops):
    return {frozenset(pair) for pair in zip(stops, stops[1:])}


def loop_via(network, start, turn):
    if turn == start:
        return None
    out = shortest_avoiding(network, start, turn, set())
    if out is None:
        return None
    back = shortest_avoiding(network, turn, start, paths_of(out.stops))
    if back is None:
        return None
    stops = out.stops + back.stops[1:]
    return Route(stops, round(out.km + back.km, 2), network.climb(stops))

## Growth, not timing

A performance test that asserts "under 50 milliseconds" fails on a slow machine and passes on a fast one whatever the code does. Test the **shape** instead: run the algorithm on inputs of doubling size, count its work, and look at the ratio between successive sizes. A ratio near 2 when the size doubles means the work grows **linearly**; near 4, **quadratically**; a ratio that itself keeps growing means something worse. A test can assert "doubling the map at most roughly doubles the round-walk searches", and that holds on any machine.

## Every combination

When options can interact, test all their combinations, generated with `itertools.product` (or `pytest.mark.parametrize` stacked, Week 72). Hand-written expected answers for twelve combinations and many routes are impractical, so check **invariants** that hold for every one: the route starts and ends where asked; its cost is at least its distance (factors are at least 1); no other route (in particular, the plain shortest one) is cheaper under the same preferences; and the reported distance equals the sum of the paths.

## Metamorphic tests

Sometimes you can't say what the right answer is, but you can say how it must **change**. Those relations are **metamorphic** tests:

- adding a path can never make any shortest distance **longer**;
- doubling every path's length must exactly double every shortest distance (and leave the routes the same);
- the shortest distance from A to B equals the one from B to A (paths are two-way), though the times may differ because of the climb.

Each relation is checked on many inputs, including random ones, and a failure points straight at a bug in the algorithm.

## Regression suites

When you've checked a route by hand (or it's in a scenario), **record** it: a file of cases (start, end, preferences, expected stops) that a test replays after every change. A changed answer isn't necessarily wrong (a better algorithm may find a better route), but it must be **noticed** and accepted on purpose, like a golden file. Name each case after its story, so a failure says which promise to a user has changed.

:::key
Test performance by how counted work grows as input size doubles (ratio near 2 is linear, near 4 quadratic), not by elapsed time. def growth_table(sizes, measure):
    table, previous = [], None
    for size in sizes:
        work = measure(size)
        table.append((size, work, None if previous is None else round(work / previous, 2)))
        previous = work
    return table

def classify(table):
    ratios = [r for _, _, r in table[1:]]
    if all(r <= 2.5 for r in ratios):
        return "linear"
    if all(r <= 5 for r in ratios):
        return "quadratic"
    return "worse than quadratic"

from waymark_prefs import Preferences, best_route, step_cost

def all_preferences():

def invariant_failures(network, start, end, prefs):
    route, cost = waymark_prefs.best_route(network, start, end, prefs)
    failures = []
    if route.stops[0] != start or route.stops[-1] != end:
        failures.append("wrong ends")
    if cost < route.km - 0.001:
        failures.append("cost below distance")
    steps = list(zip(route.stops, route.stops[1:]))
    if abs(sum(network.paths[a][b] for a, b in steps) - route.km) > 0.01:

    plain = network.shortest(start, end).stops
    plain_cost = sum(waymark_prefs.step_cost(network, a, b, prefs) for a, b in zip(plain, plain[1:]))
    if plain_cost < cost - 0.001:

def relation_failures(make_network, pairs, shortest):
    failures = []
    for a, b in pairs:
        base = make_network()
        d = shortest(base, a, b)
        if abs(d - shortest(base, b, a)) > 0.001:
            failures.append(f"symmetric: {a} to {b}")
        doubled = make_network()
        for x in doubled.paths:
            for y in doubled.paths[x]:
                doubled.paths[x][y] *= 2
        if abs(shortest(doubled, a, b) - 2 * d) > 0.001:
            failures.append(f"scaling: {a} to {b}")
        joined = make_network()
        if b not in joined.paths[a]:
            joined.add_path(a, b, 0.1)
            if shortest(joined, a, b) > 0.1 + 0.001:
                failures.append(f"adding: {a} to {b}")
    return failures

import copy
import json
from waymark_prefs import Preferences, best_route

def replay(cases, plan):
    passed, changed = 0, []
    for case in cases:
        actual = plan(case["from"], case["to"], case["avoid"])
        if actual == case["stops"]:
            passed += 1
        else:
            changed.append({"name": case["name"], "expected": case["stops"], "actual": actual})
    return passed, changed

def record(cases, plan):
    fresh = copy.deepcopy(cases)
    for case in fresh:
        case["stops"] = plan(case["from"], case["to"], case["avoid"])
    return fresh

def plan_with_prefs(network):
    def plan(start, end, avoid):
        route, _ = best_route(network, start, end, Preferences.from_query({"avoid": avoid}))
        return route.stops
    return plan

def report(passed, changed):
    lines = [f"{passed} passed, {len(changed)} changed"]
    lines += [f"CHANGED {c['name']}: {' > '.join(c['expected'])} became {' > '.join(c['actual'])}" for c in changed]
    return "\n".join(lines)
