def test_astar():
    "a small grid"
    positions = {"A": (0, 0), "B": (1, 0), "C": (1, 1), "D": (0, 1)}
    paths = {"A": {"B": 1.0, "D": 1.2}, "B": {"A": 1.0, "C": 1.5}, "C": {"B": 1.5, "D": 1.0}, "D": {"A": 1.2, "C": 1.0}}

@@solution
import heapq
import math

def astar_km(paths, positions, start, end):
    g, done = {start: 0}, set()
    heap = [(math.dist(positions[start], positions[end]), start)]
    while heap:
        _, node = heapq.heappop(heap)
        if node in done:
            continue
        done.add(node)
        if node == end:
            return round(g[end], 2)
        for nxt, km in sorted(paths[node].items()):
            if g[node] + km < g.get(nxt, float("inf")):
                g[nxt] = g[node] + km
                heapq.heappush(heap, (g[nxt] + math.dist(positions[nxt], positions[end]), nxt))
    return None
:::

import ast

def _asserts(node, in_loop):
    found = []
    for child in ast.iter_child_nodes(node):
        if isinstance(child, ast.Assert):
            found.append(in_loop)
        found += _asserts(child, in_loop or isinstance(child, (ast.For, ast.While)))
    return found

def asserts_in_loops(source):
    names = []
    for node in ast.parse(source).body:
        if isinstance(node, ast.FunctionDef) and node.name.startswith("test"):
            flags = _asserts(node, False)
            if flags and all(flags):
                names.append(node.name)
    return names

## Fit to release?

A release decision is a small design problem: decide **in advance** what evidence a release needs, so the decision on the day is a check, not a debate. Waymark's rules:

- **Blocking** (no release): any failing test; any failing demo step; any **unexplained** change in the regression suite (a change that hasn't been recorded on purpose); any open blocking review comment; a performance regression above the agreed limit.
- **Warnings** (release, but note them): skipped tests; a quality score below 80; stories carried over; slower but within the limit.

The output is a verdict (**GO** or **NO GO**), the reasons, and the warnings, short enough to read in ten seconds and specific enough to act on. It goes in the sprint review, and in CI it can block a release tag. The same idea scales up to what large projects call a release checklist.

:::exercise id=forge points=50 title="A go/no-go check"
Write `release_check(evidence, quality_limit=80)`. `evidence` is a dict with `tests` (`{"passed", "failed", "skipped"}`), `demo` (report lines starting `PASS` or `FAIL`), `regressions` (a list of changed case names) and `accepted` (case names whose change was accepted on purpose), `blocking_comments` (a number), `perf` (a dict from benchmark name to the ratio new work / old work), `perf_limit` (the largest acceptable ratio), `quality` (a score) and `carried` (story ids). Return a dict with `verdict` (`"GO"` or `"NO GO"`), `blocking` and `warnings` (lists of strings, in the order below), and `text`. Blocking, in order: `"N failing tests"`; `"demo failed: STEP"` for each failing demo line (the text after `FAIL `); `"unexplained change: NAME"` for each regression not accepted; `"N open blocking comments"`; `"slower: NAME (xR)"` for each benchmark above the limit (sorted by name, ratio to 2 places). Counts appear only when above 0. `text` is `"VERDICT"` then a line per blocking reason prefixed `"- "` and then per warning prefixed `"~ "`.

@@tests
GOOD = {"tests": {"passed": 214, "failed": 0, "skipped": 2}, "demo": ["PASS 1: waymark plan Mill Ridge", "PASS 2: waymark walks"],
        "regressions": ["S1 shortest to the ridge"], "accepted": ["S1 shortest to the ridge"], "blocking_comments": 0,
        "perf": {"route search": 0.57, "round walk": 1.1, "import": 1.0}, "perf_limit": 1.25, "quality": 75, "carried": ["S10"]}

def test_go():
    "release, with warnings"
    r = release_check(GOOD)
    assert r["verdict"] == "GO" and r["blocking"] == []
    assert r["warnings"] == ["2 skipped tests", "quality score 75 below 80", "carried over: S10", "slower but within limit: round walk (x1.1)"]
    assert r["text"] == "GO\n~ 2 skipped tests\n~ quality score 75 below 80\n~ carried over: S10\n~ slower but within limit: round walk (x1.1)"

def test_no_go():
    "every reason, in order"
    bad = dict(GOOD, tests={"passed": 210, "failed": 3, "skipped": 0}, demo=["PASS 1: a", "FAIL 2: waymark plan Mill Moon: output differs"],
               regressions=["S1 shortest to the ridge", "S9 flattest to the ridge"], blocking_comments=2,
               perf={"round walk": 1.65, "route search": 0.9}, quality=90, carried=[])
    r = release_check(bad)
    assert r["verdict"] == "NO GO"
    assert r["blocking"] == ["3 failing tests", "demo failed: 2: waymark plan Mill Moon: output differs",
                             "unexplained change: S9 flattest to the ridge", "2 open blocking comments", "slower: round walk (x1.65)"]
    assert r["warnings"] == [] and r["text"].startswith("NO GO\n- 3 failing tests\n")
@@solution
def release_check(evidence, quality_limit=80):
    blocking, warnings = [], []
    tests = evidence["tests"]
    if tests["failed"]:
        blocking.append(f"{tests['failed']} failing tests")
    blocking += [f"demo failed: {line[5:]}" for line in evidence["demo"] if line.startswith("FAIL")]
    blocking += [f"unexplained change: {name}" for name in evidence["regressions"] if name not in evidence["accepted"]]
    if evidence["blocking_comments"]:
        blocking.append(f"{evidence['blocking_comments']} open blocking comments")
    limit = evidence["perf_limit"]
    for name in sorted(evidence["perf"]):
        ratio = evidence["perf"][name]
        if ratio > limit:
            blocking.append(f"slower: {name} (x{round(ratio, 2)})")
    if tests["skipped"]:
        warnings.append(f"{tests['skipped']} skipped tests")
    if evidence["quality"] < quality_limit:
        warnings.append(f"quality score {evidence['quality']} below {quality_limit}")
    if evidence["carried"]:
        warnings.append("carried over: " + ", ".join(evidence["carried"]))
    for name in sorted(evidence["perf"]):
        ratio = evidence["perf"][name]
        if 1 < ratio <= limit:
            warnings.append(f"slower but within limit: {name} (x{round(ratio, 2)})")
    verdict = "NO GO" if blocking else "GO"
    text = "\n".join([verdict] + [f"- {b}" for b in blocking] + [f"~ {w}" for w in warnings])
    return {"verdict": verdict, "blocking": blocking, "warnings": warnings, "text": text}

## Bonus: benchmarks between versions

Week 89's growth tests check the shape; a **benchmark suite** checks the numbers, version against version. For a fixed set of scenarios (Mill to Ridge, a grid corner to corner, a two-hour round walk), record the **work** each takes (junctions finished, searches made, queries run) for the released version, and compare the new version's counts with them. A ratio above the agreed limit is a regression to explain or fix; well below 1 is an improvement worth celebrating in the changelog. Counting work rather than time makes the comparison repeatable on any machine, which is what lets it run in CI.

:::exercise id=forge2 points=30 title="Bonus: benchmarks between versions"
Write `compare_benchmarks(old, new, limit=1.25)`: `old` and `new` map scenario names to work counts. Return a dict with `ratios` (each scenario in both, sorted by name, to `new / old` rounded to 2 places), `regressions` (names with a ratio above `limit`, sorted by ratio descending then name), `improvements` (ratio at most 0.8, sorted by ratio ascending then name), `added` and `removed` (sorted names in only `new` or only `old`), and `report`: lines `"NAME: OLD -> NEW (xRATIO)"` for every compared scenario in name order, each followed by `" REGRESSION"` or `" improved"` where it applies, then `"added: A, B"` and `"removed: C"` lines if any.
@@starter

@@tests
def test_compare():
    "version against version"
    old = {"Mill to Ridge": 7, "grid corner": 900, "round walk": 97, "import 1000": 1001}
    new = {"Mill to Ridge": 4, "grid corner": 890, "round walk": 160, "flattest route": 12}
    r = compare_benchmarks(old, new)
    assert r["ratios"] == {"Mill to Ridge": 0.57, "grid corner": 0.99, "round walk": 1.65}
    assert r["regressions"] == ["round walk"] and r["improvements"] == ["Mill to Ridge"]
    assert r["added"] == ["flattest route"] and r["removed"] == ["import 1000"]
    assert r["report"] == ["Mill to Ridge: 7 -> 4 (x0.57) improved", "grid corner: 900 -> 890 (x0.99)",
                           "round walk: 97 -> 160 (x1.65) REGRESSION", "added: flattest route", "removed: import 1000"]
    same = compare_benchmarks({"a": 10}, {"a": 10})
    assert same["report"] == ["a: 10 -> 10 (x1.0)"] and same["regressions"] == []
@@solution
def compare_benchmarks(old, new, limit=1.25):
    both = sorted(set(old) & set(new))
    ratios = {name: round(new[name] / old[name], 2) for name in both}
    regressions = sorted((n for n in both if ratios[n] > limit), key=lambda n: (-ratios[n], n))
    improvements = sorted((n for n in both if ratios[n] <= 0.8), key=lambda n: (ratios[n], n))
    added, removed = sorted(set(new) - set(old)), sorted(set(old) - set(new))
    report = []
    for name in both:
        line = f"{name}: {old[name]} -> {new[name]} (x{ratios[name]})"
        if name in regressions:
            line += " REGRESSION"
        elif name in improvements:
            line += " improved"
        report.append(line)
    if added:
        report.append("added: " + ", ".join(added))
    if removed:
        report.append("removed: " + ", ".join(removed))
    return {"ratios": ratios, "regressions": regressions, "improvements": improvements,
            "added": added, "removed": removed, "report": report}
