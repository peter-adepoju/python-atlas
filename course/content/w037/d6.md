class SettingsError(Exception):
    pass

def parse_settings(lines):
    settings = {}
    where = {}
    for n, line in enumerate(lines, start=1):
        if not line.strip() or line.lstrip().startswith("#"):
            continue
        key, eq, value = line.partition("=")
        if not eq:
            raise SettingsError(f"line {n}: no '='")
        key = key.strip()
        if not key:
            raise SettingsError(f"line {n}: no name")
        if key in settings:
            raise SettingsError(f"line {n}: {key!r} was already set on line {where[key]}")
        settings[key] = value.strip()
        where[key] = n
    return settings

    for text in texts:
        try:
            return float(text)
        except ValueError:

    for text in texts:

## Forge: a statement importer

A bank lets people download their statements as text, one transaction per line, `date|description|amount`:

```text
# March
2026-03-01|Rent|-750.00
2026-03-02|Salary|£1,850.00
2026-03-02| Coffee |-2.8
```

Real files are messy: blank lines, comments, pound signs and commas in amounts, and the odd broken row. The importer should accept reasonable variations, report every broken row precisely, give up if there are too many, and log what it does. Here's the specification, from the bottom up.

**Errors.** `LedgerError(Exception)` is the base. `SourceError(LedgerError)` is for a file that can't be read. `RowError(LedgerError, ValueError)` has `__init__(self, line_no, reason)`, stores both, and has the message `"line N: reason"`. `TooManyErrors(LedgerError)` is for giving up.

**Domain.** `Transaction`, a **frozen** dataclass with `date` (a `datetime.date`), `description` (text) and `pence` (an int).

**Parsing** (generous at the edge, strict inside):

- `parse_amount(text)` returns pence. It strips spaces, removes commas, and accepts an optional sign (`-` or `+`) followed by an optional `£`, then whole pounds and optionally a `.` with one or two digits: `"-£3.20"` is `-320`, `"£1,850"` is `185000`, `"12.5"` is `1250`. Anything else (`""`, `"£"`, `"1.234"`, `"1."`, `".5"`, `"ten"`) raises `ValueError`.
- `parse_date(text)` returns a `datetime.date` from exactly `YYYY-MM-DD` (spaces around are allowed). Anything else raises `ValueError`, including impossible dates like `2026-02-30` (let `datetime.date` decide those).
- `parse_row(line_no, line)` splits on `|` and returns a `Transaction`, or raises `RowError` with one of these reasons: `"expected 3 fields, got K"`; `"bad date 'TEXT'"` and `"bad amount 'TEXT'"` (the stripped text, chained **from** the `ValueError`); `"no description"` (the description is stripped); `"amount is zero"`.

**Reading.** `read_lines(path)` returns the file's lines (UTF-8, without line endings), or raises `SourceError(f"can't read {path}")` chained from the `OSError` or `UnicodeDecodeError`.

**The importer.** `Report` is a frozen dataclass with `transactions` and `errors` (both tuples; the errors are the `RowError` messages) and a `balance` property, the sum of the pence. `Importer(log, max_errors=5)` has `run(path)`, which returns a `Report`:

- It skips blank lines and lines starting with `#` (after leading spaces). Each other line is a **row**.
- A row identical to an earlier transaction (same date, description and amount) is a `RowError` with the reason `"duplicate of line M"`.
- Each bad row is logged as a `WARNING` with its message and collected. As soon as there are **more** than `max_errors`, it raises `TooManyErrors(f"more than {max_errors} bad rows")` chained **from** an `ExceptionGroup("bad rows", ...)` of all the `RowError`s so far.
- Each good row logs `DEBUG` `"line N ok"`. At the end it logs `INFO` `"imported G of R rows"` (good rows, all rows).

**Presentation.** `money(pence)` formats pence as `£1,850.00` or `-£2.80`. `main(path, log, max_errors=5)` runs an importer and returns `(exit_code, text)`:

- Success: the text is `"N transactions, balance MONEY"`, followed (if there were bad rows) by a line `"problems:"` and one line per error, each starting with two spaces. The exit code is 0 with no problems, 1 with some.
- A `SourceError` or `TooManyErrors`: the text is `"error: "` and the message, the exit code is 2, and `main` logs it once as an `ERROR`. Anything else (a bug) isn't caught.

import datetime
import logging
from dataclasses import dataclass


class LedgerError(Exception):
    """Base class for the importer's errors."""


class SourceError(LedgerError):
    """The statement file can't be read."""


class TooManyErrors(LedgerError):
    """Too many bad rows to trust the file."""


class RowError(LedgerError, ValueError):
    def __init__(self, line_no, reason):
        super().__init__(f"line {line_no}: {reason}")
        self.line_no = line_no
        self.reason = reason


@dataclass(frozen=True)
class Transaction:
    date: datetime.date
    description: str
    pence: int


def _digits(text):
    return text.isascii() and text.isdigit()


def parse_amount(text):
    cleaned = text.strip().replace(",", "")
    sign = 1
    if cleaned.startswith("-"):
        sign, cleaned = -1, cleaned[1:]
    elif cleaned.startswith("+"):
        cleaned = cleaned[1:]
    if cleaned.startswith("£"):
        cleaned = cleaned[1:]
    pounds, dot, pence = cleaned.partition(".")
    if not _digits(pounds) or (dot and not (_digits(pence) and len(pence) <= 2)):
        raise ValueError(f"not an amount: {text!r}")
    return sign * (int(pounds) * 100 + (int(pence.ljust(2, "0")) if dot else 0))


def parse_date(text):
    cleaned = text.strip()
    parts = cleaned.split("-")
    if len(cleaned) != 10 or [len(p) for p in parts] != [4, 2, 2] or not all(_digits(p) for p in parts):
        raise ValueError(f"not a date: {text!r}")
    year, month, day = (int(p) for p in parts)
    return datetime.date(year, month, day)


def parse_row(line_no, line):
    fields = line.split("|")
    if len(fields) != 3:
        raise RowError(line_no, f"expected 3 fields, got {len(fields)}")
    date_text, description, amount_text = fields
    try:
        date = parse_date(date_text)
    except ValueError as err:
        raise RowError(line_no, f"bad date {date_text.strip()!r}") from err
    description = description.strip()
    if not description:
        raise RowError(line_no, "no description")
    try:
        pence = parse_amount(amount_text)
    except ValueError as err:
        raise RowError(line_no, f"bad amount {amount_text.strip()!r}") from err
    if pence == 0:
        raise RowError(line_no, "amount is zero")
    return Transaction(date, description, pence)


def read_lines(path):
    try:
        with open(path, encoding="utf-8") as f:
            return f.read().splitlines()
    except (OSError, UnicodeDecodeError) as err:
        raise SourceError(f"can't read {path}") from err


@dataclass(frozen=True)
class Report:
    transactions: tuple
    errors: tuple

    @property
    def balance(self):
        return sum(t.pence for t in self.transactions)


class Importer:
    def __init__(self, log, max_errors=5):
        self.log = log
        self.max_errors = max_errors

    def run(self, path):
        transactions, errors, seen = [], [], {}
        rows = 0
        for line_no, line in enumerate(read_lines(path), start=1):
            if not line.strip() or line.lstrip().startswith("#"):
                continue
            rows += 1
            try:
                transaction = parse_row(line_no, line)
                if transaction in seen:
                    raise RowError(line_no, f"duplicate of line {seen[transaction]}")
            except RowError as err:
                self.log.warning("%s", err)
                errors.append(err)
                if len(errors) > self.max_errors:
                    raise TooManyErrors(f"more than {self.max_errors} bad rows") from ExceptionGroup("bad rows", errors)
                continue
            seen[transaction] = line_no
            transactions.append(transaction)
            self.log.debug("line %d ok", line_no)
        self.log.info("imported %d of %d rows", len(transactions), rows)
        return Report(tuple(transactions), tuple(str(e) for e in errors))


def money(pence):
    sign = "-" if pence < 0 else ""
    pounds, rest = divmod(abs(pence), 100)
    return f"{sign}£{pounds:,}.{rest:02d}"


def main(path, log, max_errors=5):
    try:
        report = Importer(log, max_errors).run(path)
    except (SourceError, TooManyErrors) as err:
        log.error("%s", err)
        return 2, f"error: {err}"
    lines = [f"{len(report.transactions)} transactions, balance {money(report.balance)}"]
    if report.errors:
        lines.append("problems:")
        lines.extend(f"  {e}" for e in report.errors)
    return (1 if report.errors else 0), "\n".join(lines)

:::think title="A failure for every layer"
Look at where each kind of failure ends up. A broken amount becomes a `ValueError` in `parse_amount`, is translated into a `RowError` with a line number in `parse_row` (keeping the original as its cause), is logged once as a warning and collected by the importer, and reaches the user as one line under "problems:". A missing file becomes a `SourceError` at the reading layer and an exit code of 2 at the front end. Too much bad data becomes `TooManyErrors`, carrying every row's error in a group. And a bug, such as a `TypeError`, is caught nowhere, so it crashes with a full traceback and gets fixed. Each layer handles what it understands and translates the rest: that's what makes a program robust rather than merely quiet.
:::

## Forge bonus: retries and a circuit breaker

:::exercise id=forge2 points=30 title="Forge bonus: retries and a circuit breaker"
Programs that call other services over a network need two classic defences:

**Retrying.** Write `retry(func, attempts, sleep, log, retry_on=(ConnectionError, TimeoutError), first_delay=1)`. It calls `func()` and returns its result. If it raises one of `retry_on`, it logs a `WARNING` `"attempt N of A failed: MESSAGE"`, calls `sleep(delay)` and tries again; the delay starts at `first_delay` and doubles each time. It never sleeps after the last attempt. If every attempt fails, it raises `ExceptionGroup(f"all {attempts} attempts failed", errors)`. An exception **not** in `retry_on` goes straight through on the first occurrence. `attempts` below 1 raises `ValueError`. (`sleep` is passed in, as in Week 36, so the checks don't have to wait.)

**A circuit breaker.** When a service is down, retrying every call makes things worse. Write `CircuitOpen(Exception)` and `CircuitBreaker(threshold, cool_down)` with a `state` attribute (`"closed"`, `"open"` or `"half-open"`) and `call(func)`:

- **closed** (the start): call `func()`. A success resets the count of failures in a row; a failure adds one, and at `threshold` failures in a row the breaker becomes **open**. The failure's exception is re-raised either way.
- **open**: refuse the next `cool_down` calls without calling `func`, raising `CircuitOpen("circuit open")`. The call after those becomes a trial: the state becomes **half-open** and `func` is called.
- **half-open**: a success closes the breaker (failures back to 0); a failure opens it again at once (and its exception is re-raised).

class CircuitOpen(Exception):

def retry(func, attempts, sleep, log, retry_on=(ConnectionError, TimeoutError), first_delay=1):
    if attempts < 1:
        raise ValueError("attempts must be at least 1")
    errors = []
    delay = first_delay
    for attempt in range(1, attempts + 1):
        try:
            return func()
        except retry_on as err:
            errors.append(err)
            if attempt < attempts:
                log.warning("attempt %d of %d failed: %s", attempt, attempts, err)
                sleep(delay)
                delay *= 2
    raise ExceptionGroup(f"all {attempts} attempts failed", errors)


class CircuitBreaker:
    def __init__(self, threshold, cool_down):
        self.threshold = threshold
        self.cool_down = cool_down
        self.state = "closed"
        self._failures = 0
        self._refused = 0

    def call(self, func):
        if self.state == "open":
            if self._refused < self.cool_down:
                self._refused += 1
                raise CircuitOpen("circuit open")
            self.state = "half-open"
        try:
            result = func()
        except Exception:
            self._failures += 1
            if self.state == "half-open" or self._failures >= self.threshold:
                self._open()
            raise
        self._failures = 0
        self.state = "closed"
        return result

    def _open(self):
        self.state = "open"
        self._refused = 0
        self._failures = 0
