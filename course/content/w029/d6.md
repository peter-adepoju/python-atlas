class Score:
    def __init__(self, home, away):
        self._home, self._away = home, away

    @property
    def home(self):
        return self._home

    @property
    def away(self):
        return self._away

    @property
    def result(self):
        if self._home > self._away:
            return "home"
        if self._home < self._away:
            return "away"
        return "draw"

    def __repr__(self):
        return f"Score({self._home}, {self._away})"

    def __str__(self):
        return f"{self._home}-{self._away}"

    def __eq__(self, other):
        if not isinstance(other, Score):
            return NotImplemented
        return (self._home, self._away) == (other._home, other._away)

    def __hash__(self):
        return hash((self._home, self._away))

    def __add__(self, other):
        if not isinstance(other, Score):
            return NotImplemented
        return Score(self._home + other._home, self._away + other._away)

class GuestList:
    def __init__(self, limit):
        self._guests = set()
        self.limit = limit

    @property
    def limit(self):
        return self._limit

    @limit.setter
    def limit(self, value):
        if value < len(self._guests):
            raise ValueError(f"{len(self._guests)} guests are already invited")
        self._limit = value

    @property
    def guests(self):
        return tuple(sorted(self._guests))

    def invite(self, name):
        name = name.strip().title()
        if name in self._guests:
            raise ValueError(f"{name} is already invited")
        if len(self._guests) >= self._limit:
            raise ValueError("the list is full")
        self._guests.add(name)

    def __repr__(self):
        return f"<GuestList {len(self._guests)}/{self._limit}>"

## Forge: a recipe scaler

Recipes are written for a number of servings. Cooking for a different number means scaling every amount, and a shopping list for several recipes means adding up amounts in different units: 500 g of flour and 1.2 kg of flour make 1.7 kg. Here's the specification.

**`Amount(value, unit)`**, an immutable value class.

- Units come in families, in the class attribute `UNITS`: mass (`g` 1, `kg` 1000), volume (`ml` 1, `l` 1000, `tsp` 5, `tbsp` 15), and count (`""`, meaning a number of things, like 3 eggs). Each unit's number is its size in the family's base unit (g or ml). An unknown unit, or a negative value, raises `ValueError`.
- Read-only properties `value`, `unit` and `family`, and `base` (the value in the base unit: `Amount(1.2, "kg").base` is 1200).
- `==` compares **base amounts in the same family**: `Amount(1, "kg") == Amount(1000, "g")`. Different families are never equal. Hashing must agree (hash the family and the base amount, rounded to 6 places to avoid float noise).
- `+` with an `Amount` of the **same family** gives a result in the **left** operand's unit; different families raise `TypeError` (return `NotImplemented`). `sum()` works.
- `*` by a number (either side) scales it, keeping the unit.
- `__repr__` gives `Amount(1.5, 'kg')`. `__str__` gives the value rounded to 2 places with trailing zeros removed, then the unit: `1.5 kg`, `3 tsp`, and just `3` for counts. `__format__` with spec `"best"` gives the str of the amount in the most readable unit of its family: mass of 1000 g or more in kg, otherwise g; volume of 1000 ml or more in l, otherwise ml (teaspoons don't count as most readable); counts unchanged.

**`Recipe(name, servings)`**, with `servings` a positive whole number (a property: `ValueError` otherwise).

- `add(ingredient, amount)`: adding an ingredient already in the recipe **adds** the amounts (raising `TypeError` if their families differ).
- `ingredients`: a read-only property, a **tuple** of `(ingredient, amount)` pairs in the order first added.
- `scaled(servings)`: a **new** recipe for that many servings, every amount scaled.
- `__repr__` gives `<Recipe 'Pancakes' for 4: 3 ingredients>`.

And a function **`shopping_list(recipes)`**: combine the ingredients of all the recipes, and return lines like `flour: 1.7 kg`, sorted by ingredient, each amount shown with the `"best"` format.

@@solution
class Amount:
    UNITS = {
        "g": ("mass", 1), "kg": ("mass", 1000),
        "ml": ("volume", 1), "l": ("volume", 1000), "tsp": ("volume", 5), "tbsp": ("volume", 15),
        "": ("count", 1),
    }

    def __init__(self, value, unit):
        if unit not in Amount.UNITS:
            raise ValueError(f"unknown unit {unit!r}")
        if value < 0:
            raise ValueError("an amount can't be negative")
        self._value, self._unit = value, unit

    @property
    def value(self):
        return self._value

    @property
    def unit(self):
        return self._unit

    @property
    def family(self):
        return Amount.UNITS[self._unit][0]

    @property
    def base(self):
        return self._value * Amount.UNITS[self._unit][1]

    @staticmethod
    def _number(x):
        text = f"{round(x, 2):.2f}".rstrip("0").rstrip(".")
        return text

    def __repr__(self):
        return f"Amount({self._value!r}, {self._unit!r})"

    def __str__(self):
        n = self._number(self._value)
        return f"{n} {self._unit}" if self._unit else n

    def __format__(self, spec):
        if spec == "best":
            if self.family == "mass":
                unit = "kg" if self.base >= 1000 else "g"
            elif self.family == "volume":
                unit = "l" if self.base >= 1000 else "ml"
            else:
                return str(self)
            return str(Amount(self.base / Amount.UNITS[unit][1], unit))
        return format(str(self), spec)

    def _key(self):
        return (self.family, round(self.base, 6))

    def __eq__(self, other):
        if not isinstance(other, Amount):
            return NotImplemented
        return self._key() == other._key()

    def __hash__(self):
        return hash(self._key())

    def __add__(self, other):
        if not isinstance(other, Amount) or other.family != self.family:
            return NotImplemented
        return Amount((self.base + other.base) / Amount.UNITS[self._unit][1], self._unit)

    def __radd__(self, other):
        if other == 0:
            return self
        return NotImplemented

    def __mul__(self, k):
        if type(k) not in (int, float):
            return NotImplemented
        return Amount(self._value * k, self._unit)

    __rmul__ = __mul__


class Recipe:
    def __init__(self, name, servings):
        self.name = name
        self.servings = servings
        self._ingredients = {}

    @property
    def servings(self):
        return self._servings

    @servings.setter
    def servings(self, value):
        if type(value) is not int or value <= 0:
            raise ValueError("servings must be a positive whole number")
        self._servings = value

    def add(self, ingredient, amount):
        if ingredient in self._ingredients:
            self._ingredients[ingredient] = self._ingredients[ingredient] + amount
        else:
            self._ingredients[ingredient] = amount

    @property
    def ingredients(self):
        return tuple(self._ingredients.items())

    def scaled(self, servings):
        new = Recipe(self.name, servings)
        factor = servings / self._servings
        for ingredient, amount in self._ingredients.items():
            new.add(ingredient, amount * factor)
        return new

    def __repr__(self):
        return f"<Recipe {self.name!r} for {self._servings}: {len(self._ingredients)} ingredients>"


def shopping_list(recipes):
    totals = {}
    for recipe in recipes:
        for ingredient, amount in recipe.ingredients:
            totals[ingredient] = totals[ingredient] + amount if ingredient in totals else amount
    return [f"{ingredient}: {amount:best}" for ingredient, amount in sorted(totals.items())]
:::

:::think title="Small values, big payoff"
Most of today's work went into `Amount`, a class of about 80 lines that does one thing: be a quantity with a unit. Once it existed, `Recipe` and `shopping_list` hardly needed to think about units at all: they add amounts with `+` and print them with `:best`, and the value class handles every conversion. Good value classes (money, dates, amounts, coordinates) pay for themselves many times over, because every piece of code that uses them gets simpler and safer.

@@hint
`int("ff", 16)` reads hex. Python's `round` sends .5 to the nearest even number (`round(127.5)` is 128, `round(128.5)` is 128), which these checks expect. Moving towards white: `c + (255 - c) × fraction`; towards black: `c × (1 − fraction)`.
@@solution
import re

class Colour:
    def __init__(self, r, g, b):
        for v in (r, g, b):
            if type(v) is not int or not 0 <= v <= 255:
                raise ValueError("each channel must be a whole number 0 to 255")
        self._rgb = (r, g, b)

    @property
    def r(self):
        return self._rgb[0]

    @property
    def g(self):
        return self._rgb[1]

    @property
    def b(self):
        return self._rgb[2]

    @property
    def hex(self):
        return "#" + "".join(f"{v:02x}" for v in self._rgb)

    @property
    def brightness(self):
        r, g, b = self._rgb
        return round(0.299 * r + 0.587 * g + 0.114 * b, 1)

    @classmethod
    def from_hex(cls, text):
        m = re.fullmatch(r"#?([0-9a-fA-F]{6}|[0-9a-fA-F]{3})", text.strip())
        if m is None:
            raise ValueError(f"not a hex colour: {text!r}")
        digits = m.group(1)
        if len(digits) == 3:
            digits = "".join(d * 2 for d in digits)
        return cls(*(int(digits[i:i + 2], 16) for i in (0, 2, 4)))

    @classmethod
    def grey(cls, level):
        return cls(level, level, level)

    def lighter(self, fraction):
        return Colour(*(round(v + (255 - v) * fraction) for v in self._rgb))

    def darker(self, fraction):
        return Colour(*(round(v * (1 - fraction)) for v in self._rgb))

    def __add__(self, other):
        if not isinstance(other, Colour):
            return NotImplemented
        return Colour(*(round((a + b) / 2) for a, b in zip(self._rgb, other._rgb)))

    def __repr__(self):
        return f"Colour({self.r}, {self.g}, {self.b})"

    def __str__(self):
        return self.hex

    def __eq__(self, other):
        if not isinstance(other, Colour):
            return NotImplemented
        return self._rgb == other._rgb

    def __hash__(self):
        return hash(self._rgb)
