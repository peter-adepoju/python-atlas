import re

def parse_legs(text):
    legs = []
    for n, line in enumerate(text.splitlines(), start=1):
        line = line.strip()
        if not line or line.startswith("#"):
            continue
        m = re.fullmatch(r"(.+?) to (.+?): (\d+(?:\.\d+)?) km", line)
        if not m:
            raise ValueError(f"line {n}: can't read {line}")
        legs.append((m.group(1), m.group(2), float(m.group(3))))
    return legs

def test_busiest():
    "sort, then a window"
    assert busiest_hour([600, 610, 659, 660, 700, 30]) == (600, 3) and busiest_hour([]) == (None, 0) and busiest_hour([5]) == (5, 1)

    "100,000 bookings"

    times = [rng.randint(0, 1439) for _ in range(100_000)]
    start = time.perf_counter()

@@hint
Sort; keep a left index; for each right index, move left forward while `times[right] - times[left] > 59`.
@@solution
def busiest_hour(minutes):
    times = sorted(minutes)
    best, left = (None, 0), 0
    for right in range(len(times)):
        while times[right] - times[left] > 59:
            left += 1
        count = right - left + 1
        if count > best[1]:
            best = (times[left], count)
    return best

class Kit:
    def __init__(self):
        self.counts = {}

    def add(self, item, n=1):
        self.counts[item] = self.counts.get(item, 0) + n

    def take(self, item, n=1):
        if item not in self.counts:
            raise KeyError(f"no {item}")
        if self.counts[item] < n:
            raise ValueError(f"only {self.counts[item]} {item} left")
        self.counts[item] -= n
        if self.counts[item] == 0:
            del self.counts[item]

    def __getitem__(self, item):
        return self.counts.get(item, 0)

    def __len__(self):
        return len(self.counts)

    def __iter__(self):
        return iter(sorted(self.counts))

    def __repr__(self):
        return "Kit(" + ", ".join(f"{k}={self.counts[k]}" for k in self) + ")"

    "lazy, with context in errors"
    lines = ["# walks", "2026-01-04,6.0,105", "", "2026-01-18,7.5,98", "2026-02-01,11.0,150"]

def records(lines):
    for n, line in enumerate(lines, start=1):
        line = line.strip()
        if not line or line.startswith("#"):
            continue
        try:
            day, km, minutes = line.split(",")
            yield {"day": day, "km": float(km), "minutes": int(minutes)}
        except ValueError as err:
            err.add_note(f"line {n}")
            raise

def fastest(lines):
    best = None
    for r in records(lines):
        if best is None or r["minutes"] / r["km"] < best["minutes"] / best["km"]:
            best = r
    return best

import hashlib
import hmac

def _sig(text, key):
    return hmac.new(key, text.encode(), hashlib.sha256).hexdigest()

def share_link(route_id, expires, key):
    body = f"{route_id}.{expires}"
    return f"{body}.{_sig(body, key)}"

def open_link(link, now, key):
    parts = link.split(".")
    if len(parts) != 3 or not parts[0].isdigit() or not parts[1].isdigit():
        return 403, {"error": "invalid link"}
    route_id, expires, signature = parts
    if not hmac.compare_digest(_sig(f"{route_id}.{expires}", key), signature):
        return 403, {"error": "invalid link"}
    if now > int(expires):
        return 410, {"error": "expired"}
    return 200, {"route": int(route_id)}

Store each walk as `{"places": n, "booked": [...], "waiting": [...]}` in a dict; JSON round-trips that directly.
@@solution
import json

class Bookings:
    def __init__(self):
        self.walks = {}

    def add_walk(self, walk_id, places):
        if walk_id in self.walks:
            raise ValueError(f"walk {walk_id} exists")
        self.walks[walk_id] = {"places": places, "booked": [], "waiting": []}

    def book(self, walk_id, name):
        walk = self.walks[walk_id]
        if name in walk["booked"] or name in walk["waiting"]:
            return "already booked"
        if len(walk["booked"]) < walk["places"]:
            walk["booked"].append(name)
            return "booked"
        walk["waiting"].append(name)
        return "waiting"

    def cancel(self, walk_id, name):
        walk = self.walks[walk_id]
        if name in walk["waiting"]:
            walk["waiting"].remove(name)
            return None
        if name in walk["booked"]:
            walk["booked"].remove(name)
            if walk["waiting"]:
                promoted = walk["waiting"].pop(0)
                walk["booked"].append(promoted)
                return promoted
        return None

    def status(self, walk_id):
        walk = self.walks[walk_id]
        return {"booked": list(walk["booked"]), "waiting": list(walk["waiting"]), "free": walk["places"] - len(walk["booked"])}

    def to_json(self):
        return json.dumps(self.walks)

    @classmethod
    def from_json(cls, text):
        b = cls()
        b.walks = json.loads(text)
        return b
