URLs, queries and their encoding are where web bugs start: `&` and spaces inside a value must be encoded, and a query can repeat a key. The standard library gets all of it right, which is a recurring theme of today: use the tool that already handles the edge cases.

## Tests and packages

pytest: plain `assert`, `@pytest.mark.parametrize` for many cases, fixtures for setup (with `yield` for clean-up), `pytest.raises` for errors, `pytest.approx` for floats, mocks and fakes for slow or external things, and properties for invariants. A good suite fails on plausible mistakes (mutants). Packages: `pyproject.toml` with name, version (semver), dependencies with bounds, and scripts; builds give an sdist and a wheel.

:::example
from fares import fare

cases = [(10, 30, False, 500), (10, 10, False, 250), (10, 65, False, 250), (10, 64, False, 500), (10, 30, True, 450), (3, 70, True, 67)]
print([fare(*c[:3]) == c[3] for c in cases])
:::

## HTTP and the web

A request has a method, a path with a query, headers (case-insensitive names) and an optional body; a response has a status, headers and a body. Status codes: 200 OK, 201 Created, 204 No Content, 400 Bad Request, 401/403 for authentication and permission, 404 Not Found, 405 Method Not Allowed, 409 Conflict, 422 for invalid content, 500 for server errors. A web backend routes method and path to a handler, parses and validates the body (usually JSON), and returns JSON with the right status.

:::example
raw = "POST /v1/walks?draft=1 HTTP/1.1\r\nHost: waymark.example\r\nContent-Type: application/json\r\n\r\n{\"km\": 6.0}"
head, body = raw.split("\r\n\r\n", 1)
request_line, *header_lines = head.split("\r\n")
method, target, version = request_line.split(" ")
headers = {k.lower(): v.strip() for k, v in (h.split(":", 1) for h in header_lines)}
print(method, target, headers["content-type"], body)
:::

## Data, security and design

Databases: SQL with **parameters**, never string formatting; transactions so changes happen completely or not at all; `GROUP BY` and aggregates. Security: secrets from `secrets`, comparisons with `hmac.compare_digest`, HMAC signatures to detect tampering, escaping output, least privilege. Design: estimate, cache, queue, replicate, and say the trade-offs. Data: `csv`, `statistics`, and pandas for bigger tables.

:::example
import hmac, hashlib

key = b"waymark-demo-key"
payload = b'{"route": "Ridge walk", "user": 7}'
signature = hmac.new(key, payload, hashlib.sha256).hexdigest()
tampered = payload.replace(b"7", b"8")
print(len(signature), hmac.compare_digest(signature, hmac.new(key, tampered, hashlib.sha256).hexdigest()))
:::

from urllib.parse import parse_qs, urlsplit

def parse_request(raw):
    head, _, body = raw.partition("\r\n\r\n")
    request_line, *header_lines = head.split("\r\n")
    method, target, version = request_line.split(" ")
    url = urlsplit(target)
    headers = {}
    for line in header_lines:
        name, _, value = line.partition(":")
        headers[name.strip().lower()] = value.strip()
    return {"method": method, "path": url.path, "query": parse_qs(url.query), "headers": headers, "body": body}

@@tests

    "patterns, 404 and 405"
    r = Router()
    r.add("GET", "/v1/walks/<int:walk_id>", lambda walk_id: (200, {"id": walk_id}))
    r.add("DELETE", "/v1/walks/<int:walk_id>", lambda walk_id: (204, None))

@@solution
import re

class Router:
    def __init__(self):
        self.routes = []

    def add(self, method, pattern, handler):
        converters = {}
        def replace(m):
            kind, name = (m.group(1) or "str"), m.group(2)
            converters[name] = int if kind == "int" else str
            return rf"(?P<{name}>\d+)" if kind == "int" else rf"(?P<{name}>[^/]+)"
        regex = re.compile("^" + re.sub(r"<(?:(int):)?(\w+)>", replace, pattern) + "$")
        self.routes.append((method, regex, converters, handler))

    def dispatch(self, method, path):
        path_matched = False
        for m, regex, converters, handler in self.routes:
            found = regex.match(path)
            if not found:
                continue
            path_matched = True
            if m == method:
                return handler(**{k: converters[k](v) for k, v in found.groupdict().items()})
        if path_matched:
            return 405, {"error": "method not allowed"}
        return 404, {"error": "not found"}

import base64
import hashlib
import hmac
import json

def sign(data, key):
    payload = base64.urlsafe_b64encode(json.dumps(data, sort_keys=True).encode()).decode().rstrip("=")
    signature = hmac.new(key, payload.encode(), hashlib.sha256).hexdigest()
    return f"{payload}.{signature}"

def unsign(token, key):
    payload, dot, signature = token.partition(".")
    if not dot:
        return None
    expected = hmac.new(key, payload.encode(), hashlib.sha256).hexdigest()
    if not hmac.compare_digest(expected, signature):
        return None
    try:
        return json.loads(base64.urlsafe_b64decode(payload + "=" * (-len(payload) % 4)))
    except ValueError:
        return None

:::exercise id=challenge points=25 title="Challenge: a walks service in memory"
Write `WalksService` with `handle(method, path, body=None)` returning `(status, data)`, using a router of your own (or Day 4's): `POST /v1/walks` with a JSON body `{"km": number > 0, "route": non-empty string}` creates a walk and returns `201` and `{"id": N, "km": ..., "route": ...}` (ids from 1); invalid JSON gives `400 {"error": "invalid JSON"}`, invalid fields `422 {"error": "km must be positive"}` or `422 {"error": "route is required"}`; `GET /v1/walks/<id>` returns `200` and the walk or `404 {"error": "not found"}`; `GET /v1/walks` returns `200` and `{"walks": [...], "total_km": T}` (T to 1 place); `DELETE /v1/walks/<id>` returns `204` and `None`, or `404`; any other method on those paths `405 {"error": "method not allowed"}`; other paths `404`.
@@starter

import re

    ___
@@tests
def test_service():
    "a backend, end to end"

    assert s.handle("POST", "/v1/walks", '{"km": 6.0, "route": "Ridge walk"}') == (201, {"id": 1, "km": 6.0, "route": "Ridge walk"})

    assert s.handle("POST", "/v1/walks", '{"km": 2, "route": ""}') == (422, {"error": "route is required"})
    assert s.handle("GET", "/v1/walks/2") == (200, {"id": 2, "km": 3.5, "route": "Lake loop"})
    assert s.handle("GET", "/v1/walks") == (200, {"walks": [{"id": 1, "km": 6.0, "route": "Ridge walk"}, {"id": 2, "km": 3.5, "route": "Lake loop"}], "total_km": 9.5})
    assert s.handle("DELETE", "/v1/walks/1") == (204, None) and s.handle("DELETE", "/v1/walks/1") == (404, {"error": "not found"})
    assert s.handle("PUT", "/v1/walks/2") == (405, {"error": "method not allowed"}) and s.handle("GET", "/v2/x") == (404, {"error": "not found"})
    assert s.handle("POST", "/v1/walks", '{"km": true, "route": "x"}') == (422, {"error": "km must be positive"})
@@solution
import json
import re

class WalksService:
    def __init__(self):
        self.walks, self.next_id = {}, 1
        self.routes = [("POST", r"^/v1/walks$", self.create), ("GET", r"^/v1/walks$", self.list_walks),
                       ("GET", r"^/v1/walks/(\d+)$", self.get), ("DELETE", r"^/v1/walks/(\d+)$", self.delete)]

    def handle(self, method, path, body=None):
        matched = False
        for m, pattern, handler in self.routes:
            found = re.match(pattern, path)
            if not found:
                continue
            matched = True
            if m == method:
                args = [int(g) for g in found.groups()]
                return handler(*args, body=body) if m == "POST" else handler(*args)
        if matched:
            return 405, {"error": "method not allowed"}
        return 404, {"error": "not found"}

    def create(self, body=None):
        try:
            data = json.loads(body or "")
        except ValueError:
            return 400, {"error": "invalid JSON"}
        km = data.get("km") if isinstance(data, dict) else None
        if isinstance(km, bool) or not isinstance(km, (int, float)) or km <= 0:
            return 422, {"error": "km must be positive"}
        route = data.get("route")
        if not isinstance(route, str) or not route.strip():
            return 422, {"error": "route is required"}
        walk = {"id": self.next_id, "km": km, "route": route}
        self.walks[self.next_id] = walk
        self.next_id += 1
        return 201, dict(walk)

    def list_walks(self):
        walks = [dict(w) for _, w in sorted(self.walks.items())]
        return 200, {"walks": walks, "total_km": round(sum(w["km"] for w in walks), 1)}

    def get(self, walk_id):
        walk = self.walks.get(walk_id)
        return (200, dict(walk)) if walk else (404, {"error": "not found"})

    def delete(self, walk_id):
        if walk_id not in self.walks:
            return 404, {"error": "not found"}
        del self.walks[walk_id]
        return 204, None
