    """The grade for a mark: distinction from 80%, merit from 70%, pass from 60%, otherwise fail."""
    if out_of <= 0 or not 0 <= score <= out_of:

import re

def timetable(text):
    stops = {}
    for n, line in enumerate(text.splitlines(), start=1):
        line = line.strip()
        if not line or line.startswith("#"):
            continue
        stop, colon, rest = line.partition(":")
        if not colon:
            raise ValueError(f"line {n}: no stop")
        for t in (part.strip() for part in rest.split(",")):
            m = re.fullmatch(r"(\d\d):(\d\d)", t)
            if not m or int(m.group(1)) > 23 or int(m.group(2)) > 59:
                raise ValueError(f"line {n}: bad time {t}")
            stops.setdefault(stop.strip(), set()).add(t)
    return {stop: sorted(times) for stop, times in stops.items()}

import heapq

def route_today(paths, closed, start, end):
    links = {}
    for a, b, km in paths:
        if frozenset({a, b}) in closed:
            continue
        links.setdefault(a, []).append((b, km))
        links.setdefault(b, []).append((a, km))
    heap, done = [(0, [start])], set()
    while heap:
        km, stops = heapq.heappop(heap)
        node = stops[-1]
        if node == end:
            return round(km, 2), stops
        if node in done:
            continue
        done.add(node)
        for nxt, d in links.get(node, ()):
            if nxt not in done:
                heapq.heappush(heap, (km + d, stops + [nxt]))
    return None

import functools
from decimal import Decimal, ROUND_HALF_UP

PENNY = Decimal("0.01")

@functools.total_ordering
class Money:
    def __init__(self, amount, currency="GBP"):
        if isinstance(amount, float):
            raise TypeError("use a string or Decimal, not a float")
        self.amount = Decimal(amount).quantize(PENNY, rounding=ROUND_HALF_UP)
        self.currency = currency

    def _check(self, other):
        if not isinstance(other, Money):
            return NotImplemented
        if other.currency != self.currency:
            raise ValueError("currency mismatch")
        return other

    def __add__(self, other):
        other = self._check(other)
        return Money(self.amount + other.amount, self.currency)

    def __sub__(self, other):
        other = self._check(other)
        return Money(self.amount - other.amount, self.currency)

    def __eq__(self, other):
        if not isinstance(other, Money):
            return NotImplemented
        return (self.amount, self.currency) == (other.amount, other.currency)

    def __lt__(self, other):
        other = self._check(other)
        return self.amount < other.amount

    def __hash__(self):
        return hash((self.amount, self.currency))

    def __repr__(self):
        return f"Money('{self.amount}', '{self.currency}')"

    def split(self, n):
        pennies = int(self.amount * 100)
        base, extra = divmod(pennies, n)
        return [Money(Decimal(base + (1 if i < extra else 0)) / 100, self.currency) for i in range(n)]

import functools
import inspect

CHECKED = (int, float, str, bool)

def _ok(value, kind):
    if kind is bool:
        return isinstance(value, bool)
    if isinstance(value, bool):
        return False
    if kind is float:
        return isinstance(value, (int, float))
    return isinstance(value, kind)

def checked(fn):
    signature = inspect.signature(fn)

    @functools.wraps(fn)
    def wrapper(*args, **kwargs):
        bound = signature.bind(*args, **kwargs)
        bound.apply_defaults()
        for name, value in bound.arguments.items():
            kind = signature.parameters[name].annotation
            if kind in CHECKED and not _ok(value, kind):
                raise TypeError(f"{name} must be {kind.__name__}")
        return fn(*args, **kwargs)
    return wrapper

import math
import re
from collections import Counter

LINE = re.compile(r"([A-Z]+) (\S+) (\d{3}) (\d+)")

def log_summary(lines):
    statuses, times, paths, malformed = [], [], Counter(), 0
    for line in lines:
        m = LINE.fullmatch(line.strip())
        if not m:
            malformed += 1
            continue
        method, path, status, ms = m.groups()
        statuses.append(int(status))
        times.append(int(ms))
        paths[path.split("?")[0]] += 1
    classes = Counter(f"{s // 100}xx" for s in statuses)
    ordered = sorted(times)
    return {"requests": len(statuses), "malformed": malformed, "by_class": dict(sorted(classes.items())),
            "error_rate": round(100 * classes.get("5xx", 0) / len(statuses), 1) if statuses else 0.0,
            "p95_ms": ordered[math.ceil(0.95 * len(ordered)) - 1] if ordered else None,
            "top_paths": [p for p, _ in sorted(paths.items(), key=lambda kv: (-kv[1], kv[0]))[:3]]}
