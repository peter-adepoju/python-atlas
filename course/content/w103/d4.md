## The programme

A showcase runs on time or not at all: the last speaker shouldn't lose their slot because the first one ran over. So the programme lists each talk's start time, with a **gap** between talks for the next speaker to set up (two minutes is enough) and a **break** after every few talks, since listeners tire. The times are worked out from the start time in minutes, then formatted as `HH:MM` with zero padding.

For your own talk: ten minutes is about 1,200 spoken words, or one idea per minute. A common shape is **problem** (1 minute), **what it does** (2), **demo** (4), **how it works** (2) and **what you learned** (1). Rehearse with a timer and note when each section ends, then cut whatever runs long; the demo is usually the culprit.

## A demo that can fail safely

Live demos fail: the network drops, a login has expired, the projector changes the screen size. Plan for it:

- **Script the demo**: the exact commands or clicks, and what each should show.
- **Check the script** just before you go on, step by step, and stop at the first step that doesn't show what it should.
- **Keep a backup**: a recording or screenshots of each step. If a step fails live, say so calmly ("the login has expired; here's what it does") and switch to the backup. Listeners remember a calm recovery, not the failure.
- **Use sample data** you've prepared, never real personal data.

## Questions

Take questions **fairly**: if several people want to ask, let everyone ask once before anyone asks twice, keeping the order they raised their hands. Then answer:

- **Repeat the question** so everyone hears it, and so you're sure you've understood it.
- **Answer briefly**, then check: "does that answer it?"
- **Say "I don't know"** when you don't, and offer to find out. An honest "I haven't measured that" is better than a guess.
- **Take a hostile or off-topic question** politely, and offer to talk afterwards.

## Feedback

A simple form works: ratings from 1 to 5 for **clarity**, **the demo** and **depth**, and a comment. Average each rating per talk (to 1 decimal place), find each talk's **lowest** criterion (that's the one to work on), and read the comments for patterns. One "too fast" is a data point; two is a finding.

def programme(talks, start="14:00", gap=2, every=4, rest=10):
    hours, minutes = map(int, start.split(":"))
    now = hours * 60 + minutes
    slots = []
    for i, talk in enumerate(talks, start=1):
        slots.append((f"{now // 60:02d}:{now % 60:02d}", talk["presenter"], talk["project"]))
        now += talk["minutes"] + (rest if i % every == 0 else gap)
    return slots

@@starter

@@tests
PLAN = [("problem", 1), ("what it does", 2), ("demo", 4), ("how it works", 2), ("what I learned", 1)]

def test_pace():
    "each section's own length"
    assert over_time(PLAN, [1.1, 3.0, 8.5, 11.0, 11.8]) == [("demo", 1.5), ("how it works", 0.5)]
    assert over_time(PLAN, [1, 3, 7, 9, 10]) == []

    except ValueError:
        pass
    else:
        raise AssertionError("lengths")
@@solution
def over_time(plan, marks):
    if len(plan) != len(marks):
        raise ValueError("one mark per section")
    late, previous = [], 0
    for (section, minutes), mark in zip(plan, marks):
        took = mark - previous
        previous = mark
        if took > 1.2 * minutes + 1e-9:
            late.append((section, round(took - minutes, 1)))
    return late

@@tests
SCRIPT = [("waymark route Mill Ridge", "6.0 km"), ("waymark time Mill Ridge", "1 h 52 min"), ("waymark share 7", "code ready")]

def test_demo():
    "first failure, then stop"
    calls = []
    def good(command):
        calls.append(command)
        return dict(SCRIPT)[command]
    assert check_demo(SCRIPT, good) == ("ok", None) and len(calls) == 3
    calls.clear()
    def wrong(command):
        calls.append(command)
        return "6.5 km" if "route" in command else "?"
    assert check_demo(SCRIPT, wrong) == ("use backup", 1) and len(calls) == 1
    def offline(command):
        if "share" in command:
            raise ConnectionError("no network")
        return dict(SCRIPT)[command]
    assert check_demo(SCRIPT, offline) == ("use backup", 3) and check_demo([], good) == ("ok", None)
@@solution
def check_demo(script, run_step):
    for n, (command, expected) in enumerate(script, start=1):
        try:
            shown = run_step(command)
        except Exception:
            return "use backup", n
        if shown != expected:
            return "use backup", n
    return "ok", None

@@starter

    ___
@@tests
def test_order():
    "rounds"
    hands = ["Kofi", "Lin", "Kofi", "Kofi", "Mo", "Lin", "Nia"]
    assert question_order(hands) == ["Kofi", "Lin", "Mo", "Nia", "Kofi", "Lin", "Kofi"]

@@solution
def question_order(hands, limit=None):
    seen = {}
    rounds = []
    for name in hands:
        r = seen.get(name, 0)
        seen[name] = r + 1
        if r == len(rounds):
            rounds.append([])
        rounds[r].append(name)
    order = [name for round_ in rounds for name in round_]
    return order if limit is None else order[:limit]

from showcase import TALKS, FEEDBACK

CRITERIA = ["clarity", "demo", "depth"]

def summary(feedback, talks):
    out = {}
    for talk in talks:
        forms = [f for f in feedback if f["talk"] == talk["presenter"]]
        if not forms:
            out[talk["presenter"]] = {**{c: None for c in CRITERIA}, "work_on": None, "n": 0}
            continue
        averages = {c: round(sum(f[c] for f in forms) / len(forms), 1) for c in CRITERIA}
        lowest = min(CRITERIA, key=lambda c: averages[c])
        out[talk["presenter"]] = {**averages, "work_on": lowest, "n": len(forms)}
    return out

def programme(talks, start="14:00", gap=2, every=4, rest=10):
    hours, minutes = map(int, start.split(":"))
    now = hours * 60 + minutes
    slots = []
    for i, talk in enumerate(talks, start=1):
        slots.append((f"{now // 60:02d}:{now % 60:02d}", talk["presenter"], talk["project"]))
        now += talk["minutes"] + (rest if i % every == 0 else gap)

def summary(feedback, talks):
    out = {}
    for talk in talks:
        forms = [f for f in feedback if f["talk"] == talk["presenter"]]
        if not forms:
            out[talk["presenter"]] = {**{c: None for c in CRITERIA}, "work_on": None, "n": 0}
            continue
        averages = {c: round(sum(f[c] for f in forms) / len(forms), 1) for c in CRITERIA}
        lowest = min(CRITERIA, key=lambda c: averages[c])
        out[talk["presenter"]] = {**averages, "work_on": lowest, "n": len(forms)}
    return out

def report(talks, feedback, start="14:00"):
    s = summary(feedback, talks)
    blocks = []
    for time, presenter, project in programme(talks, start):
        lines = [f"{time} {presenter}: {project}"]
        info = s[presenter]
        if info["n"]:
            forms = "form" if info["n"] == 1 else "forms"
            lines.append(f"  clarity {info['clarity']:.1f}, demo {info['demo']:.1f}, depth {info['depth']:.1f} ({info['n']} {forms})")
            lines.append(f"  work on: {info['work_on']}")
        else:
            lines.append("  no feedback yet")
        blocks.append("\n".join(lines))
    too = sum(1 for f in feedback if "too" in f["comment"].lower().split())
    blocks.append(f'Comments mentioning "too": {too}')
    return "\n\n".join(blocks)
