## Where you started

On the first day you learned what a computer and a program are, used the console, and read your first error message. Think back to what was hard then: perhaps the difference between `=` and `==`, why `"3" + 4` fails, or what a loop variable is doing. Each of those is now something you'd explain to a beginner in a sentence, as you did in Week 101.

That's what learning looks like from the inside: the hard things become invisible. It's worth stopping, once, to see them again.

from journey import PHASES, DAYS_PER_WEEK

def course_counts(phases, days_per_week):
    weeks = sum(p["last"] - p["first"] + 1 for p in phases)
    rest = sum(len(p["rest"]) for p in phases)
    longest = max(phases, key=lambda p: (p["last"] - p["first"] + 1 - len(p["rest"]), -p["n"]))
    return {"weeks": weeks, "rest": rest, "teaching": weeks - rest, "days": (weeks - rest) * days_per_week,
            "longest": longest["title"]}

def certificate(name, date):
    lines = ["Python Atlas", "This certifies that", name, f"completed the course on {date}"]
    width = max(len(line) for line in lines) + 2
    edge = "+" + "-" * width + "+"
    return "\n".join([edge] + ["|" + line.center(width) + "|" for line in lines] + [edge])

import ast
from journey import FIRST

SKIP = {"Module", "Load", "Store"}

def kinds_used(code):
    return sorted({type(node).__name__ for node in ast.walk(ast.parse(code))} - SKIP)

def growth(then, now):
    a, b = kinds_used(then), kinds_used(now)
    return len(a), len(b), sorted(set(b) - set(a))

def test_timeline():
    "the picture"
    lines = timeline(PHASES).split("\n")
    assert len(lines) == 8 and lines[-1] == "Total: 96 teaching weeks, 8 rest weeks, 576 days."
    assert lines[0] == "1 Foundations                    " + "#" * 16 + "." * 1 + " 12"
    assert lines[6] == "7 Mastery                        " + "#" * 21 + "." * 3 + " 16"
    assert lines[3].endswith(" 17") and lines[3].count("#") == 23

@@solution
from journey import PHASES

def timeline(phases, width=24):
    longest_phase = max(p["last"] - p["first"] + 1 for p in phases)
    title_width = max(len(p["title"]) for p in phases)
    lines, teaching_total, rest_total = [], 0, 0
    for p in phases:
        weeks = p["last"] - p["first"] + 1
        rest = len(p["rest"])
        teaching = weeks - rest
        scale = width / longest_phase
        bar = "#" * round(teaching * scale) + "." * round(rest * scale)
        lines.append(f"{p['n']} {p['title']:<{title_width}} {bar} {teaching}")
        teaching_total += teaching
        rest_total += rest
    lines.append(f"Total: {teaching_total} teaching weeks, {rest_total} rest weeks, {teaching_total * 6} days.")
    return "\n".join(lines)
