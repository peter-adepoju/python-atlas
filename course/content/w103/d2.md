## Part C: a larger program

:::exercise id=c1 points=50 title="C: A library's loans" exam=yes
Write `Library`, the loans system for a small community library. Days are whole numbers (day 1, day 2 and so on); money is in **pence**, as integers. Raise `LoanError` (a subclass of `Exception` you define) with exactly the message given for each broken rule.

- **`add_book(isbn, title, copies=1)`** adds copies of a book; adding an ISBN that exists adds to its copies (and keeps the first title). **`join(member_id, name)`** adds a member; a member id that exists raises `"member exists"`.
- **`lend(isbn, member_id, day)`** lends a copy and returns the day it's due back (14 days later). Check, in this order: `"no such book"`, `"no such member"`, `"already borrowed"` (this member has this book), `"limit reached"` (3 loans at once), `"fines unpaid"` (the member owes more than 500 pence), `"no copies left"`. **`available(isbn)`** is the number of copies on the shelf.
- **`give_back(isbn, member_id, day)`** ends the loan (`"no such loan"` if there isn't one) and returns the fine for it: 20 pence for each day after the due day, at most 500 pence for one loan. Fines add to what the member owes; **`pay(member_id, pence)`** reduces it (never below 0) and returns what's left. **`owes(member_id)`** is the amount owed.
- **`overdue(day)`**: a list of `(member_id, isbn, days_late)` for loans past their due day on `day`, the most overdue first, then by member id, then by ISBN.
- **`save()`** returns the whole library as a JSON string; **`Library.load(text)`** (a class method) builds an equal library from it, loans and fines included.
- **`command(line)`** runs one line and returns a string: `add ISBN COPIES TITLE...`, `join ID NAME...`, `lend ISBN ID DAY` (gives `due DAY`), `return ISBN ID DAY` (gives `fine PENCE`), `pay ID PENCE` (gives `owes PENCE`) and `overdue DAY` (gives one line per loan, `ID ISBN DAYS`, joined with newlines, or `none`). A `LoanError` gives `error: MESSAGE`; an unknown command or the wrong number of words gives `error: bad command`.

import json


class LoanError(Exception):
    pass


LOAN_DAYS, LIMIT, MAX_OWED = 14, 3, 500
FINE_PER_DAY, FINE_CAP = 20, 500


class Library:
    def __init__(self):
        self.books = {}      # isbn -> {"title", "copies"}
        self.members = {}    # id -> {"name", "owes"}
        self.loans = {}      # (isbn, member_id) -> due day

    def add_book(self, isbn, title, copies=1):
        book = self.books.setdefault(isbn, {"title": title, "copies": 0})
        book["copies"] += copies

    def join(self, member_id, name):
        if member_id in self.members:
            raise LoanError("member exists")
        self.members[member_id] = {"name": name, "owes": 0}

    def available(self, isbn):
        out = sum(1 for (i, _) in self.loans if i == isbn)
        return self.books[isbn]["copies"] - out

    def lend(self, isbn, member_id, day):
        if isbn not in self.books:
            raise LoanError("no such book")
        if member_id not in self.members:
            raise LoanError("no such member")
        if (isbn, member_id) in self.loans:
            raise LoanError("already borrowed")
        if sum(1 for (_, m) in self.loans if m == member_id) >= LIMIT:
            raise LoanError("limit reached")
        if self.members[member_id]["owes"] > MAX_OWED:
            raise LoanError("fines unpaid")
        if self.available(isbn) <= 0:
            raise LoanError("no copies left")
        self.loans[(isbn, member_id)] = day + LOAN_DAYS
        return day + LOAN_DAYS

    def give_back(self, isbn, member_id, day):
        if (isbn, member_id) not in self.loans:
            raise LoanError("no such loan")
        due = self.loans.pop((isbn, member_id))
        fine = min(FINE_CAP, FINE_PER_DAY * max(0, day - due))
        self.members[member_id]["owes"] += fine
        return fine

    def owes(self, member_id):
        return self.members[member_id]["owes"]

    def pay(self, member_id, pence):
        member = self.members[member_id]
        member["owes"] = max(0, member["owes"] - pence)
        return member["owes"]

    def overdue(self, day):
        late = [(m, isbn, day - due) for (isbn, m), due in self.loans.items() if day > due]
        return sorted(late, key=lambda t: (-t[2], t[0], t[1]))

    def save(self):
        loans = [[isbn, m, due] for (isbn, m), due in self.loans.items()]
        return json.dumps({"books": self.books, "members": self.members, "loans": loans})

    @classmethod
    def load(cls, text):
        data = json.loads(text)
        lib = cls()
        lib.books = data["books"]
        lib.members = data["members"]
        lib.loans = {(isbn, m): due for isbn, m, due in data["loans"]}
        return lib

    def command(self, line):
        words = line.split()
        try:
            if not words:
                raise ValueError
            name, args = words[0], words[1:]
            if name == "add" and len(args) >= 3:
                self.add_book(args[0], " ".join(args[2:]), int(args[1]))
                return "ok"
            if name == "join" and len(args) >= 2:
                self.join(args[0], " ".join(args[1:]))
                return "ok"
            if name == "lend" and len(args) == 3:
                return f"due {self.lend(args[0], args[1], int(args[2]))}"
            if name == "return" and len(args) == 3:
                return f"fine {self.give_back(args[0], args[1], int(args[2]))}"
            if name == "pay" and len(args) == 2:
                return f"owes {self.pay(args[0], int(args[1]))}"
            if name == "overdue" and len(args) == 1:
                rows = self.overdue(int(args[0]))
                return "\n".join(f"{m} {isbn} {days}" for m, isbn, days in rows) or "none"
            raise ValueError
        except LoanError as e:
            return f"error: {e}"
        except ValueError:
            return "error: bad command"

import json

ROOMS = {"Oak": 8, "Elm": 4, "Ash": 12}


def parse_time(text):
    """'09:30' -> minutes after midnight."""
    hours, minutes = text.split(":")
    return int(hours) * 60 + int(minutes)


def overlaps(a, b):
    """Do two bookings (dicts with start and end in minutes) overlap?"""
    # Was <=: a booking ending at 10:00 clashed with one starting at 10:00.
    return a["start"] < b["end"] and b["start"] < a["end"]


def book(bookings, room, start, end, people, notes=None):
    """Add a booking if the room exists, is big enough and is free; return it, or None."""
    # An unknown room raised KeyError; it should simply fail.
    if room not in ROOMS or people > ROOMS[room]:
        return None
    # The default list was shared by every booking made without notes.
    notes = [] if notes is None else notes
    new = {"room": room, "start": parse_time(start), "end": parse_time(end), "people": people, "notes": notes}
    # Only bookings for the same room can clash.
    for old in bookings:
        if old["room"] == room and overlaps(old, new):
            return None
    bookings.append(new)
    return new


def usage(bookings, room):
    """The percentage of the working day (08:00 to 18:00) a room is booked, to 1 place."""
    total = 0
    for b in bookings:
        if b["room"] == room:
            total += b["end"] - b["start"]
    # round() with no places gave a whole number, not 1 decimal place.
    return round(total / 600 * 100, 1)


def load(path):
    # A bare except hid broken JSON and other bugs, and the file was never closed.
    try:
        with open(path) as f:
            return json.load(f)
    except FileNotFoundError:
        return []
