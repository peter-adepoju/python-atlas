    assert slow_square(4) == 16 and slow_square(4) == 16 and slow_square(5) == 25 and slow_square.calls == 2

@@solution
import functools
import threading

def locked_cache(func):
    store = {}
    lock = threading.Lock()

    @functools.wraps(func)
    def wrapper(*args):
        with lock:
            if args not in store:
                wrapper.calls += 1
                store[args] = func(*args)
            return store[args]

    wrapper.calls = 0
    wrapper.lock = lock
    return wrapper
:::

@@solution
def parallel_batches(deps):
    remaining = {task: set(needs) for task, needs in deps.items()}
    for needs in deps.values():
        for n in needs:
            remaining.setdefault(n, set())
    done, batches = set(), []
    while remaining:
        ready = sorted(t for t, needs in remaining.items() if needs <= done)
        if not ready:
            raise ValueError("cycle")
        batches.append(ready)
        done.update(ready)
        for t in ready:
            del remaining[t]
    return batches

def shortest_time(deps, durations):
    finish = {}
    for batch in parallel_batches(deps):
        for task in batch:
            start = max((finish[n] for n in deps.get(task, ())), default=0)
            finish[task] = start + durations[task]
    return max(finish.values(), default=0)

## Forge: a crawler

A **site** is a dict from URL (like `"a.org/shop"`, whose **host** is the part before the first `/`) to `(latency, links)`: fetching it takes `latency` time units and finds those links. URLs not in the dict are missing: fetching one takes 1 unit and fails. Write `crawl(site, start, workers=2, per_host=1, max_pages=None)`:

- The crawler keeps a **queue** of URLs to fetch, in the order they were discovered, starting with `start`, and a **seen** set so that no URL is queued twice.
- At most `workers` fetches run at once, and at most `per_host` of them for the same host. Whenever a worker is free, it takes the **first** URL in the queue whose host is below its limit (others wait their turn). At most `max_pages` fetches are started in total, if given.
- When a fetch finishes, its links are added to the queue in order (skipping seen ones), or the URL is recorded as an error if it was missing. Then free workers start new fetches.
- Fetches finishing at the same time are handled in the order they started.

Return a dict with `"order"` (URLs in the order their fetches finished), `"finished_at"` (URL to finish time), `"errors"` (missing URLs, in the order they finished) and `"time"` (when the last fetch finished).

@@hint
Write a helper `launch()` that, while a worker is free and the page limit isn't reached, scans the queue for the first URL whose host has room, removes it, counts it as active for its host and pushes its finish event. Call it once at the start and again after handling each finished fetch.
@@solution
import heapq
from collections import deque

def host_of(url):
    return url.split("/")[0]

def crawl(site, start, workers=2, per_host=1, max_pages=None):
    queue = deque([start])
    seen = {start}
    events = []
    active = {}
    busy = 0
    started = 0
    seq = 0
    now = 0
    order, finished_at, errors = [], {}, []

    def launch():
        nonlocal busy, started, seq
        while busy < workers and (max_pages is None or started < max_pages):
            pick = None
            for url in queue:
                if active.get(host_of(url), 0) < per_host:
                    pick = url
                    break
            if pick is None:
                return
            queue.remove(pick)
            latency = site[pick][0] if pick in site else 1
            active[host_of(pick)] = active.get(host_of(pick), 0) + 1
            busy += 1
            started += 1
            seq += 1
            heapq.heappush(events, (now + latency, seq, pick))

    launch()
    while events:
        now, _, url = heapq.heappop(events)
        busy -= 1
        active[host_of(url)] -= 1
        finished_at[url] = now
        order.append(url)
        if url in site:
            for link in site[url][1]:
                if link not in seen:
                    seen.add(link)
                    queue.append(link)
        else:
            errors.append(url)
        launch()
    return {"order": order, "finished_at": finished_at, "errors": errors, "time": now}
:::

:::think title="The same design with real threads"
Each part of the simulation has a real counterpart. The work queue is a `queue.Queue`, the workers are threads in a `ThreadPoolExecutor`, the per-host limit is a `Semaphore` per host, and the seen set is shared, so adding to it must happen under a `Lock` (check-then-act: "if not seen, mark seen and queue it" must be atomic, or two workers can both queue the same page). Fetching is I/O-bound, so the GIL doesn't get in the way: while one thread waits for a reply, others run. What the simulation adds is determinism: with real threads, the order of finishing would change from run to run, and a test that passed today could fail tomorrow. Designing so that results don't depend on that order is half of concurrent programming.
:::

## Forge bonus: rate limiting

:::exercise id=forge2 points=30 title="Forge bonus: rate limiting"
A polite crawler limits its **rate** as well as its concurrency. Write two thread-safe classes, each protecting its state with a real `threading` primitive:

- `TokenBucket(rate, capacity, clock=time.monotonic)`: holds up to `capacity` tokens, starting full, and gains `rate` tokens per second (by the clock), never above capacity. `try_acquire(n=1)` takes `n` tokens if there are enough and returns `True`, otherwise returns `False` (raising `ValueError` if `n` is more than the capacity); `tokens()` is the current number (rounded to 6 places); `wait_time(n=1)` is how long until `n` tokens would be available (0 if they are now, rounded to 6 places); `granted` and `refused` count the results of `try_acquire`. All state changes happen under one `threading.Lock`.
- `Limiter(max_running)`: a context manager allowing at most `max_running` blocks to be inside it at once, using a `threading.BoundedSemaphore` acquired **without waiting** (raising `RuntimeError("too many running")` if none is free), with `running` and `peak` counters updated under a lock, and the slot always released when the block ends.

    "a token bucket"
    now = [0.0]
    b = TokenBucket(rate=2, capacity=4, clock=lambda: now[0])
    assert [b.try_acquire() for _ in range(5)] == [True, True, True, True, False]
    assert b.tokens() == 0 and b.wait_time() == 0.5 and b.wait_time(3) == 1.5

    assert b.tokens() == 4 and b.wait_time(4) == 0.0

        b.try_acquire(5)
        assert False
    except ValueError:
        pass
    assert not b._lock.locked()

    "a concurrency limit"
    lim = Limiter(2)
    with lim:
        with lim:
            assert lim.running == 2
            try:
                with lim:
                    pass
                assert False
            except RuntimeError as e:
                assert str(e) == "too many running"
            assert lim.running == 2
        assert lim.running == 1
    assert lim.running == 0 and lim.peak == 2
    try:
        with lim:
            raise KeyError
    except KeyError:
        pass
    assert lim.running == 0
@@hint
Refill lazily: whenever the bucket is used, add `(now - last) * rate` tokens (capped), then set `last = now`. That way no background thread is needed to add tokens.
@@solution
import threading
import time

class TokenBucket:
    def __init__(self, rate, capacity, clock=time.monotonic):
        self.rate, self.capacity, self.clock = rate, capacity, clock
        self._tokens = float(capacity)
        self._last = clock()
        self._lock = threading.Lock()
        self.granted = self.refused = 0

    def _refill(self):
        now = self.clock()
        self._tokens = min(self.capacity, self._tokens + (now - self._last) * self.rate)
        self._last = now

    def tokens(self):
        with self._lock:
            self._refill()
            return round(self._tokens, 6)

    def try_acquire(self, n=1):
        if n > self.capacity:
            raise ValueError("more tokens than the bucket can hold")
        with self._lock:
            self._refill()
            if self._tokens >= n:
                self._tokens -= n
                self.granted += 1
                return True
            self.refused += 1
            return False

    def wait_time(self, n=1):
        with self._lock:
            self._refill()
            missing = n - self._tokens
            return 0.0 if missing <= 0 else round(missing / self.rate, 6)

class Limiter:
    def __init__(self, max_running):
        self._slots = threading.BoundedSemaphore(max_running)
        self._lock = threading.Lock()
        self.running = 0
        self.peak = 0

    def __enter__(self):
        if not self._slots.acquire(blocking=False):
            raise RuntimeError("too many running")
        with self._lock:
            self.running += 1
            self.peak = max(self.peak, self.running)
        return self

    def __exit__(self, *exc):
        with self._lock:
            self.running -= 1
        self._slots.release()
        return False
