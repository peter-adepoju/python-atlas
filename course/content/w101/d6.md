    md = "# Title here\n\n" + "word " * 390 + "\n\n" + fence + "python\n" + "code " * 500 + "\n" + fence + "\n"
    assert reading_time(md) == 2 and reading_time("") == 1 and reading_time("word " * 401) == 3
@@solution
import math

def reading_time(markdown, wpm=200):
    words, in_code = 0, False
    for line in markdown.splitlines():
        if line.startswith("`" * 3):
            in_code = not in_code
            continue
        if not in_code:
            words += len(line.lstrip("#").split())
    return max(1, math.ceil(words / wpm))

    "a mentor's habits"

@@solution
def question_summary(transcript, speaker="Mentor"):
    counts = {"open": 0, "leading": 0, "closed": 0}
    for line in transcript.splitlines():
        who, _, text = line.partition(":")
        text = text.strip().lower()
        if who != speaker or not text.endswith("?"):
            continue
        if any(p in text for p in ("don't you think", "isn't it", "shouldn't you")):
            counts["leading"] += 1
        elif text.startswith(("what", "how", "why", "which", "where")):
            counts["open"] += 1
        else:
            counts["closed"] += 1
    return counts
:::

## How this course is made

Every day of this course is a Markdown file with blocks like the ones you've used all along: predictions, exercises with checks, quizzes. A quiz block is a list of questions, each with `q`, `options`, the index of the right `answer`, and an `explain`. Writing good ones is exactly Day 2's work: run the code for the right answer, take wrong options from real misconceptions, and explain the gap in a sentence. Generating the block from data means the right answer can't be wrong, which, as Day 1 showed, is easy to get wrong by hand.

import contextlib
import io
import json

def quiz_block(quiz_id, items):
    lines = [f":::quiz id={quiz_id}"]
    for n, item in enumerate(items, start=1):
        if len(item["q"]) > 120:
            raise ValueError(f"question {n} is too long")
        code = item.get("code")
        if code is not None:
            buf = io.StringIO()
            with contextlib.redirect_stdout(buf):
                exec(code, {})
            right = buf.getvalue().strip()
            options = sorted({right, *[w for w in item["wrong"] if w != right]})
            answer = options.index(right)
        else:
            options, answer = item["options"], item["answer"]
        if not 2 <= len(options) <= 5:
            raise ValueError(f"question {n} needs 2 to 5 options")
        if not item.get("explain", "").strip():
            raise ValueError(f"question {n} needs an explanation")
        lines += [f"- q: {json.dumps(item['q'])}", f"  options: {json.dumps(options)}", f"  answer: {answer}",
                  f"  explain: {json.dumps(item['explain'])}"]
        if code is not None:
            lines.append("  code: |")
            lines += ["    " + line for line in code.splitlines()]
    lines.append(":::")
    return "\n".join(lines)

@@solution
import re

HARSH = ["wrong", "bad", "terrible", "obviously", "just"]

def review_message_problems(message):
    lines = message.strip().splitlines()
    problems = []
    if not lines or "thank" not in lines[0].lower():
        problems.append("start with thanks")
    if not any(line.startswith("What works well") for line in lines):
        problems.append("name what works")
    if sum(1 for line in lines if line.startswith("- line ")) > 3:
        problems.append("too many suggestions")
    last = next((line for line in reversed(lines) if line.strip()), "")
    if "?" not in last:
        problems.append("end with a question")
    lower = message.lower()
    problems += [f"sounds harsh: {w}" for w in HARSH if re.search(rf"\b{w}\b", lower)]
    return problems
