Escaping first means `<` in the text can't become a tag; then the Markdown syntax (backticks, brackets) is turned into tags, which are safe because they were added by the converter, not taken from the text. The order matters: escape, then add markup.

## Static sites

A **static site** is a folder of HTML files, built once and served as they are: no server code, no database, nothing to hack or keep running. Generators (MkDocs and Sphinx for documentation, and many more, all free) turn Markdown into HTML with a template. GitHub Pages serves a folder from a repository for free, which is how this course is published. The build is a program like any other, so it can be tested: that every page renders, every link resolves, and every code example runs (Day 1's checker).

## From Markdown to HTML

A small useful subset: headings (`#`, `##`, `###`), paragraphs separated by blank lines, fenced code blocks (with a language for highlighting), inline code, and links. Give each heading an **anchor** (a slug of its text, made unique with `-2`, `-3` for repeats), so a contents list and other pages can link straight to a section. Escape everything that came from the text.

## Links that work

Broken links are the commonest fault of documentation sites. Before publishing, collect every `href` that points inside the site (not starting with `http`), resolve it against the page it's on, and check the target exists, and, if it has a `#fragment`, that the target page has that anchor. A **sitemap** (`sitemap.xml`, one `<url><loc>...</loc></url>` per page) tells search engines what's there.

:::key

import html
import re

FENCE = "`" * 3

def inline(text):
    text = html.escape(text, quote=False)
    text = re.sub(r"`([^`]+)`", r"<code>\1</code>", text)
    return re.sub(r"\[([^\]]+)\]\(([^)\s]+)\)", lambda m: f'<a href="{html.escape(m.group(2))}">{m.group(1)}</a>', text)

def md_to_html(text):
    out, paragraph, code, lang = [], [], None, ""
    def flush():
        if paragraph:
            out.append("<p>" + inline(" ".join(paragraph)) + "</p>")
            paragraph.clear()
    for line in text.splitlines():
        if code is not None:
            if line.startswith(FENCE):
                cls = f' class="language-{lang}"' if lang else ""
                out.append(f"<pre><code{cls}>" + html.escape("\n".join(code), quote=False) + "</code></pre>")
                code = None
            else:
                code.append(line)
        elif line.startswith(FENCE):
            flush()
            code, lang = [], line[3:].strip()
        elif m := re.match(r"(#{1,3}) (.*)", line):
            flush()
            level = len(m.group(1))
            out.append(f"<h{level}>{inline(m.group(2).strip())}</h{level}>")
        elif not line.strip():
            flush()
        else:
            paragraph.append(line.strip())
    flush()
    return "\n".join(out)

import re

def contents(markdown):
    out, used = [], {}
    for line in markdown.splitlines():
        m = re.match(r"(#{1,3}) (.*)", line)
        if not m:
            continue
        text = m.group(2).strip()
        base = re.sub(r"[^a-z0-9]+", "-", text.lower()).strip("-")
        used[base] = used.get(base, 0) + 1
        anchor = base if used[base] == 1 else f"{base}-{used[base]}"
        out.append((len(m.group(1)), text, anchor))
    return out

from xml.sax.saxutils import escape

def sitemap(base_url, paths):
    base = base_url.rstrip("/")
    lines = ['<?xml version="1.0" encoding="UTF-8"?>', '<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">']
    lines += [f"<url><loc>{escape(base + '/' + p.lstrip('/'))}</loc></url>" for p in sorted(paths)]
    lines.append("</urlset>")
    return "\n".join(lines)

import posixpath
import re

def broken_links(site):
    problems = []
    for page in sorted(site):
        for href in re.findall(r'href="([^"]+)"', site[page]):
            if href.startswith(("http://", "https://", "mailto:")):
                continue
            path, _, fragment = href.partition("#")
            target = page if not path else posixpath.normpath(posixpath.join(posixpath.dirname(page), path))
            if target not in site:
                problems.append(f"{page}: {href} (missing page)")
            elif fragment and f'id="{fragment}"' not in site[target]:
                problems.append(f"{page}: {href} (missing anchor)")
    return problems
