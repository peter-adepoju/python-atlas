import io
import json
import re
from contextvars import ContextVar
from http import HTTPStatus
from urllib.parse import parse_qs, urlencode, urlsplit

_current = ContextVar("request")


class HTTPException(Exception):
    def __init__(self, code, description=None):
        self.code = code
        self.name = HTTPStatus(code).phrase
        self.description = description or self.name
        super().__init__(f"{code} {self.name}: {self.description}")


def abort(code, description=None):
    raise HTTPException(code, description)


class Headers:
    """Case-insensitive headers."""
    def __init__(self, items=()):
        self._items = [(k, str(v)) for k, v in (items.items() if isinstance(items, dict) else items)]

    def get(self, name, default=None):
        for k, v in self._items:
            if k.lower() == name.lower():
                return v
        return default

    def __getitem__(self, name):
        value = self.get(name)
        if value is None:
            raise KeyError(name)
        return value

    def __contains__(self, name):
        return self.get(name) is not None

    def __setitem__(self, name, value):
        self._items = [(k, v) for k, v in self._items if k.lower() != name.lower()] + [(name, str(value))]

    def items(self):
        return list(self._items)


class Args(dict):
    """Query or form values: .get(name) gives the first value; .getlist(name) all of them."""
    def __init__(self, pairs):
        super().__init__()
        self._lists = {}
        for k, v in pairs:
            self._lists.setdefault(k, []).append(v)
            dict.setdefault(self, k, v)

    def get(self, key, default=None, type=None):
        if key not in self:
            return default
        value = self[key]
        if type is not None:
            try:
                return type(value)
            except (TypeError, ValueError):
                return default
        return value

    def getlist(self, key):
        return list(self._lists.get(key, []))


class Request:
    def __init__(self, environ):
        self.environ = environ
        self.method = environ["REQUEST_METHOD"].upper()
        self.path = environ.get("PATH_INFO", "/") or "/"
        self.args = Args(_pairs(environ.get("QUERY_STRING", "")))
        items = [(k[5:].replace("_", "-").title(), v) for k, v in environ.items() if k.startswith("HTTP_")]
        if environ.get("CONTENT_TYPE"):
            items.append(("Content-Type", environ["CONTENT_TYPE"]))
        self.headers = Headers(items)
        length = int(environ.get("CONTENT_LENGTH") or 0)
        self.data = environ["wsgi.input"].read(length) if length else b""
        self.content_type = environ.get("CONTENT_TYPE", "")
        self.form = Args(_pairs(self.data.decode("utf-8"))) if self.content_type.startswith("application/x-www-form-urlencoded") else Args([])
        self.view_args = {}
        cookie = environ.get("HTTP_COOKIE", "")
        self.cookies = dict(p.strip().split("=", 1) for p in cookie.split(";") if "=" in p)

    @property
    def is_json(self):
        return self.content_type.split(";")[0].strip() == "application/json"

    def get_json(self, silent=False, force=False):
        if not (self.is_json or force):
            if silent:
                return None
            abort(415, "expected a JSON body (Content-Type: application/json)")
        try:
            return json.loads(self.data or b"null")
        except ValueError:
            if silent:
                return None
            abort(400, "the body isn't valid JSON")

    @property
    def json(self):
        return self.get_json()


def _pairs(text):
    return [(k, v) for k, vs in parse_qs(text, keep_blank_values=True).items() for v in vs]


class _Proxy:
    def __init__(self, getter):
        object.__setattr__(self, "_getter", getter)

    def __getattr__(self, name):
        return getattr(self._getter(), name)

    def __setattr__(self, name, value):
        setattr(self._getter(), name, value)

    except LookupError:
        raise RuntimeError("working outside of request context") from None


def _g():

class _G:
    def get(self, name, default=None):
        return self.__dict__.get(name, default)


request = _Proxy(_request)
g = _Proxy(_g)


class Response:
    def __init__(self, response=b"", status=200, headers=None, mimetype=None, content_type=None):
        if isinstance(response, str):
            response = response.encode("utf-8")
        self.data = response
        self.status_code = status
        self.headers = Headers(headers or {})
        if content_type or mimetype or "Content-Type" not in self.headers:
            ctype = content_type or (f"{mimetype}; charset=utf-8" if mimetype and mimetype.startswith("text/") else mimetype) or "text/html; charset=utf-8"
            self.headers["Content-Type"] = ctype

    @property
    def status(self):
        return f"{self.status_code} {HTTPStatus(self.status_code).phrase}"

    @property
    def text(self):
        return self.data.decode("utf-8")

    def get_json(self):
        return json.loads(self.data)

    @property
    def json(self):
        return self.get_json()

    def set_cookie(self, key, value="", max_age=None, path="/", secure=False, httponly=False, samesite=None):
        parts = [f"{key}={value}"]
        if max_age is not None:
            parts.append(f"Max-Age={max_age}")
        parts.append(f"Path={path}")
        if secure:
            parts.append("Secure")
        if httponly:
            parts.append("HttpOnly")
        if samesite:
            parts.append(f"SameSite={samesite}")
        self.headers._items.append(("Set-Cookie", "; ".join(parts)))


def jsonify(*args, **kwargs):
    data = args[0] if len(args) == 1 else (list(args) if args else kwargs)
    return Response(json.dumps(data), headers={"Content-Type": "application/json"})


def redirect(location, code=302):
    return Response(b"", code, {"Location": location})


def make_response(*args):
    rv = args[0] if len(args) == 1 else args
    return _App._to_response(rv)


def url_for(endpoint, **values):
    return _current.get()[2].url_for(endpoint, **values)


_CONVERTERS = {"int": (r"\d+", int), "float": (r"\d+(?:\.\d+)?", float), "path": (r".+", str), "string": (r"[^/]+", str)}


class _Rule:
    def __init__(self, rule, endpoint, view, methods):
        self.rule, self.endpoint, self.view = rule, endpoint, view
        self.methods = {m.upper() for m in methods} | ({"HEAD"} if "GET" in {m.upper() for m in methods} else set())
        regex, self.converters = "", {}
        for piece in re.split(r"(<[^>]+>)", rule):
            m = re.fullmatch(r"<(?:(\w+):)?(\w+)>", piece)
            if m:
                kind, name = m.group(1) or "string", m.group(2)
                pattern, convert = _CONVERTERS[kind]
                regex += f"(?P<{name}>{pattern})"
                self.converters[name] = convert
            else:
                regex += re.escape(piece)
        self.regex = re.compile(regex)

    def match(self, path):
        m = self.regex.fullmatch(path)
        if not m:
            return None
        return {k: self.converters[k](v) for k, v in m.groupdict().items()}

    def build(self, values):
        def fill(m):
            return str(values.pop(m.group(2)))
        path = re.sub(r"<(?:(\w+):)?(\w+)>", fill, self.rule)
        return path + (f"?{urlencode(values)}" if values else "")


class _Setup:
    def route(self, rule, methods=("GET",), endpoint=None):
        def decorate(view):
            self._add(rule, endpoint or view.__name__, view, methods)
            return view
        return decorate

    def get(self, rule, **kw):
        return self.route(rule, methods=["GET"], **kw)

    def post(self, rule, **kw):
        return self.route(rule, methods=["POST"], **kw)

    def put(self, rule, **kw):
        return self.route(rule, methods=["PUT"], **kw)

    def patch(self, rule, **kw):
        return self.route(rule, methods=["PATCH"], **kw)

    def delete(self, rule, **kw):
        return self.route(rule, methods=["DELETE"], **kw)


class Blueprint(_Setup):
    def __init__(self, name, import_name, url_prefix=""):
        self.name, self.url_prefix = name, url_prefix
        self.deferred = []

    def _add(self, rule, endpoint, view, methods):
        self.deferred.append((rule, endpoint, view, methods))


class _App(_Setup):
    def __init__(self, import_name):
        self.import_name = import_name
        self.config = {"TESTING": False}
        self.rules = []
        self.before_funcs, self.after_funcs, self.error_handlers = [], [], {}

    def _add(self, rule, endpoint, view, methods):
        self.rules.append(_Rule(rule, endpoint, view, methods))

    def register_blueprint(self, bp, url_prefix=None):
        prefix = url_prefix if url_prefix is not None else bp.url_prefix
        for rule, endpoint, view, methods in bp.deferred:
            self._add(prefix + rule, f"{bp.name}.{endpoint}", view, methods)

    def before_request(self, f):
        self.before_funcs.append(f)
        return f

    def after_request(self, f):
        self.after_funcs.append(f)
        return f

    def errorhandler(self, code):
        def decorate(f):
            self.error_handlers[code] = f
            return f
        return decorate

    def url_for(self, endpoint, **values):
        for r in self.rules:
            if r.endpoint == endpoint:
                return r.build(dict(values))
        raise LookupError(f"no endpoint {endpoint!r}")

    @staticmethod
    def _to_response(rv):
        status, headers = None, None
        if isinstance(rv, tuple):
            if len(rv) == 3:
                rv, status, headers = rv
            elif isinstance(rv[1], (dict, list)):
                rv, headers = rv
            else:
                rv, status = rv
        if isinstance(rv, Response):
            response = rv
        elif isinstance(rv, (dict, list)):
            response = jsonify(rv)
        elif isinstance(rv, (str, bytes)):
            response = Response(rv)
        elif rv is None:
            raise TypeError("the view function did not return a valid response")
        else:
            raise TypeError(f"can't turn {type(rv).__name__} into a response")
        if status is not None:
            response.status_code = status
        for k, v in (headers.items() if isinstance(headers, dict) else headers or []):
            response.headers[k] = v
        return response

    def _error(self, e):
        handler = self.error_handlers.get(e.code)
        if handler:
            return self._to_response(handler(e))
        body = f"<h1>{e.code} {e.name}</h1><p>{e.description}</p>"
        return Response(body, e.code)

    def _dispatch(self, req):
        path = req.path
        allowed = set()
        for r in self.rules:
            values = r.match(path)
            if values is None:
                continue
            if req.method in r.methods:
                req.view_args = values
                return r.view(**values)
            allowed |= r.methods
        if allowed:
            raise _MethodNotAllowed(sorted(allowed | {"OPTIONS"}))
        abort(404, "The requested URL was not found on the server.")

    def full_dispatch(self, environ):
        req = Request(environ)
        token = _current.set((req, _G(), self))
        try:
            try:
                rv = None
                for f in self.before_funcs:
                    rv = f()
                    if rv is not None:
                        break
                if rv is None:
                    rv = self._dispatch(req)
                response = self._to_response(rv)
            except _MethodNotAllowed as e:
                response = self._error(HTTPException(405))
                response.headers["Allow"] = ", ".join(e.allowed)
            except HTTPException as e:
                response = self._error(e)
            except Exception as e:
                if self.config.get("TESTING") and 500 not in self.error_handlers:
                    raise
                response = self._error(HTTPException(500)) if 500 not in self.error_handlers else self._to_response(self.error_handlers[500](e))
                response.status_code = 500
            for f in self.after_funcs:
                response = f(response) or response
            if req.method == "HEAD":
                response.headers["Content-Length"] = len(response.data)
                response.data = b""
            return response
        finally:
            _current.reset(token)

    def __call__(self, environ, start_response):
        response = self.full_dispatch(environ)
        if "Content-Length" not in response.headers:
            response.headers["Content-Length"] = len(response.data)
        start_response(response.status, response.headers.items())
        return [response.data]

    def test_client(self):
        return _TestClient(self)

    def __init__(self, allowed):
        self.allowed = allowed

    def __init__(self, app):
        self.app = app
        self.cookies = {}

    def open(self, path, method="GET", json=None, data=None, headers=None, query_string=None):
        url = urlsplit(path)
        query = url.query
        if query_string:
            query = (query + "&" if query else "") + urlencode(query_string, doseq=True)
        body, ctype = b"", None
        if json is not None:
            import json as _json
            body, ctype = _json.dumps(json).encode("utf-8"), "application/json"
        elif isinstance(data, dict):
            body, ctype = urlencode(data, doseq=True).encode("utf-8"), "application/x-www-form-urlencoded"
        elif data is not None:
            body = data.encode("utf-8") if isinstance(data, str) else data
        environ = {"REQUEST_METHOD": method.upper(), "PATH_INFO": url.path or "/", "QUERY_STRING": query,
                   "SERVER_NAME": "localhost", "SERVER_PORT": "80", "wsgi.url_scheme": "http",
                   "wsgi.input": io.BytesIO(body), "CONTENT_LENGTH": str(len(body))}
        headers = dict(headers or {})
        if ctype and not any(k.lower() == "content-type" for k in headers):
            environ["CONTENT_TYPE"] = ctype
        for k, v in headers.items():
            if k.lower() == "content-type":
                environ["CONTENT_TYPE"] = v
            else:
                environ["HTTP_" + k.upper().replace("-", "_")] = v
        if self.cookies:
            environ["HTTP_COOKIE"] = "; ".join(f"{k}={v}" for k, v in self.cookies.items())
        response = self.app.full_dispatch(environ)
        for k, v in response.headers.items():
            if k.lower() == "set-cookie":
                name, _, rest = v.partition("=")
                self.cookies[name] = rest.split(";")[0]
        return response

## Routes and views

```python
from flask import Flask

app = Flask(__name__)

@app.route("/books/<int:book_id>", methods=["GET", "DELETE"])
def book(book_id):
    ...
```

- `@app.route(rule, methods=[...])` registers a **view function** (also called an endpoint) for a URL rule; `methods` defaults to `["GET"]`. Flask 2 added shortcuts: `@app.get(rule)`, `@app.post(rule)`, `@app.put`, `@app.patch`, `@app.delete`.
- **Converters** in the rule turn path parts into arguments: `<name>` (any text without `/`), `<int:book_id>`, `<float:x>`, `<path:rest>` (slashes allowed). A path that doesn't fit, such as `/books/abc` for an `int`, is simply a `404`.
- A view **returns** the response: a `str` (sent as HTML), a `dict` or `list` (sent as JSON), a `Response` object, or a **tuple**: `(body, status)`, `(body, headers)` or `(body, status, headers)`. Returning `None` is an error: the classic forgotten `return`.
- Flask answers `405 Method Not Allowed` (with `Allow`) for a known path with the wrong method, and `HEAD` and `OPTIONS` automatically.

## Requests and responses

The current request is available as **`flask.request`**, a "context-local" object: it looks like a global, but each request (and each thread) sees its own.

- `request.method`, `request.path`, `request.headers` (case-insensitive), `request.cookies`.
- `request.args`: the query string, as a dict of first values: `request.args.get("page", 1, type=int)` converts, falling back to the default on bad input; `.getlist("tag")` gives every value.
- `request.form`: an HTML form's fields; `request.get_json()`: the JSON body (Flask refuses a body without `Content-Type: application/json` with `415`, and invalid JSON with `400`; `get_json(silent=True)` returns `None` instead).
- `jsonify(...)` makes a JSON response explicitly; `redirect(url, code=302)` a redirect (use `303` after a form); `make_response(body, status, headers)` a `Response` you can change before returning (adding cookies with `set_cookie`).
- **`url_for(endpoint, **values)`** builds a URL from a view's name and its parameters, so URLs are written in one place: `url_for("book", book_id=7)` gives `/books/7`.

## Errors and hooks

- **`abort(404)`** (or any code) stops the view at once and sends that error. `abort(422, "title is required")` adds a description.
- **`@app.errorhandler(404)`** replaces Flask's default error page: an API returns JSON, such as `{"error": e.description}, 404`. A `500` handler catches unexpected exceptions (with the details logged, not sent).
- **`@app.before_request`** functions run before every view; if one returns a response, the view isn't called at all, which suits authentication checks. **`@app.after_request`** functions receive the response and return it (perhaps with headers added) after every view.
- **`g`** holds data for the current request only: a `before_request` function can set `g.user`, and views read it.

:::example
from miniflask import Flask, request, jsonify, abort, g

app = Flask(__name__)
KEYS = {"k-ada": "ada", "k-bob": "bob"}
BOOKS = {1: {"id": 1, "title": "Emma"}}

@app.before_request
def identify():
    g.user = KEYS.get(request.headers.get("X-Api-Key"))
    if request.path.startswith("/api") and g.user is None:
        return jsonify(error="an API key is required"), 401

@app.after_request
def stamp(response):
    response.headers["X-Served-By"] = "atlas-shop"
    return response

@app.errorhandler(404)
def not_found(e):
    return {"error": e.description}, 404

@app.get("/api/books/<int:book_id>")
def get_book(book_id):
    if book_id not in BOOKS:
        abort(404, f"no book {book_id}")
    return {**BOOKS[book_id], "viewer": g.user}

client = app.test_client()
print(client.get("/api/books/1").status, client.get("/api/books/1").json)
r = client.get("/api/books/1", headers={"X-Api-Key": "k-ada"})
print(r.status, r.json, r.headers["X-Served-By"])
print(client.get("/api/books/9", headers={"X-Api-Key": "k-bob"}).json)

## Factories, blueprints and tests

As an app grows, two patterns keep it manageable:

- An **application factory**, `create_app(config=None)`, builds and returns a new app, applying the configuration it's given. Tests can then make a fresh app with test settings (an in-memory database, `TESTING = True`) for each test, and the real app reads its settings from the environment (Week 79).
- **Blueprints** group related routes into modules: `api = Blueprint("api", __name__, url_prefix="/api/v1")`, routes registered with `@api.get(...)`, and `app.register_blueprint(api)` in the factory. Endpoint names gain the blueprint's name: `url_for("api.book", book_id=7)`.

Testing uses the **test client** with pytest fixtures (Week 72):

```python
import pytest
from shop import create_app

@pytest.fixture
def client():
    app = create_app({"TESTING": True})
    return app.test_client()

def test_missing_book(client):
    response = client.get("/api/v1/books/999")
    assert response.status_code == 404 and response.get_json() == {"error": "no book 999"}
```

`client.get`, `.post(json=...)`, `.post(data={...})` (a form), `.put`, `.delete`, each with optional `headers=` and `query_string=`, return a response with `status_code`, `headers`, `data`, `text` and `get_json()`.

:::key
Flask apps register view functions with @app.route(rule, methods=...) (or @app.get, @app.post ...); converters (<int:id>, <path:p>) turn path parts into arguments, and non-matching paths are 404s. Views return a str, a dict or list (JSON), a Response, or a tuple with a status and headers; never None. flask.request gives method, path, headers, args (with get(..., type=int)), form, get_json() and cookies; jsonify, redirect, make_response and url_for build responses and URLs. abort(code, description) stops a view; errorhandler replaces error pages; before_request (which may answer early), after_request and g share per-request work. Application factories and blueprints structure larger apps, and test_client with pytest fixtures tests them without a server.
:::

        return {"error": e.description}, 404

    @app.get("/notes")
    def list_notes():
        q = request.args.get("q", "").lower()
        return [notes[k] for k in sorted(notes) if q in notes[k]["text"].lower()]

    @app.post("/notes")
    def add_note():
        data = request.get_json() or {}
        text = data.get("text") if isinstance(data, dict) else None
        if not isinstance(text, str) or not text.strip():
            return {"error": "text is required"}, 422
        new_id = max(notes, default=0) + 1
        notes[new_id] = {"id": new_id, "text": text}
        return notes[new_id], 201, {"Location": url_for("get_note", note_id=new_id)}

    @app.get("/notes/<int:note_id>")
    def get_note(note_id):
        if note_id not in notes:

        return notes[note_id]

    def delete_note(note_id):

from miniflask import Flask, request, g

def create_app(keys):

    count = [0]

    @app.before_request
    def authenticate():
        count[0] += 1
        g.user = keys.get(request.headers.get("X-Api-Key"))
        if request.path.startswith("/api/") and g.user is None:
            return {"error": "unauthorised"}, 401

        response.headers["X-Request-Count"] = str(count[0])
        return response

    @app.get("/")
    def home():
        return {"hello": g.user}

    @app.get("/api/me")
    def me():
        return {"user": g.user}

    return app

@@hint
Read each number with `request.args.get(name)` and convert it inside `try`, returning the 400 yourself. Filter, sort (`key=lambda b: b["year"], reverse=True` for `-year`), count, then slice. For the links, start from the query parameters the request actually had (`dict(request.args)`), set `page`, and pass them to `url_for` as keyword arguments.
@@solution
import math
from miniflask import Flask, request, url_for

def create_app(books):
    app = Flask(__name__)

    def number(name, default, low=None, high=None):
        raw = request.args.get(name)
        if raw is None:
            return default
        try:
            value = int(raw)
        except ValueError:
            raise ValueError(f"invalid {name}")
        if (low is not None and value < low) or (high is not None and value > high):
            raise ValueError(f"invalid {name}")
        return value

    @app.get("/books")
    def list_books():
        try:
            page = number("page", 1, low=1)
            per_page = number("per_page", 10, low=1, high=50)
            since = number("since", None)
        except ValueError as e:
            return {"error": str(e)}, 400
        sort = request.args.get("sort", "id")
        if sort not in ("id", "title", "year", "-year"):
            return {"error": "invalid sort"}, 400
        found = list(books)
        author = request.args.get("author")
        if author:
            found = [b for b in found if b["author"].lower() == author.lower()]
        if since is not None:
            found = [b for b in found if b["year"] >= since]
        found.sort(key=lambda b: b[sort.lstrip("-")], reverse=sort.startswith("-"))
        total = len(found)
        pages = max(1, math.ceil(total / per_page))
        items = found[(page - 1) * per_page: page * per_page]
        links = []
        params = dict(request.args)
        if page < pages:
            links.append(f'<{url_for("list_books", **{**params, "page": page + 1})}>; rel="next"')
        if 1 < page <= pages + 1:
            links.append(f'<{url_for("list_books", **{**params, "page": page - 1})}>; rel="prev"')
        headers = {"Link": ", ".join(links)} if links else {}
        return {"items": items, "page": page, "per_page": per_page, "total": total, "pages": pages}, 200, headers

    return app
