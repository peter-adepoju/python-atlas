class HTTPException(Exception):
    def __init__(self, code, description=None):
        self.code = code
        self.name = HTTPStatus(code).phrase
        self.description = description or self.name
        super().__init__(f"{code} {self.name}: {self.description}")


def abort(code, description=None):
    raise HTTPException(code, description)


class Headers:
    """Case-insensitive headers."""
    def __init__(self, items=()):
        self._items = [(k, str(v)) for k, v in (items.items() if isinstance(items, dict) else items)]

    def get(self, name, default=None):
        for k, v in self._items:
            if k.lower() == name.lower():
                return v
        return default

    def __getitem__(self, name):
        value = self.get(name)
        if value is None:
            raise KeyError(name)
        return value

    def __contains__(self, name):
        return self.get(name) is not None

    def __setitem__(self, name, value):
        self._items = [(k, v) for k, v in self._items if k.lower() != name.lower()] + [(name, str(value))]

    def items(self):
        return list(self._items)


class Args(dict):
    """Query or form values: .get(name) gives the first value; .getlist(name) all of them."""
    def __init__(self, pairs):
        super().__init__()
        self._lists = {}
        for k, v in pairs:
            self._lists.setdefault(k, []).append(v)
            dict.setdefault(self, k, v)

    def get(self, key, default=None, type=None):
        if key not in self:
            return default
        value = self[key]
        if type is not None:
            try:
                return type(value)
            except (TypeError, ValueError):
                return default
        return value

    def getlist(self, key):
        return list(self._lists.get(key, []))


class Request:
    def __init__(self, environ):
        self.environ = environ
        self.method = environ["REQUEST_METHOD"].upper()
        self.path = environ.get("PATH_INFO", "/") or "/"
        self.args = Args(_pairs(environ.get("QUERY_STRING", "")))
        items = [(k[5:].replace("_", "-").title(), v) for k, v in environ.items() if k.startswith("HTTP_")]
        if environ.get("CONTENT_TYPE"):
            items.append(("Content-Type", environ["CONTENT_TYPE"]))
        self.headers = Headers(items)
        length = int(environ.get("CONTENT_LENGTH") or 0)
        self.data = environ["wsgi.input"].read(length) if length else b""
        self.content_type = environ.get("CONTENT_TYPE", "")
        self.form = Args(_pairs(self.data.decode("utf-8"))) if self.content_type.startswith("application/x-www-form-urlencoded") else Args([])
        self.view_args = {}
        cookie = environ.get("HTTP_COOKIE", "")
        self.cookies = dict(p.strip().split("=", 1) for p in cookie.split(";") if "=" in p)

    @property
    def is_json(self):
        return self.content_type.split(";")[0].strip() == "application/json"

    def get_json(self, silent=False, force=False):
        if not (self.is_json or force):
            if silent:
                return None
            abort(415, "expected a JSON body (Content-Type: application/json)")
        try:
            return json.loads(self.data or b"null")
        except ValueError:
            if silent:
                return None
            abort(400, "the body isn't valid JSON")

    @property
    def json(self):
        return self.get_json()

class Response:
    def __init__(self, response=b"", status=200, headers=None, mimetype=None, content_type=None):

_CONVERTERS = {"int": (r"\d+", int), "float": (r"\d+(?:\.\d+)?", float), "path": (r".+", str), "string": (r"[^/]+", str)}


class _Rule:
    def __init__(self, rule, endpoint, view, methods):
        self.rule, self.endpoint, self.view = rule, endpoint, view
        self.methods = {m.upper() for m in methods} | ({"HEAD"} if "GET" in {m.upper() for m in methods} else set())
        regex, self.converters = "", {}
        for piece in re.split(r"(<[^>]+>)", rule):
            m = re.fullmatch(r"<(?:(\w+):)?(\w+)>", piece)
            if m:
                kind, name = m.group(1) or "string", m.group(2)
                pattern, convert = _CONVERTERS[kind]
                regex += f"(?P<{name}>{pattern})"
                self.converters[name] = convert
            else:
                regex += re.escape(piece)
        self.regex = re.compile(regex)

    def match(self, path):
        m = self.regex.fullmatch(path)
        if not m:
            return None
        return {k: self.converters[k](v) for k, v in m.groupdict().items()}

    def build(self, values):
        def fill(m):
            return str(values.pop(m.group(2)))
        path = re.sub(r"<(?:(\w+):)?(\w+)>", fill, self.rule)
        return path + (f"?{urlencode(values)}" if values else "")


class _Setup:
    def route(self, rule, methods=("GET",), endpoint=None):
        def decorate(view):
            self._add(rule, endpoint or view.__name__, view, methods)
            return view
        return decorate

    def get(self, rule, **kw):
        return self.route(rule, methods=["GET"], **kw)

    def post(self, rule, **kw):
        return self.route(rule, methods=["POST"], **kw)

    def put(self, rule, **kw):
        return self.route(rule, methods=["PUT"], **kw)

    def patch(self, rule, **kw):
        return self.route(rule, methods=["PATCH"], **kw)

    def delete(self, rule, **kw):
        return self.route(rule, methods=["DELETE"], **kw)


class Blueprint(_Setup):
    def __init__(self, name, import_name, url_prefix=""):
        self.name, self.url_prefix = name, url_prefix
        self.deferred = []

    def _add(self, rule, endpoint, view, methods):
        self.deferred.append((rule, endpoint, view, methods))


class _App(_Setup):
    def __init__(self, import_name):
        self.import_name = import_name
        self.config = {"TESTING": False}
        self.rules = []
        self.before_funcs, self.after_funcs, self.error_handlers = [], [], {}

    def _add(self, rule, endpoint, view, methods):
        self.rules.append(_Rule(rule, endpoint, view, methods))

    def register_blueprint(self, bp, url_prefix=None):
        prefix = url_prefix if url_prefix is not None else bp.url_prefix
        for rule, endpoint, view, methods in bp.deferred:
            self._add(prefix + rule, f"{bp.name}.{endpoint}", view, methods)

    def before_request(self, f):
        self.before_funcs.append(f)
        return f

    def after_request(self, f):
        self.after_funcs.append(f)
        return f

    def errorhandler(self, code):
        def decorate(f):
            self.error_handlers[code] = f
            return f
        return decorate

    def url_for(self, endpoint, **values):
        for r in self.rules:
            if r.endpoint == endpoint:
                return r.build(dict(values))
        raise LookupError(f"no endpoint {endpoint!r}")

    @staticmethod
    def _to_response(rv):
        status, headers = None, None
        if isinstance(rv, tuple):
            if len(rv) == 3:
                rv, status, headers = rv
            elif isinstance(rv[1], (dict, list)):
                rv, headers = rv
            else:
                rv, status = rv
        if isinstance(rv, Response):
            response = rv
        elif isinstance(rv, (dict, list)):
            response = jsonify(rv)
        elif isinstance(rv, (str, bytes)):
            response = Response(rv)
        elif rv is None:
            raise TypeError("the view function did not return a valid response")
        else:
            raise TypeError(f"can't turn {type(rv).__name__} into a response")
        if status is not None:
            response.status_code = status
        for k, v in (headers.items() if isinstance(headers, dict) else headers or []):
            response.headers[k] = v
        return response

    def _error(self, e):
        handler = self.error_handlers.get(e.code)
        if handler:
            return self._to_response(handler(e))
        body = f"<h1>{e.code} {e.name}</h1><p>{e.description}</p>"
        return Response(body, e.code)

    def _dispatch(self, req):
        path = req.path
        allowed = set()
        for r in self.rules:
            values = r.match(path)
            if values is None:
                continue
            if req.method in r.methods:
                req.view_args = values
                return r.view(**values)
            allowed |= r.methods
        if allowed:
            raise _MethodNotAllowed(sorted(allowed | {"OPTIONS"}))
        abort(404, "The requested URL was not found on the server.")

    def full_dispatch(self, environ):
        req = Request(environ)
        token = _current.set((req, _G(), self))
        try:
            try:
                rv = None
                for f in self.before_funcs:
                    rv = f()
                    if rv is not None:
                        break
                if rv is None:
                    rv = self._dispatch(req)
                response = self._to_response(rv)
            except _MethodNotAllowed as e:
                response = self._error(HTTPException(405))
                response.headers["Allow"] = ", ".join(e.allowed)
            except HTTPException as e:
                response = self._error(e)
            except Exception as e:
                if self.config.get("TESTING") and 500 not in self.error_handlers:
                    raise
                response = self._error(HTTPException(500)) if 500 not in self.error_handlers else self._to_response(self.error_handlers[500](e))
                response.status_code = 500
            for f in self.after_funcs:
                response = f(response) or response
            if req.method == "HEAD":
                response.headers["Content-Length"] = len(response.data)
                response.data = b""
            return response
        finally:
            _current.reset(token)

    def __call__(self, environ, start_response):
        response = self.full_dispatch(environ)
        if "Content-Length" not in response.headers:
            response.headers["Content-Length"] = len(response.data)
        start_response(response.status, response.headers.items())
        return [response.data]

"""A small subset of FastAPI (and of Pydantic's BaseModel) that runs in the page (Week 77).

Names and behaviour follow FastAPI's own, so code written against this module also runs with the
real FastAPI (pip install "fastapi[standard]"): FastAPI, BaseModel, Field, Query, Header, Depends,
HTTPException, JSONResponse, Response and TestClient (from fastapi.testclient on the real thing).
Parameters are read from your type hints: path parameters, query parameters, a JSON body from a
BaseModel, and dependencies. Bad input gets FastAPI's 422 response. It's a teaching copy: no
async, no OpenAPI page, and only common types (int, float, str, bool, lists, optional, models).

_MISSING = object()


# ---------- validation -------------------------------------------------------------------------

class ValidationError(Exception):
    def __init__(self, errors):
        super().__init__(f"{len(errors)} validation error(s)")
        self._errors = errors

    def errors(self):
        return list(self._errors)


class FieldInfo:
    def __init__(self, default=_MISSING, ge=None, le=None, gt=None, lt=None, min_length=None, max_length=None, alias=None):
        self.default, self.ge, self.le, self.gt, self.lt = default, ge, le, gt, lt
        self.min_length, self.max_length, self.alias = min_length, max_length, alias


def Field(default=_MISSING, **kw):

def _err(loc, msg, kind):
    return {"loc": list(loc), "msg": msg, "type": kind}


def _optional(tp):
    args = typing.get_args(tp)
    if (typing.get_origin(tp) is typing.Union or type(tp).__name__ == "UnionType") and type(None) in args:
        rest = [a for a in args if a is not type(None)]
        return True, rest[0] if len(rest) == 1 else typing.Union[tuple(rest)]
    return False, tp


def _convert(value, tp, loc, errors, from_text=False):
    """Convert value to tp, appending errors; returns the value or _MISSING."""
    optional, tp = _optional(tp)
    if value is None:
        if optional:
            return None
        errors.append(_err(loc, "Input should not be None", "none_required"))
        return _MISSING
    origin = typing.get_origin(tp)
    if tp is typing.Any or tp is inspect.Parameter.empty:
        return value
    if isinstance(tp, type) and issubclass(tp, BaseModel):
        if isinstance(value, tp):
            return value
        if not isinstance(value, dict):
            errors.append(_err(loc, "Input should be a valid dictionary or object", "model_type"))
            return _MISSING
        return tp._validate(value, loc, errors)
    if origin in (list, tuple, set):
        if not isinstance(value, (list, tuple)):
            errors.append(_err(loc, "Input should be a valid list", "list_type"))
            return _MISSING
        (item_type,) = typing.get_args(tp)[:1] or (typing.Any,)
        out = [_convert(v, item_type, list(loc) + [i], errors, from_text) for i, v in enumerate(value)]
        return _MISSING if any(v is _MISSING for v in out) else origin(out)
    if tp is bool:
        if isinstance(value, bool):
            return value
        if isinstance(value, str) and value.lower() in ("true", "1", "yes", "on", "false", "0", "no", "off"):
            return value.lower() in ("true", "1", "yes", "on")
        errors.append(_err(loc, "Input should be a valid boolean", "bool_parsing"))
        return _MISSING
    if tp is int:
        if isinstance(value, int) and not isinstance(value, bool):
            return value
        if isinstance(value, str) and re.fullmatch(r"[+-]?\d+", value.strip()):
            return int(value)
        if isinstance(value, float) and value.is_integer():
            return int(value)
        errors.append(_err(loc, "Input should be a valid integer", "int_parsing"))
        return _MISSING
    if tp is float:
        if isinstance(value, (int, float)) and not isinstance(value, bool):
            return float(value)
        if isinstance(value, str):
            try:
                return float(value)
            except ValueError:
                pass
        errors.append(_err(loc, "Input should be a valid number", "float_parsing"))
        return _MISSING
    if tp is str:
        if isinstance(value, str):
            return value
        if from_text:
            return str(value)
        errors.append(_err(loc, "Input should be a valid string", "string_type"))
        return _MISSING
    if tp is dict or origin is dict:
        if isinstance(value, dict):
            return value
        errors.append(_err(loc, "Input should be a valid dictionary", "dict_type"))
        return _MISSING
    return value


def _constraints(value, info, loc, errors):
    if value is _MISSING or value is None or info is None:
        return value
    checks = [("ge", lambda v, n: v >= n, "greater than or equal to", "greater_than_equal"),
              ("gt", lambda v, n: v > n, "greater than", "greater_than"),
              ("le", lambda v, n: v <= n, "less than or equal to", "less_than_equal"),
              ("lt", lambda v, n: v < n, "less than", "less_than")]
    for attr, ok, words, kind in checks:
        limit = getattr(info, attr)
        if limit is not None and not ok(value, limit):
            errors.append(_err(loc, f"Input should be {words} {limit}", kind))
            return _MISSING
    if info.min_length is not None and len(value) < info.min_length:
        errors.append(_err(loc, f"Value should have at least {info.min_length} item{'s' if info.min_length != 1 else ''}"
                           if not isinstance(value, str) else
                           f"String should have at least {info.min_length} character{'s' if info.min_length != 1 else ''}",
                           "string_too_short" if isinstance(value, str) else "too_short"))
        return _MISSING
    if info.max_length is not None and len(value) > info.max_length:
        errors.append(_err(loc, f"String should have at most {info.max_length} characters", "string_too_long"))
        return _MISSING
    return value


class BaseModel:
    """Declare fields as annotations, with defaults or Field(...)."""

    @classmethod
    def _fields(cls):
        hints = typing.get_type_hints(cls)
        fields = {}
        for name, tp in hints.items():
            if name.startswith("_"):
                continue
            raw = getattr(cls, name, _MISSING)
            info = raw if isinstance(raw, FieldInfo) else FieldInfo(raw)
            fields[name] = (tp, info)
        return fields

    @classmethod
    def _validate(cls, data, loc, errors):
        values = {}
        before = len(errors)
        for name, (tp, info) in cls._fields().items():
            key = info.alias or name
            if key in data:
                value = _constraints(_convert(data[key], tp, list(loc) + [key], errors), info, list(loc) + [key], errors)
            elif info.default is not _MISSING:
                value = copy.deepcopy(info.default)
            else:
                errors.append(_err(list(loc) + [key], "Field required", "missing"))
                value = _MISSING
            values[name] = value
        if len(errors) > before:
            return _MISSING
        obj = cls.__new__(cls)
        obj.__dict__.update(values)
        return obj

    def __init__(self, **data):
        errors = []
        obj = type(self)._validate(data, [], errors)
        if errors:
            raise ValidationError(errors)
        self.__dict__.update(obj.__dict__)

    @classmethod
    def model_validate(cls, data):
        errors = []
        obj = cls._validate(data, [], errors)
        if errors:
            raise ValidationError(errors)
        return obj

    def model_dump(self):
        def dump(v):
            if isinstance(v, BaseModel):
                return v.model_dump()
            if isinstance(v, (list, tuple, set)):
                return [dump(x) for x in v]
            return v
        return {name: dump(getattr(self, name)) for name in type(self)._fields()}

    def __eq__(self, other):
        return type(other) is type(self) and self.model_dump() == other.model_dump()

    def __repr__(self):
        inside = " ".join(f"{k}={v!r}" for k, v in self.model_dump().items())
        return f"{type(self).__name__}({inside})"

    def __init__(self, content=b"", status_code=200, headers=None, media_type=None):
        self.body = content.encode("utf-8") if isinstance(content, str) else content
        self.status_code = status_code
        self.headers = {k.lower(): v for k, v in (headers or {}).items()}
        if self.body:
            self.headers.setdefault("content-type", media_type or self.media_type)


class JSONResponse(Response):
    media_type = "application/json"

    def __init__(self, content=None, status_code=200, headers=None):
        super().__init__(json.dumps(content).encode("utf-8"), status_code, headers)


class Request:
    def __init__(self, method, path, query, headers, body):
        self.method, self.url_path, self.query_params = method, path, dict(query)
        self.headers = {k.lower(): v for k, v in headers.items()}
        self.body = body

    def json(self):
        return json.loads(self.body or b"null")

class FastAPI:
    def __init__(self, title="FastAPI", version="0.1.0"):
        self.title, self.version = title, version
        self.routes = []
        self.dependency_overrides = {}

    def api_route(self, path, methods, status_code=200, response_model=None):
        names = re.findall(r"{(\w+)(?::path)?}", path)
        regex = re.compile("^" + re.sub(r"{(\w+)(:path)?}", lambda m: f"(?P<{m.group(1)}>{'.+' if m.group(2) else '[^/]+'})", path) + "$")

        def decorate(func):
            for method in methods:
                self.routes.append((method, regex, names, func, status_code, response_model))
            return func
        return decorate

    def get(self, path, **kw):
        return self.api_route(path, ["GET"], **kw)

    def post(self, path, **kw):
        return self.api_route(path, ["POST"], **kw)

    def put(self, path, **kw):
        return self.api_route(path, ["PUT"], **kw)

    def patch(self, path, **kw):
        return self.api_route(path, ["PATCH"], **kw)

    def delete(self, path, **kw):
        return self.api_route(path, ["DELETE"], **kw)

    def _solve(self, func, request, path_values, errors, cleanups, cache):
        func = self.dependency_overrides.get(func, func)
        if func in cache:
            return cache[func]
        hints = typing.get_type_hints(func)
        kwargs = {}
        for name, param in inspect.signature(func).parameters.items():
            tp = hints.get(name, str if param.annotation is inspect.Parameter.empty else param.annotation)
            default = param.default
            if isinstance(default, _Depends):
                kwargs[name] = self._solve(default.dependency, request, path_values, errors, cleanups, cache)
            elif tp is Request:
                kwargs[name] = request
            elif name in path_values:
                kwargs[name] = _constraints(_convert(path_values[name], tp, ["path", name], errors, True),
                                            default if isinstance(default, FieldInfo) else None, ["path", name], errors)
            elif isinstance(tp, type) and issubclass(tp, BaseModel):
                try:
                    data = json.loads(request.body) if request.body else _MISSING
                except ValueError:
                    errors.append(_err(["body"], "JSON decode error", "json_invalid"))
                    continue
                if data is _MISSING:
                    if default is not inspect.Parameter.empty:
                        kwargs[name] = default
                    else:
                        errors.append(_err(["body"], "Field required", "missing"))
                    continue
                kwargs[name] = _convert(data, tp, ["body"], errors)
            else:
                info = default if isinstance(default, FieldInfo) else FieldInfo(default if default is not inspect.Parameter.empty else _MISSING)
                where = "header" if isinstance(default, Header) else "query"
                key = info.alias or (name.replace("_", "-") if where == "header" else name)
                if where == "header":
                    raw = request.headers.get(key.lower(), _MISSING)
                elif typing.get_origin(_optional(tp)[1]) is list:
                    found = [v for k, v in request._pairs if k == key]
                    raw = found if found else _MISSING
                else:
                    raw = request.query_params.get(key, _MISSING)
                if raw is _MISSING:
                    if info.default is _MISSING:
                        errors.append(_err([where, key], "Field required", "missing"))
                    else:
                        kwargs[name] = info.default
                    continue
                kwargs[name] = _constraints(_convert(raw, tp, [where, key], errors, True), info, [where, key], errors)
        if errors:
            return _MISSING
        result = func(**kwargs)
        if inspect.isgenerator(result):
            gen = result
            result = next(gen)
            cleanups.append(gen)
        cache[func] = result
        return result

    def handle(self, method, path, query_pairs, headers, body):
        request = Request(method, path, query_pairs, headers, body)
        request._pairs = list(query_pairs)
        allowed = []
        for m, regex, names, func, status_code, response_model in self.routes:
            match = regex.match(path)
            if not match:
                continue
            if m != method:
                allowed.append(m)
                continue
            errors, cleanups = [], []
            try:
                try:
                    result = self._solve(func, request, match.groupdict(), errors, cleanups, {})
                    if errors:
                        return JSONResponse({"detail": errors}, 422)
                except HTTPException as e:
                    return JSONResponse({"detail": e.detail}, e.status_code, e.headers)
                return self._respond(result, status_code, response_model)
            finally:
                for gen in reversed(cleanups):
                    try:
                        next(gen)
                    except StopIteration:
                        pass
        if allowed:
            return JSONResponse({"detail": "Method Not Allowed"}, 405, {"allow": ", ".join(sorted(set(allowed)))})
        return JSONResponse({"detail": "Not Found"}, 404)

    @staticmethod
    def _respond(result, status_code, response_model):
        if isinstance(result, Response):
            return result
        if response_model is not None:
            data = result.model_dump() if isinstance(result, BaseModel) else result
            if isinstance(data, list):
                result = [{k: d[k] for k in response_model._fields()} if isinstance(d, dict) else d.model_dump() for d in
                          [x.model_dump() if isinstance(x, BaseModel) else x for x in data]]
            else:
                result = {k: data[k] for k in response_model._fields()}
        if status_code == 204:
            return Response(b"", 204)

        def plain(v):
            if isinstance(v, BaseModel):
                return v.model_dump()
            if isinstance(v, (list, tuple)):
                return [plain(x) for x in v]
            if isinstance(v, dict):
                return {k: plain(x) for k, x in v.items()}
            return v
        return JSONResponse(plain(result), status_code)

from miniflask import Flask, Response, abort

def problem_errors(app):
    def respond(status, title, detail):
        body = {"type": "about:blank", "title": title, "status": status, "detail": detail}
        return Response(json.dumps(body), status, {"Content-Type": "application/problem+json"})

    for code in (400, 401, 403, 404, 405, 409, 422):
        app.errorhandler(code)(lambda e: respond(e.code, e.name, e.description))

from minifast import BaseModel, Field, ValidationError

class Settings(BaseModel):
    secret_key: str = Field(min_length=16)
    debug: bool = False
    port: int = Field(8000, ge=1, le=65535)
    allowed_hosts: list[str] = ["localhost"]

def from_env(env):
    data = {}
    for name in ("secret_key", "debug", "port"):
        if name.upper() in env:
            data[name] = env[name.upper()]
    if "ALLOWED_HOSTS" in env:
        data["allowed_hosts"] = [h.strip() for h in env["ALLOWED_HOSTS"].split(",") if h.strip()]
    try:
        return Settings.model_validate(data)
    except ValidationError as e:
        raise RuntimeError("; ".join(f"{str(err['loc'][0]).upper()}: {err['msg']}" for err in e.errors())) from None

## Forge: an events-booking API

Build `create_app(env, clock)`: a miniflask app for booking places at events. The starter gives you `hash_password`, `verify_password`, `encode_jwt`, `decode_jwt` (which raises `ValueError` for any bad or expired token) and `problem(status, detail)`, a problem-details response. Every error the API itself reports uses `problem`; a `404` or `405` from routing does too.

- **Settings**: `SECRET_KEY` is required (raise `RuntimeError` at once if it's missing); `TOKEN_MINUTES` (default `15`) must be an integer (`RuntimeError` otherwise).
- `GET /healthz` returns `{"status": "ok"}`.
- `POST /users` with `{"email", "password"}`: emails lower-cased and containing `@`, passwords at least 8 characters (`422`, detail `"an email and a password of at least 8 characters are needed"`), duplicates `409` (`"email already registered"`). Returns `201` with `{"id", "email"}`. The **first** user is an admin.
- `POST /tokens` with the same fields returns `{"token": ...}` (a JWT with `sub` the user id and `exp` `clock() + TOKEN_MINUTES * 60`), or `401` (`"wrong email or password"`).
- `GET /events` (public) with optional `city` (any case), `page` (from 1) and `per_page` (1 to 50, default 10), else `400`; returns `{"items": [...], "page": P, "total": T}`, events by id.
- `POST /events` (admins only: `401` `"log in first"` without a valid token, `403` `"only admins can create events"`) with `{"title", "city", "capacity"}` (non-empty strings, and an integer from 1 to 1000, else `422`); `201` with the event and a `Location`.
- `GET /events/<id>`: the event (`404` `"no event N"`). Every event shown anywhere has `id`, `title`, `city`, `capacity` and `seats_left`.
- `POST /events/<id>/bookings` (logged in): `201` with `{"event_id", "user_id"}`; `404` for no event; `409` `"already booked"` or `"sold out"`.
- `DELETE /events/<id>/bookings/me` (logged in): `204`, or `404` `"no booking to cancel"`.

    u = hmac.new(password.encode(), salt + b"\x00\x00\x00\x01", hashlib.sha256).digest()

    """The payload of a valid, unexpired token; raises ValueError otherwise."""
    try:
        h, p, s = token.split(".")
        ok_alg = json.loads(_unb64(h)).get("alg") == "HS256"

    if now > payload.get("exp", 0):

    return payload

    users, events, bookings = {}, {}, set()

    def current_user():
        scheme, _, token = request.headers.get("Authorization", "").partition(" ")
        if scheme != "Bearer":
            return None
        try:
            payload = decode_jwt(token, key, clock())
        except ValueError:
            return None
        return payload.get("sub") if payload.get("sub") in users else None

    def public_event(event):
        taken = sum(1 for e, _ in bookings if e == event["id"])
        return {**event, "seats_left": event["capacity"] - taken}

    @app.errorhandler(404)
    def not_found(e):
        return problem(404, e.description)

    @app.errorhandler(405)
    def not_allowed(e):
        return problem(405, e.description)

    @app.get("/healthz")
    def healthz():
        return {"status": "ok"}

    @app.post("/users")
    def register():
        data = request.get_json(silent=True) or {}
        email = str(data.get("email", "")).strip().lower()
        password = data.get("password")
        if "@" not in email or not isinstance(password, str) or len(password) < 8:
            return problem(422, "an email and a password of at least 8 characters are needed")
        if any(u["email"] == email for u in users.values()):
            return problem(409, "email already registered")
        user_id = len(users) + 1
        users[user_id] = {"email": email, "hash": hash_password(password), "admin": user_id == 1}
        return {"id": user_id, "email": email}, 201

    @app.post("/tokens")
    def login():
        data = request.get_json(silent=True) or {}
        email = str(data.get("email", "")).strip().lower()
        password = data.get("password")
        match = next((uid for uid, u in users.items() if u["email"] == email), None)
        if match is None or not isinstance(password, str) or not verify_password(password, users[match]["hash"]):
            return problem(401, "wrong email or password")
        return {"token": encode_jwt({"sub": match, "exp": clock() + minutes * 60}, key)}

    @app.get("/events")
    def list_events():
        try:
            page = int(request.args.get("page", "1"))
            per_page = int(request.args.get("per_page", "10"))
        except ValueError:
            return problem(400, "page and per_page must be integers")
        if page < 1 or not 1 <= per_page <= 50:
            return problem(400, "page must be at least 1 and per_page from 1 to 50")
        city = request.args.get("city")
        found = [events[k] for k in sorted(events) if not city or events[k]["city"].lower() == city.lower()]
        items = found[(page - 1) * per_page: page * per_page]
        return {"items": [public_event(e) for e in items], "page": page, "total": len(found)}

    @app.post("/events")
    def create_event():
        user = current_user()
        if user is None:
            return problem(401, "log in first")
        if not users[user]["admin"]:
            return problem(403, "only admins can create events")
        data = request.get_json(silent=True) or {}
        title, city, capacity = data.get("title"), data.get("city"), data.get("capacity")
        if not (isinstance(title, str) and title.strip() and isinstance(city, str) and city.strip()
                and type(capacity) is int and 1 <= capacity <= 1000):
            return problem(422, "title, city and a capacity from 1 to 1000 are needed")
        event_id = max(events, default=0) + 1
        events[event_id] = {"id": event_id, "title": title, "city": city, "capacity": capacity}
        return public_event(events[event_id]), 201, {"Location": url_for("get_event", event_id=event_id)}

    @app.get("/events/<int:event_id>")
    def get_event(event_id):
        if event_id not in events:
            return problem(404, f"no event {event_id}")
        return public_event(events[event_id])

    @app.post("/events/<int:event_id>/bookings")
    def book(event_id):
        user = current_user()
        if user is None:
            return problem(401, "log in first")
        if event_id not in events:
            return problem(404, f"no event {event_id}")
        if (event_id, user) in bookings:
            return problem(409, "already booked")
        if public_event(events[event_id])["seats_left"] <= 0:
            return problem(409, "sold out")
        bookings.add((event_id, user))
        return {"event_id": event_id, "user_id": user}, 201

    @app.delete("/events/<int:event_id>/bookings/me")
    def cancel(event_id):
        user = current_user()
        if user is None:
            return problem(401, "log in first")
        if (event_id, user) not in bookings:
            return problem(404, "no booking to cancel")
        bookings.discard((event_id, user))
        return "", 204

    return app

from minifast import FastAPI, BaseModel, Field, Query, Header, Depends, HTTPException

def create_api(tokens, events):
    app = FastAPI()
    bookings = set()

    class EventIn(BaseModel):
        title: str = Field(min_length=1, max_length=100)
        city: str = Field(min_length=1)
        capacity: int = Field(ge=1, le=1000)

    class EventOut(BaseModel):
        id: int
        title: str
        city: str
        capacity: int
        seats_left: int

    def current_user(authorization: str | None = Header(None)):
        scheme, _, token = (authorization or "").partition(" ")
        if scheme != "Bearer" or token not in tokens:
            raise HTTPException(401, "log in first")
        return tokens[token]

    def pagination(page: int = Query(1, ge=1), per_page: int = Query(10, ge=1, le=50)):
        return page, per_page

    def show(event):
        taken = sum(1 for e, _ in bookings if e == event["id"])
        return {**event, "seats_left": event["capacity"] - taken}

    @app.get("/events")
    def list_events(city: str | None = None, page: tuple = Depends(pagination)):
        number, size = page
        found = [events[k] for k in sorted(events) if not city or events[k]["city"].lower() == city.lower()]
        return [EventOut(**show(e)) for e in found[(number - 1) * size: number * size]]

    @app.post("/events", status_code=201, response_model=EventOut)
    def create_event(event: EventIn, user: tuple = Depends(current_user)):
        if not user[1]:
            raise HTTPException(403, "only admins can create events")
        new_id = max(events, default=0) + 1
        events[new_id] = {"id": new_id, **event.model_dump()}
        return show(events[new_id])

    @app.get("/events/{event_id}", response_model=EventOut)
    def get_event(event_id: int):
        if event_id not in events:
            raise HTTPException(404, f"no event {event_id}")
        return show(events[event_id])

    @app.post("/events/{event_id}/bookings", status_code=201)
    def book(event_id: int, user: tuple = Depends(current_user)):
        if event_id not in events:
            raise HTTPException(404, f"no event {event_id}")
        if (event_id, user[0]) in bookings:
            raise HTTPException(409, "already booked")
        if show(events[event_id])["seats_left"] <= 0:
            raise HTTPException(409, "sold out")
        bookings.add((event_id, user[0]))
        return {"event_id": event_id, "user_id": user[0]}

    return app
