DOOR = {("closed", "open"): "opened", ("opened", "close"): "closed", ("closed", "lock"): "locked", ("locked", "unlock"): "closed"}

class Door:
    def __init__(self):
        self.state = "closed"

    def handle(self, event):
        key = (self.state, event)
        if key not in DOOR:
            allowed = sorted(e for (s, e) in DOOR if s == self.state)
            raise ValueError(f"can't {event} when {self.state}; allowed: {allowed}")
        self.state = DOOR[key]
        return self.state

class Chunks:
    def __init__(self, text, size):
        if size < 1:
            raise ValueError("size must be at least 1")
        self.text, self.size = text, size

    def __iter__(self):
        return ChunkIterator(self.text, self.size)

class ChunkIterator:
    def __init__(self, text, size):
        self.text, self.size, self.pos = text, size, 0

    def __iter__(self):
        return self

    def __next__(self):
        if self.pos >= len(self.text):
            raise StopIteration
        piece = self.text[self.pos:self.pos + self.size]
        self.pos += self.size
        return piece

## Forge: a task board

A team keeps its work on a board: tasks move from to-do, to doing, to done, and every change can be undone. Here's the specification, layer by layer.

**Domain.**

- `Status`, an Enum: `TODO = "todo"`, `DOING = "doing"`, `DONE = "done"`, `ARCHIVED = "archived"`.
- `FLOW`, a transition table from `(Status, event)` to `Status`: todo + `start` → doing; doing + `pause` → todo; doing + `finish` → done; done + `reopen` → todo; done + `archive` → archived.
- `Task`, a dataclass: `task_id` (int), `title`, `status` (default `Status.TODO`) and `assignee` (default `None`).
- `TaskError`, with `NoSuchTask` and `BadMove` as kinds of it.

**Infrastructure.** `MemoryTasks`, a repository with `add(task)`, `get(task_id)` (a **copy**, or `None`), `save(task)`, `delete(task_id)` and `all()` (copies, by id).

**Commands** (each with `execute()` and `undo()`, working through the repository):

- `Create(repo, title)`: adds a task with the next id (1 more than the largest ever used by this repository's tasks, starting at 1) and remembers it; undo deletes it. A blank title raises `TaskError`.
- `Move(repo, task_id, event)`: applies `FLOW`, remembering the old status; an unknown task raises `NoSuchTask`, an event not allowed raises `BadMove`; undo restores the old status.
- `Assign(repo, task_id, person)`: sets the assignee (`None` to unassign), remembering the old one.

**Application.** `Board(repo)`, the service: `create(title)` (returns the new id), `move(task_id, event)` (returns the new `Status`), `assign(task_id, person)`, `undo()` and `redo()` (return `True` or `False`, as in Day 2's `History`), all done through commands so every change can be undone; and `columns()`, a dictionary from each `Status`'s value, in the Enum's order, to the titles of its tasks, by id, leaving out empty statuses.

**Presentation.** `handle(line, board)`, returning text:

| Line | Result |
|---|---|
| `add Write report` | `added 1` |
| `start 1` (or any event) | `1 is now doing` |
| `assign 1 Ada` / `assign 1 -` | `1 assigned to Ada` / `1 unassigned` |
| `undo` / `redo` | `undone` / `redone`, or `nothing to undo` / `nothing to redo` |
| `show` | one line per non-empty column, like `todo: Write report, Book room` (or `empty board`) |

A `TaskError` becomes `error: ` and its message; anything else unrecognised gives `unknown command`.

**Composition root.** `build()` returns a `Board` with a new `MemoryTasks`.

from dataclasses import dataclass, replace
from enum import Enum

class Status(Enum):
    TODO = "todo"
    DOING = "doing"
    DONE = "done"
    ARCHIVED = "archived"

FLOW = {
    (Status.TODO, "start"): Status.DOING,
    (Status.DOING, "pause"): Status.TODO,
    (Status.DOING, "finish"): Status.DONE,
    (Status.DONE, "reopen"): Status.TODO,
    (Status.DONE, "archive"): Status.ARCHIVED,
}

EVENTS = {event for _, event in FLOW}

@dataclass
class Task:
    task_id: int
    title: str
    status: Status = Status.TODO
    assignee: str | None = None

class TaskError(Exception):
    pass

class NoSuchTask(TaskError):
    pass

class BadMove(TaskError):
    pass

class MemoryTasks:
    def __init__(self):
        self._tasks = {}

    def add(self, task):
        self._tasks[task.task_id] = replace(task)

    def get(self, task_id):
        t = self._tasks.get(task_id)
        return replace(t) if t else None

    def save(self, task):
        self._tasks[task.task_id] = replace(task)

    def delete(self, task_id):
        del self._tasks[task_id]

    def all(self):
        return [replace(self._tasks[k]) for k in sorted(self._tasks)]

def find(repo, task_id):
    task = repo.get(task_id)
    if task is None:
        raise NoSuchTask(f"no task {task_id}")
    return task

class Create:
    def __init__(self, repo, title):
        if not title.strip():
            raise TaskError("a task needs a title")
        self.repo, self.title = repo, title.strip()
        self.task_id = None

    def execute(self):
        if self.task_id is None:
            self.task_id = max((t.task_id for t in self.repo.all()), default=0) + 1
        self.repo.add(Task(self.task_id, self.title))

    def undo(self):
        self.repo.delete(self.task_id)

class Move:
    def __init__(self, repo, task_id, event):
        self.repo, self.task_id, self.event = repo, task_id, event
        self.old = None

    def execute(self):
        task = find(self.repo, self.task_id)
        key = (task.status, self.event)
        if key not in FLOW:
            raise BadMove(f"can't {self.event} a task that is {task.status.value}")
        self.old = task.status
        task.status = FLOW[key]
        self.repo.save(task)

    def undo(self):
        task = find(self.repo, self.task_id)
        task.status = self.old
        self.repo.save(task)

class Assign:
    def __init__(self, repo, task_id, person):
        self.repo, self.task_id, self.person = repo, task_id, person
        self.old = None

    def execute(self):
        task = find(self.repo, self.task_id)
        self.old = task.assignee
        task.assignee = self.person
        self.repo.save(task)

    def undo(self):
        task = find(self.repo, self.task_id)
        task.assignee = self.old
        self.repo.save(task)

class Board:
    def __init__(self, repo):
        self.repo = repo
        self._undo = []
        self._redo = []

    def _run(self, command):
        command.execute()
        self._undo.append(command)
        self._redo.clear()

    def create(self, title):
        command = Create(self.repo, title)
        self._run(command)
        return command.task_id

    def move(self, task_id, event):
        self._run(Move(self.repo, task_id, event))
        return self.repo.get(task_id).status

    def assign(self, task_id, person):
        self._run(Assign(self.repo, task_id, person))

    def undo(self):
        if not self._undo:
            return False
        command = self._undo.pop()
        command.undo()
        self._redo.append(command)
        return True

    def redo(self):
        if not self._redo:
            return False
        command = self._redo.pop()
        command.execute()
        self._undo.append(command)
        return True

    def columns(self):
        result = {}
        for status in Status:
            titles = [t.title for t in self.repo.all() if t.status is status]
            if titles:
                result[status.value] = titles
        return result

def handle(line, board):
    word, _, rest = line.strip().partition(" ")
    try:
        if word == "add":
            return f"added {board.create(rest)}"
        if word in EVENTS:
            task_id = int(rest)
            return f"{task_id} is now {board.move(task_id, word).value}"
        if word == "assign":
            task_id, person = rest.split()
            board.assign(int(task_id), None if person == "-" else person)
            return f"{task_id} unassigned" if person == "-" else f"{task_id} assigned to {person}"
        if word == "undo":
            return "undone" if board.undo() else "nothing to undo"
        if word == "redo":
            return "redone" if board.redo() else "nothing to redo"
        if word == "show":
            cols = board.columns()
            if not cols:
                return "empty board"
            return "\n".join(f"{status}: {', '.join(titles)}" for status, titles in cols.items())
        return "unknown command"
    except TaskError as err:
        return f"error: {err}"

def build():
    return Board(MemoryTasks())
:::

:::think title="Each piece in its place"
Follow one typed line through the program: `handle` turns `finish 1` into `board.move(1, "finish")`; the service wraps it in a `Move` command and runs it; the command loads the task from the repository, asks the domain's `FLOW` whether the move is allowed, saves it, and remembers how to undo it. No piece knows more than it needs: the presentation knows nothing about `FLOW`, the domain knows nothing about undo, and none of them knows the tasks live in a dictionary. Swapping the repository for a file, the presentation for a web page, or adding a new event to `FLOW` would each touch one place. That's what this whole phase has been building towards.
:::

## Forge bonus: a music player

:::exercise id=forge2 points=30 title="Forge bonus: a music player"
Build a small music player:

- **State**: a table `PLAYER` from `(state, event)` to a new state, for the states `"stopped"`, `"playing"` and `"paused"`: stopped + play → playing; playing + pause → paused; paused + play → playing; playing + stop → stopped; paused + stop → stopped. `Player.press(event)` changes state (or raises `ValueError`) and returns the new state.
- **Queue with undo**: `Player` has a `queue` (a list of titles) changed only through commands: `add(title)`, `remove(index)` (an `IndexError` for a bad index) and `move(old_index, new_index)`, each undoable with `undo()` (which returns `False` if there's nothing to undo).
- **Iterator**: `Player.upcoming(repeat=False)` returns an iterator over the queue's titles. With `repeat=True` it starts again from the beginning after the last title, forever (an empty queue gives nothing, even with repeat). It's a snapshot: changing the queue afterwards doesn't affect an iterator already made.

@@hint
The iterator class keeps a copy of the queue, a position, and the `repeat` flag; when the position reaches the end it goes back to 0 if `repeat` (and the copy isn't empty), otherwise it raises `StopIteration`.
@@solution
from itertools import islice

PLAYER = {
    ("stopped", "play"): "playing",
    ("playing", "pause"): "paused",
    ("paused", "play"): "playing",
    ("playing", "stop"): "stopped",
    ("paused", "stop"): "stopped",
}

class AddTrack:
    def __init__(self, queue, title):
        self.queue, self.title = queue, title
    def execute(self):
        self.queue.append(self.title)
    def undo(self):
        self.queue.pop()

class RemoveTrack:
    def __init__(self, queue, index):
        self.queue, self.index = queue, index
        self.title = None
    def execute(self):
        self.title = self.queue.pop(self.index)
    def undo(self):
        self.queue.insert(self.index, self.title)

class MoveTrack:
    def __init__(self, queue, old, new):
        self.queue, self.old, self.new = queue, old, new
    def execute(self):
        self.queue.insert(self.new, self.queue.pop(self.old))
    def undo(self):
        self.queue.insert(self.old, self.queue.pop(self.new))

class Upcoming:
    def __init__(self, titles, repeat):
        self.titles, self.repeat, self.i = list(titles), repeat, 0
    def __iter__(self):
        return self
    def __next__(self):
        if self.i >= len(self.titles):
            if not self.repeat or not self.titles:
                raise StopIteration
            self.i = 0
        title = self.titles[self.i]
        self.i += 1
        return title

class Player:
    def __init__(self):
        self.state = "stopped"
        self.queue = []
        self._history = []

    def press(self, event):
        key = (self.state, event)
        if key not in PLAYER:
            raise ValueError(f"can't {event} when {self.state}")
        self.state = PLAYER[key]
        return self.state

    def _run(self, command):
        command.execute()
        self._history.append(command)

    def add(self, title):
        self._run(AddTrack(self.queue, title))

    def remove(self, index):
        self._run(RemoveTrack(self.queue, index))

    def move(self, old_index, new_index):
        self._run(MoveTrack(self.queue, old_index, new_index))

    def undo(self):
        if not self._history:
            return False
        self._history.pop().undo()
        return True

    def upcoming(self, repeat=False):
        return Upcoming(self.queue, repeat)
