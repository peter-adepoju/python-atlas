from collections.abc import Callable, Iterable

def group_by[T, K](items: Iterable[T], key: Callable[[T], K]) -> dict[K, list[T]]:
    groups: dict[K, list[T]] = {}
    for item in items:
        groups.setdefault(key(item), []).append(item)
    return groups

## Forge: a typed stock keeper

A small shop keeps its products in a JSON file and wants a program that tracks stock. The program is to be written **fully typed**: every function and method has hints on every parameter (except `self`) and on its return value, exactly as given below. Data from the file is checked once, when it's loaded; after that, the hints are trusted.

**`Product`**, a frozen dataclass with `sku: str`, `name: str`, `pence: int` and `tags: tuple[str, ...] = ()`.

**`money(pence: int) -> str`** gives `£4.50` (and `-£3.05` for negative amounts).

**`Repository[K, V]`**, a generic class that stores items by a key worked out from each item.

- `__init__(self, key: Callable[[V], K]) -> None`.
- `add(self, item: V) -> None`: raises `ValueError` if an item with the same key is already there.
- `get(self, key: K) -> V | None`.
- `find(self, test: Callable[[V], bool]) -> list[V]`: the items for which `test` is True, in the order added.
- `all(self) -> list[V]`, in the order added, and `__len__(self) -> int`.

**`load_products(text: str) -> list[Product]`** reads JSON text that must be a list of objects. Each object has exactly the keys `sku` (three capital letters, a hyphen and three digits, like `RIC-001`), `name` (a non-empty string), `pence` (a whole number of 0 or more, not a `bool`), and optionally `tags` (a list of strings). Any problem raises `ValueError` with a message starting `product 2:` (counting from 0) for a problem with one product, or `products must be a list` if the top level isn't a list. Invalid JSON raises `ValueError` too.

**`Inventory`**, built on a `Repository[str, Product]` keyed by SKU, and a stock count for each product, starting at 0.

- `__init__(self, products: Iterable[Product]) -> None`: a duplicate SKU raises `ValueError`.
- `restock(self, sku: str, qty: int) -> int` and `sell(self, sku: str, qty: int) -> int` return the new stock level. An unknown SKU raises `KeyError`, a `qty` below 1 raises `ValueError`, and selling more than is in stock raises `ValueError` and changes nothing.
- `value(self) -> int`: the total price of everything in stock.
- `low_stock(self, limit: int = 5) -> list[str]`: the SKUs with fewer than `limit` in stock, sorted.
- `tagged(self, tag: str) -> list[Product]`: the products with that tag, in the order loaded.
- `price_list(self, fmt: Callable[[int], str] = money) -> list[str]`: one line per product, in the order loaded, like `RIC-001 Rice £4.50`, with the price formatted by `fmt`.

import json
import re
from collections.abc import Callable, Iterable
from dataclasses import dataclass
from typing import Any

@dataclass(frozen=True)
class Product:
    sku: str
    name: str
    pence: int
    tags: tuple[str, ...] = ()

def money(pence: int) -> str:
    sign = "-" if pence < 0 else ""
    pounds, p = divmod(abs(pence), 100)
    return f"{sign}£{pounds}.{p:02}"

class Repository[K, V]:
    def __init__(self, key: Callable[[V], K]) -> None:
        self.key = key
        self._items: dict[K, V] = {}

    def add(self, item: V) -> None:
        k = self.key(item)
        if k in self._items:
            raise ValueError(f"already have {k!r}")
        self._items[k] = item

    def get(self, key: K) -> V | None:
        return self._items.get(key)

    def find(self, test: Callable[[V], bool]) -> list[V]:
        return [item for item in self._items.values() if test(item)]

    def all(self) -> list[V]:
        return list(self._items.values())

    def __len__(self) -> int:
        return len(self._items)

def check_product(i: int, data: Any) -> Product:
    if not isinstance(data, dict):
        raise ValueError(f"product {i}: must be an object")
    unknown = set(data) - {"sku", "name", "pence", "tags"}
    if unknown:
        raise ValueError(f"product {i}: unknown keys {sorted(unknown)}")
    sku, name, pence, tags = data.get("sku"), data.get("name"), data.get("pence"), data.get("tags", [])
    if not isinstance(sku, str) or not re.fullmatch(r"[A-Z]{3}-\d{3}", sku):
        raise ValueError(f"product {i}: bad sku {sku!r}")
    if not isinstance(name, str) or not name:
        raise ValueError(f"product {i}: name must be a non-empty string")
    if not isinstance(pence, int) or isinstance(pence, bool) or pence < 0:
        raise ValueError(f"product {i}: pence must be a whole number of 0 or more")
    if not isinstance(tags, list) or not all(isinstance(t, str) for t in tags):
        raise ValueError(f"product {i}: tags must be a list of strings")
    return Product(sku, name, pence, tuple(tags))

def load_products(text: str) -> list[Product]:
    data: Any = json.loads(text)
    if not isinstance(data, list):
        raise ValueError("products must be a list")
    return [check_product(i, item) for i, item in enumerate(data)]

class Inventory:
    def __init__(self, products: Iterable[Product]) -> None:
        self.products: Repository[str, Product] = Repository(lambda p: p.sku)
        self.stock: dict[str, int] = {}
        for p in products:
            self.products.add(p)
            self.stock[p.sku] = 0

    def _find(self, sku: str) -> Product:
        product = self.products.get(sku)
        if product is None:
            raise KeyError(sku)
        return product

    def restock(self, sku: str, qty: int) -> int:
        self._find(sku)
        if qty < 1:
            raise ValueError("qty must be at least 1")
        self.stock[sku] += qty
        return self.stock[sku]

    def sell(self, sku: str, qty: int) -> int:
        self._find(sku)
        if qty < 1:
            raise ValueError("qty must be at least 1")
        if qty > self.stock[sku]:
            raise ValueError(f"only {self.stock[sku]} of {sku} in stock")
        self.stock[sku] -= qty
        return self.stock[sku]

    def value(self) -> int:
        return sum(p.pence * self.stock[p.sku] for p in self.products.all())

    def low_stock(self, limit: int = 5) -> list[str]:
        return sorted(sku for sku, qty in self.stock.items() if qty < limit)

    def tagged(self, tag: str) -> list[Product]:
        return self.products.find(lambda p: tag in p.tags)

    def price_list(self, fmt: Callable[[int], str] = money) -> list[str]:
        return [f"{p.sku} {p.name} {fmt(p.pence)}" for p in self.products.all()]
:::

:::think title="Hints as a design tool"
Notice how much the hints said before you wrote a line of the bodies. `Repository[K, V]` with `key: Callable[[V], K]` told you the whole design: items in, keys worked out from them. `get(...) -> V | None` told every caller to deal with a missing item; `load_products(text: str) -> list[Product]` drew the line between untrusted text and trusted objects. Writing signatures first, and filling in bodies after, is a good way to design: if a signature is hard to write, the function probably does too much or too little. Next week turns that kind of thinking into principles.
:::

## Forge bonus: typed signals

A **signal** lets one part of a program announce events to any number of others without knowing who they are: "a payment arrived", "a parcel was scanned". Each listener is a function. Typed generically, a `Signal[T]` only accepts listeners that take a `T`, and only emits `T`s.

:::exercise id=forge2 points=30 title="Forge bonus: typed signals"
Write a generic class `Signal[T]`, fully annotated:

- `__init__(self) -> None`.
- `connect(self, handler: Callable[[T], None]) -> Callable[[], None]`: adds a listener, and returns a **function** that, when called, disconnects that listener (calling it again does nothing).
- `emit(self, value: T) -> int`: calls every connected listener with the value, in the order connected, and returns how many were called.
- `__len__(self) -> int`: how many listeners are connected.
- A generic method `map[R](self, f: Callable[[T], R]) -> "Signal[R]"` (the return hint in quotes, because the class isn't finished while its body is being read): returns a **new** signal that emits `f(value)` whenever this one emits `value`.
- A method `filter(self, test: Callable[[T], bool]) -> "Signal[T]"`: returns a new signal that emits only the values for which `test` is True.

Connecting the same function twice connects it twice. A listener that disconnects itself (or others) while `emit` is running must not cause any listener to be skipped or called twice during that emit.

Store the listeners as a list of `(id_number, handler)` pairs, with a counter giving each connection its own number; the disconnect function removes the pair with its number. In `emit`, loop over a **copy** of the list (`list(self._handlers)`) so changes during the loop don't affect it, but skip any pair that's been removed meanwhile.
@@hint
`map` makes `out: Signal[R] = Signal()`, connects `lambda v: out.emit(f(v))` to `self`, and returns `out`.
@@solution
from collections.abc import Callable

class Signal[T]:
    def __init__(self) -> None:
        self._handlers: list[tuple[int, Callable[[T], None]]] = []
        self._next = 0

    def connect(self, handler: Callable[[T], None]) -> Callable[[], None]:
        number = self._next
        self._next += 1
        self._handlers.append((number, handler))

        def disconnect() -> None:
            self._handlers = [pair for pair in self._handlers if pair[0] != number]

        return disconnect

    def emit(self, value: T) -> int:
        called = 0
        for pair in list(self._handlers):
            if pair in self._handlers:
                pair[1](value)
                called += 1
        return called

    def __len__(self) -> int:
        return len(self._handlers)

    def map[R](self, f: Callable[[T], R]) -> "Signal[R]":
        out: Signal[R] = Signal()
        self.connect(lambda v: out.emit(f(v)))
        return out

    def filter(self, test: Callable[[T], bool]) -> "Signal[T]":
        out: Signal[T] = Signal()
        self.connect(lambda v: out.emit(v) if test(v) else None)
        return out
