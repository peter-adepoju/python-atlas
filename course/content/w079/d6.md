class HTTPException(Exception):
    def __init__(self, code, description=None):
        self.code = code
        self.name = HTTPStatus(code).phrase
        self.description = description or self.name
        super().__init__(f"{code} {self.name}: {self.description}")


def abort(code, description=None):
    raise HTTPException(code, description)


class Headers:
    """Case-insensitive headers."""
    def __init__(self, items=()):
        self._items = [(k, str(v)) for k, v in (items.items() if isinstance(items, dict) else items)]

    def get(self, name, default=None):
        for k, v in self._items:
            if k.lower() == name.lower():
                return v
        return default

    def __getitem__(self, name):
        value = self.get(name)
        if value is None:
            raise KeyError(name)
        return value

    def __contains__(self, name):
        return self.get(name) is not None

    def __setitem__(self, name, value):
        self._items = [(k, v) for k, v in self._items if k.lower() != name.lower()] + [(name, str(value))]

    def items(self):
        return list(self._items)


class Args(dict):
    """Query or form values: .get(name) gives the first value; .getlist(name) all of them."""
    def __init__(self, pairs):
        super().__init__()
        self._lists = {}
        for k, v in pairs:
            self._lists.setdefault(k, []).append(v)
            dict.setdefault(self, k, v)

    def get(self, key, default=None, type=None):
        if key not in self:
            return default
        value = self[key]
        if type is not None:
            try:
                return type(value)
            except (TypeError, ValueError):
                return default
        return value

    def getlist(self, key):
        return list(self._lists.get(key, []))


class Request:
    def __init__(self, environ):
        self.environ = environ
        self.method = environ["REQUEST_METHOD"].upper()
        self.path = environ.get("PATH_INFO", "/") or "/"
        self.args = Args(_pairs(environ.get("QUERY_STRING", "")))
        items = [(k[5:].replace("_", "-").title(), v) for k, v in environ.items() if k.startswith("HTTP_")]
        if environ.get("CONTENT_TYPE"):
            items.append(("Content-Type", environ["CONTENT_TYPE"]))
        self.headers = Headers(items)
        length = int(environ.get("CONTENT_LENGTH") or 0)
        self.data = environ["wsgi.input"].read(length) if length else b""
        self.content_type = environ.get("CONTENT_TYPE", "")
        self.form = Args(_pairs(self.data.decode("utf-8"))) if self.content_type.startswith("application/x-www-form-urlencoded") else Args([])
        self.view_args = {}
        cookie = environ.get("HTTP_COOKIE", "")
        self.cookies = dict(p.strip().split("=", 1) for p in cookie.split(";") if "=" in p)

    @property
    def is_json(self):
        return self.content_type.split(";")[0].strip() == "application/json"

    def get_json(self, silent=False, force=False):
        if not (self.is_json or force):
            if silent:
                return None
            abort(415, "expected a JSON body (Content-Type: application/json)")
        try:
            return json.loads(self.data or b"null")
        except ValueError:
            if silent:
                return None
            abort(400, "the body isn't valid JSON")

    @property
    def json(self):
        return self.get_json()

_CONVERTERS = {"int": (r"\d+", int), "float": (r"\d+(?:\.\d+)?", float), "path": (r".+", str), "string": (r"[^/]+", str)}


class _Rule:
    def __init__(self, rule, endpoint, view, methods):
        self.rule, self.endpoint, self.view = rule, endpoint, view
        self.methods = {m.upper() for m in methods} | ({"HEAD"} if "GET" in {m.upper() for m in methods} else set())
        regex, self.converters = "", {}
        for piece in re.split(r"(<[^>]+>)", rule):
            m = re.fullmatch(r"<(?:(\w+):)?(\w+)>", piece)
            if m:
                kind, name = m.group(1) or "string", m.group(2)
                pattern, convert = _CONVERTERS[kind]
                regex += f"(?P<{name}>{pattern})"
                self.converters[name] = convert
            else:
                regex += re.escape(piece)
        self.regex = re.compile(regex)

    def match(self, path):
        m = self.regex.fullmatch(path)
        if not m:
            return None
        return {k: self.converters[k](v) for k, v in m.groupdict().items()}

    def build(self, values):
        def fill(m):
            return str(values.pop(m.group(2)))
        path = re.sub(r"<(?:(\w+):)?(\w+)>", fill, self.rule)
        return path + (f"?{urlencode(values)}" if values else "")


class _Setup:
    def route(self, rule, methods=("GET",), endpoint=None):
        def decorate(view):
            self._add(rule, endpoint or view.__name__, view, methods)
            return view
        return decorate

    def get(self, rule, **kw):
        return self.route(rule, methods=["GET"], **kw)

    def post(self, rule, **kw):
        return self.route(rule, methods=["POST"], **kw)

    def put(self, rule, **kw):
        return self.route(rule, methods=["PUT"], **kw)

    def patch(self, rule, **kw):
        return self.route(rule, methods=["PATCH"], **kw)

    def delete(self, rule, **kw):
        return self.route(rule, methods=["DELETE"], **kw)


class Blueprint(_Setup):
    def __init__(self, name, import_name, url_prefix=""):
        self.name, self.url_prefix = name, url_prefix
        self.deferred = []

    def _add(self, rule, endpoint, view, methods):
        self.deferred.append((rule, endpoint, view, methods))


class _App(_Setup):
    def __init__(self, import_name):
        self.import_name = import_name
        self.config = {"TESTING": False}
        self.rules = []
        self.before_funcs, self.after_funcs, self.error_handlers = [], [], {}

    def _add(self, rule, endpoint, view, methods):
        self.rules.append(_Rule(rule, endpoint, view, methods))

    def register_blueprint(self, bp, url_prefix=None):
        prefix = url_prefix if url_prefix is not None else bp.url_prefix
        for rule, endpoint, view, methods in bp.deferred:
            self._add(prefix + rule, f"{bp.name}.{endpoint}", view, methods)

    def before_request(self, f):
        self.before_funcs.append(f)
        return f

    def after_request(self, f):
        self.after_funcs.append(f)
        return f

    def errorhandler(self, code):
        def decorate(f):
            self.error_handlers[code] = f
            return f
        return decorate

    def url_for(self, endpoint, **values):
        for r in self.rules:
            if r.endpoint == endpoint:
                return r.build(dict(values))
        raise LookupError(f"no endpoint {endpoint!r}")

    @staticmethod
    def _to_response(rv):
        status, headers = None, None
        if isinstance(rv, tuple):
            if len(rv) == 3:
                rv, status, headers = rv
            elif isinstance(rv[1], (dict, list)):
                rv, headers = rv
            else:
                rv, status = rv
        if isinstance(rv, Response):
            response = rv
        elif isinstance(rv, (dict, list)):
            response = jsonify(rv)
        elif isinstance(rv, (str, bytes)):
            response = Response(rv)
        elif rv is None:
            raise TypeError("the view function did not return a valid response")
        else:
            raise TypeError(f"can't turn {type(rv).__name__} into a response")
        if status is not None:
            response.status_code = status
        for k, v in (headers.items() if isinstance(headers, dict) else headers or []):
            response.headers[k] = v
        return response

    def _error(self, e):
        handler = self.error_handlers.get(e.code)
        if handler:
            return self._to_response(handler(e))
        body = f"<h1>{e.code} {e.name}</h1><p>{e.description}</p>"
        return Response(body, e.code)

    def _dispatch(self, req):
        path = req.path
        allowed = set()
        for r in self.rules:
            values = r.match(path)
            if values is None:
                continue
            if req.method in r.methods:
                req.view_args = values
                return r.view(**values)
            allowed |= r.methods
        if allowed:
            raise _MethodNotAllowed(sorted(allowed | {"OPTIONS"}))
        abort(404, "The requested URL was not found on the server.")

    def full_dispatch(self, environ):
        req = Request(environ)
        token = _current.set((req, _G(), self))
        try:
            try:
                rv = None
                for f in self.before_funcs:
                    rv = f()
                    if rv is not None:
                        break
                if rv is None:
                    rv = self._dispatch(req)
                response = self._to_response(rv)
            except _MethodNotAllowed as e:
                response = self._error(HTTPException(405))
                response.headers["Allow"] = ", ".join(e.allowed)
            except HTTPException as e:
                response = self._error(e)
            except Exception as e:
                if self.config.get("TESTING") and 500 not in self.error_handlers:
                    raise
                response = self._error(HTTPException(500)) if 500 not in self.error_handlers else self._to_response(self.error_handlers[500](e))
                response.status_code = 500
            for f in self.after_funcs:
                response = f(response) or response
            if req.method == "HEAD":
                response.headers["Content-Length"] = len(response.data)
                response.data = b""
            return response
        finally:
            _current.reset(token)

    def __call__(self, environ, start_response):
        response = self.full_dispatch(environ)
        if "Content-Length" not in response.headers:
            response.headers["Content-Length"] = len(response.data)
        start_response(response.status, response.headers.items())
        return [response.data]

import hashlib
import hmac
from urllib.parse import parse_qs, urlsplit

def _sig(path, expires, key):
    return hmac.new(key, f"{path}|{expires}".encode(), hashlib.sha256).hexdigest()

def make_link(path, expires, key):
    return f"{path}?expires={expires}&sig={_sig(path, expires, key)}"

def check_link(url, now, key):
    parts = urlsplit(url)
    query = parse_qs(parts.query)
    expires, sig = query.get("expires", [""])[0], query.get("sig", [""])[0]
    if not expires.isdigit() or not sig or not hmac.compare_digest(sig, _sig(parts.path, expires, key)):
        return "bad signature"
    return "expired" if now > int(expires) else "ok"

import html

ALLOWED = ["name", "email", "plan", "joined"]

def report_table(rows, columns):
    for name in columns:
        if name not in ALLOWED:
            raise ValueError(f"unknown column: {name!r}")

## Forge: a security review

The starter is a small notes service, `create_app(env, conn, upload_dir, log=print)`, built on miniflask (Week 77) and sqlite3 (Week 75). It works: people can sign up, log in, write notes, search them, view a note as a page, and download files from their own upload folder. It has also failed a security review. Fix it so that it keeps working **and**:

- **Settings**: it refuses to start without a `SECRET_KEY` of at least 32 characters (`RuntimeError("SECRET_KEY must be set (at least 32 characters)")`), and `DEBUG` is on only for `"1"` or `"true"` (any case).
- **Passwords**: stored with the given `hash_password` (salted, slow) and checked with `check_password`; a duplicate email gets `409`.
- **Login**: no SQL injection anywhere (every query uses parameters); one message, `{"error": "wrong email or password"}` with `401`, whether the email or the password was wrong; passwords never appear in the log.
- **Sessions**: the `session` cookie is `USER_ID.SIGNATURE`, the signature being the hex HMAC-SHA256 of the id with the secret key, checked with `hmac.compare_digest`; a missing, unsigned or forged cookie gets `401`. Set it with `httponly=True`, `secure=True` and `samesite="Lax"`.
- **Access**: people see only their **own** notes: someone else's note is `404` (as if it didn't exist), in the JSON and on the page, and searching never shows anyone else's notes, whatever the search text. Search text is matched literally (escape `LIKE` wildcards).
- **Output**: the note page escapes the title and body.
- **Files**: `/files?name=...` serves only files inside the user's own folder (`403` for anything outside, or an empty name; `404` for a missing file).
- **Errors**: an unexpected error returns `500` with only `{"error": "something went wrong", "incident": ID}` (an 8-character hex id), logs the details with that id, and adds `"detail"` only when `DEBUG` is on.

ITERATIONS = 1000   # far too few for real use (around 600,000 is the guidance); kept low so the checks run quickly


def pbkdf2_sha256(password, salt, iterations, dklen=32):
    base = hmac.new(password, digestmod=hashlib.sha256)
    out, block = b"", 1
    while len(out) < dklen:
        h = base.copy()
        h.update(salt + block.to_bytes(4, "big"))
        u = h.digest()
        total = int.from_bytes(u, "big")
        for _ in range(iterations - 1):
            h = base.copy()
            h.update(u)
            u = h.digest()
            total ^= int.from_bytes(u, "big")
        out += total.to_bytes(32, "big")
        block += 1
    return out[:dklen]


def hash_password(password):
    salt = secrets.token_bytes(16)
    digest = pbkdf2_sha256(password.encode(), salt, ITERATIONS)

ITERATIONS = 1000   # far too few for real use (around 600,000 is the guidance); kept low so the checks run quickly


def pbkdf2_sha256(password, salt, iterations, dklen=32):
    base = hmac.new(password, digestmod=hashlib.sha256)
    out, block = b"", 1
    while len(out) < dklen:
        h = base.copy()
        h.update(salt + block.to_bytes(4, "big"))
        u = h.digest()
        total = int.from_bytes(u, "big")
        for _ in range(iterations - 1):
            h = base.copy()
            h.update(u)
            u = h.digest()
            total ^= int.from_bytes(u, "big")
        out += total.to_bytes(32, "big")
        block += 1
    return out[:dklen]


def hash_password(password):
    salt = secrets.token_bytes(16)
    digest = pbkdf2_sha256(password.encode(), salt, ITERATIONS)
    return f"pbkdf2_sha256${ITERATIONS}${base64.b64encode(salt).decode()}${base64.b64encode(digest).decode()}"

import ast
import re

SECRET_NAME = re.compile(r"password|secret|token|api_?key", re.I)


def _dotted(node):
    if isinstance(node, ast.Name):
        return node.id
    if isinstance(node, ast.Attribute):
        inner = _dotted(node.value)
        return f"{inner}.{node.attr}" if inner else None
    return None


def _keyword(call, name):
    return next((k.value for k in call.keywords if k.arg == name), None)


def _is_const(node, value):
    return isinstance(node, ast.Constant) and node.value is value


def _built_string(node):
    if isinstance(node, ast.JoinedStr):
        return any(isinstance(v, ast.FormattedValue) for v in node.values)
    if isinstance(node, ast.BinOp) and isinstance(node.op, (ast.Add, ast.Mod)):
        return True
    return isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute) and node.func.attr == "format"


def review(source):
    tree = ast.parse(source)
    found = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Call):
            name = _dotted(node.func) or ""
            short = name.rsplit(".", 1)[-1]
            if name in ("eval", "exec"):
                found.add((node.lineno, "eval or exec"))
            if name in ("pickle.loads", "pickle.load"):
                found.add((node.lineno, "unpickling"))
            if _is_const(_keyword(node, "shell"), True):
                found.add((node.lineno, "shell=True"))
            if _is_const(_keyword(node, "verify"), False):
                found.add((node.lineno, "certificate checks off"))
            if short in ("execute", "executemany") and node.args and _built_string(node.args[0]):
                found.add((node.lineno, "SQL built from strings"))
            if name in ("hashlib.md5", "hashlib.sha1") or (name == "hashlib.new" and node.args and isinstance(node.args[0], ast.Constant)
                                                           and str(node.args[0].value).lower() in ("md5", "sha1")):
                found.add((node.lineno, "weak hash"))
            if name == "yaml.load":
                loader = _keyword(node, "Loader")
                if loader is None or "Safe" not in (_dotted(loader) or ""):
                    found.add((node.lineno, "unsafe yaml"))
        elif isinstance(node, ast.Assign):
            if isinstance(node.value, ast.Constant) and isinstance(node.value.value, str) and len(node.value.value) >= 8:
                for target in node.targets:
                    label = target.id if isinstance(target, ast.Name) else target.attr if isinstance(target, ast.Attribute) else ""
                    if SECRET_NAME.search(label):
                        found.add((node.lineno, "hard-coded secret"))
        elif isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and re.search(r"token|password|secret|key", node.name, re.I):
            for inner in ast.walk(node):
                if isinstance(inner, ast.Call) and (_dotted(inner.func) or "").startswith("random."):
                    found.add((inner.lineno, "random used for a secret"))
    return sorted(found)
