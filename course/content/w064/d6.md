def test_closure_vars():
    "closures and code objects"
    def make(rate):
        count = 0
        def bump(x):
            nonlocal count
            count += 1
            return x * rate
        return bump
    f = make(3)
    assert closure_vars(f) == {"count": 0, "rate": 3}

@@solution
def closure_vars(func):
    code = func.__code__
    cells = func.__closure__ or ()
    values = {}
    for name, cell in zip(code.co_freevars, cells):
        try:
            values[name] = cell.cell_contents
        except ValueError:
            values[name] = None
    return values

import ast

def walk_tree(node, depth=0):
    yield depth, type(node).__name__
    for child in ast.iter_child_nodes(node):
        yield from walk_tree(child, depth + 1)

def deepest(source):
    return max(depth for depth, _ in walk_tree(ast.parse(source)))

## Forge: a code checker

A **linter** reads code without running it and reports likely mistakes. Write `check(source)`, returning a sorted list of `(line, code, message)` tuples with no duplicates, and `report(source)`, the same problems as lines `"line: CODE message"` joined by newlines (an empty string if there are none). Checks:

| code | finds | message |
|---|---|---|
| `F401` | an imported name that's never read anywhere (`import a.b` binds `a`; `as` names count; names listed in a module-level `__all__` count as used) | `'os' imported but unused` |
| `F841` | inside a function, a name assigned with `=` anywhere in the function's body that's never read in that body, ignoring `_` and names declared `global` or `nonlocal`; report the line of the first assignment | `local variable 'x' is assigned to but never used` |
| `B006` | a default argument that's a list, dict or set display, or a call to `list`, `dict` or `set` (positional and keyword-only defaults) | `mutable default argument` |
| `E722` | `except:` with no exception type | `do not use bare 'except'` |
| `A001` | a name that's assigned, or a function, class or parameter name, that's also a built-in (`dir(builtins)`, ignoring names starting with `_`) | `'list' shadows a built-in` |
| `E711` | `== None` or `!= None` | `comparison to None should use 'is' or 'is not'` |

A line with a `# noqa` comment hides every problem on that line; `# noqa: E711, F401` hides only those codes (case-insensitive `noqa`). Find comments with `tokenize`, so `"# noqa"` inside a string doesn't count. Code with a syntax error gives `[(line, "E999", "syntax error")]`.

Write one small function per check, each appending to a shared list of problems, then filter by the `noqa` rules (a dict from line number to `None` for "everything" or a set of codes). For F401, collect every `Name` with a `Load` context in the whole tree. For F841, walk each function's body for `Assign` targets and `Load` names separately.
@@solution
import ast
import builtins
import io
import tokenize

BUILTINS = {name for name in dir(builtins) if not name.startswith("_")}

def _noqa(source):
    rules = {}
    try:
        for tok in tokenize.generate_tokens(io.StringIO(source).readline):
            if tok.type == tokenize.COMMENT:
                text = tok.string.lstrip("#").strip()
                if text.lower().startswith("noqa"):
                    rest = text[4:].strip()
                    if rest.startswith(":"):
                        codes = {c.strip() for c in rest[1:].split(",") if c.strip()}
                        rules[tok.start[0]] = codes
                    else:
                        rules[tok.start[0]] = None
    except tokenize.TokenError:
        pass
    return rules

def _loaded(node):
    return {n.id for n in ast.walk(node) if isinstance(n, ast.Name) and isinstance(n.ctx, ast.Load)}

def _check_imports(tree, problems):
    used = _loaded(tree)
    for node in ast.walk(tree):
        if isinstance(node, ast.Assign):
            for target in node.targets:
                if isinstance(target, ast.Name) and target.id == "__all__" and isinstance(node.value, (ast.List, ast.Tuple)):
                    used |= {e.value for e in node.value.elts if isinstance(e, ast.Constant)}
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for alias in node.names:
                name = alias.asname or alias.name.split(".")[0]
                if name not in used:
                    problems.append((node.lineno, "F401", f"'{name}' imported but unused"))
        elif isinstance(node, ast.ImportFrom):
            for alias in node.names:
                name = alias.asname or alias.name
                if name != "*" and name not in used:
                    problems.append((node.lineno, "F401", f"'{name}' imported but unused"))

def _check_locals(tree, problems):
    for func in ast.walk(tree):
        if not isinstance(func, (ast.FunctionDef, ast.AsyncFunctionDef)):
            continue
        declared = set()
        for n in ast.walk(func):
            if isinstance(n, (ast.Global, ast.Nonlocal)):
                declared.update(n.names)
        used = set()
        for stmt in func.body:
            used |= _loaded(stmt)
        first = {}
        for stmt in func.body:
            for n in ast.walk(stmt):
                if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef, ast.Lambda)):
                    continue
                if isinstance(n, ast.Assign):
                    for target in n.targets:
                        if isinstance(target, ast.Name):
                            first.setdefault(target.id, n.lineno)
        for name, line in first.items():
            if name not in used and name not in declared and name != "_":
                problems.append((line, "F841", f"local variable '{name}' is assigned to but never used"))

def _check_defaults(tree, problems):
    for func in ast.walk(tree):
        if isinstance(func, (ast.FunctionDef, ast.AsyncFunctionDef, ast.Lambda)):
            for default in func.args.defaults + [d for d in func.args.kw_defaults if d is not None]:
                mutable = isinstance(default, (ast.List, ast.Dict, ast.Set)) or (
                    isinstance(default, ast.Call) and isinstance(default.func, ast.Name)
                    and default.func.id in ("list", "dict", "set"))
                if mutable:
                    problems.append((default.lineno, "B006", "mutable default argument"))

def _check_misc(tree, problems):
    for node in ast.walk(tree):
        if isinstance(node, ast.ExceptHandler) and node.type is None:
            problems.append((node.lineno, "E722", "do not use bare 'except'"))
        elif isinstance(node, ast.Compare):
            for op, right in zip(node.ops, node.comparators):
                if isinstance(op, (ast.Eq, ast.NotEq)) and isinstance(right, ast.Constant) and right.value is None:
                    problems.append((node.lineno, "E711", "comparison to None should use 'is' or 'is not'"))
        names = []
        if isinstance(node, ast.Name) and isinstance(node.ctx, ast.Store):
            names.append(node.id)
        elif isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            names.append(node.name)
        elif isinstance(node, ast.arg):
            names.append(node.arg)
        for name in names:
            if name in BUILTINS:
                problems.append((node.lineno, "A001", f"'{name}' shadows a built-in"))

def check(source):
    try:
        tree = ast.parse(source)
    except SyntaxError as e:
        return [(e.lineno or 1, "E999", "syntax error")]
    problems = []
    for checker in (_check_imports, _check_locals, _check_defaults, _check_misc):
        checker(tree, problems)
    rules = _noqa(source)
    kept = []
    for line, code, message in problems:
        if line in rules and (rules[line] is None or code in rules[line]):
            continue
        kept.append((line, code, message))
    return sorted(set(kept))

def report(source):
    return "\n".join(f"{line}: {code} {message}" for line, code, message in check(source))
:::

:::think title="Programs that read programs"
The checker never runs the code it checks, yet it finds real bugs, because the syntax tree holds the program's structure precisely. That's the idea behind every tool in this family: linters (flake8, pylint, Ruff), formatters (Black), type checkers (mypy, Week 33), refactoring tools and IDE features such as "find all references". Each starts from the same stages you met on Monday, tokens for what the tree forgets (comments, exact layout) and the AST for everything else, and adds a symbol table when it needs to know which name means what. When a tool reports a strange warning, you can now picture what it saw.

The tracer must return itself both for the module-level frame and for function frames from `"<covered>"`, and `None` for any other file. Format each margin with `f"{mark:>3}"`.
@@solution
import ast
import sys

FILENAME = "<covered>"

def statement_lines(source):
    lines = set()
    for node in ast.walk(ast.parse(source)):
        if isinstance(node, ast.stmt):
            if isinstance(node, ast.Expr) and isinstance(node.value, ast.Constant) and isinstance(node.value.value, str):
                continue
            lines.add(node.lineno)
    return sorted(lines)

def measure(source, namespace=None):
    namespace = {} if namespace is None else namespace
    hits = {}

    def tracer(frame, event, arg):
        if frame.f_code.co_filename != FILENAME:
            return None
        if event == "line":
            hits[frame.f_lineno] = hits.get(frame.f_lineno, 0) + 1
        return tracer

    code = compile(source, FILENAME, "exec")
    previous = sys.gettrace()
    sys.settrace(tracer)
    try:
        exec(code, namespace)
    finally:
        sys.settrace(previous)
    statements = statement_lines(source)
    executed = sorted(line for line in statements if line in hits)
    missed = sorted(line for line in statements if line not in hits)
    percent = round(100 * len(executed) / len(statements)) if statements else 100
    return {"statements": statements, "executed": executed, "missed": missed,
            "percent": percent, "hits": {line: hits[line] for line in executed}}

def annotate(source, result):
    out = []
    for number, text in enumerate(source.split("\n"), 1):
        if number in result["missed"]:
            mark = "!"
        elif number in result["executed"]:
            mark = str(result["hits"][number])
        else:
            mark = " "
        out.append(f"{mark:>3} | {text}")
    return "\n".join(out)
