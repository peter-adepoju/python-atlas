Keep a heap of at most k entries `(-distance, negated point)`, so the top is the **worst** kept; replace it when a shop beats it. Sort what's left at the end.
@@solution
import heapq

def nearest(shops, home, k):
    if k <= 0:
        return []
    heap = []
    for p in shops:
        d = (p[0] - home[0]) ** 2 + (p[1] - home[1]) ** 2
        entry = (-d, -p[0], -p[1])
        if len(heap) < k:
            heapq.heappush(heap, entry)
        elif entry > heap[0]:
            heapq.heapreplace(heap, entry)
    kept = [(-x, -y) for _, x, y in heap]
    return sorted(kept, key=lambda p: ((p[0] - home[0]) ** 2 + (p[1] - home[1]) ** 2, p))

@@solution
from collections import deque

def two_teams(n, rivals):
    adj = [[] for _ in range(n)]
    for a, b in rivals:
        adj[a].append(b)
        adj[b].append(a)
    team = [None] * n
    for s in range(n):
        if team[s] is not None:
            continue
        team[s] = 0
        queue = deque([s])
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if team[v] is None:
                    team[v] = 1 - team[u]
                    queue.append(v)
                elif team[v] == team[u]:
                    return None
    return team

import itertools, random
def test_hidden():
    "two-string DP"
    assert count_hidden("rabbbit", "rabbit") == 3 and count_hidden("babgbag", "bag") == 5
    assert count_hidden("abc", "") == 1 and count_hidden("", "a") == 0 and count_hidden("aaa", "aa") == 3

        w = "".join(rng.choice("ab") for _ in range(rng.randint(0, 3)))

@@hint
Let ways[j] be the number of ways to spell `word[:j]` from the text so far, with ways[0] = 1. For each letter of the text, go through j **downwards** and add ways[j − 1] to ways[j] when the letter matches word[j − 1] (downwards, like the 0/1 knapsack, so a letter is used once).
@@solution
def count_hidden(text, word):
    ways = [1] + [0] * len(word)
    for ch in text:
        for j in range(len(word), 0, -1):
            if word[j - 1] == ch:
                ways[j] += ways[j - 1]
    return ways[len(word)]

@@tests
import itertools, random
def test_fuel():
    "greedy with a max-heap"
    assert fewest_stops(100, 10, [(10, 60), (20, 30), (30, 30), (60, 40)]) == 2
    assert fewest_stops(1, 1, []) == 0 and fewest_stops(100, 1, [(10, 100)]) == -1 and fewest_stops(10, 5, [(5, 5)]) == 1
    rng = random.Random(4)
    for _ in range(100):
        target = rng.randint(1, 50)
        stations = sorted({rng.randint(1, target): rng.randint(1, 20) for _ in range(rng.randint(0, 6))}.items())
        fuel = rng.randint(1, 30)
        best = -1
        for k in range(len(stations) + 1):
            for chosen in itertools.combinations(stations, k):
                tank, pos, ok = fuel, 0, True
                for mile, f in chosen:
                    if tank < mile - pos:
                        ok = False
                        break
                    tank -= mile - pos
                    pos = mile
                    tank += f
                if ok and tank >= target - pos:
                    best = k
                    break
            if best != -1:
                break
        assert fewest_stops(target, fuel, stations) == best, (target, fuel, stations)
@@hint
Walk through the stations (and finally the target). Before reaching each, while the tank can't get you there, pop the largest fuel from a heap of passed stations (a max-heap of negated values) and count a stop; if the heap is empty, return -1. Then push this station's fuel.
@@solution
import heapq

def fewest_stops(target, fuel, stations):
    passed, stops, reach = [], 0, fuel
    for mile, f in list(stations) + [(target, 0)]:
        while reach < mile:
            if not passed:
                return -1
            reach += -heapq.heappop(passed)
            stops += 1
        heapq.heappush(passed, -f)
    return stops

    P = 10 ** 9 + 7
    count = [0] * 1024
    count[0] = 1
    for a in nums:
        count = [(count[v] + count[v ^ a]) % P for v in range(1024)]
    return count[x]

:::exercise id=c1 points=30 title="C: A metro planner"
Write a class `Metro()` for planning journeys on a metro of lines and stations.

- `add_line(name, stations, minutes)`: a line runs through the stations in order, both ways; `minutes[i]` is the time between `stations[i]` and `stations[i + 1]`. `ValueError` if the lengths don't fit (there must be one fewer time than stations) or the line name is already used.

Store `self.adj[(station, line)]` as a list of `((next_station, line), minutes)` for travel along a line, and `self.lines[station]` as a set of lines. In Dijkstra, from `(s, line)` you can also move to `(s, other)` for `change` minutes. Start with every `(a, line)` at 0 and stop at the first popped state whose station is b. For fewest changes, run the same search with cost 0 for travel and 1 for a change, using a deque (0-1 BFS: push 0-cost moves at the front).
@@solution
import heapq
from collections import deque

class Metro:
    def __init__(self):
        self.adj = {}
        self.lines = {}
        self.names = set()

    def add_line(self, name, stations, minutes):
        if name in self.names:
            raise ValueError(f"line {name} already exists")
        if len(minutes) != len(stations) - 1 or len(stations) < 2:
            raise ValueError("there must be one fewer time than stations")
        self.names.add(name)
        for s in stations:
            self.lines.setdefault(s, set()).add(name)
            self.adj.setdefault((s, name), [])
        for (a, b), t in zip(zip(stations, stations[1:]), minutes):
            self.adj[(a, name)].append(((b, name), t))
            self.adj[(b, name)].append(((a, name), t))

    def lines_at(self, station):
        if station not in self.lines:
            raise KeyError(station)
        return sorted(self.lines[station])

    def _next(self, state, change):
        station, line = state
        out = list(self.adj[state])
        for other in self.lines[station]:
            if other != line:
                out.append(((station, other), change))
        return out

    def _search(self, a, change):
        if a not in self.lines:
            raise KeyError(a)
        dist, parent, heap = {}, {}, []
        for line in sorted(self.lines[a]):
            dist[(a, line)] = 0
            parent[(a, line)] = None
            heap.append((0, a, line))
        heapq.heapify(heap)
        while heap:
            d, s, line = heapq.heappop(heap)
            if d > dist[(s, line)]:
                continue
            for nxt, t in self._next((s, line), change):
                if d + t < dist.get(nxt, float("inf")):
                    dist[nxt] = d + t
                    parent[nxt] = (s, line)
                    heapq.heappush(heap, (d + t, nxt[0], nxt[1]))
        return dist, parent

    def fastest(self, a, b, change=5):
        if b not in self.lines:
            raise KeyError(b)
        if a == b:
            if a not in self.lines:
                raise KeyError(a)
            return 0, [(a, None)]
        dist, parent = self._search(a, change)
        ends = [s for s in dist if s[0] == b]
        if not ends:
            return None
        end = min(ends, key=lambda s: (dist[s], s[1]))
        route, state = [], end
        while state is not None:
            route.append(state)
            state = parent[state]
        return dist[end], route[::-1]

    def fewest_changes(self, a, b):
        if a not in self.lines or b not in self.lines:
            raise KeyError(a if a not in self.lines else b)
        if a == b:
            return 0
        dist, queue = {}, deque()
        for line in self.lines[a]:
            dist[(a, line)] = 0
            queue.append((a, line))
        while queue:
            state = queue.popleft()
            d = dist[state]
            if state[0] == b:
                return d
            for nxt, t in self.adj[state]:
                if d < dist.get(nxt, float("inf")):
                    dist[nxt] = d
                    queue.appendleft(nxt)
            for other in self.lines[state[0]]:
                nxt = (state[0], other)
                if d + 1 < dist.get(nxt, float("inf")):
                    dist[nxt] = d + 1
                    queue.append(nxt)
        return None

    def within(self, a, limit, change=5):
        dist, _ = self._search(a, change)
        return sorted({s for (s, _), d in dist.items() if d <= limit})
