def valid_tree(n, edges):
    if len(edges) != n - 1:
        return False
    parent = list(range(n))
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    for a, b in edges:
        ra, rb = find(a), find(b)
        if ra == rb:
            return False
        parent[ra] = rb
    return True

from collections import deque

JUMPS = [(1, 2), (2, 1), (2, -1), (1, -2), (-1, -2), (-2, -1), (-2, 1), (-1, 2)]

def knight_map(start):
    dist = {start: 0}
    queue = deque([start])
    while queue:
        r, c = queue.popleft()
        for dr, dc in JUMPS:
            nxt = (r + dr, c + dc)
            if 0 <= nxt[0] < 8 and 0 <= nxt[1] < 8 and nxt not in dist:
                dist[nxt] = dist[(r, c)] + 1
                queue.append(nxt)
    return [[dist[(r, c)] for c in range(8)] for r in range(8)]

def knight_moves(start, goal):
    return knight_map(start)[goal[0]][goal[1]]

## Forge: a transport network

A city's transport network is a set of **lines**, each a list of stations in order; trains run both ways, and a station can be on several lines. Build a planner that answers the questions travellers and planners ask.

Write `Network(lines)`, where `lines` is a dict from line name to its list of stations:

- Raise `ValueError` for a line with fewer than two stations or one that visits a station twice.
- Store the station graph as `adj`, a dict from each station to the **set** of stations next to it on any line.
- `stations()` (sorted) and `lines_at(station)` (sorted line names). Any method given an unknown station raises `KeyError`.
- `route(a, b)`: a route with the fewest stops, as a list of stations (`[a]` when `a == b`; `None` if unreachable). Explore neighbours in **alphabetical** order, so the route is the one BFS finds that way. `fewest_stops(a, b)`: its number of stops (edges), or `None`.
- `within(a, k)`: every station at most k stops from a, sorted.
- `fewest_changes(a, b)`: the fewest times a traveller must change line (0 if one line serves both, or `a == b`; `None` if unreachable). Run BFS on a **different graph**: the lines are the vertices, and two lines are joined when they share a station. Start from every line at a (distance 0) and stop at the first line that serves b.
- `split_if_closed(station)`: if that station closed, the pieces the rest of the network would fall into, each a sorted list, the list sorted.
- `critical()`: the sorted stations whose closure would split their part of the network into more pieces (compare the number of pieces with and without the station). A simple approach that closes each station in turn is fine for networks of a few hundred stations. (A single DFS can find them all in O(V + E): they're called **articulation points**, and Robert Tarjan's 1972 algorithm finds them from discovery times.)

Build `adj` and `at` (station to its set of lines) in `__init__`. Write one `_bfs(start, closed)` that returns a parent dict and skips closed stations; `route`, `split_if_closed` and the piece-counting in `critical` all use it. For `fewest_changes`, first build `meets[line]`, the set of other lines sharing a station with it.
@@solution
from collections import deque

class Network:
    def __init__(self, lines):
        self.lines = {}
        self.adj = {}
        self.at = {}
        for name, stops in lines.items():
            if len(stops) < 2:
                raise ValueError(f"line {name!r} needs at least two stations")
            if len(set(stops)) != len(stops):
                raise ValueError(f"line {name!r} visits a station twice")
            self.lines[name] = list(stops)
            for s in stops:
                self.adj.setdefault(s, set())
                self.at.setdefault(s, set()).add(name)
            for a, b in zip(stops, stops[1:]):
                self.adj[a].add(b)
                self.adj[b].add(a)

    def _check(self, *stations):
        for s in stations:
            if s not in self.adj:
                raise KeyError(s)

    def stations(self):
        return sorted(self.adj)

    def lines_at(self, station):
        self._check(station)
        return sorted(self.at[station])

    def _bfs(self, start, closed=frozenset()):
        parent = {start: None}
        queue = deque([start])
        while queue:
            v = queue.popleft()
            for w in sorted(self.adj[v]):
                if w not in parent and w not in closed:
                    parent[w] = v
                    queue.append(w)
        return parent

    def route(self, a, b):
        self._check(a, b)
        parent = self._bfs(a)
        if b not in parent:
            return None
        path = []
        while b is not None:
            path.append(b)
            b = parent[b]
        return path[::-1]

    def fewest_stops(self, a, b):
        r = self.route(a, b)
        return None if r is None else len(r) - 1

    def within(self, a, k):
        self._check(a)
        dist = {a: 0}
        queue = deque([a])
        while queue:
            v = queue.popleft()
            if dist[v] == k:
                continue
            for w in self.adj[v]:
                if w not in dist:
                    dist[w] = dist[v] + 1
                    queue.append(w)
        return sorted(dist)

    def fewest_changes(self, a, b):
        self._check(a, b)
        if a == b:
            return 0
        meets = {name: set() for name in self.lines}
        for s, names in self.at.items():
            for x in names:
                meets[x] |= names - {x}
        dist = {name: 0 for name in self.at[a]}
        queue = deque(sorted(dist))
        while queue:
            x = queue.popleft()
            if x in self.at[b]:
                return dist[x]
            for y in sorted(meets[x]):
                if y not in dist:
                    dist[y] = dist[x] + 1
                    queue.append(y)
        return None

    def split_if_closed(self, station):
        self._check(station)
        seen, parts = {station}, []
        for s in sorted(self.adj):
            if s in seen:
                continue
            comp = self._bfs(s, closed={station})
            seen |= set(comp)
            parts.append(sorted(comp))
        return parts

    def _pieces(self):
        seen, count = set(), 0
        for s in self.adj:
            if s not in seen:
                count += 1
                seen |= set(self._bfs(s))
        return count

    def critical(self):
        base = self._pieces()
        out = []
        for s in sorted(self.adj):
            if self.adj[s] and len(self.split_if_closed(s)) > base:
                out.append(s)
        return out

:::think title="The right graph for the question"
The same city gave two different graphs today. For **fewest stops**, stations are the vertices and track between them the edges. For **fewest changes**, lines are the vertices and shared stations the edges: a route with few changes may have many stops, and BFS on the station graph can't see changes at all. Choosing what the vertices and edges are is usually the real work in a graph problem; once the graph is right, the algorithm is often one you already know. (Week 49's word ladder did the same: words as vertices, one-letter changes as edges.) When a question doesn't fit, ask whether a different graph, with richer vertices such as "station and current line" or "cell and doors used", would make it a plain BFS.

class Installer:
    def __init__(self, index):
        self.index = {name: list(deps) for name, deps in index.items()}
        self.installed = set()
        self.explicit = set()

    def install_order(self, package):
        if package not in self.index:
            raise KeyError(package)
        order, state, path = [], {}, []
        stack = [(package, iter(self.index[package]))]
        state[package] = "open"
        path.append(package)
        while stack:
            name, deps = stack[-1]
            for d in deps:
                if d not in self.index:
                    raise ValueError(f"{name} needs {d}, which isn't in the index")
                if state.get(d) == "open":
                    cycle = path[path.index(d):] + [d]
                    raise ValueError("dependency cycle: " + " -> ".join(cycle))
                if d not in state:
                    state[d] = "open"
                    path.append(d)
                    stack.append((d, iter(self.index[d])))
                    break
            else:
                state[name] = "done"
                order.append(name)
                path.pop()
                stack.pop()
        return order

    def install(self, package):
        new = [p for p in self.install_order(package) if p not in self.installed]
        self.installed.update(new)
        self.explicit.add(package)
        return new

    def _needs(self, package):
        seen, stack = set(), [package]
        while stack:
            p = stack.pop()
            for d in self.index[p]:
                if d not in seen:
                    seen.add(d)
                    stack.append(d)
        return seen

    def dependents(self, package):
        return sorted(p for p in self.installed if p != package and package in self._needs(p))

    def uninstall(self, package):
        if package not in self.installed:
            raise KeyError(package)
        users = self.dependents(package)
        if users:
            raise ValueError(f"{package} is needed by {', '.join(users)}")
        self.installed.discard(package)
        self.explicit.discard(package)

    def autoremove(self):
        keep = set()
        for p in self.explicit:
            keep.add(p)
            keep |= self._needs(p)
        removed = sorted(self.installed - keep)
        self.installed &= keep
        return removed
