@@tests
import types
def test_paths():
    "recursive generators"
    t = [5, [4, [11, [7], [2]]], [8, [13], [4, [1]]]]
    assert list(root_paths(t)) == [[5, 4, 11, 7], [5, 4, 11, 2], [5, 8, 13], [5, 8, 4, 1]]

@@solution
def root_paths(tree, prefix=()):
    value, children = tree[0], tree[1:]
    path = prefix + (value,)
    if not children:
        yield list(path)
        return
    for child in children:
        yield from root_paths(child, path)

def path_sums(tree, target):
    return [p for p in root_paths(tree) if sum(p) == target]

    assert k_smallest([1, 7, 11], [2, 4, 6], 3) == [(1, 2), (1, 4), (1, 6)]
    assert k_smallest([1, 1, 2], [1, 2, 3], 2) == [(1, 1), (1, 1)] and k_smallest([], [1], 3) == []

@@solution
import heapq
from itertools import islice

def pair_sums(a, b):
    if not a or not b:
        return
    heap = [(a[0] + b[j], 0, j) for j in range(len(b))]
    heapq.heapify(heap)
    while heap:
        _, i, j = heapq.heappop(heap)
        yield a[i], b[j]
        if i + 1 < len(a):
            heapq.heappush(heap, (a[i + 1] + b[j], i + 1, j))

def k_smallest(a, b, k):
    return list(islice(pair_sums(a, b), k))

## Forge: a stream library

Write a class `Stream(iterable)` that wraps an iterable and offers **stages** (each returning a new `Stream`, without doing any work) and **sinks** (each pulling values through and returning a result). Every stage must stay lazy, so a `Stream` over an endless source works as long as a sink stops early. Use generators and `itertools` inside.

- `iter(stream)` iterates over the values (so a `Stream` works in `for`, `list` and so on; a stream over a list or range can be iterated again).
- Constructors: `Stream.of(*items)` and `Stream.lines(text)`: the non-blank lines of a string.
- Stages: `map(func)`, `filter(pred)`, `take(n)`, `skip(n)`, `take_while(pred)`, `drop_while(pred)`, `chunk(n)` (lists of n, the last possibly shorter), `window(n)` (tuples of n neighbours), `unique(key=None)` (first occurrences), `runs(key=None)` (pairs `(key, list_of_items)` for each run of neighbouring items with the same key), `flat_map(func)` (each item's iterable, flattened), `enumerate(start=0)`. `chunk` and `window` raise `ValueError` for n < 1.
- Sinks: `to_list()`, `first(default)` (the first value; with no default, `ValueError` on an empty stream), `count()`, `sum()`, `reduce(func, initial)` (fold left; with no initial, start from the first value, and `ValueError` if empty).

Store the iterable; `__iter__` returns `iter(self._source)`. Each stage builds a new `Stream` around a lazy object: `Stream(map(func, self))`, `Stream(itertools.islice(self, n))`, `Stream(itertools.batched(self, n))` turned into lists, or a small generator function for `window` and `unique`. Use a private sentinel `_MISSING = object()` to tell "no default given" from `None`.
@@solution
import itertools
from collections import deque

_MISSING = object()

class Stream:
    def __init__(self, iterable):
        self._source = iterable

    def __iter__(self):
        return iter(self._source)

    @classmethod
    def of(cls, *items):
        return cls(items)

    @classmethod
    def lines(cls, text):
        return cls(line for line in text.splitlines() if line.strip())

    # stages: each returns a new Stream
    def map(self, func):
        return Stream(map(func, self))

    def filter(self, pred):
        return Stream(filter(pred, self))

    def take(self, n):
        return Stream(itertools.islice(self, n))

    def skip(self, n):
        return Stream(itertools.islice(self, n, None))

    def take_while(self, pred):
        return Stream(itertools.takewhile(pred, self))

    def drop_while(self, pred):
        return Stream(itertools.dropwhile(pred, self))

    def chunk(self, n):
        if n < 1:
            raise ValueError("chunk size must be at least 1")
        return Stream(list(b) for b in itertools.batched(self, n))

    def window(self, n):
        if n < 1:
            raise ValueError("window size must be at least 1")
        def windows(it):
            recent = deque(maxlen=n)
            for x in it:
                recent.append(x)
                if len(recent) == n:
                    yield tuple(recent)
        return Stream(windows(self))

    def unique(self, key=None):
        def gen(it):
            seen = set()
            for x in it:
                k = x if key is None else key(x)
                if k not in seen:
                    seen.add(k)
                    yield x
        return Stream(gen(self))

    def runs(self, key=None):
        return Stream((k, list(g)) for k, g in itertools.groupby(self, key))

    def flat_map(self, func):
        return Stream(itertools.chain.from_iterable(map(func, self)))

    def enumerate(self, start=0):
        return Stream(enumerate(self, start))

    # sinks
    def to_list(self):
        return list(self)

    def first(self, default=_MISSING):
        for x in self:
            return x
        if default is _MISSING:
            raise ValueError("the stream is empty")
        return default

    def count(self):
        return sum(1 for _ in self)

    def sum(self):
        return sum(self)

    def reduce(self, func, initial=_MISSING):
        it = iter(self)
        if initial is _MISSING:
            try:
                acc = next(it)
            except StopIteration:
                raise ValueError("reduce of an empty stream with no initial value") from None
        else:
            acc = initial
        for x in it:
            acc = func(acc, x)
        return acc
:::

:::think title="Why the stages don't run"
Every stage returns a Stream wrapping another lazy object, so building `Stream(data).map(f).filter(g).take(5)` costs nothing: it's a chain of generators waiting to be asked. Only a sink pulls, and it pulls one value at a time through the whole chain. That's why the laziness test can count exactly how many items of the source were read (four), and why an endless source is fine. It's also the design of Java's streams, of LINQ in C#, and of Python packages such as more-itertools, all built on the same protocol you met on Monday.
:::

## Forge bonus: merging logs

:::exercise id=forge2 points=30 title="Forge bonus: merging logs"
Several programs write logs with lines `"HH:MM:SS source message"`, each log already in time order. Write, lazily throughout:

- `parse_line(line)`: `(time, source, message)`, or `None` for a line without all three parts or whose time isn't exactly two digits, two digits, two digits separated by colons. `records(lines)`: a generator of the parsed records, skipping the rest.
- `merged(*logs)`: one stream of records from all the logs in time order, using `heapq.merge` with a `key` (equal times keep the order of the logs given).
- `seconds(time)`: seconds since midnight.
- `per_minute(records)`: a generator of `(minute, count)`, where minute is `"HH:MM"`, using `groupby` (the records arrive in time order).
- `busy_minutes(records, threshold)`: the list of minutes with at least `threshold` records.
- `quiet_gaps(records, limit)`: a generator of `(time_a, time_b)` for neighbouring records more than `limit` seconds apart (`pairwise`).

@@tests
import itertools, types

WEB = ["09:00:01 web start", "09:00:30 web GET /", "09:01:05 web GET /about", "bad", "09:03:00 web stop"]
DB = ["09:00:02 db connect", "09:00:30 db query", "9:1:0 db broken time", "09:01:59 db query"]

def test_parse():
    "parsing"
    assert parse_line("09:00:01 web start up") == ("09:00:01", "web", "start up")
    assert parse_line("bad") is None and parse_line("9:00:01 web x") is None and parse_line("09:00:01 web") is None
    assert isinstance(records(WEB), types.GeneratorType) and len(list(records(WEB))) == 4

def test_merge():
    "heapq.merge"
    times = [r[:2] for r in merged(WEB, DB)]
    assert times == [("09:00:01", "web"), ("09:00:02", "db"), ("09:00:30", "web"), ("09:00:30", "db"),
                     ("09:01:05", "web"), ("09:01:59", "db"), ("09:03:00", "web")]
    assert list(merged()) == [] and list(merged([], WEB))[0][0] == "09:00:01"

def test_minutes():
    "groupby and pairwise"
    assert list(per_minute(merged(WEB, DB))) == [("09:00", 4), ("09:01", 2), ("09:03", 1)]
    assert busy_minutes(merged(WEB, DB), 2) == ["09:00", "09:01"] and busy_minutes([], 1) == []
    assert list(quiet_gaps(merged(WEB, DB), 30)) == [("09:00:30", "09:01:05"), ("09:01:05", "09:01:59"), ("09:01:59", "09:03:00")]
    assert seconds("01:02:03") == 3723

def test_lazy():
    "endless logs"
    def ticker(source, start, step):
        for t in itertools.count(start, step):
            h, rest = divmod(t, 3600)
            m, s = divmod(rest, 60)
            yield f"{h:02d}:{m:02d}:{s:02d} {source} tick"
    first = list(itertools.islice(merged(ticker("a", 0, 7), ticker("b", 3, 5)), 5))
    assert [r[0] for r in first] == ["00:00:00", "00:00:03", "00:00:07", "00:00:08", "00:00:13"]
    gen = per_minute(merged(ticker("a", 0, 7), ticker("b", 3, 5)))
    assert next(gen) == ("00:00", 9 + 12)
@@solution
import heapq
from itertools import groupby, pairwise

def parse_line(line):
    parts = line.strip().split(" ", 2)
    if len(parts) < 3:
        return None
    time, source, message = parts
    bits = time.split(":")
    if len(bits) != 3 or not all(b.isdigit() and len(b) == 2 for b in bits):
        return None
    return time, source, message

def records(lines):
    for line in lines:
        rec = parse_line(line)
        if rec is not None:
            yield rec

def merged(*logs):
    return heapq.merge(*(records(log) for log in logs), key=lambda r: r[0])

def seconds(time):
    h, m, s = map(int, time.split(":"))
    return h * 3600 + m * 60 + s

def per_minute(recs):
    for minute, group in groupby(recs, key=lambda r: r[0][:5]):
        yield minute, sum(1 for _ in group)

def busy_minutes(recs, threshold):
    return [m for m, n in per_minute(recs) if n >= threshold]

def quiet_gaps(recs, limit):
    for a, b in pairwise(recs):
        if seconds(b[0]) - seconds(a[0]) > limit:
            yield a[0], b[0]
