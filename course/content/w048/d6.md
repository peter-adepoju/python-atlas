    pi, k = [0] * len(s), 0

def _suffix_trie(s):
    root = {}
    for i in range(len(s)):
        node = root
        for ch in s[i:]:
            node = node.setdefault(ch, {})
    return root

def count_distinct_substrings(s):
    count, stack = 0, [_suffix_trie(s)]
    while stack:
        node = stack.pop()
        count += len(node)
        stack.extend(node.values())
    return count

def nth_substring(s, n):
    stack = [(ch, child) for ch, child in sorted(_suffix_trie(s).items(), reverse=True)]
    while stack:
        word, node = stack.pop()
        n -= 1
        if n == 0:
            return word
        for ch, child in sorted(node.items(), reverse=True):
            stack.append((word + ch, child))
    raise IndexError("not that many distinct substrings")

def repeated_sequences(dna, k=10):
    code = {"A": 0, "C": 1, "G": 2, "T": 3}
    for ch in dna:
        if ch not in code:
            raise ValueError(f"not a DNA letter: {ch!r}")
    if k > len(dna):
        return []
    top = 4 ** (k - 1)
    v = 0
    for ch in dna[:k]:
        v = v * 4 + code[ch]
    counts, first = {v: 1}, {v: 0}
    for i in range(k, len(dna)):
        v = (v - code[dna[i - k]] * top) * 4 + code[dna[i]]
        counts[v] = counts.get(v, 0) + 1
        first.setdefault(v, i - k + 1)
    return [dna[first[v]:first[v] + k] for v in first if counts[v] > 1]

## Forge: many patterns at once

A message filter has a list of a thousand banned phrases, and must check every message for all of them. Running KMP once per phrase reads each message a thousand times. In 1975 Alfred Aho and Margaret Corasick, working on a bibliographic search tool, combined this week's two big ideas so the text is read **once**:

1. Put all the patterns in a **trie** (Day 1). Each node is "the last few characters read match this prefix of some pattern".
2. Give every node a **failure link** (Day 2's prefix function, generalised): the node for the **longest proper suffix** of its string that is also a prefix of some pattern (a path in the trie). On a mismatch, follow failure links until a node has the needed child, or the root is reached, exactly as KMP falls back with `pi`.
3. Compute the links **breadth-first** (Week 43): a node's failure link is found from its parent's, which is shallower and so already done. For a child reached from `node` by character `c`: start at `node.fail`, follow failure links until one has a child `c` (or you reach the root); that child (or the root, if even the root has none) is the link.
4. Give every node an **output list**: the patterns ending exactly there, followed by the output list of its failure link (patterns that end at the same place but are shorter, like "he" inside "she").

Then scanning the text is KMP with a trie: for each character, step (falling back as needed), and report everything in the current node's output list. The time is O(text length + total pattern length + number of matches).

```text
patterns: he, she, his, hers
          (root)
          /    \
         h      s
        / \      \
       e*  i      h
       |    \      \
       r     s*     e*     failure links: "sh" -> "h", "she" -> "he",
       |                    "hi" -> root, "his" -> "s", "her" -> root, ...
       s*                   output of "she": she, he
```

Write `AhoCorasick(patterns)`:

- Patterns must be non-empty strings (`ValueError` otherwise); repeats are ignored. Keep the distinct patterns in `patterns`, in first-seen order.
- `find_all(text)`: every match as `(start, pattern)`, sorted.
- `count(text)`: a dict from each pattern to how many times it occurs (overlaps count).
- `feed(ch)` and `reset()`: a streaming interface; `feed` returns the list of patterns that end with this character, longest first.
- `censor(text, mask="*")`: the text with every character that's part of **any** match replaced by `mask`. Do it in O(n): the longest match ending at each position covers all shorter ones, so mark its start and end in a difference array (Week 41's prefix sums) and sweep once.

No recursion. The checks compare with brute force on random patterns and texts, and time 1,000 patterns against a 100,000-character text.

Build the trie first; mark each node that ends a pattern by putting the pattern in its `out`. Then: set every child of the root to fail to the root and queue them; for each node popped, for each `(ch, child)`, find `f = node.fail`, walk `f = f.fail` while `f` isn't the root and has no child `ch`, and set `child.fail` to `f.children[ch]` if it exists, otherwise the root; then `child.out = child.out + child.fail.out`, and queue the child. A `_step(node, ch)` helper that falls back and then moves is used by everything else.
@@solution
from collections import deque

class ACNode:
    def __init__(self):
        self.children = {}
        self.fail = None
        self.out = []          # patterns ending exactly here, longest first

class AhoCorasick:
    def __init__(self, patterns):
        self.root = ACNode()
        self.patterns = []
        for p in patterns:
            if not isinstance(p, str) or not p:
                raise ValueError(f"bad pattern: {p!r}")
            node = self.root
            for ch in p:
                node = node.children.setdefault(ch, ACNode())
            if not node.out:
                node.out.append(p)
                self.patterns.append(p)
        self._link()
        self._state = self.root

    def _link(self):
        root = self.root
        root.fail = root
        queue = deque()
        for child in root.children.values():
            child.fail = root
            queue.append(child)
        while queue:
            node = queue.popleft()
            for ch, child in node.children.items():
                f = node.fail
                while f is not root and ch not in f.children:
                    f = f.fail
                child.fail = f.children[ch] if ch in f.children and f.children[ch] is not child else root
                child.out = child.out + child.fail.out
                queue.append(child)

    def _step(self, node, ch):
        while node is not self.root and ch not in node.children:
            node = node.fail
        return node.children.get(ch, self.root)

    def feed(self, ch):
        self._state = self._step(self._state, ch)
        return list(self._state.out)

    def reset(self):
        self._state = self.root

    def find_all(self, text):
        node, found = self.root, []
        for i, ch in enumerate(text):
            node = self._step(node, ch)
            for p in node.out:
                found.append((i - len(p) + 1, p))
        return sorted(found)

    def count(self, text):
        counts = {p: 0 for p in self.patterns}
        node = self.root
        for ch in text:
            node = self._step(node, ch)
            for p in node.out:
                counts[p] += 1
        return counts

    def censor(self, text, mask="*"):
        diff = [0] * (len(text) + 1)            # +1 where a match starts, -1 just after it ends
        node = self.root
        for i, ch in enumerate(text):
            node = self._step(node, ch)
            if node.out:
                diff[i - len(node.out[0]) + 1] += 1     # the longest match ending here covers the rest
                diff[i + 1] -= 1
        out, depth = [], 0
        for i, ch in enumerate(text):
            depth += diff[i]
            out.append(mask if depth > 0 else ch)
        return "".join(out)

:::think title="Two ideas, one structure"
Aho-Corasick is what you get when two of this week's ideas meet. The trie shares work between patterns that **start** the same way; the failure links share work between positions in the text, never re-reading a character, as KMP did for a single pattern. Building the links breadth-first is the same "use what you've already computed" move as the prefix function's `k = pi[k - 1]`, applied to a tree instead of a line. The algorithm is still at the heart of tools that scan text for many fixed strings at once: the original `fgrep` command was built on it, and network intrusion detection and virus scanners use descendants of it to check traffic against thousands of signatures.
:::

## Forge bonus: shared passages

:::exercise id=forge2 points=30 title="Forge bonus: shared passages"
Yesterday's winnowing gave a similarity score; now find the actual passages two documents share, measured in **words**.

- `words(text)`: the lower-cased runs of letters and digits, using `re.findall(r"[a-z0-9]+", text.lower())`.
- `shared_passages(a, b, min_words)`: every **maximal** shared passage of at least `min_words` words, as `(i, j, length)`: `words(a)[i:i + length] == words(b)[j:j + length]`, and it can't be extended (the words just before both, or just after both, differ or don't exist). Sorted. Give each distinct word a number, hash every `min_words`-word window of both documents with a rolling hash (random base, modulus 2⁶¹ − 1), and look a's windows up in a dict of b's. For each matching pair that **starts** a passage (no equal words just before both), confirm the window really matches (a collision is possible), then extend it word by word.
- `coverage(a, b, min_words)`: the fraction of a's words that lie in at least one shared passage (0.0 if a has no words).
- `report(a, b, min_words)`: one line per passage: `words 0-4 of a = words 1-5 of b: the quick brown fox jumps`.

import random
def brute(a, b, m):

    out = []
    for i in range(len(wa)):
        for j in range(len(wb)):

import random, re

MOD = (1 << 61) - 1

def words(text):
    return re.findall(r"[a-z0-9]+", text.lower())

def _window_hashes(ids, m, base):
    """The hash of every length-m window of a list of whole numbers."""
    if m > len(ids):
        return []
    high = pow(base, m - 1, MOD)
    v = 0
    for x in ids[:m]:
        v = (v * base + x) % MOD
    out = [v]
    for i in range(m, len(ids)):
        v = ((v - ids[i - m] * high) * base + ids[i]) % MOD
        out.append(v)
    return out

def shared_passages(a, b, min_words):
    wa, wb = words(a), words(b)
    vocab = {}
    ia = [vocab.setdefault(w, len(vocab) + 1) for w in wa]
    ib = [vocab.setdefault(w, len(vocab) + 1) for w in wb]
    base = random.randrange(1 << 20, 1 << 40)
    where = {}
    for j, h in enumerate(_window_hashes(ib, min_words, base)):
        where.setdefault(h, []).append(j)
    found = []
    for i, h in enumerate(_window_hashes(ia, min_words, base)):
        for j in where.get(h, ()):
            if i > 0 and j > 0 and ia[i - 1] == ib[j - 1]:
                continue                       # not the start of a maximal run
            if ia[i:i + min_words] != ib[j:j + min_words]:
                continue                       # a hash collision
            length = min_words
            while i + length < len(ia) and j + length < len(ib) and ia[i + length] == ib[j + length]:
                length += 1
            found.append((i, j, length))
    return sorted(found)

def coverage(a, b, min_words):
    n = len(words(a))
    if n == 0:
        return 0.0
    covered = set()
    for i, _, length in shared_passages(a, b, min_words):
        covered.update(range(i, i + length))
    return len(covered) / n

def report(a, b, min_words):
    wa = words(a)
    lines = []
    for i, j, length in shared_passages(a, b, min_words):
        lines.append(f"words {i}-{i + length - 1} of a = words {j}-{j + length - 1} of b: {' '.join(wa[i:i + length])}")
    return lines
