class HTTPException(Exception):
    def __init__(self, code, description=None):
        self.code = code
        self.name = HTTPStatus(code).phrase
        self.description = description or self.name
        super().__init__(f"{code} {self.name}: {self.description}")


def abort(code, description=None):
    raise HTTPException(code, description)


class Headers:
    """Case-insensitive headers."""
    def __init__(self, items=()):
        self._items = [(k, str(v)) for k, v in (items.items() if isinstance(items, dict) else items)]

    def get(self, name, default=None):
        for k, v in self._items:
            if k.lower() == name.lower():
                return v
        return default

    def __getitem__(self, name):
        value = self.get(name)
        if value is None:
            raise KeyError(name)
        return value

    def __contains__(self, name):
        return self.get(name) is not None

    def __setitem__(self, name, value):
        self._items = [(k, v) for k, v in self._items if k.lower() != name.lower()] + [(name, str(value))]

    def items(self):
        return list(self._items)


class Args(dict):
    """Query or form values: .get(name) gives the first value; .getlist(name) all of them."""
    def __init__(self, pairs):
        super().__init__()
        self._lists = {}
        for k, v in pairs:
            self._lists.setdefault(k, []).append(v)
            dict.setdefault(self, k, v)

    def get(self, key, default=None, type=None):
        if key not in self:
            return default
        value = self[key]
        if type is not None:
            try:
                return type(value)
            except (TypeError, ValueError):
                return default
        return value

    def getlist(self, key):
        return list(self._lists.get(key, []))


class Request:
    def __init__(self, environ):
        self.environ = environ
        self.method = environ["REQUEST_METHOD"].upper()
        self.path = environ.get("PATH_INFO", "/") or "/"
        self.args = Args(_pairs(environ.get("QUERY_STRING", "")))
        items = [(k[5:].replace("_", "-").title(), v) for k, v in environ.items() if k.startswith("HTTP_")]
        if environ.get("CONTENT_TYPE"):
            items.append(("Content-Type", environ["CONTENT_TYPE"]))
        self.headers = Headers(items)
        length = int(environ.get("CONTENT_LENGTH") or 0)
        self.data = environ["wsgi.input"].read(length) if length else b""
        self.content_type = environ.get("CONTENT_TYPE", "")
        self.form = Args(_pairs(self.data.decode("utf-8"))) if self.content_type.startswith("application/x-www-form-urlencoded") else Args([])
        self.view_args = {}
        cookie = environ.get("HTTP_COOKIE", "")
        self.cookies = dict(p.strip().split("=", 1) for p in cookie.split(";") if "=" in p)

    @property
    def is_json(self):
        return self.content_type.split(";")[0].strip() == "application/json"

    def get_json(self, silent=False, force=False):
        if not (self.is_json or force):
            if silent:
                return None
            abort(415, "expected a JSON body (Content-Type: application/json)")
        try:
            return json.loads(self.data or b"null")
        except ValueError:
            if silent:
                return None
            abort(400, "the body isn't valid JSON")

    @property
    def json(self):
        return self.get_json()

_CONVERTERS = {"int": (r"\d+", int), "float": (r"\d+(?:\.\d+)?", float), "path": (r".+", str), "string": (r"[^/]+", str)}


class _Rule:
    def __init__(self, rule, endpoint, view, methods):
        self.rule, self.endpoint, self.view = rule, endpoint, view
        self.methods = {m.upper() for m in methods} | ({"HEAD"} if "GET" in {m.upper() for m in methods} else set())
        regex, self.converters = "", {}
        for piece in re.split(r"(<[^>]+>)", rule):
            m = re.fullmatch(r"<(?:(\w+):)?(\w+)>", piece)
            if m:
                kind, name = m.group(1) or "string", m.group(2)
                pattern, convert = _CONVERTERS[kind]
                regex += f"(?P<{name}>{pattern})"
                self.converters[name] = convert
            else:
                regex += re.escape(piece)
        self.regex = re.compile(regex)

    def match(self, path):
        m = self.regex.fullmatch(path)
        if not m:
            return None
        return {k: self.converters[k](v) for k, v in m.groupdict().items()}

    def build(self, values):
        def fill(m):
            return str(values.pop(m.group(2)))
        path = re.sub(r"<(?:(\w+):)?(\w+)>", fill, self.rule)
        return path + (f"?{urlencode(values)}" if values else "")


class _Setup:
    def route(self, rule, methods=("GET",), endpoint=None):
        def decorate(view):
            self._add(rule, endpoint or view.__name__, view, methods)
            return view
        return decorate

    def get(self, rule, **kw):
        return self.route(rule, methods=["GET"], **kw)

    def post(self, rule, **kw):
        return self.route(rule, methods=["POST"], **kw)

    def put(self, rule, **kw):
        return self.route(rule, methods=["PUT"], **kw)

    def patch(self, rule, **kw):
        return self.route(rule, methods=["PATCH"], **kw)

    def delete(self, rule, **kw):
        return self.route(rule, methods=["DELETE"], **kw)


class Blueprint(_Setup):
    def __init__(self, name, import_name, url_prefix=""):
        self.name, self.url_prefix = name, url_prefix
        self.deferred = []

    def _add(self, rule, endpoint, view, methods):
        self.deferred.append((rule, endpoint, view, methods))


class _App(_Setup):
    def __init__(self, import_name):
        self.import_name = import_name
        self.config = {"TESTING": False}
        self.rules = []
        self.before_funcs, self.after_funcs, self.error_handlers = [], [], {}

    def _add(self, rule, endpoint, view, methods):
        self.rules.append(_Rule(rule, endpoint, view, methods))

    def register_blueprint(self, bp, url_prefix=None):
        prefix = url_prefix if url_prefix is not None else bp.url_prefix
        for rule, endpoint, view, methods in bp.deferred:
            self._add(prefix + rule, f"{bp.name}.{endpoint}", view, methods)

    def before_request(self, f):
        self.before_funcs.append(f)
        return f

    def after_request(self, f):
        self.after_funcs.append(f)
        return f

    def errorhandler(self, code):
        def decorate(f):
            self.error_handlers[code] = f
            return f
        return decorate

    def url_for(self, endpoint, **values):
        for r in self.rules:
            if r.endpoint == endpoint:
                return r.build(dict(values))
        raise LookupError(f"no endpoint {endpoint!r}")

    @staticmethod
    def _to_response(rv):
        status, headers = None, None
        if isinstance(rv, tuple):
            if len(rv) == 3:
                rv, status, headers = rv
            elif isinstance(rv[1], (dict, list)):
                rv, headers = rv
            else:
                rv, status = rv
        if isinstance(rv, Response):
            response = rv
        elif isinstance(rv, (dict, list)):
            response = jsonify(rv)
        elif isinstance(rv, (str, bytes)):
            response = Response(rv)
        elif rv is None:
            raise TypeError("the view function did not return a valid response")
        else:
            raise TypeError(f"can't turn {type(rv).__name__} into a response")
        if status is not None:
            response.status_code = status
        for k, v in (headers.items() if isinstance(headers, dict) else headers or []):
            response.headers[k] = v
        return response

    def _error(self, e):
        handler = self.error_handlers.get(e.code)
        if handler:
            return self._to_response(handler(e))
        body = f"<h1>{e.code} {e.name}</h1><p>{e.description}</p>"
        return Response(body, e.code)

    def _dispatch(self, req):
        path = req.path
        allowed = set()
        for r in self.rules:
            values = r.match(path)
            if values is None:
                continue
            if req.method in r.methods:
                req.view_args = values
                return r.view(**values)
            allowed |= r.methods
        if allowed:
            raise _MethodNotAllowed(sorted(allowed | {"OPTIONS"}))
        abort(404, "The requested URL was not found on the server.")

    def full_dispatch(self, environ):
        req = Request(environ)
        token = _current.set((req, _G(), self))
        try:
            try:
                rv = None
                for f in self.before_funcs:
                    rv = f()
                    if rv is not None:
                        break
                if rv is None:
                    rv = self._dispatch(req)
                response = self._to_response(rv)
            except _MethodNotAllowed as e:
                response = self._error(HTTPException(405))
                response.headers["Allow"] = ", ".join(e.allowed)
            except HTTPException as e:
                response = self._error(e)
            except Exception as e:
                if self.config.get("TESTING") and 500 not in self.error_handlers:
                    raise
                response = self._error(HTTPException(500)) if 500 not in self.error_handlers else self._to_response(self.error_handlers[500](e))
                response.status_code = 500
            for f in self.after_funcs:
                response = f(response) or response
            if req.method == "HEAD":
                response.headers["Content-Length"] = len(response.data)
                response.data = b""
            return response
        finally:
            _current.reset(token)

    def __call__(self, environ, start_response):
        response = self.full_dispatch(environ)
        if "Content-Length" not in response.headers:
            response.headers["Content-Length"] = len(response.data)
        start_response(response.status, response.headers.items())
        return [response.data]

class ValidationError(Exception):
    """Raised with a dict of field -> message."""

    def __init__(self, errors):
        super().__init__("; ".join(f"{k}: {v}" for k, v in sorted(errors.items())))
        self.errors = errors


def validate_ticket(data, partial=False):
    """Check a ticket body from a client. partial=True for PATCH: only the fields present are checked."""
    if not isinstance(data, dict):
        raise ValidationError({"_": "expected a JSON object"})
    errors, clean = {}, {}
    unknown = sorted(set(data) - {"title", "body", "priority"})
    if unknown:
        errors["_"] = f"unexpected field(s): {', '.join(unknown)}"
    for field in ("title", "body"):
        if field not in data:
            if not partial:
                errors[field] = "required"
            continue
        value = data[field]
        limit = (3, 120) if field == "title" else (1, 5000)
        if not isinstance(value, str) or not limit[0] <= len(value.strip()) <= limit[1]:
            errors[field] = f"{limit[0]} to {limit[1]} characters"
        else:
            clean[field] = value.strip()
    if "priority" in data:
        if data["priority"] not in PRIORITIES:
            errors["priority"] = f"one of {', '.join(PRIORITIES)}"
        else:
            clean["priority"] = data["priority"]
    elif not partial:
        clean["priority"] = "normal"
    if partial and not data:
        errors["_"] = "nothing to change"
    if errors:
        raise ValidationError(errors)
    return clean


def can_transition(current, new, role):
    """True if someone with this role may move a ticket from `current` to `new`."""
    return role in TRANSITIONS.get(current, {}).get(new, set())


def allowed(role, action, is_owner=False):
    """True if the role may do the action (is_owner: the ticket is the user's own)."""
    always, own = PERMISSIONS[action]
    return role in always or (is_owner and role in own)


def problem(status, detail, title=None, **extra):
    """An RFC 9457 problem-details response: (body, status, headers)."""
    body = {"type": "about:blank", "title": title or HTTPStatus(status).phrase, "status": status, "detail": detail}
    body.update(extra)
    return body, status, {"Content-Type": "application/problem+json"}

def connect(path=":memory:"):

def migrate(conn, migrations=MIGRATIONS):
    """Apply the migrations the database hasn't had yet, each in its own transaction. Returns their versions."""
    conn.execute("CREATE TABLE IF NOT EXISTS schema_migrations (version INTEGER PRIMARY KEY, name TEXT NOT NULL)")
    conn.commit()
    versions = [v for v, _, _ in migrations]
    if versions != list(range(1, len(versions) + 1)):
        raise ValueError("migrations must be numbered 1, 2, 3... with no gaps")
    current = conn.execute("SELECT COALESCE(MAX(version), 0) FROM schema_migrations").fetchone()[0]
    if current > len(versions):
        raise RuntimeError(f"the database is at version {current}, newer than this code")
    applied = []
    for version, name, sql in migrations[current:]:
        try:
            conn.execute("BEGIN")
            for statement in sql.split(";"):
                if statement.strip():
                    conn.execute(statement)

import base64
import hashlib
import hmac
import secrets


def pbkdf2_sha256(password, salt, iterations, dklen=32):
    base = hmac.new(password, digestmod=hashlib.sha256)
    out, block = b"", 1
    while len(out) < dklen:
        h = base.copy()
        h.update(salt + block.to_bytes(4, "big"))
        u = h.digest()
        total = int.from_bytes(u, "big")
        for _ in range(iterations - 1):
            h = base.copy()
            h.update(u)
            u = h.digest()
            total ^= int.from_bytes(u, "big")
        out += total.to_bytes(32, "big")
        block += 1
    return out[:dklen]


def hash_password(password, iterations):
    salt = secrets.token_bytes(16)
    digest = pbkdf2_sha256(password.encode(), salt, iterations)
    return f"pbkdf2_sha256${iterations}${base64.b64encode(salt).decode()}${base64.b64encode(digest).decode()}"


def check_password(password, stored):
    try:
        algorithm, iterations, salt, digest = stored.split("$")
        expected = pbkdf2_sha256(password.encode(), base64.b64decode(salt), int(iterations))
        return algorithm == "pbkdf2_sha256" and hmac.compare_digest(expected, base64.b64decode(digest))
    except ValueError:
        return False


def issue_token(user_id, expires, key):
    """A signed bearer token: USER_ID.EXPIRES.SIGNATURE."""
    payload = f"{user_id}.{expires}"
    return f"{payload}.{hmac.new(key, payload.encode(), hashlib.sha256).hexdigest()}"


def verify_token(token, key, now):
    """The user id in a valid, unexpired token, else None."""
    parts = token.split(".")
    if len(parts) != 3 or not parts[0].isdigit() or not parts[1].isdigit():
        return None
    payload = f"{parts[0]}.{parts[1]}"
    expected = hmac.new(key, payload.encode(), hashlib.sha256).hexdigest()
    if not hmac.compare_digest(parts[2], expected) or now >= int(parts[1]):
        return None
    return int(parts[0])

## Step 11: errors and health

Before any feature, decide what every **failure** looks like, so no endpoint has to. Two pieces:

- An exception, **`ApiError(code, detail, **extra)`**, a subclass of miniflask's `HTTPException`, that any view can raise: `raise ApiError(404, "no ticket 7")`. Raising (instead of returning an error) means helper functions such as "load this ticket or fail" can stop a request from anywhere.
- An **error handler** for each status code the API uses, turning the exception into `problem(...)` from the domain module. The `500` handler is special: it also receives exceptions that **aren't** `HTTPException`s (a bug, a database failure), and must not show them to the client: log `incident ID: TYPE: MESSAGE` and return only the incident id (Week 79).

Then two endpoints for whoever runs the service (Week 77 Day 5): **`/healthz`** says the process is alive, and **`/readyz`** says it can do its job, here meaning the database answers and every migration has been applied; otherwise `503`, so a load balancer sends traffic elsewhere until it's ready.

## Step 12: accounts and tokens

Three endpoints and one hook give the helpdesk its accounts:

- **`POST /users`** signs up a **customer** (staff accounts are created by admins, never through a public form): a valid email, lower-cased, and a password of 12 to 128 characters, else `422`; a taken email is `409`. The password is stored with `hash_password` (Week 79), never as it was sent.
- **`POST /tokens`** logs in: the right password returns `201` with `{"token", "expires_at"}`, where the token expires after `TOKEN_SECONDS`; a wrong email **or** password gives the same `401`, and an unknown email is still checked against a dummy hash, so both take the same time.
- A **`before_request`** hook reads `Authorization: Bearer TOKEN` on every request and, if the token is valid and unexpired, loads the user into `g.user`; otherwise `g.user` is `None`. Endpoints that need a user call `user()`, which raises `401` if there isn't one. Expiry uses the injected clock, so a test can step past it.
- **`GET /me`** returns the logged-in user (from `db.get_user`, which never includes the password).

A small helper, `body()`, reads the JSON body and raises `400` if it isn't a JSON object, so no view crashes on `None.get`.

## Step 13: opening and reading tickets

Now the tickets themselves, with the permission rules from Monday. Three helpers keep every endpoint short and consistent:

- **`checked(data, partial)`** runs `validate_ticket` and turns a `ValidationError` into a `422` whose `errors` field lists the problems.
- **`visible_ticket(ticket_id)`** loads a ticket for the current user, and raises **`404`** if it doesn't exist **or** the user may not view it, so a customer can't even confirm that someone else's ticket exists.
- **`require(me, action, ticket, message)`** raises `403` with a message unless `allowed(...)` says yes (with `is_owner` worked out from the ticket).

The list endpoint validates its query parameters (the status and priority must be known values; `after` and `limit` whole numbers in range, `limit` from 1 to 50), and **customers only ever get their own tickets**, whatever they ask for: the web layer adds `customer_id` itself, rather than trusting a parameter.

## Step 14: changing tickets

Three endpoints change tickets, each mapping a rule onto a status code:

- **`PATCH /tickets/{id}`** edits the title, body or priority. Only staff may change the **priority**; only the customer who opened it (or an admin) may change the **title or body**, and a customer only while it's still `open` (`409` once someone is working on it, because the request conflicts with the ticket's current state).
- **`POST /tickets/{id}/transitions`** with `{"status": NEW}` moves a ticket, if Monday's state machine allows that role to make that move (`403` otherwise, saying which move was refused); an unknown status is `422`. It passes the status the ticket had when it was loaded as `expected`, so if someone else moved it in the meantime, `ConflictError` becomes `409`.
- **`PUT /tickets/{id}/assignee`** with `{"agent_id": N}` assigns it (staff only); an id that isn't a whole number, or isn't staff, is `422`.

class ApiError(HTTPException):
    def __init__(self, code, detail, **extra):
        super().__init__(code, detail)
        self.extra = extra


def iso(t):
    return datetime.fromtimestamp(t, timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


def create_app(settings, conn, clock=time.time, log=print):

    key = settings["SECRET_KEY"].encode()
    token_seconds = settings.get("TOKEN_SECONDS", 3600)
    iterations = settings.get("PASSWORD_ITERATIONS", 600_000)
    dummy = hash_password(secrets.token_hex(8), iterations)   # checked for unknown emails, so they take as long as real ones

    for code in ERROR_CODES:
        @app.errorhandler(code)
        def handle(e, code=code):
            if code == 500 and not isinstance(e, HTTPException):
                incident = secrets.token_hex(4)
                log(f"incident {incident}: {type(e).__name__}: {e}")
                return problem(500, "something went wrong", incident=incident)
            return problem(e.code, e.description, **getattr(e, "extra", {}))

    @app.get("/healthz")
    def healthz():
        return {"status": "ok"}

    @app.get("/readyz")
    def readyz():
        try:
            version = conn.execute("SELECT MAX(version) FROM schema_migrations").fetchone()[0]
        except Exception:
            version = None
        if version != len(db.MIGRATIONS):
            raise ApiError(503, "the database isn't ready")
        return {"status": "ready", "schema": version}

    @app.before_request
    def authenticate():
        g.user = None
        header = request.headers.get("Authorization", "")
        if header.startswith("Bearer "):
            user_id = verify_token(header[7:], key, int(clock()))
            if user_id is not None:
                g.user = db.get_user(conn, user_id)

    def user():
        if g.user is None:
            raise ApiError(401, "log in first")
        return g.user

    def body():
        data = request.get_json(silent=True)
        if not isinstance(data, dict):
            raise ApiError(400, "send a JSON object")
        return data

    @app.post("/users")
    def signup():
        data = body()

        password = data.get("password")

            raise ApiError(422, "a valid email and a password of 12 to 128 characters are needed")
        try:
            user_id = db.create_user(conn, email, hash_password(password, iterations), "customer", iso(clock()))
        except ValueError:
            raise ApiError(409, "email already registered") from None
        return {"id": user_id, "email": email, "role": "customer"}, 201

    @app.post("/tokens")
    def login():
        data = body()
        found = db.get_user_by_email(conn, str(data.get("email", "")).strip().lower())
        stored = found["password"] if found else dummy
        if not check_password(str(data.get("password", "")), stored) or found is None:
            raise ApiError(401, "wrong email or password")
        expires = int(clock()) + token_seconds
        return {"token": issue_token(found["id"], expires, key), "expires_at": iso(expires)}, 201

    @app.get("/me")
    def me():
        return user()

    def checked(data, partial=False):
        try:

        except ValidationError as e:
            raise ApiError(422, "the request has invalid fields", errors=e.errors) from None

    def visible_ticket(ticket_id):
        me = user()
        ticket = db.get_ticket(conn, ticket_id)
        if ticket is None or not allowed(me["role"], "view_ticket", ticket["customer_id"] == me["id"]):
            raise ApiError(404, f"no ticket {ticket_id}")
        return me, ticket

    def require(me, action, ticket=None, message="you can't do that"):
        if not allowed(me["role"], action, ticket is not None and ticket["customer_id"] == me["id"]):
            raise ApiError(403, message)

    @app.post("/tickets")
    def open_ticket():

        require(me, "create_ticket", message="only customers can open tickets")
        ticket = db.create_ticket(conn, me["id"], checked(body()), iso(clock()))

                    raise ApiError(400, f"{name} must be one of {', '.join(choices)}")
                filters[name] = args[name]
        for name, low, high, default in (("after", 0, 10 ** 9, 0), ("limit", 1, 50, 20)):

    def read_ticket(ticket_id):

        if "title" in changes or "body" in changes:
            require(me, "edit_ticket", ticket, "only the customer who opened it can edit it")
            if me["role"] == "customer" and ticket["status"] != "open":
                raise ApiError(409, "only open tickets can be edited")
        return db.update_ticket(conn, ticket_id, changes, me["id"], iso(clock()))

    @app.post("/tickets/<int:ticket_id>/transitions")
    def transition(ticket_id):
        me, ticket = visible_ticket(ticket_id)
        new = body().get("status")
        if new not in STATUSES:
            raise ApiError(422, f"status must be one of {', '.join(STATUSES)}")
        if not can_transition(ticket["status"], new, me["role"]):
            raise ApiError(403, f"you can't move a ticket from {ticket['status']} to {new}")
        try:
            return db.change_status(conn, ticket_id, ticket["status"], new, me["id"], iso(clock()))
        except db.ConflictError:
            raise ApiError(409, "the ticket changed; reload it and try again") from None

    @app.put("/tickets/<int:ticket_id>/assignee")
    def assign(ticket_id):
        me, ticket = visible_ticket(ticket_id)
        require(me, "assign", ticket, "only staff can assign tickets")
        agent_id = body().get("agent_id")
        if not isinstance(agent_id, int) or isinstance(agent_id, bool):
            raise ApiError(422, "agent_id must be a user id")
        try:
            return db.assign(conn, ticket_id, agent_id, me["id"], iso(clock()))
        except ValueError as e:
            raise ApiError(422, str(e)) from None

    return app

## Step 15: comments, statistics and deletion

The last endpoints:

- **`GET /tickets/{id}/comments`** lists a ticket's comments, with **internal notes only for staff**: the filter happens in the query, so a customer's response never contains them at all.
- **`POST /tickets/{id}/comments`** with `{"body", "internal"}` adds one (the owner or staff may comment; only staff may add internal notes).
- **`GET /stats`** returns Tuesday's statistics, to staff only.
- **`DELETE /tickets/{id}`** (admins only) answers `204` with an empty body; the database cascades to the ticket's comments and events.

        super().__init__(code, detail)
        self.extra = extra


def iso(t):
    return datetime.fromtimestamp(t, timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")

    token_seconds = settings.get("TOKEN_SECONDS", 3600)
    iterations = settings.get("PASSWORD_ITERATIONS", 600_000)
    dummy = hash_password(secrets.token_hex(8), iterations)   # checked for unknown emails, so they take as long as real ones

    for code in ERROR_CODES:
        @app.errorhandler(code)
        def handle(e, code=code):
            if code == 500 and not isinstance(e, HTTPException):
                incident = secrets.token_hex(4)
                log(f"incident {incident}: {type(e).__name__}: {e}")
                return problem(500, "something went wrong", incident=incident)
            return problem(e.code, e.description, **getattr(e, "extra", {}))

    @app.get("/healthz")
    def healthz():
        return {"status": "ok"}

    @app.get("/readyz")
    def readyz():
        try:
            version = conn.execute("SELECT MAX(version) FROM schema_migrations").fetchone()[0]
        except Exception:
            version = None
        if version != len(db.MIGRATIONS):
            raise ApiError(503, "the database isn't ready")
        return {"status": "ready", "schema": version}

    @app.before_request
    def authenticate():
        g.user = None
        header = request.headers.get("Authorization", "")
        if header.startswith("Bearer "):
            user_id = verify_token(header[7:], key, int(clock()))
            if user_id is not None:
                g.user = db.get_user(conn, user_id)

    def user():
        if g.user is None:
            raise ApiError(401, "log in first")
        return g.user

    def body():
        data = request.get_json(silent=True)
        if not isinstance(data, dict):
            raise ApiError(400, "send a JSON object")
        return data

    @app.post("/users")
    def signup():
        data = body()
        email = data.get("email", "").strip().lower() if isinstance(data.get("email"), str) else ""
        password = data.get("password")
        if not re.fullmatch(r"[^@\s]+@[^@\s]+\.[a-z]{2,}", email) or not isinstance(password, str) or not 12 <= len(password) <= 128:
            raise ApiError(422, "a valid email and a password of 12 to 128 characters are needed")
        try:
            user_id = db.create_user(conn, email, hash_password(password, iterations), "customer", iso(clock()))
        except ValueError:
            raise ApiError(409, "email already registered") from None
        return {"id": user_id, "email": email, "role": "customer"}, 201

    @app.post("/tokens")
    def login():

        found = db.get_user_by_email(conn, str(data.get("email", "")).strip().lower())
        stored = found["password"] if found else dummy
        if not check_password(str(data.get("password", "")), stored) or found is None:
            raise ApiError(401, "wrong email or password")
        expires = int(clock()) + token_seconds
        return {"token": issue_token(found["id"], expires, key), "expires_at": iso(expires)}, 201

            if name in args:
