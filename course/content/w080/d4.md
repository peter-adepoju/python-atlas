class HTTPException(Exception):

        self.code = code

def abort(code, description=None):

class Headers:
    """Case-insensitive headers."""
    def __init__(self, items=()):
        self._items = [(k, str(v)) for k, v in (items.items() if isinstance(items, dict) else items)]

    def get(self, name, default=None):
        for k, v in self._items:
            if k.lower() == name.lower():
                return v
        return default

    def __getitem__(self, name):
        value = self.get(name)
        if value is None:
            raise KeyError(name)
        return value

    def __contains__(self, name):
        return self.get(name) is not None

    def __setitem__(self, name, value):
        self._items = [(k, v) for k, v in self._items if k.lower() != name.lower()] + [(name, str(value))]

    def items(self):
        return list(self._items)


class Args(dict):
    """Query or form values: .get(name) gives the first value; .getlist(name) all of them."""
    def __init__(self, pairs):
        super().__init__()
        self._lists = {}
        for k, v in pairs:
            self._lists.setdefault(k, []).append(v)
            dict.setdefault(self, k, v)

    def get(self, key, default=None, type=None):
        if key not in self:
            return default
        value = self[key]
        if type is not None:
            try:
                return type(value)
            except (TypeError, ValueError):
                return default
        return value

    def getlist(self, key):
        return list(self._lists.get(key, []))


class Request:
    def __init__(self, environ):
        self.environ = environ
        self.method = environ["REQUEST_METHOD"].upper()
        self.path = environ.get("PATH_INFO", "/") or "/"
        self.args = Args(_pairs(environ.get("QUERY_STRING", "")))
        items = [(k[5:].replace("_", "-").title(), v) for k, v in environ.items() if k.startswith("HTTP_")]
        if environ.get("CONTENT_TYPE"):
            items.append(("Content-Type", environ["CONTENT_TYPE"]))
        self.headers = Headers(items)
        length = int(environ.get("CONTENT_LENGTH") or 0)
        self.data = environ["wsgi.input"].read(length) if length else b""
        self.content_type = environ.get("CONTENT_TYPE", "")
        self.form = Args(_pairs(self.data.decode("utf-8"))) if self.content_type.startswith("application/x-www-form-urlencoded") else Args([])
        self.view_args = {}
        cookie = environ.get("HTTP_COOKIE", "")
        self.cookies = dict(p.strip().split("=", 1) for p in cookie.split(";") if "=" in p)

    @property
    def is_json(self):
        return self.content_type.split(";")[0].strip() == "application/json"

    def get_json(self, silent=False, force=False):
        if not (self.is_json or force):
            if silent:
                return None
            abort(415, "expected a JSON body (Content-Type: application/json)")
        try:
            return json.loads(self.data or b"null")
        except ValueError:
            if silent:
                return None
            abort(400, "the body isn't valid JSON")

    @property
    def json(self):
        return self.get_json()

_CONVERTERS = {"int": (r"\d+", int), "float": (r"\d+(?:\.\d+)?", float), "path": (r".+", str), "string": (r"[^/]+", str)}


class _Rule:
    def __init__(self, rule, endpoint, view, methods):
        self.rule, self.endpoint, self.view = rule, endpoint, view
        self.methods = {m.upper() for m in methods} | ({"HEAD"} if "GET" in {m.upper() for m in methods} else set())
        regex, self.converters = "", {}
        for piece in re.split(r"(<[^>]+>)", rule):
            m = re.fullmatch(r"<(?:(\w+):)?(\w+)>", piece)
            if m:
                kind, name = m.group(1) or "string", m.group(2)
                pattern, convert = _CONVERTERS[kind]
                regex += f"(?P<{name}>{pattern})"
                self.converters[name] = convert
            else:
                regex += re.escape(piece)
        self.regex = re.compile(regex)

    def match(self, path):
        m = self.regex.fullmatch(path)
        if not m:
            return None
        return {k: self.converters[k](v) for k, v in m.groupdict().items()}

    def build(self, values):
        def fill(m):
            return str(values.pop(m.group(2)))
        path = re.sub(r"<(?:(\w+):)?(\w+)>", fill, self.rule)
        return path + (f"?{urlencode(values)}" if values else "")


class _Setup:
    def route(self, rule, methods=("GET",), endpoint=None):
        def decorate(view):
            self._add(rule, endpoint or view.__name__, view, methods)
            return view
        return decorate

    def get(self, rule, **kw):
        return self.route(rule, methods=["GET"], **kw)

    def post(self, rule, **kw):
        return self.route(rule, methods=["POST"], **kw)

    def put(self, rule, **kw):
        return self.route(rule, methods=["PUT"], **kw)

    def patch(self, rule, **kw):
        return self.route(rule, methods=["PATCH"], **kw)

    def delete(self, rule, **kw):
        return self.route(rule, methods=["DELETE"], **kw)


class Blueprint(_Setup):
    def __init__(self, name, import_name, url_prefix=""):
        self.name, self.url_prefix = name, url_prefix
        self.deferred = []

    def _add(self, rule, endpoint, view, methods):
        self.deferred.append((rule, endpoint, view, methods))


class _App(_Setup):
    def __init__(self, import_name):
        self.import_name = import_name
        self.config = {"TESTING": False}
        self.rules = []
        self.before_funcs, self.after_funcs, self.error_handlers = [], [], {}

    def _add(self, rule, endpoint, view, methods):
        self.rules.append(_Rule(rule, endpoint, view, methods))

    def register_blueprint(self, bp, url_prefix=None):
        prefix = url_prefix if url_prefix is not None else bp.url_prefix
        for rule, endpoint, view, methods in bp.deferred:
            self._add(prefix + rule, f"{bp.name}.{endpoint}", view, methods)

    def before_request(self, f):
        self.before_funcs.append(f)
        return f

    def after_request(self, f):
        self.after_funcs.append(f)
        return f

    def errorhandler(self, code):
        def decorate(f):
            self.error_handlers[code] = f
            return f
        return decorate

    def url_for(self, endpoint, **values):
        for r in self.rules:
            if r.endpoint == endpoint:
                return r.build(dict(values))
        raise LookupError(f"no endpoint {endpoint!r}")

    @staticmethod
    def _to_response(rv):
        status, headers = None, None
        if isinstance(rv, tuple):
            if len(rv) == 3:
                rv, status, headers = rv
            elif isinstance(rv[1], (dict, list)):
                rv, headers = rv
            else:
                rv, status = rv
        if isinstance(rv, Response):
            response = rv
        elif isinstance(rv, (dict, list)):
            response = jsonify(rv)
        elif isinstance(rv, (str, bytes)):
            response = Response(rv)
        elif rv is None:
            raise TypeError("the view function did not return a valid response")
        else:
            raise TypeError(f"can't turn {type(rv).__name__} into a response")
        if status is not None:
            response.status_code = status
        for k, v in (headers.items() if isinstance(headers, dict) else headers or []):
            response.headers[k] = v
        return response

    def _error(self, e):
        handler = self.error_handlers.get(e.code)
        if handler:
            return self._to_response(handler(e))
        body = f"<h1>{e.code} {e.name}</h1><p>{e.description}</p>"
        return Response(body, e.code)

    def _dispatch(self, req):
        path = req.path
        allowed = set()
        for r in self.rules:
            values = r.match(path)
            if values is None:
                continue
            if req.method in r.methods:
                req.view_args = values
                return r.view(**values)
            allowed |= r.methods
        if allowed:
            raise _MethodNotAllowed(sorted(allowed | {"OPTIONS"}))
        abort(404, "The requested URL was not found on the server.")

    def full_dispatch(self, environ):
        req = Request(environ)
        token = _current.set((req, _G(), self))
        try:
            try:
                rv = None
                for f in self.before_funcs:
                    rv = f()
                    if rv is not None:
                        break
                if rv is None:
                    rv = self._dispatch(req)
                response = self._to_response(rv)
            except _MethodNotAllowed as e:
                response = self._error(HTTPException(405))
                response.headers["Allow"] = ", ".join(e.allowed)
            except HTTPException as e:
                response = self._error(e)
            except Exception as e:
                if self.config.get("TESTING") and 500 not in self.error_handlers:
                    raise
                response = self._error(HTTPException(500)) if 500 not in self.error_handlers else self._to_response(self.error_handlers[500](e))
                response.status_code = 500
            for f in self.after_funcs:
                response = f(response) or response
            if req.method == "HEAD":
                response.headers["Content-Length"] = len(response.data)
                response.data = b""
            return response
        finally:
            _current.reset(token)

    def __call__(self, environ, start_response):
        response = self.full_dispatch(environ)
        if "Content-Length" not in response.headers:
            response.headers["Content-Length"] = len(response.data)
        start_response(response.status, response.headers.items())
        return [response.data]

class ValidationError(Exception):
    """Raised with a dict of field -> message."""

    def __init__(self, errors):
        super().__init__("; ".join(f"{k}: {v}" for k, v in sorted(errors.items())))

def validate_ticket(data, partial=False):
    """Check a ticket body from a client. partial=True for PATCH: only the fields present are checked."""
    if not isinstance(data, dict):
        raise ValidationError({"_": "expected a JSON object"})
    errors, clean = {}, {}
    unknown = sorted(set(data) - {"title", "body", "priority"})
    if unknown:
        errors["_"] = f"unexpected field(s): {', '.join(unknown)}"
    for field in ("title", "body"):
        if field not in data:
            if not partial:
                errors[field] = "required"
            continue
        value = data[field]
        limit = (3, 120) if field == "title" else (1, 5000)
        if not isinstance(value, str) or not limit[0] <= len(value.strip()) <= limit[1]:
            errors[field] = f"{limit[0]} to {limit[1]} characters"
        else:
            clean[field] = value.strip()
    if "priority" in data:
        if data["priority"] not in PRIORITIES:
            errors["priority"] = f"one of {', '.join(PRIORITIES)}"
        else:
            clean["priority"] = data["priority"]
    elif not partial:
        clean["priority"] = "normal"
    if partial and not data:
        errors["_"] = "nothing to change"
    if errors:
        raise ValidationError(errors)
    return clean


def can_transition(current, new, role):
    """True if someone with this role may move a ticket from `current` to `new`."""
    return role in TRANSITIONS.get(current, {}).get(new, set())


def allowed(role, action, is_owner=False):
    """True if the role may do the action (is_owner: the ticket is the user's own)."""
    always, own = PERMISSIONS[action]
    return role in always or (is_owner and role in own)


def problem(status, detail, title=None, **extra):
    """An RFC 9457 problem-details response: (body, status, headers)."""
    body = {"type": "about:blank", "title": title or HTTPStatus(status).phrase, "status": status, "detail": detail}
    body.update(extra)
    return body, status, {"Content-Type": "application/problem+json"}

def connect(path=":memory:"):
    conn = sqlite3.connect(path)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn


def migrate(conn, migrations=MIGRATIONS):
    """Apply the migrations the database hasn't had yet, each in its own transaction. Returns their versions."""
    conn.execute("CREATE TABLE IF NOT EXISTS schema_migrations (version INTEGER PRIMARY KEY, name TEXT NOT NULL)")
    conn.commit()
    versions = [v for v, _, _ in migrations]
    if versions != list(range(1, len(versions) + 1)):
        raise ValueError("migrations must be numbered 1, 2, 3... with no gaps")
    current = conn.execute("SELECT COALESCE(MAX(version), 0) FROM schema_migrations").fetchone()[0]
    if current > len(versions):
        raise RuntimeError(f"the database is at version {current}, newer than this code")
    applied = []
    for version, name, sql in migrations[current:]:
        try:
            conn.execute("BEGIN")
            for statement in sql.split(";"):
                if statement.strip():
                    conn.execute(statement)
            conn.execute("INSERT INTO schema_migrations (version, name) VALUES (?, ?)", (version, name))
            conn.commit()
        except Exception:
            conn.rollback()
            raise
        applied.append(version)
    return applied

import base64
import hashlib
import hmac
import secrets


def pbkdf2_sha256(password, salt, iterations, dklen=32):
    base = hmac.new(password, digestmod=hashlib.sha256)
    out, block = b"", 1
    while len(out) < dklen:
        h = base.copy()
        h.update(salt + block.to_bytes(4, "big"))
        u = h.digest()
        total = int.from_bytes(u, "big")
        for _ in range(iterations - 1):
            h = base.copy()
            h.update(u)
            u = h.digest()
            total ^= int.from_bytes(u, "big")
        out += total.to_bytes(32, "big")
        block += 1
    return out[:dklen]


def hash_password(password, iterations):
    salt = secrets.token_bytes(16)
    digest = pbkdf2_sha256(password.encode(), salt, iterations)
    return f"pbkdf2_sha256${iterations}${base64.b64encode(salt).decode()}${base64.b64encode(digest).decode()}"


def check_password(password, stored):
    try:
        algorithm, iterations, salt, digest = stored.split("$")
        expected = pbkdf2_sha256(password.encode(), base64.b64decode(salt), int(iterations))
        return algorithm == "pbkdf2_sha256" and hmac.compare_digest(expected, base64.b64decode(digest))
    except ValueError:
        return False


def issue_token(user_id, expires, key):
    """A signed bearer token: USER_ID.EXPIRES.SIGNATURE."""
    payload = f"{user_id}.{expires}"
    return f"{payload}.{hmac.new(key, payload.encode(), hashlib.sha256).hexdigest()}"


def verify_token(token, key, now):
    """The user id in a valid, unexpired token, else None."""
    parts = token.split(".")
    if len(parts) != 3 or not parts[0].isdigit() or not parts[1].isdigit():
        return None
    payload = f"{parts[0]}.{parts[1]}"
    expected = hmac.new(key, payload.encode(), hashlib.sha256).hexdigest()
    if not hmac.compare_digest(parts[2], expected) or now >= int(parts[1]):
        return None
    return int(parts[0])

## Step 17: tables as tests

## Step 18: testing the API

API tests go through HTTP, so they need an app to talk to, and every test should get a **fresh** one, so tests can't affect each other through leftover data. That's what pytest **fixtures** are for (Week 72): a function marked `@pytest.fixture` builds something, and any test that names it as a parameter receives a new one. Fixtures can use other fixtures: a `client` fixture can use a `clock` fixture, and a test that asks for both gets the **same** clock its client uses, so it can move time forward and watch a token expire.

## Step 19: a CI workflow

Tests only protect you if they **run**, on every change, before it's merged. That's continuous integration (Week 74): on GitHub, a workflow file in `.github/workflows/` runs the tests on every push and pull request, and a red cross on the pull request stops the merge. The helpdesk's workflow, as data (the YAML file holds exactly this structure):

```python
def workflow():
    """The helpdesk's CI workflow, as the data a .github/workflows/ci.yml file would hold."""
    return {
        "name": "CI",
        "on": {"push": {"branches": ["main"]}, "pull_request": {}},
        "permissions": {"contents": "read"},
        "jobs": {
            "test": {
                "runs-on": "ubuntu-latest",
                "timeout-minutes": 15,
                "strategy": {"matrix": {"python-version": ["3.12", "3.13"]}},
                "steps": [
                    {"uses": "actions/checkout@v4"},
                    {"uses": "actions/setup-python@v5", "with": {"python-version": "${{ matrix.python-version }}"}},
                    {"run": "pip install --require-hashes -r requirements.txt"},
                    {"run": "ruff check ."},
                    {"run": "pytest -q"},
                ],
            }
        },
    }
```

Each line carries a lesson from this phase: run on **pull requests** as well as pushes (the point is to catch problems before they're merged); **least privilege** for the workflow's token (`contents: read`, Week 79); a **timeout** so a hung test can't run for six hours; a **matrix** of Python versions, because users run more than one; actions **pinned** to a version, never a moving branch like `main` (a compromised action is a supply-chain attack, Week 79); dependencies installed from the **hashed** lock file; and the **linter** as well as the tests (Week 74).

Because the workflow is data, it can be checked by code, which is how teams keep dozens of repositories' workflows consistent.

def check_workflow(wf):
    """Problems with a CI workflow, as a sorted list of strings."""
    problems = []
    triggers = wf.get("on", {})
    if not ("push" in triggers and "pull_request" in triggers):
        problems.append("run on push and pull_request")
    if wf.get("permissions") != {"contents": "read"}:
        problems.append("set permissions to contents: read")
    job = wf.get("jobs", {}).get("test")
    if job is None:
        return sorted(problems + ["add a test job"])
    if not str(job.get("runs-on", "")).startswith("ubuntu"):
        problems.append("run on ubuntu")
    timeout = job.get("timeout-minutes")
    if not isinstance(timeout, int) or not 1 <= timeout <= 30:
        problems.append("set timeout-minutes to at most 30")
    versions = job.get("strategy", {}).get("matrix", {}).get("python-version", [])
    if not {"3.12", "3.13"} <= set(versions):

## Step 20: what the tests never reach

Line coverage (Week 72) tells you which lines ran. For an API there's a simpler, very useful question: which **routes** did the test suite never call, and which never once **succeeded**? A route whose tests only ever see `403` and `422` has had its refusals tested but never its main job. Because the route table is data (Monday), a small after-request hook can record every call, and Monday's `match_route` can tie each recorded path back to its route.

    """Make the app add (method, path, status) to `calls` for every response."""
    @app.after_request
    def remember(response):
        calls.append((request.method, request.path, response.status_code))
        return response
    return app

def untested(routes, calls):
    """Routes the calls never reached, or never saw succeed with the status the table promises."""
    seen = {}
    for method, path, status in calls:
        found, info = match_route(routes, method, path)
        if found == 200:

    problems = []

        elif code not in statuses:
