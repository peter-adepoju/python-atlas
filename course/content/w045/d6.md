from collections import deque

class TreeNode:
    def __init__(self, val, left=None, right=None):
        self.val, self.left, self.right = val, left, right

def invert(root):
    queue = deque([root] if root else [])
    while queue:
        node = queue.popleft()
        node.left, node.right = node.right, node.left
        for child in (node.left, node.right):
            if child:
                queue.append(child)
    return root

## Forge: an organisation chart

A company's staff list says who reports to whom, as text lines `name,manager` (the head of the company has an empty manager: `Ngozi,`). People ask the chart questions all day: who's in someone's team, who manages both of these people, how deep does the company go. Real data has mistakes, and the company is tall in places (a long chain of single reports), so **nothing may use recursion**: use explicit stacks and queues.

Build `OrgChart(lines)`:

- **Loading.** Each line is `name,manager` with spaces around each part ignored; blank lines are skipped. Raise `ValueError` for: a name listed twice, not exactly one head, a manager who isn't on the list, or a **cycle** (someone who, following managers upwards, never reaches the head). Keep each person's direct reports in the order their lines appear.
- `manager(name)` and `reports(name)` (direct reports, as a list); `chain(name)`: the names from `name` up to the head; unknown names raise `KeyError`.
- `team_size(name)`: everyone below `name` (not counting them).
- `common_manager(a, b)`: the lowest person who is `a` or above `a`, and also `b` or above `b`.
- `levels()`: a list of lists of names by depth (level order, reports in order).
- `outline()`: lines of the chart, depth-first (each person then their teams, in order), indented two spaces per level.
- `move(name, new_manager)`: reassign `name` (with their whole team) to report to `new_manager`, at the end of the new manager's reports. Moving the head, or moving someone under themselves or anyone in their own team, raises `ValueError` and changes nothing.

Store `self._manager` (name to manager) and `self._reports` (name to a list). To check for cycles, walk up from each person until you reach someone already known to reach the head; meeting someone twice on the same walk means a loop. Remembering who reaches the head keeps this fast on a tall chart. `team_size`, `outline` and `levels` are a stack walk, a stack walk (push reports in reverse), and a queue walk. For `move`, refuse if `new_manager` is `name` or appears in `name`'s team (walk up from `new_manager`: if you pass `name`, refuse).
@@solution
from collections import deque

class OrgChart:
    def __init__(self, lines):
        self._manager = {}
        self._reports = {}
        order = []
        for line in lines:
            if not line.strip():
                continue
            name, _, boss = line.partition(",")
            name, boss = name.strip(), boss.strip() or None
            if name in self._manager:
                raise ValueError(f"{name} is listed twice")
            self._manager[name] = boss
            order.append(name)
        heads = [n for n, m in self._manager.items() if m is None]
        if len(heads) != 1:
            raise ValueError(f"expected one head, found {len(heads)}")
        self.head = heads[0]
        for name in order:
            self._reports[name] = []
        for name in order:
            boss = self._manager[name]
            if boss is None:
                continue
            if boss not in self._manager:
                raise ValueError(f"{name}'s manager {boss} isn't on the list")
            self._reports[boss].append(name)
        reaches_head = {self.head}
        for start in order:
            walked, name = set(), start
            while name not in reaches_head:
                if name in walked:
                    raise ValueError("the reporting lines form a loop")
                walked.add(name)
                name = self._manager[name]
                if name is None:
                    break
            reaches_head.update(walked)

    def manager(self, name):
        return self._manager[name]

    def reports(self, name):
        return list(self._reports[name])

    def chain(self, name):
        if name not in self._manager:
            raise KeyError(name)
        out = []
        while name is not None:
            out.append(name)
            name = self._manager[name]
        return out

    def team_size(self, name):
        count, stack = 0, list(self._reports[name])
        while stack:
            person = stack.pop()
            count += 1
            stack.extend(self._reports[person])
        return count

    def common_manager(self, a, b):
        above_a = set(self.chain(a))
        for name in self.chain(b):
            if name in above_a:
                return name

    def levels(self):
        out, queue = [], deque([self.head])
        while queue:
            level = []
            for _ in range(len(queue)):
                person = queue.popleft()
                level.append(person)
                queue.extend(self._reports[person])
            out.append(level)
        return out

    def outline(self):
        lines, stack = [], [(self.head, 0)]
        while stack:
            person, depth = stack.pop()
            lines.append("  " * depth + person)
            for r in reversed(self._reports[person]):
                stack.append((r, depth + 1))
        return lines

    def move(self, name, new_manager):
        if name == self.head:
            raise ValueError("the head can't be moved")
        if new_manager not in self._manager or name not in self._manager:
            raise KeyError(name if name not in self._manager else new_manager)
        if name in self.chain(new_manager):
            raise ValueError(f"{new_manager} is in {name}'s own team")
        self._reports[self._manager[name]].remove(name)
        self._reports[new_manager].append(name)
        self._manager[name] = new_manager

:::think title="Trusting the data"
Most of the work in `OrgChart` isn't the questions; it's making sure the data really is a tree before answering them. A loop in the reporting lines would make `chain` run for ever and `team_size` count for ever; two heads would make "the company" two companies. Those are Week 37's lessons applied to a data structure: check at the boundary, refuse clearly, and keep the structure's rules true afterwards (which is why `move` checks for loops too). And because real organisations are sometimes very deep, every walk here uses its own stack or queue: a tree algorithm that works on your test data but crashes on a real chart 1,500 people deep isn't finished.
:::

## Forge bonus: Huffman coding

:::exercise id=forge2 points=30 title="Forge bonus: Huffman coding"
**Huffman coding** (David Huffman, 1952) compresses text by giving common characters short codes of bits and rare ones long codes, with no code the start of another, so the bits can be read back without separators. The codes come from a binary tree:

1. Count each character. Make a leaf for each, weighted by its count.
2. Repeatedly take the **two lightest** trees, join them under a new node whose weight is their sum (the lighter on the left), until one tree is left. Use `heapq` (Week 19) with entries `(weight, order, node)`, where `order` is a counter that breaks ties: a new node gets the next number, and leaves get their numbers in order of the characters sorted alphabetically.
3. A character's code is its path from the root: `0` for left, `1` for right. (A text with only one distinct character gives it the code `"0"`.)

Build the heap from `(count, i, Node(count, ch))` for the characters in sorted order. While more than one entry is left, pop two, push `(w1 + w2, next_order, Node(w1 + w2, left=first, right=second))`. To make codes, walk the tree with a stack of `(node, code_so_far)`. To decode, start at the root and follow `0` left and `1` right; at a leaf, write its character and go back to the root (if the root itself is a leaf, every bit is that character).
@@solution
import heapq

class Node:
    def __init__(self, weight, char=None, left=None, right=None):
        self.weight, self.char, self.left, self.right = weight, char, left, right

def huffman_tree(text):
    counts = {}
    for ch in text:
        counts[ch] = counts.get(ch, 0) + 1
    if not counts:
        return None
    heap = [(n, i, Node(n, ch)) for i, (ch, n) in enumerate(sorted(counts.items()))]
    heapq.heapify(heap)
    order = len(heap)
    while len(heap) > 1:
        w1, _, a = heapq.heappop(heap)
        w2, _, b = heapq.heappop(heap)
        heapq.heappush(heap, (w1 + w2, order, Node(w1 + w2, left=a, right=b)))
        order += 1
    return heap[0][2]

def huffman_codes(text):
    root = huffman_tree(text)
    if root is None:
        return {}
    if root.char is not None:
        return {root.char: "0"}
    codes, stack = {}, [(root, "")]
    while stack:
        node, code = stack.pop()
        if node.char is not None:
            codes[node.char] = code
            continue
        stack.append((node.right, code + "1"))
        stack.append((node.left, code + "0"))
    return codes

def encode(text, codes):
    return "".join(codes[ch] for ch in text)

def decode(bits, root):
    if root.char is not None:
        return root.char * len(bits)
    out, node = [], root
    for bit in bits:
        node = node.left if bit == "0" else node.right
        if node.char is not None:
            out.append(node.char)
            node = root
    return "".join(out)

def saving(text):
    return 8 * len(text) - len(encode(text, huffman_codes(text)))
