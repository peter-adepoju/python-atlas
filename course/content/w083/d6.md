import numpy as np

def moving_average(values, k):
    if k < 1:
        raise ValueError("k must be positive")
    a = np.asarray(values, dtype=float)
    if k > len(a):
        return []
    c = np.concatenate([[0.0], np.cumsum(a)])
    return ((c[k:] - c[:-k]) / k).round(2).tolist()

def top_per_group(df, group, value, n):
    top = df.sort_values(value, ascending=False).groupby(group).head(n)
    return {g: part[value].tolist() for g, part in sorted(top.groupby(group), key=lambda kv: kv[0])}

    return (df[value] / df.groupby(group)[value].transform("sum")).round(3).tolist()

## Forge: a bike-hire analysis

The scheme has six docking stations (in `stations.csv`, with the area each is in and its number of docks) and a week of trips (in `trips.csv`: an id, start and end times, the two stations, whether the rider is a `member` or `casual`, and whether the bike was `classic` or `electric`). Fares are:

| Rider | Classic | Electric |
| --- | --- | --- |
| casual | £1.50 to unlock, then £0.10 a minute | £1.50 to unlock, then £0.20 a minute |
| member | first 30 minutes free, then £0.05 a minute | first 30 minutes free, then £0.10 a minute |

Write these functions:

- `load_trips(path)`: read the trips and parse `start` and `end` (format `"%Y-%m-%d %H:%M"`); add `minutes` (whole minutes from start to end, as integers; negative if the end is before the start), `hour` (of the start) and `day` (the start's day name).
- `split_valid(trips, max_minutes=180)`: return `(valid, rejected)`: `valid` is the trips that end after or when they start and last at most `max_minutes` (index renumbered); `rejected` is a sorted list of `(trip_id, reason)` with reasons `"ends before it starts"` and `"too long"`.
- `fare(rider, bike, minutes)`: one trip's fare, rounded to 2 places.
- `add_fares(trips)`: a new DataFrame with a `fare` column, computed **without a loop** (`np.select` or `np.where`), equal to `fare` for every row.
- `summary(trips, stations)` for valid trips: a dict with `trips` (how many), `busiest_hour` (the start hour with most trips; ties go to the earliest), `median_minutes` (a float), `by_rider` (a dict from rider type to `{"trips", "avg_minutes", "revenue"}`, the last two rounded to 2 places), `top_routes` (the three most common `((start, end), count)` pairs, ties by station names), `revenue_by_area` (by the **start** station's area, 2 places) and `net_flow` (for **every** station in `stations`, sorted by name: arrivals minus departures, as integers).
- `charts(trips)`: a figure with two axes side by side: on the left, a bar chart of the number of trips starting in each hour from 0 to 23 (24 bars, including empty hours), titled `"Trips by hour"`; on the right, a histogram of `minutes` with bins `[0, 10, 20, 30, 45, 60, 90, 180]`, titled `"Trip length"`. Label both axes of both charts.

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

def load_trips(path):
    df = pd.read_csv(path)
    df["start"] = pd.to_datetime(df["start"], format="%Y-%m-%d %H:%M")
    df["end"] = pd.to_datetime(df["end"], format="%Y-%m-%d %H:%M")
    df["minutes"] = ((df["end"] - df["start"]).dt.total_seconds() // 60).astype(int)
    df["hour"] = df["start"].dt.hour
    df["day"] = df["start"].dt.day_name()
    return df

def split_valid(trips, max_minutes=180):
    backwards = trips["minutes"] < 0
    too_long = trips["minutes"] > max_minutes
    rejected = [(t, "ends before it starts") for t in trips.loc[backwards, "trip_id"]]
    rejected += [(t, "too long") for t in trips.loc[too_long, "trip_id"]]
    return trips[~(backwards | too_long)].reset_index(drop=True), sorted(rejected)

RATES = {("casual", "classic"): 0.10, ("casual", "electric"): 0.20,
         ("member", "classic"): 0.05, ("member", "electric"): 0.10}

def fare(rider, bike, minutes):
    rate = RATES[(rider, bike)]
    if rider == "casual":
        return round(1.50 + minutes * rate, 2)
    return round(max(0, minutes - 30) * rate, 2)

def add_fares(trips):
    rate = np.select([(trips["rider"] == r) & (trips["bike"] == b) for r, b in RATES], list(RATES.values()))
    casual = trips["rider"] == "casual"
    charged = np.where(casual, trips["minutes"], np.maximum(trips["minutes"] - 30, 0))
    return trips.assign(fare=(np.where(casual, 1.50, 0) + charged * rate).round(2))

def summary(trips, stations):
    t = add_fares(trips)
    by_rider = (t.groupby("rider")
                  .agg(trips=("trip_id", "count"), avg_minutes=("minutes", "mean"), revenue=("fare", "sum"))
                  .round(2))
    routes = t.groupby(["start_station", "end_station"]).size().reset_index(name="n")
    routes = routes.sort_values(["n", "start_station", "end_station"], ascending=[False, True, True]).head(3)
    area = t.merge(stations, left_on="start_station", right_on="station", how="left", validate="many_to_one")
    flow = (t["end_station"].value_counts()
              .sub(t["start_station"].value_counts(), fill_value=0)
              .reindex(stations["station"], fill_value=0)
              .astype(int))
    return {
        "trips": len(t),
        "busiest_hour": int(t["hour"].value_counts().sort_index().idxmax()),
        "median_minutes": float(t["minutes"].median()),
        "by_rider": by_rider.to_dict("index"),
        "top_routes": [((a, b), int(n)) for a, b, n in routes.itertuples(index=False)],
        "revenue_by_area": area.groupby("area")["fare"].sum().round(2).to_dict(),
        "net_flow": dict(sorted(flow.to_dict().items())),
    }

def charts(trips):
    fig, (left, right) = plt.subplots(1, 2, figsize=(10, 4))
    counts = trips["hour"].value_counts().reindex(range(24), fill_value=0)
    left.bar(counts.index, counts.values)
    left.set_title("Trips by hour")
    left.set_xlabel("Hour of day")
    left.set_ylabel("Trips")
    right.hist(trips["minutes"], bins=[0, 10, 20, 30, 45, 60, 90, 180])
    right.set_title("Trip length")
    right.set_xlabel("Minutes")
    right.set_ylabel("Trips")
    fig.tight_layout()
    return fig

## Forge bonus: docks and vans

A station can only lend a bike it has, and only take one back into a free dock. Running the week's trips in time order against real dock counts shows where the scheme fails, and a simple greedy plan says how to fix it overnight.

:::exercise id=forge2 points=30 title="Forge bonus: docks and vans"

    df["start"] = pd.to_datetime(df["start"])
    df["end"] = pd.to_datetime(df["end"])

    return dict(zip(s["station"], s["docks"]))

def test_simulate():

    bikes = {"Castle": 6, "Market": 7, "Riverside": 5, "Station": 10, "University": 9, "Park": 4}
    before = dict(bikes)
    r = simulate(_trips(), bikes, _docks())

    r = simulate(_trips(), tight, _docks())
    assert r["no_bike"] == ["T019", "T028"] and r["full_dock"] == ["T018", "T015", "T021", "T022", "T023", "T029"]
    assert r["bikes"]["University"] == 18 and sum(r["bikes"].values()) == sum(tight.values())

    "greedy van runs"
    end = {"Castle": 0, "Market": 8, "Riverside": 5, "Station": 8, "University": 17, "Park": 3}
    assert rebalance(end, _docks(), 5) == [("University", "Castle", 5), ("University", "Station", 2),
                                            ("Market", "Castle", 1), ("University", "Park", 1)]
    assert rebalance({"A": 5, "B": 5}, {"A": 10, "B": 10}, 3) == []
    assert rebalance({"A": 10, "B": 0}, {"A": 10, "B": 10}, 3) == [("A", "B", 3), ("A", "B", 2)]
@@solution
import pandas as pd

def simulate(trips, bikes, docks):
    bikes = dict(bikes)
    events = []
    for t in trips.itertuples(index=False):
        events.append((t.start, 1, t.trip_id, t.start_station, "out"))
        events.append((t.end, 0, t.trip_id, t.end_station, "in"))
    events.sort()
    cancelled, no_bike, full_dock = set(), [], []
    for when, order, trip, station, kind in events:
        if kind == "out":
            if bikes[station] == 0:
                no_bike.append(trip)
                cancelled.add(trip)
            else:
                bikes[station] -= 1
        elif trip not in cancelled:
            if bikes[station] >= docks[station]:
                full_dock.append(trip)
                station = min(docks, key=lambda s: (-(docks[s] - bikes[s]), s))
            bikes[station] += 1
    return {"bikes": bikes, "no_bike": no_bike, "full_dock": full_dock}

def rebalance(bikes, docks, capacity, share=0.5):
    target = {s: int(docks[s] * share) for s in docks}
    surplus = {s: bikes[s] - target[s] for s in docks if bikes[s] > target[s]}
    deficit = {s: target[s] - bikes[s] for s in docks if bikes[s] < target[s]}
    runs = []
    while surplus and deficit:
        src = min(surplus, key=lambda s: (-surplus[s], s))
        dst = min(deficit, key=lambda s: (-deficit[s], s))
        n = min(surplus[src], deficit[dst], capacity)
        runs.append((src, dst, n))
        for side, station in ((surplus, src), (deficit, dst)):
            side[station] -= n
            if side[station] == 0:
                del side[station]
    return runs
