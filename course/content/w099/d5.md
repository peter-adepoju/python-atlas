@@hint
First pass: each position gets the product of everything to its left. Second pass, from the right: multiply in the product of everything to its right.
@@solution
def products_except_self(nums):
    out, left = [], 1
    for x in nums:
        out.append(left)
        left *= x
    right = 1
    for i in range(len(nums) - 1, -1, -1):
        out[i] *= right
        right *= nums[i]
    return out

@@starter

    ___
@@tests
import time

def test_hops():
    "greedy, level by level"
    assert fewest_hops([2, 3, 1, 1, 4]) == 2 and fewest_hops([2, 3, 0, 1, 4]) == 2
    assert fewest_hops([3, 2, 1, 0, 4]) == -1 and fewest_hops([0]) == 0 and fewest_hops([1, 1, 1]) == 2

def test_fast():
    "100,000 junctions"
    start = time.perf_counter()
    assert fewest_hops([1] * 100_000) == 99_999 and time.perf_counter() - start < 1.5
@@hint
Treat it like breadth-first search over ranges: keep the end of the current hop's range and the furthest reachable from inside it; when you pass the end, take a hop.
@@solution
def fewest_hops(reach):
    hops, end, furthest = 0, 0, 0
    for i in range(len(reach) - 1):
        if i > furthest:
            return -1
        furthest = max(furthest, i + reach[i])
        if i == end:
            hops += 1
            end = furthest
            if end <= i:
                return -1
    return hops if furthest >= len(reach) - 1 or len(reach) == 1 else -1
:::

:::exercise id=d2 points=10 title="Debugging: a name that wasn't equal"
`is_member(name, members)` uses `is` to compare strings, which compares identity: it happens to work for some strings and not others. Fix it to compare values, ignoring case and surrounding spaces.
@@starter
def is_member(name, members):
    return any(name is m for m in members)
@@tests
def test_member():
    "== compares values"
    members = ["Ada Walker", "Bo Hill"]
    built = " ".join(["Ada", "Walker"])
    assert is_member(built, members) and is_member("  bo hill ", members) and not is_member("Cy", members)
@@solution
def is_member(name, members):
    wanted = name.strip().lower()
    return any(wanted == m.strip().lower() for m in members)

@@tests
def test_iterable():
    "an iterable makes fresh iterators"
    r = Route(["Mill", "Weir", "Ridge"])
    assert list(r) == ["Mill", "Weir", "Ridge"] and list(r) == ["Mill", "Weir", "Ridge"] and len(r) == 3
    it = iter(r)
    assert isinstance(it, RouteIterator) and iter(it) is it and next(it) == "Mill"
    assert [a + "-" + b for a in r for b in r][:2] == ["Mill-Mill", "Mill-Weir"]
    rest = list(it)
    assert rest == ["Weir", "Ridge"]
    try:
        next(it)
    except StopIteration:
        pass
    else:
        raise AssertionError("exhausted")
@@solution
class RouteIterator:
    def __init__(self, stops):
        self.stops, self.i = stops, 0

    def __iter__(self):
        return self

    def __next__(self):
        if self.i >= len(self.stops):
            raise StopIteration
        stop = self.stops[self.i]
        self.i += 1
        return stop

class Route:
    def __init__(self, stops):
        self.stops = list(stops)

    def __iter__(self):
        return RouteIterator(self.stops)

    def __len__(self):
        return len(self.stops)

@@starter

    ___
@@tests
import random, time

def test_rectangles():
    "a stack of rising bars"
    assert largest_rectangle([2, 1, 5, 6, 2, 3]) == 10 and largest_rectangle([2, 4]) == 4
    assert largest_rectangle([]) == 0 and largest_rectangle([3]) == 3 and largest_rectangle([1, 1, 1, 1]) == 4

def test_against_slow():
    "matches a simple O(n squared) version"
    rng = random.Random(8)
    for _ in range(50):
        h = [rng.randint(0, 9) for _ in range(rng.randint(0, 30))]
        slow = max([min(h[i:j]) * (j - i) for i in range(len(h)) for j in range(i + 1, len(h) + 1)], default=0)
        assert largest_rectangle(h) == slow, h

def test_fast():
    "200,000 bars"
    start = time.perf_counter()
    assert largest_rectangle(list(range(200_000))) == 10_000_000_000 and time.perf_counter() - start < 2
@@hint
Keep a stack of indices with rising heights. When a lower bar arrives, pop taller ones: each popped bar's rectangle stretches from just after the new top of the stack to just before the current index. Add a 0-height bar at the end to empty the stack.
@@solution
def largest_rectangle(heights):
    stack, best = [], 0
    for i, h in enumerate(list(heights) + [0]):
        while stack and heights[stack[-1]] > h:
            height = heights[stack.pop()]
            left = stack[-1] + 1 if stack else 0
            best = max(best, height * (i - left))
        stack.append(i)
    return best
:::
