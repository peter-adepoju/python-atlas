@@solution
def longest_zero_sum(nums):
    first, prefix, best = {0: -1}, 0, 0
    for i, x in enumerate(nums):
        prefix += x
        if prefix in first:
            best = max(best, i - first[prefix])
        else:
            first[prefix] = i
    return best
:::

def hashable_kinds(values):
    kinds = []
    for v in values:
        try:
            hash(v)
            kinds.append("hashable")
        except TypeError:
            kinds.append("unhashable")
    return kinds

## Practising what's weak

It's natural to practise what you're good at, because it feels like progress. Interview practice works the other way: find the topics where you solve least, take longest and need most hints, and drill those, while keeping the strong ones warm with an occasional problem. And **space** the practice: coming back to a topic after a day or two (interleaved with others) makes it stick better than doing ten of the same in a row, which is one of the best-supported findings about learning.

:::exercise id=forge points=50 title="The practice plan"
Write `practice_plan(log, loops)`. Return a list of lines:

1. one line per topic, in order of first appearance: `"TOPIC: S/N solved, median M min, H hints"` (M the median minutes in the `g` format, H the total hints), followed by `" WEAK"` if fewer than 70% were solved, the median is over 30 minutes, or the hints average more than 1;
2. `"drill: A, B, C"`: up to three weak topics, worst first: lowest solve rate, then highest median minutes, then name (or `"drill: none"`);
3. `"keep warm: ..."`: the topics that aren't weak, in order of first appearance (or `none`);
4. one line per round in the **last** loop, in its order: `"ROUND: FIRST to LAST"` with the mean ratings (2 decimal places) in the first loop that had the round and in the last loop, or `"ROUND: new, LAST"` if only the last loop had it;
5. `"next loop focus: ROUND (CRITERION)"`: the round with the lowest mean in the last loop (first on ties), and its lowest-rated criterion (first on ties).

def test_plan():
    "from evidence to a plan"

    assert practice_plan(log, loops) == ["arrays: 1/1 solved, median 10 min, 0 hints", "drill: none", "keep warm: arrays",
                                         "coding: 4.00 to 4.00", "next loop focus: coding (solving)"]
@@solution
def _median(values):
    values = sorted(values)
    n, mid = len(values), len(values) // 2
    return values[mid] if n % 2 else (values[mid - 1] + values[mid]) / 2

def _mean(scores):
    return sum(scores.values()) / len(scores)

def practice_plan(log, loops):
    topics = {}
    for p in log:
        topics.setdefault(p["topic"], []).append(p)
    lines, weak, strong = [], [], []
    for topic, items in topics.items():
        solved = sum(p["solved"] for p in items)
        median = _median([p["minutes"] for p in items])
        hints = sum(p["hints"] for p in items)
        rate = solved / len(items)
        is_weak = rate < 0.7 or median > 30 or hints / len(items) > 1
        lines.append(f"{topic}: {solved}/{len(items)} solved, median {median:g} min, {hints} hints" + (" WEAK" if is_weak else ""))
        (weak if is_weak else strong).append((rate, -median, topic))
    worst = sorted(weak, key=lambda w: (w[0], w[1], w[2]))[:3]
    lines.append("drill: " + (", ".join(t for _, _, t in worst) or "none"))
    lines.append("keep warm: " + (", ".join(t for _, _, t in strong) or "none"))
    last = loops[-1]
    for name, scores in last.items():
        first = next((loop[name] for loop in loops[:-1] if name in loop), None)
        if first is None and len(loops) > 1:
            lines.append(f"{name}: new, {_mean(scores):.2f}")
        else:
            lines.append(f"{name}: {_mean(first or scores):.2f} to {_mean(scores):.2f}")
    focus = min(last, key=lambda n: _mean(last[n]))
    criterion = min(last[focus], key=last[focus].get)
    lines.append(f"next loop focus: {focus} ({criterion})")
    return lines
:::

## Bonus: spacing the drills

Return lines `"Day N: A, B, C"`.
@@starter

    ___
@@tests
def test_schedule():
    "weak topics come back after a gap"
    assert drill_schedule(["dp", "graphs"], ["arrays", "trees"], 3, 3) == [
        "Day 1: dp, graphs, arrays", "Day 2: dp, graphs, trees", "Day 3: dp, graphs, arrays"]
    assert drill_schedule(["dp", "graphs", "design"], ["heaps"], 2, 3) == ["Day 1: dp, graphs, heaps", "Day 2: design, dp, heaps"]
    assert drill_schedule(["dp"], [], 2, 2) == ["Day 1: dp, dp", "Day 2: dp, dp"]
@@solution
from itertools import cycle

def drill_schedule(drills, keep, days, per_day):
    drill_cycle, keep_cycle = cycle(drills), cycle(keep) if keep else None
    lines = []
    for day in range(1, days + 1):
        slots = [next(drill_cycle) for _ in range(per_day - 1)]
        slots.append(next(keep_cycle) if keep_cycle else next(drill_cycle))
        lines.append(f"Day {day}: " + ", ".join(slots))
    return lines
:::
