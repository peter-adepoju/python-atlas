from dataclasses import dataclass, replace
from enum import Enum

class Theme(Enum):
    LIGHT = "light"
    DARK = "dark"
    HIGH_CONTRAST = "high contrast"

@dataclass(frozen=True)
class Settings:
    theme: Theme = Theme.LIGHT
    font_size: int = 16

    def __post_init__(self):
        if not isinstance(self.font_size, int) or not 8 <= self.font_size <= 32:
            raise ValueError(f"font size must be 8 to 32, not {self.font_size!r}")

    def with_theme(self, name):
        try:
            theme = Theme[name.upper()]
        except KeyError:
            raise ValueError(f"no theme called {name!r}")
        return replace(self, theme=theme)

    def bigger(self, step=2):
        return replace(self, font_size=min(32, self.font_size + step))

import math
from typing import NamedTuple

class LatLon(NamedTuple):
    lat: float
    lon: float

    @classmethod
    def parse(cls, text):
        parts = text.split(",")
        if len(parts) != 2:
            raise ValueError(f"expected 'lat, lon', got {text!r}")
        lat, lon = float(parts[0]), float(parts[1])
        if not (-90 <= lat <= 90 and -180 <= lon <= 180):
            raise ValueError(f"out of range: {text!r}")
        return cls(lat, lon)

    def distance_km(self, other):
        p1, p2 = math.radians(self.lat), math.radians(other.lat)
        dp = p2 - p1
        dl = math.radians(other.lon - self.lon)
        a = math.sin(dp / 2) ** 2 + math.cos(p1) * math.cos(p2) * math.sin(dl / 2) ** 2
        return round(2 * 6371 * math.asin(math.sqrt(a)), 1)

## Forge: a parcel tracker

A delivery company scans each parcel as it moves: collected, at a depot, on the van, delivered. Customers want to know where their parcel is, the company wants a bill for each customer, and several systems (text messages, a website, a log) want to hear about every scan. Here's the specification. Each piece uses the tool that fits it.

**`Status`, an Enum** with members `CREATED`, `IN_TRANSIT`, `OUT_FOR_DELIVERY`, `DELIVERED` and `RETURNED`, whose values are `"created"`, `"in transit"`, `"out for delivery"`, `"delivered"` and `"returned"`.

- `can_move_to(other)` says whether a parcel may move from this status to `other`: from `CREATED` only to `IN_TRANSIT`; from `IN_TRANSIT` to `IN_TRANSIT` (another depot), `OUT_FOR_DELIVERY` or `RETURNED`; from `OUT_FOR_DELIVERY` to `DELIVERED`, `IN_TRANSIT` (back to the depot) or `RETURNED`; from `DELIVERED` and `RETURNED`, nowhere.
- A property `finished`: True for `DELIVERED` and `RETURNED`.

**`Service`, an Enum**: `STANDARD = "standard"`, `EXPRESS = "express"`.

**`Scan`, a NamedTuple** with fields `when` (text like `"2026-11-03 09:00"`), `place` and `status`.

**`Parcel`, a dataclass** with `code`, `grams`, `service` (default `Service.STANDARD`) and `scans` (a new empty list for each parcel).

- In `__post_init__`: `code` must be two capital letters and six digits (like `"AB123456"`), and `grams` an `int` above 0; otherwise `ValueError`.
- A property `status`: `Status.CREATED` if there are no scans, otherwise the last scan's status.
- `add_scan(when, place, status)`: `status` may be a `Status` or its value (`"in transit"`); `when` must have the form `YYYY-MM-DD HH:MM` (check it with `datetime.strptime(when, "%Y-%m-%d %H:%M")`) and be **later** than the last scan's; the move must be allowed. Any problem raises `ValueError` and adds nothing. Otherwise append and return the new `Scan`.

**`Pricer`, an ABC**, for a family of pricing schemes the company designs itself.

- An abstract method `price(parcel)`, returning pence.
- An ordinary method `quote(parcels)`: the total price of all of them.
- `FlatPricer(standard, express)`: a fixed price for each service.
- `WeightPricer(per_kg, express_extra)`: `per_kg` for each kilogram or part of one (`math.ceil(grams / 1000)`), plus `express_extra` for express parcels.

**`Notifier`, a runtime-checkable Protocol**, for anything that wants to hear about scans, whoever wrote it: a method `notify(code: str, message: str)`.

**`Tracker(pricer, notifiers=())`**, which holds everything together.

- `pricer` must be a `Pricer` and every notifier must match `Notifier`; otherwise `TypeError`. The tracker keeps its own list of notifiers.
- `register(parcel)` adds a parcel (a second parcel with the same code raises `ValueError`) and returns it.
- `scan(code, when, place, status)`: an unknown code raises `KeyError`. Otherwise it adds the scan to the parcel, then calls `notify(code, message)` on every notifier with a message like `AB123456: out for delivery at Ikeja`, and returns the scan.
- `by_status()`: a dictionary from each status that at least one parcel has to a sorted list of those parcels' codes.
- `bill()`: the pricer's quote for all the registered parcels.

import math
import re
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import NamedTuple, Protocol, runtime_checkable

class Status(Enum):
    CREATED = "created"
    IN_TRANSIT = "in transit"
    OUT_FOR_DELIVERY = "out for delivery"
    DELIVERED = "delivered"
    RETURNED = "returned"

    def can_move_to(self, other):
        return other in MOVES[self]

    @property
    def finished(self):
        return self in (Status.DELIVERED, Status.RETURNED)

MOVES = {
    Status.CREATED: {Status.IN_TRANSIT},
    Status.IN_TRANSIT: {Status.IN_TRANSIT, Status.OUT_FOR_DELIVERY, Status.RETURNED},
    Status.OUT_FOR_DELIVERY: {Status.DELIVERED, Status.IN_TRANSIT, Status.RETURNED},
    Status.DELIVERED: set(),
    Status.RETURNED: set(),
}

class Service(Enum):
    STANDARD = "standard"
    EXPRESS = "express"

class Scan(NamedTuple):
    when: str
    place: str
    status: Status

@dataclass
class Parcel:
    code: str
    grams: int
    service: Service = Service.STANDARD
    scans: list = field(default_factory=list)

    def __post_init__(self):
        if not re.fullmatch(r"[A-Z]{2}\d{6}", self.code):
            raise ValueError(f"bad code: {self.code!r}")
        if not isinstance(self.grams, int) or isinstance(self.grams, bool) or self.grams <= 0:
            raise ValueError(f"bad weight: {self.grams!r}")

    @property
    def status(self):
        return self.scans[-1].status if self.scans else Status.CREATED

    def add_scan(self, when, place, status):
        status = Status(status)
        datetime.strptime(when, "%Y-%m-%d %H:%M")
        if self.scans and when <= self.scans[-1].when:
            raise ValueError(f"{when} isn't after the last scan")
        if not self.status.can_move_to(status):
            raise ValueError(f"can't go from {self.status.value} to {status.value}")
        scan = Scan(when, place, status)
        self.scans.append(scan)
        return scan

class Pricer(ABC):
    @abstractmethod
    def price(self, parcel):
        """The price of one parcel, in pence."""

    def quote(self, parcels):
        return sum(self.price(p) for p in parcels)

class FlatPricer(Pricer):
    def __init__(self, standard, express):
        self.standard = standard
        self.express = express

    def price(self, parcel):
        return self.express if parcel.service is Service.EXPRESS else self.standard

class WeightPricer(Pricer):
    def __init__(self, per_kg, express_extra):
        self.per_kg = per_kg
        self.express_extra = express_extra

    def price(self, parcel):
        extra = self.express_extra if parcel.service is Service.EXPRESS else 0
        return self.per_kg * math.ceil(parcel.grams / 1000) + extra

@runtime_checkable
class Notifier(Protocol):
    def notify(self, code: str, message: str): ...

class Tracker:
    def __init__(self, pricer, notifiers=()):
        if not isinstance(pricer, Pricer):
            raise TypeError(f"not a Pricer: {pricer!r}")
        for n in notifiers:
            if not isinstance(n, Notifier):
                raise TypeError(f"not a Notifier: {n!r}")
        self.pricer = pricer
        self.notifiers = list(notifiers)
        self._parcels = {}

    def register(self, parcel):
        if parcel.code in self._parcels:
            raise ValueError(f"already tracking {parcel.code}")
        self._parcels[parcel.code] = parcel
        return parcel

    def scan(self, code, when, place, status):
        parcel = self._parcels[code]
        scan = parcel.add_scan(when, place, status)
        for n in self.notifiers:
            n.notify(code, f"{code}: {scan.status.value} at {place}")
        return scan

    def by_status(self):
        groups = {}
        for p in self._parcels.values():
            groups.setdefault(p.status, []).append(p.code)
        return {status: sorted(codes) for status, codes in groups.items()}

    def bill(self):
        return self.pricer.quote(self._parcels.values())

:::think title="One tool per job"
Look at which tool did which job, and why. The **Enums** made the statuses and services impossible to misspell, and put the rules for moving between statuses next to them. The **named tuple** was the right weight for a scan: a small immutable record that nobody needs to change. The **dataclass** wrote the parcel's routine methods and gave `__post_init__` one place for its checks. The **ABC** suited the pricers, a family the company designs itself, where forgetting `price` should fail at once. The **Protocol** suited the notifiers, which other teams write, and which shouldn't have to inherit anything from you. None of them is better than the others; each fits a different kind of need.

In `Track.__post_init__`, the dataclass is frozen, so set the converted genre with `object.__setattr__(self, "genre", genre)`. Catch the `ValueError` from `Genre(...)` only if you want a nicer message: it's already a `ValueError`.
@@hint
For `duration`, `divmod` twice: `h, rest = divmod(total, 3600)` and `m, s = divmod(rest, 60)`. For `by_genre`, a dictionary keeps the order genres first appear.
@@solution
import random
from collections.abc import Sequence
from dataclasses import dataclass
from enum import Enum

class Genre(Enum):
    AFROBEATS = "afrobeats"
    HIGHLIFE = "highlife"
    JAZZ = "jazz"
    GRIME = "grime"
    FOLK = "folk"

@dataclass(frozen=True)
class Track:
    title: str
    artist: str
    seconds: int
    genre: Genre

    def __post_init__(self):
        if not isinstance(self.seconds, int) or isinstance(self.seconds, bool) or self.seconds <= 0:
            raise ValueError(f"bad length: {self.seconds!r}")
        object.__setattr__(self, "genre", Genre(self.genre))

    def __str__(self):
        m, s = divmod(self.seconds, 60)
        return f"{self.title} by {self.artist} ({m}:{s:02})"

class Playlist(Sequence):
    def __init__(self, name, tracks=()):
        self.name = name
        self._tracks = tuple(tracks)

    def __getitem__(self, i):
        if isinstance(i, slice):
            return Playlist(self.name, self._tracks[i])
        return self._tracks[i]

    def __len__(self):
        return len(self._tracks)

    def __add__(self, other):
        if not isinstance(other, Playlist):
            return NotImplemented
        return Playlist(self.name, self._tracks + other._tracks)

    @property
    def duration(self):
        total = sum(t.seconds for t in self._tracks)
        h, rest = divmod(total, 3600)
        m, s = divmod(rest, 60)
        return f"{h}:{m:02}:{s:02}"

    def by_genre(self):
        groups = {}
        for t in self._tracks:
            groups.setdefault(t.genre, []).append(t.title)
        return groups

    def shuffled(self, seed):
        return Playlist(self.name, random.Random(seed).sample(self._tracks, len(self._tracks)))

    def __repr__(self):
        return f"Playlist({self.name!r}, {len(self)} tracks)"
