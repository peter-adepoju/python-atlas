import re

def safe_filename(title, ext=".gpx"):
    name = re.sub(r"[^a-z0-9]+", "-", title.lower()).strip("-")
    name = name[:40].rstrip("-") or "route"
    return name + ext

import logging

class LevelCounter(logging.Handler):
    def __init__(self):
        super().__init__()
        self.counts = {}

    def emit(self, record):
        self.counts[record.levelname] = self.counts.get(record.levelname, 0) + 1

## Blockers and warnings

Hardening finds more problems than a week can fix, so the decision to ship needs two lists. A **blocker** makes Waymark wrong, unsafe or unusable for real users: an operation over its budget, an open threat that lets someone pretend to be another user, change data or see what they shouldn't, a swallowed exception, a README with no install section, a task most testers can't finish. A **warning** is worth fixing but not worth waiting for: a denial-of-service risk with a rate limit already in front of it, docstrings at 70%, a SUS score just below average. Ship with warnings written down; never ship with a blocker.

:::exercise id=forge points=50 title="The hardening report"
Write `hardening_report(evidence)`. `evidence` has five keys:

- `"performance"`: a dict from case to `(p95, budget)` in ms;
- `"security"`: a list of threats, dicts with `threat`, `category` (a STRIDE name) and `status` (`"mitigated"`, `"accepted"` or `"open"`);
- `"errors"`: a dict with `swallowed` and `untranslated` (counts) and `request_ids` (a bool);
- `"documentation"`: a dict with `docstrings_pct`, `examples_failing` and `readme_missing` (a list);
- `"usability"`: a dict with `lowest_success` (0 to 1), `sus` and `severity4` (a count).

Problems, in this order within each area:

- performance: a **blocker** `"CASE over budget (P95 > BUDGET ms)"` per case over budget, in the order given;
- security: for each open threat in order, a **blocker** `"open: THREAT"` if its category is Spoofing, Tampering, Information disclosure or Elevation of privilege, otherwise a **warning** with the same text;
- errors: blockers `"N swallowed exceptions"` and `"N errors reach users untranslated"` when the counts aren't 0; a warning `"no request ids in logs"` if `request_ids` is false;
- documentation: blockers `"README missing X, Y"` (if any) and `"N failing examples"` (if any); a warning `"docstrings P%"` below 80;
- usability: blockers `"a task at P% success"` (P rounded) when `lowest_success` is below 0.8 and `"N severity 4 findings"` if any; a warning `"SUS X"` below 68.

Return `(lines, verdict)`: one line per area in the order above, `"AREA: ok"` or `"AREA: "` and the problems joined by `"; "`, each written `"blocker: TEXT"` or `"warning: TEXT"`. `verdict` is `"HOLD: N blockers"` (or `"HOLD: 1 blocker"`) if there are any, else `"SHIP"`, or `"SHIP with N warnings"` (`"1 warning"`) when there are warnings.

        "documentation": {"docstrings_pct": 92, "examples_failing": 0, "readme_missing": []},

    assert hardening_report(fixed)[1] == "SHIP"
    fixed["errors"] = {"swallowed": 3, "untranslated": 0, "request_ids": True}
    assert hardening_report(fixed) == (["performance: ok", "security: ok", "errors: blocker: 3 swallowed exceptions",
                                        "documentation: ok", "usability: ok"], "HOLD: 1 blocker")
@@solution
SERIOUS = {"Spoofing", "Tampering", "Information disclosure", "Elevation of privilege"}

def _plural(n, word):
    return f"{n} {word}" if n == 1 else f"{n} {word}s"

def hardening_report(evidence):
    areas = {}
    perf = areas["performance"] = []
    for case, (p95, budget) in evidence["performance"].items():
        if p95 > budget:
            perf.append(("blocker", f"{case} over budget ({p95} > {budget} ms)"))
    sec = areas["security"] = []
    for threat in evidence["security"]:
        if threat["status"] == "open":
            kind = "blocker" if threat["category"] in SERIOUS else "warning"
            sec.append((kind, f"open: {threat['threat']}"))
    err = areas["errors"] = []
    errors = evidence["errors"]
    if errors["swallowed"]:
        err.append(("blocker", f"{errors['swallowed']} swallowed exceptions"))
    if errors["untranslated"]:
        err.append(("blocker", f"{errors['untranslated']} errors reach users untranslated"))
    if not errors["request_ids"]:
        err.append(("warning", "no request ids in logs"))
    doc = areas["documentation"] = []
    docs = evidence["documentation"]
    if docs["readme_missing"]:
        doc.append(("blocker", "README missing " + ", ".join(docs["readme_missing"])))
    if docs["examples_failing"]:
        doc.append(("blocker", f"{docs['examples_failing']} failing examples"))
    if docs["docstrings_pct"] < 80:
        doc.append(("warning", f"docstrings {docs['docstrings_pct']}%"))
    use = areas["usability"] = []
    users = evidence["usability"]
    if users["lowest_success"] < 0.8:
        use.append(("blocker", f"a task at {round(100 * users['lowest_success'])}% success"))
    if users["severity4"]:
        use.append(("blocker", f"{users['severity4']} severity 4 findings"))
    if users["sus"] < 68:
        use.append(("warning", f"SUS {users['sus']}"))
    lines = []
    for area, problems in areas.items():
        lines.append(f"{area}: " + ("; ".join(f"{kind}: {text}" for kind, text in problems) if problems else "ok"))
    kinds = [kind for problems in areas.values() for kind, _ in problems]
    blockers, warnings = kinds.count("blocker"), kinds.count("warning")
    if blockers:
        verdict = "HOLD: " + _plural(blockers, "blocker")
    elif warnings:
        verdict = "SHIP with " + _plural(warnings, "warning")
    else:
        verdict = "SHIP"
    return lines, verdict

        "errors fixed: warning: no request ids in logs",
        "documentation fixed: blocker: README missing Quick start",
        "documentation fixed: blocker: 1 failing examples",
        "documentation fixed: warning: docstrings 67%",
        "documentation new: warning: docstrings 75%",
        "usability fixed: blocker: a task at 60% success",
        "usability fixed: blocker: 1 severity 4 findings",
        "usability new: warning: SUS 66.5",
        "9 fixed, 2 new, 3 remaining",
    ]
    assert compare_reports(AFTER, AFTER) == ["0 fixed, 0 new, 3 remaining"]
@@solution
def _parse(lines):
    areas = {}
    for line in lines:
        area, _, rest = line.partition(": ")
        areas[area] = [] if rest == "ok" else rest.split("; ")
    return areas

def compare_reports(before, after):
    old, new = _parse(before), _parse(after)
    out, fixed, added = [], 0, 0
    for area, problems in new.items():
        previous = old.get(area, [])
        for p in previous:
            if p not in problems:
                out.append(f"{area} fixed: {p}")
                fixed += 1
        for p in problems:
            if p not in previous:
                out.append(f"{area} new: {p}")
                added += 1
    remaining = sum(len(p) for p in new.values())
    out.append(f"{fixed} fixed, {added} new, {remaining} remaining")
    return out
