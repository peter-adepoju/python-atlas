class Thermostat:
    LOW = 5
    HIGH = 30

    def __init__(self, target=20):
        self.target = None
        self.set(target)

    def set(self, target):
        if not Thermostat.LOW <= target <= Thermostat.HIGH:
            raise ValueError(f"target must be {Thermostat.LOW} to {Thermostat.HIGH}")
        self.target = target

    def up(self):
        self.target = min(self.target + 0.5, Thermostat.HIGH)

    def down(self):
        self.target = max(self.target - 0.5, Thermostat.LOW)

    def heating(self, current):
        return current < self.target

class Queue:
    def __init__(self):
        self._names = []

    def join(self, name):
        if name in self._names:
            raise ValueError(f"{name} is already queuing")
        self._names.append(name)

    def serve(self):
        if not self._names:
            raise IndexError("the queue is empty")
        return self._names.pop(0)

    def position(self, name):
        return self._names.index(name) + 1 if name in self._names else None

    def size(self):
        return len(self._names)

import re

class Run:
    LINE = re.compile(r"\s*(\d{4}-\d{2}-\d{2})\s*,\s*([\d.]+)\s*km\s*,\s*([\d.]+)\s*min\s*")

    def __init__(self, date, km, minutes):
        self.date = date
        self.km = km
        self.minutes = minutes

    @classmethod
    def from_line(cls, line):
        m = cls.LINE.fullmatch(line)
        if m is None:
            raise ValueError(f"not a run: {line!r}")
        return cls(m.group(1), float(m.group(2)), float(m.group(3)))

    def pace(self):
        return round(self.minutes / self.km, 2)

## Forge: a vending machine

A vending machine is a good small system: several kinds of thing, each with its own data and rules, working together. Here's the specification.

- A **`Product`** has a `name` and a `price` in pence (a positive whole number).
- A **`Slot`** has a `code` like `"A1"`, a `product`, and a `quantity` (0 to `CAPACITY`, a class attribute of 8). It can be `restock(n)`ed (never above capacity: `ValueError`) and can `take_one()` (`ValueError` when empty).
- The **`Machine`** accepts only the coins in its class attribute `COINS = (200, 100, 50, 20, 10, 5)` (pence). `Machine(float=None)` takes its starting float, a dictionary from coin values to counts (copied; empty if not given). It keeps:
  - `slots`, a dictionary from codes to slots;
  - `float`, a dictionary from coin values to how many of that coin it holds, for giving change;
  - `inserted`, the total of coins put in for the current purchase;
  - `sales`, a list of `(code, product name, price)` for every sale.

Its methods:

- `add_slot(code, product, quantity)`: a new slot (a code already used raises `ValueError`).
- `insert(coin)`: add a coin to `inserted` and to the float; any other value raises `ValueError` and isn't kept.
- `choose(code)`: buy from that slot. It returns a list of the coins given as change, largest first. It raises `KeyError` for an unknown code and `ValueError` (changing nothing) if the slot is empty, if not enough money has been inserted, or if the exact change can't be made from the float. On success: take one item, record the sale, give the change from the float, and reset `inserted` to 0.
- `cancel()`: give back exactly the coins inserted for the current purchase (largest first), taking them out of the float, and reset `inserted`.
- `takings()`: the total price of all sales.

To give change, use the **largest coin first** that fits and that the float still holds, repeatedly. (For these coin values, that always finds a way if one exists using the coins held, except in rare cases this exercise won't test.)

Write `Machine._make_change(amount)` returning a list of coins or `None`, working on a **copy** of the float, so a failure changes nothing. Only when everything has been checked: take the item, record the sale, remove the change coins from the float, and reset `inserted`.
@@solution
class Product:
    def __init__(self, name, price):
        if type(price) is not int or price <= 0:
            raise ValueError("price must be a positive whole number of pence")
        self.name = name
        self.price = price

class Slot:
    CAPACITY = 8

    def __init__(self, code, product, quantity=0):
        if not 0 <= quantity <= Slot.CAPACITY:
            raise ValueError(f"quantity must be 0 to {Slot.CAPACITY}")
        self.code = code
        self.product = product
        self.quantity = quantity

    def restock(self, n):
        if n <= 0 or self.quantity + n > Slot.CAPACITY:
            raise ValueError(f"slot {self.code} can only take {Slot.CAPACITY - self.quantity} more")
        self.quantity += n

    def take_one(self):
        if self.quantity == 0:
            raise ValueError(f"slot {self.code} is empty")
        self.quantity -= 1

class Machine:
    COINS = (200, 100, 50, 20, 10, 5)

    def __init__(self, float=None):
        self.slots = {}
        self.float = {} if float is None else dict(float)
        self.inserted = 0
        self._coins_in = []
        self.sales = []

    def add_slot(self, code, product, quantity):
        if code in self.slots:
            raise ValueError(f"slot {code} already exists")
        self.slots[code] = Slot(code, product, quantity)

    def insert(self, coin):
        if coin not in Machine.COINS:
            raise ValueError(f"{coin}p coins aren't accepted")
        self.inserted += coin
        self._coins_in.append(coin)
        self.float[coin] = self.float.get(coin, 0) + 1

    def _make_change(self, amount):
        available = dict(self.float)
        coins = []
        for coin in Machine.COINS:
            while amount >= coin and available.get(coin, 0) > 0:
                coins.append(coin)
                available[coin] -= 1
                amount -= coin
        return coins if amount == 0 else None

    def _give(self, coins):
        for coin in coins:
            self.float[coin] -= 1

    def choose(self, code):
        slot = self.slots[code]
        if slot.quantity == 0:
            raise ValueError(f"{slot.product.name} is sold out")
        price = slot.product.price
        if self.inserted < price:
            raise ValueError(f"insert {price - self.inserted}p more")
        change = self._make_change(self.inserted - price)
        if change is None:
            raise ValueError("can't make change: please use exact money")
        slot.take_one()
        self.sales.append((code, slot.product.name, price))
        self._give(change)
        self.inserted = 0
        self._coins_in = []
        return change

    def cancel(self):
        coins = sorted(self._coins_in, reverse=True)
        self._give(coins)
        self.inserted = 0
        self._coins_in = []
        return coins

    def takings(self):
        return sum(price for _, _, price in self.sales)
:::

:::think title="Whose job is it?"
Each rule lives with the data it's about. The slot knows its capacity, so `restock` checks it; the product knows its price, so it checks that; only the machine knows about coins and change. When you wonder where a method should go, ask which object has the data it needs. If a method uses another object's attributes more than its own, it probably belongs on that other object.
:::

`re.fullmatch(r"\s*(.+?)\s+(\d+)-(\d+)\s+(.+?)\s*", line)` reads a line. Show the goal difference with `f"{d:+d}"`. For the order, sort teams by `(-points, -difference, -scored, name)`; for positions, compare each team's `(points, difference, scored)` with the one before.
@@solution
import re

class Match:
    LINE = re.compile(r"\s*(.+?)\s+(\d+)-(\d+)\s+(.+?)\s*")

    def __init__(self, home, home_goals, away, away_goals):
        self.home = home
        self.home_goals = home_goals
        self.away = away
        self.away_goals = away_goals

    @classmethod
    def from_line(cls, line):
        m = cls.LINE.fullmatch(line)
        if m is None:
            raise ValueError(f"not a result: {line!r}")
        return cls(m.group(1), int(m.group(2)), m.group(4), int(m.group(3)))

    def points(self):
        if self.home_goals > self.away_goals:
            return 3, 0
        if self.home_goals < self.away_goals:
            return 0, 3
        return 1, 1

class Team:
    def __init__(self, name):
        self.name = name
        self.played = self.won = self.drawn = self.lost = 0
        self.scored = self.conceded = 0

    def record(self, goals_for, goals_against):
        self.played += 1
        self.scored += goals_for
        self.conceded += goals_against
        if goals_for > goals_against:
            self.won += 1
        elif goals_for == goals_against:
            self.drawn += 1
        else:
            self.lost += 1

    def points(self):
        return 3 * self.won + self.drawn

    def difference(self):
        return self.scored - self.conceded

class League:
    def __init__(self):
        self.teams = {}

    def _team(self, name):
        if name not in self.teams:
            self.teams[name] = Team(name)
        return self.teams[name]

    def add_result(self, line):
        m = Match.from_line(line)
        self._team(m.home).record(m.home_goals, m.away_goals)
        self._team(m.away).record(m.away_goals, m.home_goals)

    def table(self):
        ordered = sorted(self.teams.values(), key=lambda t: (-t.points(), -t.difference(), -t.scored, t.name))
        lines, previous, position = [], None, 0
        for i, t in enumerate(ordered, start=1):
            key = (t.points(), t.difference(), t.scored)
            if key != previous:
                position = i
            previous = key
            lines.append(f"{position}. {t.name}  P{t.played} W{t.won} D{t.drawn} L{t.lost} "
                         f"GD{t.difference():+d} Pts{t.points()}")
        return lines
