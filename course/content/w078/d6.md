from pathlib import Path

def stale_files(root, now, days):
    root = Path(root)
    found = []
    for p in root.rglob("*"):
        rel = p.relative_to(root)
        if p.is_file() and not any(part.startswith(".") for part in rel.parts):
            age = int((now - p.stat().st_mtime) // 86400)
            if age >= days:
                found.append((rel.as_posix(), age))
    return sorted(found, key=lambda f: (-f[1], f[0]))

def total_size(root, paths):
    return sum((Path(root) / p).stat().st_size for p in paths)

## Forge: pagewatch

Build `main(argv, root, fetch, now, out, err)` for **pagewatch** (`prog="pagewatch"`), a tool that watches parts of web pages. `root` is its data folder (create it if needed), `fetch(url, headers)` returns `(status, body)`, `now` is a `datetime`, and output goes to `out` (results) and `err` (problems). The starter gives you `cron_matches(expr, when)` (which raises `ValueError` for a malformed expression), `write_atomic`, and a `Parser` whose `error` raises `UsageError`. Usage errors print `pagewatch: MESSAGE` to `err` and return `2`; nothing may exit.

State lives in `root/watch.json` (`{"pages": {NAME: {"url", "id", "every", "last"}}}`, written atomically with `json.dumps(state, indent=2, sort_keys=True)`), history in `root/history.csv` (columns `time,name,value`, written with the `csv` module, header first, times as `YYYY-MM-DD HH:MM`). Subcommands:

- **`add NAME URL --id ELEMENT [--every CRON]`** (default every hour: `0 * * * *`): print `watching NAME`. Refuse, with exit code `1` and a message on `err`: a name already watched (`pagewatch: NAME is already watched`), a URL that isn't `http` or `https` with a host (`pagewatch: not a web address: URL`), and a bad schedule (`pagewatch: bad schedule: CRON`). `last` starts as `None`.
- **`list`**: one line per page in name order, `NAME URL #ELEMENT every 'CRON': LAST`, where LAST is `never checked` or the last value in quotes (use `repr`); or `nothing watched`.
- **`check [--all]`**: for each page in name order that is due now (its schedule matches `now`; `--all` checks every page): fetch the site's `robots.txt` (once per host per run; a status other than `200` means no rules) and refuse disallowed pages (`NAME: not allowed by robots.txt` on `err`); fetch the page, always sending `{"User-Agent": AGENT}` (also for robots.txt); a status other than `200` is `NAME: failed (status N)` on `err`; extract the text of the element with that `id` (all text inside it, including nested tags, whitespace collapsed), or report `NAME: no element #ID on the page` on `err`. Then print `NAME: first value 'V'`, `NAME: changed from 'OLD' to 'NEW'` or `NAME: unchanged` (values shown with `repr`), store the new value, and add a history row. Save the state and history only if anything was checked successfully. Return `1` if any page had a problem, else `0`.
- **`history NAME [--limit N]`** (default `5`): the page's last N history rows, oldest first, as `TIME  VALUE` (two spaces), or `no history yet`; an unknown name is `pagewatch: not watching NAME` on `err` with code `1`.

def parse_field(text, low, high):
    """The set of values a cron field allows; ValueError if it's malformed."""
    values = set()
    for item in text.split(","):
        base, slash, step_text = item.partition("/")
        step = int(step_text) if slash else 1
        if base == "*":
            start, end = low, high
        elif "-" in base:
            start, end = map(int, base.split("-"))
        else:
            start = int(base)
            end = high if slash else start
        if step < 1 or not low <= start <= end <= high:
            raise ValueError(f"bad cron field: {text}")
        values.update(range(start, end + 1, step))
    return values


def cron_matches(expr, when):
    """True if the cron expression (or shortcut) matches the datetime; ValueError if it's malformed."""
    fields = SHORTCUTS.get(expr, expr).split()
    if len(fields) != 5:
        raise ValueError(f"expected 5 fields: {expr}")

class ElementText(HTMLParser):
    def __init__(self, element_id):
        super().__init__()
        self.element_id = element_id
        self.tag, self.depth, self.text, self.found = None, 0, "", False

    def handle_starttag(self, tag, attrs):
        if self.depth:
            if tag == self.tag:
                self.depth += 1
        elif not self.found and dict(attrs).get("id") == self.element_id:
            self.tag, self.depth, self.found = tag, 1, True

    def handle_endtag(self, tag):
        if self.depth and tag == self.tag:
            self.depth -= 1

    def handle_data(self, data):
        if self.depth:
            self.text += data


def element_text(html, element_id):
    parser = ElementText(element_id)
    parser.feed(html)
    parser.close()
    return " ".join(parser.text.split()) if parser.found else None


def load(root):
    path = Path(root) / "watch.json"
    return json.loads(path.read_text()) if path.exists() else {"pages": {}}


def save(root, state):
    write_atomic(Path(root) / "watch.json", json.dumps(state, indent=2, sort_keys=True))

def cmd_add(args, root, fetch, now, out, err):
    state = load(root)
    if args.name in state["pages"]:
        print(f"pagewatch: {args.name} is already watched", file=err)
        return 1
    if urlsplit(args.url).scheme not in ("http", "https") or not urlsplit(args.url).netloc:
        print(f"pagewatch: not a web address: {args.url}", file=err)
        return 1
    try:
        cron_matches(args.every, now)
    except ValueError:
        print(f"pagewatch: bad schedule: {args.every}", file=err)
        return 1
    state["pages"][args.name] = {"url": args.url, "id": args.element_id, "every": args.every, "last": None}
    save(root, state)
    print(f"watching {args.name}", file=out)
    return 0


def cmd_list(args, root, fetch, now, out, err):
    pages = load(root)["pages"]
    if not pages:
        print("nothing watched", file=out)
    for name in sorted(pages):
        page = pages[name]
        last = "never checked" if page["last"] is None else repr(page["last"])
        print(f"{name} {page['url']} #{page['id']} every {page['every']!r}: {last}", file=out)
    return 0


def cmd_check(args, root, fetch, now, out, err):
    state = load(root)
    robots = {}
    failed = False
    stamp = now.strftime("%Y-%m-%d %H:%M")
    rows = []
    for name in sorted(state["pages"]):
        page = state["pages"][name]
        if not args.all and not cron_matches(page["every"], now):
            continue
        parts = urlsplit(page["url"])
        if parts.netloc not in robots:
            status, body = fetch(f"{parts.scheme}://{parts.netloc}/robots.txt", {"User-Agent": AGENT})
            rp = RobotFileParser()
            rp.parse(body.splitlines() if status == 200 else [])
            robots[parts.netloc] = rp
        if not robots[parts.netloc].can_fetch(AGENT, page["url"]):
            print(f"{name}: not allowed by robots.txt", file=err)
            failed = True
            continue
        status, body = fetch(page["url"], {"User-Agent": AGENT})
        if status != 200:
            print(f"{name}: failed (status {status})", file=err)
            failed = True
            continue
        value = element_text(body, page["id"])
        if value is None:
            print(f"{name}: no element #{page['id']} on the page", file=err)
            failed = True
            continue
        if page["last"] is None:
            print(f"{name}: first value {value!r}", file=out)
        elif page["last"] != value:
            print(f"{name}: changed from {page['last']!r} to {value!r}", file=out)
        else:
            print(f"{name}: unchanged", file=out)
        page["last"] = value
        rows.append([stamp, name, value])
    if rows:
        history = Path(root) / "history.csv"
        new = not history.exists()
        with open(history, "a", newline="", encoding="utf-8") as f:
            writer = csv.writer(f)
            if new:
                writer.writerow(["time", "name", "value"])
            writer.writerows(rows)
        save(root, state)
    return 1 if failed else 0


def cmd_history(args, root, fetch, now, out, err):
    if args.name not in load(root)["pages"]:
        print(f"pagewatch: not watching {args.name}", file=err)
        return 1
    history = Path(root) / "history.csv"
    rows = []
    if history.exists():
        with open(history, newline="", encoding="utf-8") as f:
            rows = [r for r in csv.DictReader(f) if r["name"] == args.name]
    for r in rows[-args.limit:]:
        print(f"{r['time']}  {r['value']}", file=out)
    if not rows:
        print("no history yet", file=out)
    return 0

import csv
import io
import os
import zipfile
from datetime import datetime, timedelta
from pathlib import Path

def write_atomic(path, text):
    path = Path(path)
    tmp = path.with_name(path.name + ".tmp")
    tmp.write_text(text, encoding="utf-8")
    os.replace(tmp, path)

def prune(root, now, keep_days=30):
    root = Path(root)
    history = root / "history.csv"
    if not history.exists():
        return {}
    with open(history, newline="", encoding="utf-8") as f:
        rows = list(csv.DictReader(f))
    cutoff = now - timedelta(days=keep_days)
    old, keep = {}, []
    for row in rows:
        when = datetime.strptime(row["time"], "%Y-%m-%d %H:%M")
        if when < cutoff:
            old.setdefault(when.strftime("%Y-%m"), []).append(row)
        else:
            keep.append(row)
    if not old:
        return {}
    (root / "archive").mkdir(exist_ok=True)
    member = f"rows-{now:%Y%m%d-%H%M}.csv"
    for month, month_rows in old.items():
        buf = io.StringIO()
        writer = csv.DictWriter(buf, ["time", "name", "value"], lineterminator="\n")
        writer.writeheader()
        writer.writerows(month_rows)
        with zipfile.ZipFile(root / "archive" / f"history-{month}.zip", "a", compression=zipfile.ZIP_DEFLATED) as zf:
            zf.writestr(member, buf.getvalue())
    buf = io.StringIO()
    writer = csv.DictWriter(buf, ["time", "name", "value"], lineterminator="\n")
    writer.writeheader()
    writer.writerows(keep)
    write_atomic(history, buf.getvalue())
    return {month: len(month_rows) for month, month_rows in sorted(old.items())}


def month_history(root, month):
    path = Path(root) / "archive" / f"history-{month}.zip"
    if not path.exists():
        return []
    rows = []
    with zipfile.ZipFile(path) as zf:
        for name in sorted(zf.namelist()):
            rows += list(csv.DictReader(io.StringIO(zf.read(name).decode("utf-8"))))
    return sorted(rows, key=lambda r: (r["time"], r["name"]))
