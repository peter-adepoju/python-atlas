## Warm-up

        amount -= c
        used.append(c)

Taking the largest coin each time uses three coins (4, 1, 1), but two threes would do. Greedy choices that look right locally can be wrong globally; that's the signal for dynamic programming.

## Disguised structures

- "Sorted, but…" (rotated, with an unknown length, or a yes/no question that flips once): **binary search**, O(log n). Decide at each step which half is sorted, or which side the answer lies on.
- A grid of cells, a map, "connected", "reachable": a **graph**. Explore with **breadth-first search** and a queue (`collections.deque`), which has no recursion limit; a deep recursive search on a 300 by 300 grid can exceed Python's limit of about 1,000 calls.
- "Prerequisites", "must come before", "build order": a **directed graph**; a valid order is a **topological sort**, and a cycle means none exists.
- "Fewest", "how many ways", choices whose effects overlap: **dynamic programming**: the best answer for `amount` built from the best answers for smaller amounts.
- "The k-th largest so far", "the k closest": a **heap** of size k (`heapq`), O(log k) per item.

@@hint
At each step one half of `[lo, hi]` is sorted: if `nums[lo] <= nums[mid]`, the left half is. Check whether the target lies in the sorted half; if so search there, else the other half.
@@solution
def search_rotated(nums, target):
    lo, hi = 0, len(nums) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        m = nums[mid]
        if m == target:
            return mid
        first, last = nums[lo], nums[hi]
        if first <= m:
            if first <= target < m:
                hi = mid - 1
            else:
                lo = mid + 1
        else:
            if m < target <= last:
                lo = mid + 1
            else:
                hi = mid - 1
    return -1

For each unvisited land cell, start a breadth-first search with a `deque`, marking every land cell it reaches. Recursion would hit Python's limit on the huge island.
@@solution
from collections import deque

def islands(grid):
    rows, cols = len(grid), len(grid[0]) if grid else 0
    seen, count = set(), 0
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] != "1" or (r, c) in seen:
                continue
            count += 1
            seen.add((r, c))
            queue = deque([(r, c)])
            while queue:
                y, x = queue.popleft()
                for ny, nx in ((y + 1, x), (y - 1, x), (y, x + 1), (y, x - 1)):
                    if 0 <= ny < rows and 0 <= nx < cols and grid[ny][nx] == "1" and (ny, nx) not in seen:
                        seen.add((ny, nx))
                        queue.append((ny, nx))
    return count

def fewest_coins(coins, amount):
    best = [0] + [None] * amount
    for a in range(1, amount + 1):
        options = [best[a - c] for c in coins if c <= a and best[a - c] is not None]
        best[a] = 1 + min(options) if options else None
    return -1 if best[amount] is None else best[amount]

@@hint
Keep a min-heap of the k largest numbers; its smallest item, `heap[0]`, is the k-th largest.
@@solution
import heapq

class KthLargest:
    def __init__(self, k, nums):
        self.k, self.heap = k, []
        for n in nums:
            self.add(n)

    def add(self, value):
        if len(self.heap) < self.k:
            heapq.heappush(self.heap, value)
        elif value > self.heap[0]:
            heapq.heapreplace(self.heap, value)
        return self.heap[0] if len(self.heap) == self.k else None

Count each course's unmet prerequisites; push courses with none onto a heap; pop the smallest, then release the courses that depended on it.
@@solution
import heapq

def course_order(n, pairs):
    waiting = [0] * n
    after = [[] for _ in range(n)]
    for course, before in pairs:
        waiting[course] += 1
        after[before].append(course)
    ready = [c for c in range(n) if waiting[c] == 0]
    heapq.heapify(ready)
    order = []
    while ready:
        c = heapq.heappop(ready)
        order.append(c)
        for nxt in after[c]:
            waiting[nxt] -= 1
            if waiting[nxt] == 0:
                heapq.heappush(ready, nxt)
    return order if len(order) == n else []

def test_ladders():

@@hint
Breadth-first search from `begin`. To find neighbours fast, group the words by patterns with one letter blanked (`h*t`, `*ot`, `ho*`), built once.
@@solution
from collections import deque

def ladder_length(begin, end, words):
    words = set(words)
    if end not in words:
        return 0
    buckets = {}
    for w in words | {begin}:
        for i in range(len(w)):
            buckets.setdefault(w[:i] + "*" + w[i + 1:], []).append(w)
    seen, queue = {begin}, deque([(begin, 1)])
    while queue:
        word, steps = queue.popleft()
        if word == end:
            return steps
        for i in range(len(word)):
            for nxt in buckets.get(word[:i] + "*" + word[i + 1:], []):
                if nxt not in seen and nxt in words:
                    seen.add(nxt)
                    queue.append((nxt, steps + 1))
    return 0
