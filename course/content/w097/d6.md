:::exercise id=mix_window points=10 title="The best k-day stretch"
Write `best_stretch(km, k)`: the largest total of any `k` consecutive values in `km` (a walk log's daily distances), in linear time; `0` if there are fewer than `k` values.
@@starter
def best_stretch(km, k):
    ___
@@tests
import time

def test_stretch():
    "a sliding window"
    assert best_stretch([3, 0, 8, 4, 0, 0, 12], 3) == 12 and best_stretch([1, 2], 3) == 0 and best_stretch([5], 1) == 5
    start = time.perf_counter()
    assert best_stretch(list(range(200_000)), 1000) == sum(range(199_000, 200_000))
    assert time.perf_counter() - start < 1.5, "slide, don't re-add"
@@solution
def best_stretch(km, k):
    if len(km) < k:
        return 0
    total = best = sum(km[:k])
    for i in range(k, len(km)):
        total += km[i] - km[i - k]
        best = max(best, total)
    return best
:::

:::exercise id=mix_review points=10 title="One more review fix"
This function reads a settings file but hides every error and leaves the file open. Rewrite `read_limit(path)` to return the `limit` from the JSON file, using `with`, returning `100` only when the file doesn't exist.
@@starter
import json

def read_limit(path):
    try:
        return json.load(open(path))["limit"]
    except:
        return 100
@@tests
import json

def test_limit():
    "only a missing file means the default"
    with open("settings.json", "w") as f:
        json.dump({"limit": 25}, f)
    assert read_limit("settings.json") == 25 and read_limit("missing-settings.json") == 100
    with open("settings2.json", "w") as f:
        json.dump({"other": 1}, f)
    try:
        read_limit("settings2.json")
    except KeyError:
        pass
    else:
        raise AssertionError("a missing key should raise")
    assert "with open" in source
@@solution
import json

def read_limit(path):
    try:
        with open(path) as f:
            return json.load(f)["limit"]
    except FileNotFoundError:
        return 100

@@tests
import random, time

def test_basic():
    "the classic sequence"
    c = LRUCache(2)
    c.put(1, 1)
    c.put(2, 2)
    assert c.get(1) == 1
    c.put(3, 3)
    assert c.get(2) == -1 and c.keys() == [1, 3]
    c.put(4, 4)
    assert c.get(1) == -1 and c.get(3) == 3 and c.get(4) == 4 and c.keys() == [3, 4]

def test_update():
    "put on an existing key updates and uses it"
    c = LRUCache(2)
    c.put("a", 1)
    c.put("b", 2)
    c.put("a", 10)
    c.put("c", 3)
    assert c.get("a") == 10 and c.get("b") == -1 and c.keys() == ["c", "a"]
    one = LRUCache(1)
    one.put(1, 1)
    one.put(2, 2)
    assert one.get(1) == -1 and one.get(2) == 2

def test_fast():
    "a million operations"
    rng = random.Random(7)
    c = LRUCache(1000)
    start = time.perf_counter()
    for i in range(500_000):
        k = rng.randrange(5000)
        c.put(k, i)
        c.get(rng.randrange(5000))
    assert len(c.keys()) == 1000
    assert time.perf_counter() - start < 6, "too slow: O(1) operations"
@@hint
`collections.OrderedDict` keeps keys in order and has `move_to_end(key)` and `popitem(last=False)`, both O(1). (In an interview, also be ready to explain the underlying idea: a dictionary plus a doubly linked list.)
@@solution
from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.items = OrderedDict()

    def get(self, key):
        if key not in self.items:
            return -1
        self.items.move_to_end(key)
        return self.items[key]

    def put(self, key, value):
        if key in self.items:
            self.items.move_to_end(key)
        elif len(self.items) >= self.capacity:
            self.items.popitem(last=False)
        self.items[key] = value

    def keys(self):
        return list(self.items)
:::

## Bonus: the scorecard

After a loop, interviewers rate each round against a few criteria and meet to decide together. Simulating the decision teaches you what each round is really scoring.

:::exercise id=forge2 points=30 title="Bonus: from rounds to a decision"
Write `loop_decision(rounds)`: each round has `name` and `scores`, a dict of criteria to ratings from 1 to 4. A round's **signal** is its mean rating: `"strong yes"` from 3.5, `"yes"` from 2.75, `"lean no"` from 2, else `"no"`. Return `(lines, decision)`: lines `"NAME: MEAN (SIGNAL)"` with the mean to 2 decimal places, then `"overall: MEAN"` (the mean of the round means, 2 places). The decision is `"no offer"` if any round is `"no"` or two or more are `"lean no"`; otherwise `"offer"` if the overall mean is at least 3 and at least one round is `"strong yes"`; otherwise `"another round"`.
@@starter
def loop_decision(rounds):
    ___
@@tests
ROUNDS = [
    {"name": "coding", "scores": {"problem solving": 4, "code quality": 3, "testing": 4, "communication": 4}},
    {"name": "design", "scores": {"requirements": 3, "estimates": 3, "trade-offs": 2}},
    {"name": "code review", "scores": {"severity": 4, "clarity": 3}},
    {"name": "behavioural", "scores": {"ownership": 3, "feedback": 3, "learning": 4}},
]

def test_offer():
    "a solid loop"
    lines, decision = loop_decision(ROUNDS)
    assert lines == ["coding: 3.75 (strong yes)", "design: 2.67 (lean no)", "code review: 3.50 (strong yes)",
                     "behavioural: 3.33 (yes)", "overall: 3.31"]
    assert decision == "offer"

def test_others():
    "no, and another round"
    weak = [dict(r, scores={k: 2 for k in r["scores"]}) for r in ROUNDS[:2]] + ROUNDS[2:]
    assert loop_decision(weak)[1] == "no offer"
    one_no = ROUNDS[:3] + [{"name": "behavioural", "scores": {"ownership": 1, "feedback": 2}}]
    assert loop_decision(one_no)[1] == "no offer"
    middling = [{"name": n, "scores": {"x": 3}} for n in ("a", "b", "c")]
    assert loop_decision(middling) == (["a: 3.00 (yes)", "b: 3.00 (yes)", "c: 3.00 (yes)", "overall: 3.00"], "another round")
@@solution
def _signal(mean):
    if mean >= 3.5:
        return "strong yes"
    if mean >= 2.75:
        return "yes"
    if mean >= 2:
        return "lean no"
    return "no"

def loop_decision(rounds):
    lines, means, signals = [], [], []
    for r in rounds:
        mean = sum(r["scores"].values()) / len(r["scores"])
        means.append(mean)
        signals.append(_signal(mean))
        lines.append(f"{r['name']}: {mean:.2f} ({signals[-1]})")
    overall = sum(means) / len(means)
    lines.append(f"overall: {overall:.2f}")
    if "no" in signals or signals.count("lean no") >= 2:
        decision = "no offer"
    elif overall >= 3 and "strong yes" in signals:
        decision = "offer"
    else:
        decision = "another round"
    return lines, decision
:::
