def daily_writes(follower_counts, posts_per_day):
    return sum(f * p for f, p in zip(follower_counts, posts_per_day))

def split_writes(follower_counts, posts_per_day, threshold):
    pushed = sum(f * p for f, p in zip(follower_counts, posts_per_day) if f <= threshold)
    return pushed, sum(1 for f in follower_counts if f > threshold)

from collections import OrderedDict

class ExpiringLRU:
    def __init__(self, capacity, ttl, clock):
        self.capacity, self.ttl, self.clock = capacity, ttl, clock
        self.data = OrderedDict()
        self.hits = self.misses = 0

    def put(self, key, value):
        self.data[key] = (value, self.clock())
        self.data.move_to_end(key)
        while len(self.data) > self.capacity:
            self.data.popitem(last=False)

    def get(self, key):
        if key in self.data:
            value, stored = self.data[key]
            if self.clock() - stored < self.ttl:
                self.hits += 1
                self.data.move_to_end(key)
                return value
            del self.data[key]
        self.misses += 1
        return None

## Forge: a news feed

A feed shows each user the newest posts from everyone they follow. Two ways to build it:

- **Fan-out on write (push)**: when someone posts, copy the post's id into the stored feed of each follower. Reading a feed is then one quick lookup. But a post by an account with ten million followers means ten million writes.
- **Fan-out on read (pull)**: store nothing per reader; at read time, fetch the recent posts of everyone the reader follows and merge them. Posting is cheap, but every feed read does a lot of work.

Large feeds use **both**: push for ordinary accounts, pull for the few with huge followings ("celebrities"), merged at read time. The push work goes through a **queue**, so posting stays fast.

Build `FeedService(clock, celebrity_threshold=1000, feed_size=200, cache_ttl=30)`:

- `follow(follower, followee)` and `unfollow(follower, followee)`; following yourself raises `ValueError("you can't follow yourself")`. Following or unfollowing clears the follower's cached feeds.
- `post(author, text)` stores the post with the next id (from 1) and the current time, queues a fan-out job, and returns the id.
- `process_fanout(batch=100)` handles up to `batch` queued jobs: for an author with **at most** `celebrity_threshold` followers, add the post id to each follower's stored feed (newest first, keeping only `feed_size` ids). Authors above the threshold are skipped (they'll be pulled). It returns the number of feed writes made.
- `feed(user, limit=10, before=None)` returns `{"posts": [...], "next": CURSOR}`: posts (each `{"id", "author", "text"}`) newest first by id, drawn from the user's stored feed **plus** the posts of every celebrity they currently follow (above the threshold now), excluding authors they no longer follow, only ids **below** `before` if it's given, at most `limit` of them; `next` is the last post's id if there might be more (a full page), else `None`. Results are cached per `(user, limit, before)` for `cache_ttl` seconds; count `cache_hits` and `cache_misses`.

import heapq
from collections import deque

class FeedService:
    def __init__(self, clock, celebrity_threshold=1000, feed_size=200, cache_ttl=30):
        self.clock = clock
        self.threshold = celebrity_threshold
        self.feed_size = feed_size
        self.cache_ttl = cache_ttl
        self.following = {}
        self.followers = {}
        self.posts = {}
        self.by_author = {}
        self.feeds = {}
        self.queue = deque()
        self.cache = {}
        self.cache_hits = self.cache_misses = 0
        self.next_id = 1

    def _clear(self, user):
        self.cache = {k: v for k, v in self.cache.items() if k[0] != user}

    def follow(self, follower, followee):
        if follower == followee:
            raise ValueError("you can't follow yourself")
        self.following.setdefault(follower, set()).add(followee)
        self.followers.setdefault(followee, set()).add(follower)
        self._clear(follower)

    def unfollow(self, follower, followee):
        self.following.get(follower, set()).discard(followee)
        self.followers.get(followee, set()).discard(follower)
        self._clear(follower)

    def post(self, author, text):
        post_id = self.next_id
        self.next_id += 1
        self.posts[post_id] = {"id": post_id, "author": author, "text": text, "at": self.clock()}
        self.by_author.setdefault(author, []).append(post_id)
        self.queue.append(post_id)
        return post_id

    def _celebrity(self, author):
        return len(self.followers.get(author, ())) > self.threshold

    def process_fanout(self, batch=100):
        writes = 0
        for _ in range(min(batch, len(self.queue))):
            post_id = self.queue.popleft()
            author = self.posts[post_id]["author"]
            if self._celebrity(author):
                continue
            for follower in self.followers.get(author, ()):
                feed = self.feeds.setdefault(follower, [])
                feed.insert(0, post_id)
                del feed[self.feed_size:]
                writes += 1
        return writes

    def feed(self, user, limit=10, before=None):
        key = (user, limit, before)
        now = self.clock()
        if key in self.cache and now - self.cache[key][1] < self.cache_ttl:
            self.cache_hits += 1
            return self.cache[key][0]
        self.cache_misses += 1
        follows = self.following.get(user, set())
        candidates = {i for i in self.feeds.get(user, []) if self.posts[i]["author"] in follows}
        for author in follows:
            if self._celebrity(author):
                candidates.update(self.by_author.get(author, []))
        if before is not None:
            candidates = {i for i in candidates if i < before}
        chosen = heapq.nlargest(limit, candidates)
        posts = [{"id": i, "author": self.posts[i]["author"], "text": self.posts[i]["text"]} for i in chosen]
        result = {"posts": posts, "next": chosen[-1] if len(chosen) == limit else None}
        self.cache[key] = (result, now)
        return result

## Forge bonus: a gateway that sheds load

When traffic exceeds what a service can handle, it's far better to refuse some requests quickly than to accept them all and answer everyone slowly. A **gateway** in front of the service does that: per-user rate limits stop one client hogging capacity, a **concurrency limit** caps the requests in progress, and when the service is overloaded it **sheds** the least important work first.

:::exercise id=forge2 points=30 title="Forge bonus: a gateway that sheds load"
Write a class `Gateway(clock, max_in_flight, per_user_rate, burst)`. `begin(user, kind)` decides whether a request may start, returning `(status, reason)`: requests of kind `"write"` or `"read"` (anything else is `(400, "unknown kind")`). A per-user token bucket (capacity `burst`, refilling at `per_user_rate` tokens a second, starting full) gives `(429, "rate limited")` when empty. If `in_flight` has reached `max_in_flight`, the request is shed: `(503, "overloaded")`. Also, once in-flight reaches 80% of `max_in_flight` (rounded down, at least 1), **reads** are shed with `(503, "shedding reads")` so writes, which matter more, still get through. Otherwise count it in flight and return `(200, "ok")`. Check in this order: kind, rate limit, full, shedding (a refused request doesn't use a token unless it was refused only for load). `end()` finishes one request. `stats()` returns a dict counting each status (as keys `200`, `429`, `503`, `400`, sorted, only those seen) plus `"in_flight"`.

    assert g.begin("ada", "read") == (429, "rate limited")
    assert g.begin("bob", "write") == (200, "ok") and g.begin("cy", "read") == (200, "ok")
    assert g.begin("dan", "read") == (503, "shedding reads") and g.begin("dan", "write") == (200, "ok")
    assert g.begin("eve", "write") == (503, "overloaded") and g.begin("eve", "delete") == (400, "unknown kind")

    g.end()
    assert g.begin("eve", "read") == (200, "ok")
    clock.t = 1.0
    assert g.begin("ada", "read") == (503, "shedding reads")
    assert g.stats() == {200: 6, 400: 1, 429: 1, 503: 3, "in_flight": 4}

@@solution
class Gateway:
    def __init__(self, clock, max_in_flight, per_user_rate, burst):
        self.clock = clock
        self.max_in_flight = max_in_flight
        self.shed_at = max(1, int(max_in_flight * 0.8))
        self.rate = per_user_rate
        self.burst = burst
        self.buckets = {}
        self.in_flight = 0
        self.counts = {}

    def _answer(self, status, reason):
        self.counts[status] = self.counts.get(status, 0) + 1
        return status, reason

    def _take(self, user):
        now = self.clock()
        tokens, last = self.buckets.get(user, (self.burst, now))
        tokens = min(self.burst, tokens + (now - last) * self.rate)
        if tokens < 1:
            self.buckets[user] = (tokens, now)
            return False
        self.buckets[user] = (tokens - 1, now)
        return True

    def begin(self, user, kind):
        if kind not in ("read", "write"):
            return self._answer(400, "unknown kind")
        if not self._take(user):
            return self._answer(429, "rate limited")
        if self.in_flight >= self.max_in_flight:
            return self._answer(503, "overloaded")
        if kind == "read" and self.in_flight >= self.shed_at:
            return self._answer(503, "shedding reads")
        self.in_flight += 1
        return self._answer(200, "ok")

    def end(self):
        self.in_flight -= 1

    def stats(self):
        out = {k: self.counts[k] for k in sorted(self.counts)}
        out["in_flight"] = self.in_flight
        return out
