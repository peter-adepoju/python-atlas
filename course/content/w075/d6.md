def rows(sql, params=()):
    """Run a query on a fresh copy of the database and return every row."""
    conn = connect()
    try:
        return conn.execute(sql, params).fetchall()
    finally:
        conn.close()


def show(sql, params=()):
    """Print a query's result as a small table."""
    conn = connect()
    cur = conn.execute(sql, params)
    names = [d[0] for d in cur.description]
    data = cur.fetchall()
    widths = [max(len(str(x)) for x in [n] + [r[i] for r in data]) for i, n in enumerate(names)]
    print("  ".join(n.ljust(w) for n, w in zip(names, widths)).rstrip())
    for r in data:
        print("  ".join(str(x).ljust(w) for x, w in zip(r, widths)).rstrip())
    conn.close()

def count_word(conn, word):
    with conn:
        conn.execute("INSERT INTO words (word, n) VALUES (?, 1) ON CONFLICT (word) DO UPDATE SET n = n + 1",

def top(conn, k):
    return conn.execute("SELECT word, n FROM words ORDER BY n DESC, word LIMIT ?", (k,)).fetchall()

## Forge: a library's loans

Write the data layer for a small lending library: a `Library(conn)` class that turns on foreign keys for its connection, with these methods (`LoanError` is an exception class you define):

- **`setup()`** applies the numbered migrations in `MIGRATIONS` (as in Day 3's challenge: each in its own explicit transaction, recording the version in `PRAGMA user_version`) and returns the numbers applied. Migration 1 creates `books (id, isbn TEXT NOT NULL UNIQUE, title TEXT NOT NULL, author TEXT NOT NULL, copies INTEGER NOT NULL CHECK (copies >= 0))`, `members (id, name TEXT NOT NULL, email TEXT NOT NULL UNIQUE)` and `loans (id, book_id, member_id, out, due, returned)`, with the two ids as foreign keys, the dates as ISO text, and `returned` NULL until the book comes back. Migration 2 creates the indexes `loans_book_id` and `loans_member_id`.
- **`add_book(isbn, title, author, copies=1)`** and **`add_member(name, email)`** (storing the email lower-cased) insert and return the new id.
- **`lend(book_id, member_id, today)`** (with `today` a `date`) checks, in this order, inside one transaction: that the book exists (`"no such book"`), the member exists (`"no such member"`), a copy is free (books on loan are those with `returned IS NULL`; `"no copies available"`), the member has no overdue loans (unreturned, with `due` before today; `"member has overdue books"`) and fewer than 3 active loans (`"member has too many loans"`); raising `LoanError` with that message on the first failure. Otherwise it records a loan due 14 days later and returns its id.
- **`give_back(loan_id, today)`** marks the loan returned (`LoanError("no such loan")` or `LoanError("already returned")`) and returns the fine in pence: 10 per day late, at most 500.
- **`overdue(today)`** lists `(member name, title, days late)` for unreturned overdue loans, most days late first, then by name and title.
- **`popular(limit=3)`** gives `(title, number of loans ever)` for the most-borrowed books (only books that have been borrowed), most first, ties by title.
- **`summary(member_id, today)`** returns `{"total": ..., "active": ..., "overdue": ...}` counts of the member's loans.

import sqlite3
from datetime import date, timedelta

MIGRATIONS = [
    [
        """CREATE TABLE books (id INTEGER PRIMARY KEY, isbn TEXT NOT NULL UNIQUE, title TEXT NOT NULL,
                              author TEXT NOT NULL, copies INTEGER NOT NULL CHECK (copies >= 0))""",
        """CREATE TABLE members (id INTEGER PRIMARY KEY, name TEXT NOT NULL, email TEXT NOT NULL UNIQUE)""",
        """CREATE TABLE loans (id INTEGER PRIMARY KEY,
                              book_id INTEGER NOT NULL REFERENCES books(id),
                              member_id INTEGER NOT NULL REFERENCES members(id),
                              out TEXT NOT NULL, due TEXT NOT NULL, returned TEXT)""",
    ],
    [
        "CREATE INDEX loans_book_id ON loans (book_id)",
        "CREATE INDEX loans_member_id ON loans (member_id)",
    ],
]


class LoanError(Exception):
    pass


class Library:
    MAX_LOANS = 3
    LOAN_DAYS = 14
    FINE_PER_DAY = 10
    MAX_FINE = 500

    def __init__(self, conn):
        self.conn = conn
        conn.execute("PRAGMA foreign_keys = ON")

    def setup(self):
        current = self.conn.execute("PRAGMA user_version").fetchone()[0]
        applied = []
        for n, statements in enumerate(MIGRATIONS, start=1):
            if n <= current:
                continue
            try:
                self.conn.execute("BEGIN")
                for s in statements:
                    self.conn.execute(s)
                self.conn.execute(f"PRAGMA user_version = {n}")
                self.conn.commit()
            except sqlite3.Error:
                self.conn.rollback()
                raise
            applied.append(n)
        return applied

    def add_book(self, isbn, title, author, copies=1):
        with self.conn:
            return self.conn.execute("INSERT INTO books (isbn, title, author, copies) VALUES (?, ?, ?, ?)",
                                     (isbn, title, author, copies)).lastrowid

    def add_member(self, name, email):
        with self.conn:
            return self.conn.execute("INSERT INTO members (name, email) VALUES (?, ?)", (name, email.lower())).lastrowid

    def lend(self, book_id, member_id, today):
        day = today.isoformat()
        with self.conn:
            book = self.conn.execute("SELECT copies FROM books WHERE id = ?", (book_id,)).fetchone()
            if book is None:
                raise LoanError("no such book")
            if self.conn.execute("SELECT 1 FROM members WHERE id = ?", (member_id,)).fetchone() is None:
                raise LoanError("no such member")
            out = self.conn.execute("SELECT count(*) FROM loans WHERE book_id = ? AND returned IS NULL", (book_id,)).fetchone()[0]
            if out >= book[0]:
                raise LoanError("no copies available")
            active, overdue = self.conn.execute(
                "SELECT count(*), coalesce(sum(due < ?), 0) FROM loans WHERE member_id = ? AND returned IS NULL",
                (day, member_id)).fetchone()
            if overdue:
                raise LoanError("member has overdue books")
            if active >= self.MAX_LOANS:
                raise LoanError("member has too many loans")
            due = (today + timedelta(days=self.LOAN_DAYS)).isoformat()
            return self.conn.execute("INSERT INTO loans (book_id, member_id, out, due) VALUES (?, ?, ?, ?)",
                                     (book_id, member_id, day, due)).lastrowid

    def give_back(self, loan_id, today):
        with self.conn:
            row = self.conn.execute("SELECT due, returned FROM loans WHERE id = ?", (loan_id,)).fetchone()
            if row is None:
                raise LoanError("no such loan")
            if row[1] is not None:
                raise LoanError("already returned")
            self.conn.execute("UPDATE loans SET returned = ? WHERE id = ?", (today.isoformat(), loan_id))
        late = (today - date.fromisoformat(row[0])).days
        return min(self.MAX_FINE, max(0, late) * self.FINE_PER_DAY)

    def overdue(self, today):
        rows = self.conn.execute("""
            SELECT m.name, b.title, l.due FROM loans l
            JOIN members m ON m.id = l.member_id JOIN books b ON b.id = l.book_id
            WHERE l.returned IS NULL AND l.due < ?
        """, (today.isoformat(),)).fetchall()
        result = [(name, title, (today - date.fromisoformat(due)).days) for name, title, due in rows]
        return sorted(result, key=lambda r: (-r[2], r[0], r[1]))

    def popular(self, limit=3):
        return self.conn.execute("""
            SELECT b.title, count(l.id) AS loans FROM books b JOIN loans l ON l.book_id = b.id
            GROUP BY b.id ORDER BY loans DESC, b.title LIMIT ?
        """, (limit,)).fetchall()

    def summary(self, member_id, today):
        total, active, overdue = self.conn.execute("""
            SELECT count(*), coalesce(sum(returned IS NULL), 0), coalesce(sum(returned IS NULL AND due < ?), 0)
            FROM loans WHERE member_id = ?
        """, (today.isoformat(), member_id)).fetchone()
        return {"total": total, "active": active, "overdue": overdue}

:::think title="Where the rules live"
Look at where each rule is enforced. Some live in the **schema** (unique ISBNs and emails, non-negative copies, loans that must point at real books and members): the database refuses to break them, whatever code is calling. Others live in **`lend`** (free copies, the loan limit, no overdue books), and hold only because every loan goes through that method, inside one transaction. A second program writing to the same database could bypass them. That's the usual trade-off: put the rules that must never break into constraints where you can, and keep the rest behind one well-tested door.

@@hint
`PRAGMA foreign_key_check` returns rows of `(table, rowid, parent, fkid)`. For the first column of each index: `[r[1] for r in conn.execute("PRAGMA index_list(TABLE)")]` gives index names, and `conn.execute(f"PRAGMA index_info({name})").fetchone()[2]` the first column's name. In `table_info`, the last column (`pk`) is the position in the primary key (0 if not part of it).
@@solution
import sqlite3

def audit(conn):
    problems = []
    for (message,) in conn.execute("PRAGMA integrity_check"):
        if message != "ok":
            problems.append(f"integrity: {message}")
    for table, rowid, parent, _ in conn.execute("PRAGMA foreign_key_check"):
        problems.append(f"broken key: {table} row {rowid} -> {parent}")
    tables = sorted(r[0] for r in conn.execute("SELECT name FROM sqlite_master WHERE type = 'table' AND name NOT LIKE 'sqlite_%'"))
    unindexed = []
    for table in tables:
        info = conn.execute(f"PRAGMA table_info({table})").fetchall()
        pk = [row[1] for row in sorted(info, key=lambda r: r[5]) if row[5] > 0]
        if not pk:
            problems.append(f"no primary key: {table}")
        firsts = set()
        for index in conn.execute(f"PRAGMA index_list({table})").fetchall():
            first = conn.execute(f"PRAGMA index_info({index[1]})").fetchone()
            if first is not None:
                firsts.add(first[2])
        if len(pk) > 1:
            firsts.add(pk[0])
        keys = {}
        for row in conn.execute(f"PRAGMA foreign_key_list({table})"):
            keys.setdefault(row[0], []).append(row[3])
        for columns in keys.values():
            if len(columns) == 1 and columns[0] not in firsts:
                unindexed.append(f"unindexed foreign key: {table}.{columns[0]}")
    return problems + unindexed
