Week 80, Day 4

Project: tests and CI

The helpdesk works today. Tests are what make sure it still works after next month’s changes, made by someone in a hurry. Today you’ll write the test suite the way Week 72 taught: tests for the rules that catch real mistakes (each exercise checks your tests against deliberately broken versions of the code), parametrized tests that turn the permission table and the state machine into one readable list of cases, and API tests with fixtures that build a fresh app and database for every test. Then the automation that runs them on every change: a continuous-integration workflow, checked against the habits from Weeks 74 and 79, and a report of which routes the tests never reach.

By the end of today you can

  • Write pytest tests that pass on correct code and fail on each realistic bug
  • Express tables of rules as parametrized tests
  • Test an HTTP API end to end with fixtures for the app, the database and the clock
  • Define a CI workflow that tests every push on several Python versions with hashed dependencies
  • Measure which routes a test run never reaches

"""A small subset of Flask that runs in the page (Week 77).

The names and behaviour follow Flask's own, so code written against this module also runs with
the real Flask (pip install flask): Flask, request, g, jsonify, abort, redirect, make_response,
url_for, Blueprint, and app.test_client(). It's a teaching copy, not the real thing: no templates,
sessions, streaming or debugger.

import io
import json
import re
from contextvars import ContextVar
from http import HTTPStatus
from urllib.parse import parse_qs, urlencode, urlsplit

_current = ContextVar("request")


class HTTPException(Exception):
    def __init__(self, code, description=None):
        self.code = code
        self.name = HTTPStatus(code).phrase
        self.description = description or self.name
        super().__init__(f"{code} {self.name}: {self.description}")


def abort(code, description=None):
    raise HTTPException(code, description)


class Headers:
    """Case-insensitive headers."""
    def __init__(self, items=()):
        self._items = [(k, str(v)) for k, v in (items.items() if isinstance(items, dict) else items)]

    def get(self, name, default=None):
        for k, v in self._items:
            if k.lower() == name.lower():
                return v
        return default

    def __getitem__(self, name):
        value = self.get(name)
        if value is None:
            raise KeyError(name)
        return value

    def __contains__(self, name):
        return self.get(name) is not None

    def __setitem__(self, name, value):
        self._items = [(k, v) for k, v in self._items if k.lower() != name.lower()] + [(name, str(value))]

    def items(self):
        return list(self._items)


class Args(dict):
    """Query or form values: .get(name) gives the first value; .getlist(name) all of them."""
    def __init__(self, pairs):
        super().__init__()
        self._lists = {}
        for k, v in pairs:
            self._lists.setdefault(k, []).append(v)
            dict.setdefault(self, k, v)

    def get(self, key, default=None, type=None):
        if key not in self:
            return default
        value = self[key]
        if type is not None:
            try:
                return type(value)
            except (TypeError, ValueError):
                return default
        return value

    def getlist(self, key):
        return list(self._lists.get(key, []))


class Request:
    def __init__(self, environ):
        self.environ = environ
        self.method = environ["REQUEST_METHOD"].upper()
        self.path = environ.get("PATH_INFO", "/") or "/"
        self.args = Args(_pairs(environ.get("QUERY_STRING", "")))
        items = [(k[5:].replace("_", "-").title(), v) for k, v in environ.items() if k.startswith("HTTP_")]
        if environ.get("CONTENT_TYPE"):
            items.append(("Content-Type", environ["CONTENT_TYPE"]))
        self.headers = Headers(items)
        length = int(environ.get("CONTENT_LENGTH") or 0)
        self.data = environ["wsgi.input"].read(length) if length else b""
        self.content_type = environ.get("CONTENT_TYPE", "")
        self.form = Args(_pairs(self.data.decode("utf-8"))) if self.content_type.startswith("application/x-www-form-urlencoded") else Args([])
        self.view_args = {}
        cookie = environ.get("HTTP_COOKIE", "")
        self.cookies = dict(p.strip().split("=", 1) for p in cookie.split(";") if "=" in p)

    @property
    def is_json(self):
        return self.content_type.split(";")[0].strip() == "application/json"

    def get_json(self, silent=False, force=False):
        if not (self.is_json or force):
            if silent:
                return None
            abort(415, "expected a JSON body (Content-Type: application/json)")
        try:
            return json.loads(self.data or b"null")
        except ValueError:
            if silent:
                return None
            abort(400, "the body isn't valid JSON")

    @property
    def json(self):
        return self.get_json()


def _pairs(text):
    return [(k, v) for k, vs in parse_qs(text, keep_blank_values=True).items() for v in vs]


class _Proxy:
    def __init__(self, getter):
        object.__setattr__(self, "_getter", getter)

    def __getattr__(self, name):
        return getattr(self._getter(), name)

    def __setattr__(self, name, value):
        setattr(self._getter(), name, value)

class Response:
    def __init__(self, response=b"", status=200, headers=None, mimetype=None, content_type=None):
        if isinstance(response, str):
            response = response.encode("utf-8")
        self.data = response
        self.status_code = status
        self.headers = Headers(headers or {})
        if content_type or mimetype or "Content-Type" not in self.headers:
            ctype = content_type or (f"{mimetype}; charset=utf-8" if mimetype and mimetype.startswith("text/") else mimetype) or "text/html; charset=utf-8"
            self.headers["Content-Type"] = ctype

    @property
    def status(self):
        return f"{self.status_code} {HTTPStatus(self.status_code).phrase}"

    @property
    def text(self):
        return self.data.decode("utf-8")

    def get_json(self):
        return json.loads(self.data)

    @property
    def json(self):
        return self.get_json()

    def set_cookie(self, key, value="", max_age=None, path="/", secure=False, httponly=False, samesite=None):
        parts = [f"{key}={value}"]
        if max_age is not None:
            parts.append(f"Max-Age={max_age}")
        parts.append(f"Path={path}")
        if secure:
            parts.append("Secure")
        if httponly:
            parts.append("HttpOnly")
        if samesite:
            parts.append(f"SameSite={samesite}")
        self.headers._items.append(("Set-Cookie", "; ".join(parts)))


def jsonify(*args, **kwargs):
    data = args[0] if len(args) == 1 else (list(args) if args else kwargs)
    return Response(json.dumps(data), headers={"Content-Type": "application/json"})


def redirect(location, code=302):

def make_response(*args):

_CONVERTERS = {"int": (r"\d+", int), "float": (r"\d+(?:\.\d+)?", float), "path": (r".+", str), "string": (r"[^/]+", str)}


class _Rule:
    def __init__(self, rule, endpoint, view, methods):
        self.rule, self.endpoint, self.view = rule, endpoint, view
        self.methods = {m.upper() for m in methods} | ({"HEAD"} if "GET" in {m.upper() for m in methods} else set())
        regex, self.converters = "", {}
        for piece in re.split(r"(<[^>]+>)", rule):
            m = re.fullmatch(r"<(?:(\w+):)?(\w+)>", piece)
            if m:
                kind, name = m.group(1) or "string", m.group(2)
                pattern, convert = _CONVERTERS[kind]
                regex += f"(?P<{name}>{pattern})"
                self.converters[name] = convert
            else:
                regex += re.escape(piece)
        self.regex = re.compile(regex)

    def match(self, path):
        m = self.regex.fullmatch(path)
        if not m:
            return None
        return {k: self.converters[k](v) for k, v in m.groupdict().items()}

    def build(self, values):
        def fill(m):
            return str(values.pop(m.group(2)))
        path = re.sub(r"<(?:(\w+):)?(\w+)>", fill, self.rule)
        return path + (f"?{urlencode(values)}" if values else "")


class _Setup:
    def route(self, rule, methods=("GET",), endpoint=None):
        def decorate(view):
            self._add(rule, endpoint or view.__name__, view, methods)
            return view
        return decorate

    def get(self, rule, **kw):
        return self.route(rule, methods=["GET"], **kw)

    def post(self, rule, **kw):
        return self.route(rule, methods=["POST"], **kw)

    def put(self, rule, **kw):
        return self.route(rule, methods=["PUT"], **kw)

    def patch(self, rule, **kw):
        return self.route(rule, methods=["PATCH"], **kw)

    def delete(self, rule, **kw):
        return self.route(rule, methods=["DELETE"], **kw)


class Blueprint(_Setup):
    def __init__(self, name, import_name, url_prefix=""):
        self.name, self.url_prefix = name, url_prefix
        self.deferred = []

    def _add(self, rule, endpoint, view, methods):
        self.deferred.append((rule, endpoint, view, methods))


class _App(_Setup):
    def __init__(self, import_name):
        self.import_name = import_name
        self.config = {"TESTING": False}
        self.rules = []
        self.before_funcs, self.after_funcs, self.error_handlers = [], [], {}

    def _add(self, rule, endpoint, view, methods):
        self.rules.append(_Rule(rule, endpoint, view, methods))

    def register_blueprint(self, bp, url_prefix=None):
        prefix = url_prefix if url_prefix is not None else bp.url_prefix
        for rule, endpoint, view, methods in bp.deferred:
            self._add(prefix + rule, f"{bp.name}.{endpoint}", view, methods)

    def before_request(self, f):
        self.before_funcs.append(f)
        return f

    def after_request(self, f):
        self.after_funcs.append(f)
        return f

    def errorhandler(self, code):
        def decorate(f):
            self.error_handlers[code] = f
            return f
        return decorate

    def url_for(self, endpoint, **values):
        for r in self.rules:
            if r.endpoint == endpoint:
                return r.build(dict(values))
        raise LookupError(f"no endpoint {endpoint!r}")

    @staticmethod
    def _to_response(rv):
        status, headers = None, None
        if isinstance(rv, tuple):
            if len(rv) == 3:
                rv, status, headers = rv
            elif isinstance(rv[1], (dict, list)):
                rv, headers = rv
            else:
                rv, status = rv
        if isinstance(rv, Response):
            response = rv
        elif isinstance(rv, (dict, list)):
            response = jsonify(rv)
        elif isinstance(rv, (str, bytes)):
            response = Response(rv)
        elif rv is None:
            raise TypeError("the view function did not return a valid response")
        else:
            raise TypeError(f"can't turn {type(rv).__name__} into a response")
        if status is not None:
            response.status_code = status
        for k, v in (headers.items() if isinstance(headers, dict) else headers or []):
            response.headers[k] = v
        return response

    def _error(self, e):
        handler = self.error_handlers.get(e.code)
        if handler:
            return self._to_response(handler(e))
        body = f"<h1>{e.code} {e.name}</h1><p>{e.description}</p>"
        return Response(body, e.code)

    def _dispatch(self, req):
        path = req.path
        allowed = set()
        for r in self.rules:
            values = r.match(path)
            if values is None:
                continue
            if req.method in r.methods:
                req.view_args = values
                return r.view(**values)
            allowed |= r.methods
        if allowed:
            raise _MethodNotAllowed(sorted(allowed | {"OPTIONS"}))
        abort(404, "The requested URL was not found on the server.")

    def full_dispatch(self, environ):
        req = Request(environ)
        token = _current.set((req, _G(), self))
        try:
            try:
                rv = None
                for f in self.before_funcs:
                    rv = f()
                    if rv is not None:
                        break
                if rv is None:
                    rv = self._dispatch(req)
                response = self._to_response(rv)
            except _MethodNotAllowed as e:
                response = self._error(HTTPException(405))
                response.headers["Allow"] = ", ".join(e.allowed)
            except HTTPException as e:
                response = self._error(e)
            except Exception as e:
                if self.config.get("TESTING") and 500 not in self.error_handlers:
                    raise
                response = self._error(HTTPException(500)) if 500 not in self.error_handlers else self._to_response(self.error_handlers[500](e))
                response.status_code = 500
            for f in self.after_funcs:
                response = f(response) or response
            if req.method == "HEAD":
                response.headers["Content-Length"] = len(response.data)
                response.data = b""
            return response
        finally:
            _current.reset(token)

    def __call__(self, environ, start_response):
        response = self.full_dispatch(environ)
        if "Content-Length" not in response.headers:
            response.headers["Content-Length"] = len(response.data)
        start_response(response.status, response.headers.items())
        return [response.data]

    def test_client(self):
        return _TestClient(self)


class _MethodNotAllowed(Exception):
    def __init__(self, allowed):
        self.allowed = allowed

    def __init__(self, app):
        self.app = app
        self.cookies = {}

    def open(self, path, method="GET", json=None, data=None, headers=None, query_string=None):
        url = urlsplit(path)
        query = url.query
        if query_string:
            query = (query + "&" if query else "") + urlencode(query_string, doseq=True)
        body, ctype = b"", None
        if json is not None:
            import json as _json
            body, ctype = _json.dumps(json).encode("utf-8"), "application/json"
        elif isinstance(data, dict):
            body, ctype = urlencode(data, doseq=True).encode("utf-8"), "application/x-www-form-urlencoded"
        elif data is not None:
            body = data.encode("utf-8") if isinstance(data, str) else data
        environ = {"REQUEST_METHOD": method.upper(), "PATH_INFO": url.path or "/", "QUERY_STRING": query,
                   "SERVER_NAME": "localhost", "SERVER_PORT": "80", "wsgi.url_scheme": "http",
                   "wsgi.input": io.BytesIO(body), "CONTENT_LENGTH": str(len(body))}
        headers = dict(headers or {})
        if ctype and not any(k.lower() == "content-type" for k in headers):
            environ["CONTENT_TYPE"] = ctype
        for k, v in headers.items():
            if k.lower() == "content-type":
                environ["CONTENT_TYPE"] = v
            else:
                environ["HTTP_" + k.upper().replace("-", "_")] = v
        if self.cookies:
            environ["HTTP_COOKIE"] = "; ".join(f"{k}={v}" for k, v in self.cookies.items())
        response = self.app.full_dispatch(environ)
        for k, v in response.headers.items():
            if k.lower() == "set-cookie":
                name, _, rest = v.partition("=")
                self.cookies[name] = rest.split(";")[0]
        return response

    def get(self, path, **kw):
        return self.open(path, "GET", **kw)

        return self.open(path, "POST", **kw)

    def put(self, path, **kw):

        return self.open(path, "PATCH", **kw)

"""The helpdesk's rules, with no web or database code (Week 80, Day 1).

Everything here is plain data and pure functions, so it can be tested on its own and used by the
database layer, the web layer and the tests alike.
"""
import re
from http import HTTPStatus

ROLES = ("customer", "agent", "admin")
PRIORITIES = ("low", "normal", "high", "urgent")
STATUSES = ("open", "in_progress", "waiting", "resolved", "closed")
STAFF = {"agent", "admin"}

# status -> {new status: roles that may make the change}
TRANSITIONS = {
    "open": {"in_progress": {"agent", "admin"}, "closed": {"customer", "admin"}},
    "in_progress": {"waiting": {"agent", "admin"}, "resolved": {"agent", "admin"}},
    "waiting": {"in_progress": {"customer", "agent", "admin"}},
    "resolved": {"closed": {"customer", "admin"}, "in_progress": {"customer", "agent", "admin"}},
    "closed": {},
}

# action -> (roles always allowed, roles allowed only on their own tickets)
PERMISSIONS = {
    "create_ticket": ({"customer"}, set()),
    "view_ticket": ({"agent", "admin"}, {"customer"}),
    "list_all_tickets": ({"agent", "admin"}, set()),
    "edit_ticket": ({"admin"}, {"customer"}),
    "comment": ({"agent", "admin"}, {"customer"}),
    "internal_note": ({"agent", "admin"}, set()),
    "change_priority": ({"agent", "admin"}, set()),
    "assign": ({"agent", "admin"}, set()),
    "delete_ticket": ({"admin"}, set()),
    "view_stats": ({"agent", "admin"}, set()),
}


class ValidationError(Exception):
    """Raised with a dict of field -> message."""

    def __init__(self, errors):
        super().__init__("; ".join(f"{k}: {v}" for k, v in sorted(errors.items())))
        self.errors = errors


def validate_ticket(data, partial=False):
    """Check a ticket body from a client. partial=True for PATCH: only the fields present are checked."""
    if not isinstance(data, dict):
        raise ValidationError({"_": "expected a JSON object"})
    errors, clean = {}, {}
    unknown = sorted(set(data) - {"title", "body", "priority"})
    if unknown:
        errors["_"] = f"unexpected field(s): {', '.join(unknown)}"
    for field in ("title", "body"):
        if field not in data:
            if not partial:
                errors[field] = "required"
            continue
        value = data[field]
        limit = (3, 120) if field == "title" else (1, 5000)
        if not isinstance(value, str) or not limit[0] <= len(value.strip()) <= limit[1]:
            errors[field] = f"{limit[0]} to {limit[1]} characters"
        else:
            clean[field] = value.strip()
    if "priority" in data:
        if data["priority"] not in PRIORITIES:
            errors["priority"] = f"one of {', '.join(PRIORITIES)}"
        else:
            clean["priority"] = data["priority"]
    elif not partial:
        clean["priority"] = "normal"
    if partial and not data:
        errors["_"] = "nothing to change"
    if errors:
        raise ValidationError(errors)
    return clean


def can_transition(current, new, role):
    """True if someone with this role may move a ticket from `current` to `new`."""
    return role in TRANSITIONS.get(current, {}).get(new, set())


def allowed(role, action, is_owner=False):
    """True if the role may do the action (is_owner: the ticket is the user's own)."""
    always, own = PERMISSIONS[action]
    return role in always or (is_owner and role in own)


def problem(status, detail, title=None, **extra):
    """An RFC 9457 problem-details response: (body, status, headers)."""
    body = {"type": "about:blank", "title": title or HTTPStatus(status).phrase, "status": status, "detail": detail}
    body.update(extra)
    return body, status, {"Content-Type": "application/problem+json"}

def docs_table(routes):
    """The routes as a Markdown table, for the README."""
    lines = ["| Method | Path | What it does | Success |", "|---|---|---|---|"]
    lines += [f"| {m} | `{p}` | {what} | {code} {HTTPStatus(code).phrase} |" for m, p, what, code in routes]
    return "\n".join(lines)

"""The helpdesk's database: migrations and the queries the app needs (Week 80, Day 2).

Every query uses parameters; every change that touches more than one row runs in a transaction.
"""
import sqlite3

MIGRATIONS = [
    (1, "users", """
        CREATE TABLE users (
            id INTEGER PRIMARY KEY,
            email TEXT NOT NULL UNIQUE,
            password TEXT NOT NULL,
            role TEXT NOT NULL CHECK (role IN ('customer', 'agent', 'admin')),
            created_at TEXT NOT NULL
        )"""),
    (2, "tickets", """
        CREATE TABLE tickets (
            id INTEGER PRIMARY KEY,
            customer_id INTEGER NOT NULL REFERENCES users (id),
            assignee_id INTEGER REFERENCES users (id),
            title TEXT NOT NULL,
            body TEXT NOT NULL,
            priority TEXT NOT NULL DEFAULT 'normal' CHECK (priority IN ('low', 'normal', 'high', 'urgent')),
            status TEXT NOT NULL DEFAULT 'open'
                CHECK (status IN ('open', 'in_progress', 'waiting', 'resolved', 'closed')),
            created_at TEXT NOT NULL,
            updated_at TEXT NOT NULL
        )"""),
    (3, "comments and events", """
        CREATE TABLE comments (
            id INTEGER PRIMARY KEY,
            ticket_id INTEGER NOT NULL REFERENCES tickets (id) ON DELETE CASCADE,
            author_id INTEGER NOT NULL REFERENCES users (id),
            body TEXT NOT NULL,
            internal INTEGER NOT NULL DEFAULT 0,
            created_at TEXT NOT NULL
        );
        CREATE TABLE events (
            id INTEGER PRIMARY KEY,
            ticket_id INTEGER NOT NULL REFERENCES tickets (id) ON DELETE CASCADE,
            actor_id INTEGER NOT NULL REFERENCES users (id),
            kind TEXT NOT NULL,
            detail TEXT NOT NULL,
            at TEXT NOT NULL
        )"""),
    (4, "indexes", """
        CREATE INDEX idx_tickets_status ON tickets (status, id);
        CREATE INDEX idx_tickets_customer ON tickets (customer_id, id);
        CREATE INDEX idx_comments_ticket ON comments (ticket_id, id)"""),
]


class ConflictError(Exception):
    """The row changed since it was read."""


def connect(path=":memory:"):
    conn = sqlite3.connect(path)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn


def migrate(conn, migrations=MIGRATIONS):
    """Apply the migrations the database hasn't had yet, each in its own transaction. Returns their versions."""
    conn.execute("CREATE TABLE IF NOT EXISTS schema_migrations (version INTEGER PRIMARY KEY, name TEXT NOT NULL)")
    conn.commit()
    versions = [v for v, _, _ in migrations]
    if versions != list(range(1, len(versions) + 1)):
        raise ValueError("migrations must be numbered 1, 2, 3... with no gaps")
    current = conn.execute("SELECT COALESCE(MAX(version), 0) FROM schema_migrations").fetchone()[0]
    if current > len(versions):
        raise RuntimeError(f"the database is at version {current}, newer than this code")
    applied = []
    for version, name, sql in migrations[current:]:
        try:
            conn.execute("BEGIN")
            for statement in sql.split(";"):
                if statement.strip():
                    conn.execute(statement)
            conn.execute("INSERT INTO schema_migrations (version, name) VALUES (?, ?)", (version, name))
            conn.commit()
        except Exception:
            conn.rollback()
            raise
        applied.append(version)
    return applied

    return dict(row) if row is not None else None

                 (ticket_id, actor_id, kind, detail, now))

def _like(text):
    return "%" + text.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_") + "%"


def list_tickets(conn, customer_id=None, status=None, priority=None, assignee_id=None, q=None, after_id=0, limit=20):
    """Tickets matching the filters, by id, after a cursor. Returns (items, next_after or None)."""
    where, values = ["id > ?"], [after_id]
    for column, value in (("customer_id", customer_id), ("status", status), ("priority", priority), ("assignee_id", assignee_id)):
        if value is not None:
            where.append(f"{column} = ?")
            values.append(value)
    if q:
        where.append("title LIKE ? ESCAPE '\\'")
        values.append(_like(q))
    rows = conn.execute(f"SELECT * FROM tickets WHERE {' AND '.join(where)} ORDER BY id LIMIT ?", values + [limit + 1]).fetchall()
    items = [dict(r) for r in rows[:limit]]
    return items, (items[-1]["id"] if len(rows) > limit else None)


def update_ticket(conn, ticket_id, changes, actor_id, now):
    if not changes:
        return get_ticket(conn, ticket_id)
    columns = sorted(changes)
    with conn:
        conn.execute(f"UPDATE tickets SET {', '.join(f'{c} = ?' for c in columns)}, updated_at = ? WHERE id = ?",
                     [changes[c] for c in columns] + [now, ticket_id])
        _event(conn, ticket_id, actor_id, "edited", ",".join(columns), now)
    return get_ticket(conn, ticket_id)


def change_status(conn, ticket_id, expected, new, actor_id, now):
    """Move a ticket from `expected` to `new`, unless someone changed its status first."""
    with conn:
        cur = conn.execute("UPDATE tickets SET status = ?, updated_at = ? WHERE id = ? AND status = ?",
                           (new, now, ticket_id, expected))
        if cur.rowcount != 1:
            raise ConflictError(f"ticket {ticket_id} is no longer {expected}")
        _event(conn, ticket_id, actor_id, "status", f"{expected}->{new}", now)
    return get_ticket(conn, ticket_id)

"""Passwords and tokens for the helpdesk (Week 79's tools, gathered in one place)."""
import base64
import hashlib
import hmac
import secrets


def pbkdf2_sha256(password, salt, iterations, dklen=32):
    base = hmac.new(password, digestmod=hashlib.sha256)
    out, block = b"", 1
    while len(out) < dklen:
        h = base.copy()
        h.update(salt + block.to_bytes(4, "big"))
        u = h.digest()
        total = int.from_bytes(u, "big")
        for _ in range(iterations - 1):
            h = base.copy()
            h.update(u)
            u = h.digest()
            total ^= int.from_bytes(u, "big")
        out += total.to_bytes(32, "big")
        block += 1
    return out[:dklen]


def hash_password(password, iterations):
    salt = secrets.token_bytes(16)
    digest = pbkdf2_sha256(password.encode(), salt, iterations)
    return f"pbkdf2_sha256${iterations}${base64.b64encode(salt).decode()}${base64.b64encode(digest).decode()}"


def check_password(password, stored):
    try:
        algorithm, iterations, salt, digest = stored.split("$")
        expected = pbkdf2_sha256(password.encode(), base64.b64decode(salt), int(iterations))
        return algorithm == "pbkdf2_sha256" and hmac.compare_digest(expected, base64.b64decode(digest))
    except ValueError:
        return False


def issue_token(user_id, expires, key):
    """A signed bearer token: USER_ID.EXPIRES.SIGNATURE."""
    payload = f"{user_id}.{expires}"
    return f"{payload}.{hmac.new(key, payload.encode(), hashlib.sha256).hexdigest()}"


def verify_token(token, key, now):
    """The user id in a valid, unexpired token, else None."""
    parts = token.split(".")
    if len(parts) != 3 or not parts[0].isdigit() or not parts[1].isdigit():
        return None
    payload = f"{parts[0]}.{parts[1]}"
    expected = hmac.new(key, payload.encode(), hashlib.sha256).hexdigest()
    if not hmac.compare_digest(parts[2], expected) or now >= int(parts[1]):
        return None
    return int(parts[0])

"""The helpdesk's HTTP API (Week 80, Day 3), built with miniflask (Week 77)."""
import re
import secrets
import time
from datetime import datetime, timezone

import helpdesk_db as db
from helpdesk_domain import ROLES, STATUSES, ValidationError, allowed, can_transition, problem, validate_ticket
from helpdesk_security import check_password, hash_password, issue_token, verify_token
from miniflask import Flask, HTTPException, g, request

ERROR_CODES = (400, 401, 403, 404, 405, 409, 415, 422, 500, 503)


class ApiError(HTTPException):
    def __init__(self, code, detail, **extra):
        super().__init__(code, detail)
        self.extra = extra


def iso(t):
    return datetime.fromtimestamp(t, timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


def create_app(settings, conn, clock=time.time, log=print):
    app = Flask(__name__)
    key = settings["SECRET_KEY"].encode()
    token_seconds = settings.get("TOKEN_SECONDS", 3600)
    iterations = settings.get("PASSWORD_ITERATIONS", 600_000)
    dummy = hash_password(secrets.token_hex(8), iterations)   # checked for unknown emails, so they take as long as real ones

    # ---- step 11
    for code in ERROR_CODES:
        @app.errorhandler(code)
        def handle(e, code=code):
            if code == 500 and not isinstance(e, HTTPException):
                incident = secrets.token_hex(4)
                log(f"incident {incident}: {type(e).__name__}: {e}")
                return problem(500, "something went wrong", incident=incident)
            return problem(e.code, e.description, **getattr(e, "extra", {}))

    @app.get("/healthz")
    def healthz():
        return {"status": "ok"}

    @app.get("/readyz")
    def readyz():
        try:
            version = conn.execute("SELECT MAX(version) FROM schema_migrations").fetchone()[0]
        except Exception:
            version = None
        if version != len(db.MIGRATIONS):
            raise ApiError(503, "the database isn't ready")
        return {"status": "ready", "schema": version}

    # ---- step 12
    @app.before_request
    def authenticate():
        g.user = None
        header = request.headers.get("Authorization", "")
        if header.startswith("Bearer "):
            user_id = verify_token(header[7:], key, int(clock()))
            if user_id is not None:
                g.user = db.get_user(conn, user_id)

    def user():
        if g.user is None:
            raise ApiError(401, "log in first")
        return g.user

    def body():
        data = request.get_json(silent=True)
        if not isinstance(data, dict):
            raise ApiError(400, "send a JSON object")
        return data

    @app.post("/users")
    def signup():
        data = body()
        email = data.get("email", "").strip().lower() if isinstance(data.get("email"), str) else ""
        password = data.get("password")
        if not re.fullmatch(r"[^@\s]+@[^@\s]+\.[a-z]{2,}", email) or not isinstance(password, str) or not 12 <= len(password) <= 128:
            raise ApiError(422, "a valid email and a password of 12 to 128 characters are needed")
        try:
            user_id = db.create_user(conn, email, hash_password(password, iterations), "customer", iso(clock()))
        except ValueError:
            raise ApiError(409, "email already registered") from None
        return {"id": user_id, "email": email, "role": "customer"}, 201

    @app.post("/tokens")
    def login():
        data = body()
        found = db.get_user_by_email(conn, str(data.get("email", "")).strip().lower())
        stored = found["password"] if found else dummy
        if not check_password(str(data.get("password", "")), stored) or found is None:
            raise ApiError(401, "wrong email or password")
        expires = int(clock()) + token_seconds
        return {"token": issue_token(found["id"], expires, key), "expires_at": iso(expires)}, 201

    @app.get("/me")
    def me():
        return user()

    # ---- step 13
    def checked(data, partial=False):
        try:
            return validate_ticket(data, partial)
        except ValidationError as e:
            raise ApiError(422, "the request has invalid fields", errors=e.errors) from None

    def visible_ticket(ticket_id):
        me = user()
        ticket = db.get_ticket(conn, ticket_id)
        if ticket is None or not allowed(me["role"], "view_ticket", ticket["customer_id"] == me["id"]):
            raise ApiError(404, f"no ticket {ticket_id}")
        return me, ticket

    def require(me, action, ticket=None, message="you can't do that"):
        if not allowed(me["role"], action, ticket is not None and ticket["customer_id"] == me["id"]):
            raise ApiError(403, message)

    @app.post("/tickets")
    def open_ticket():
        me = user()
        require(me, "create_ticket", message="only customers can open tickets")
        ticket = db.create_ticket(conn, me["id"], checked(body()), iso(clock()))
        return ticket, 201, {"Location": f"/tickets/{ticket['id']}"}

    @app.get("/tickets")
    def list_tickets():
        me = user()
        args = request.args
        filters = {}
        for name, choices in (("status", STATUSES), ("priority", ("low", "normal", "high", "urgent"))):
            if name in args:
                if args[name] not in choices:
                    raise ApiError(400, f"{name} must be one of {', '.join(choices)}")
                filters[name] = args[name]
        for name, low, high, default in (("after", 0, 10 ** 9, 0), ("limit", 1, 50, 20)):
            value = args.get(name, str(default))
            if not value.isdigit() or not low <= int(value) <= high:
                raise ApiError(400, f"{name} must be a whole number from {low} to {high}")
            filters[name] = int(value)
        if not allowed(me["role"], "list_all_tickets"):
            filters["customer_id"] = me["id"]
        items, after = db.list_tickets(conn, customer_id=filters.get("customer_id"), status=filters.get("status"),
                                       priority=filters.get("priority"), q=args.get("q") or None,
                                       after_id=filters["after"], limit=filters["limit"])
        return {"items": items, "next": after}

    @app.get("/tickets/<int:ticket_id>")
    def read_ticket(ticket_id):
        return visible_ticket(ticket_id)[1]

    # ---- step 14
    @app.patch("/tickets/<int:ticket_id>")
    def edit_ticket(ticket_id):
        me, ticket = visible_ticket(ticket_id)
        changes = checked(body(), partial=True)
        if "priority" in changes:
            require(me, "change_priority", ticket, "only staff can change the priority")
        if "title" in changes or "body" in changes:
            require(me, "edit_ticket", ticket, "only the customer who opened it can edit it")
            if me["role"] == "customer" and ticket["status"] != "open":
                raise ApiError(409, "only open tickets can be edited")
        return db.update_ticket(conn, ticket_id, changes, me["id"], iso(clock()))

    @app.post("/tickets/<int:ticket_id>/transitions")
    def transition(ticket_id):
        me, ticket = visible_ticket(ticket_id)
        new = body().get("status")
        if new not in STATUSES:
            raise ApiError(422, f"status must be one of {', '.join(STATUSES)}")
        if not can_transition(ticket["status"], new, me["role"]):
            raise ApiError(403, f"you can't move a ticket from {ticket['status']} to {new}")
        try:
            return db.change_status(conn, ticket_id, ticket["status"], new, me["id"], iso(clock()))
        except db.ConflictError:
            raise ApiError(409, "the ticket changed; reload it and try again") from None

    @app.put("/tickets/<int:ticket_id>/assignee")
    def assign(ticket_id):
        me, ticket = visible_ticket(ticket_id)
        require(me, "assign", ticket, "only staff can assign tickets")
        agent_id = body().get("agent_id")
        if not isinstance(agent_id, int) or isinstance(agent_id, bool):
            raise ApiError(422, "agent_id must be a user id")
        try:
            return db.assign(conn, ticket_id, agent_id, me["id"], iso(clock()))
        except ValueError as e:
            raise ApiError(422, str(e)) from None

    # ---- step 15
    @app.delete("/tickets/<int:ticket_id>")
    def remove_ticket(ticket_id):
        me, ticket = visible_ticket(ticket_id)
        require(me, "delete_ticket", ticket, "only admins can delete tickets")
        db.delete_ticket(conn, ticket_id)
        return "", 204

    @app.get("/tickets/<int:ticket_id>/comments")
    def comments(ticket_id):
        me, ticket = visible_ticket(ticket_id)
        return {"items": db.list_comments(conn, ticket_id, include_internal=allowed(me["role"], "internal_note"))}

    @app.post("/tickets/<int:ticket_id>/comments")
    def comment(ticket_id):
        me, ticket = visible_ticket(ticket_id)
        require(me, "comment", ticket)
        data = body()
        text, internal = data.get("body"), data.get("internal", False)
        if not isinstance(text, str) or not 1 <= len(text.strip()) <= 5000 or not isinstance(internal, bool):
            raise ApiError(422, "body must be 1 to 5000 characters and internal true or false")
        if internal:
            require(me, "internal_note", ticket, "only staff can add internal notes")
        return db.add_comment(conn, ticket_id, me["id"], text.strip(), internal, iso(clock())), 201

    @app.get("/stats")
    def stats():
        require(user(), "view_stats", message="only staff can see statistics")
        return db.stats(conn)

How do you know a test suite is any good? Week 72’s mutation testing answer: break the code on purpose, in small realistic ways, and see whether some test fails. A suite that passes on the broken code would have let that bug through. Steps 16 to 18 work exactly like that: the checks run your tests against the correct module and against several broken copies, and your tests must pass on the first and fail on each of the others.

Step 16: testing the rules

Start where tests are cheapest: validate_ticket. Each rule from Monday deserves a test that would fail if that one rule were broken, and the edges are where bugs hide: the shortest title allowed and the one just too short, text that’s long enough only because of its spaces, a field nobody should send. pytest.raises (Week 24) checks that the right error is raised, and caught.value.errors lets the test look at exactly which fields were refused.

Step 17: tables as tests

The permission table and the state machine are tables, and the clearest tests for a table are another table: the expected answer for every case, written out independently, by hand, from the brief. @pytest.mark.parametrize turns one test function into one test case per row, and each case is reported separately, so a failure names the exact role and action that went wrong. Two parametrize decorators on one function multiply: every action with every role.

Writing the expectations out by hand matters: a test that computes its expected answers from PERMISSIONS itself would pass whatever the table said. The test is a second, independent statement of the rules.

Step 18: testing the API

API tests go through HTTP, so they need an app to talk to, and every test should get a fresh one, so tests can’t affect each other through leftover data. That’s what pytest fixtures are for (Week 72): a function marked @pytest.fixture builds something, and any test that names it as a parameter receives a new one. Fixtures can use other fixtures: a client fixture can use a clock fixture, and a test that asks for both gets the same clock its client uses, so it can move time forward and watch a token expire.

Step 19: a CI workflow

Tests only protect you if they run, on every change, before it’s merged. That’s continuous integration (Week 74): on GitHub, a workflow file in .github/workflows/ runs the tests on every push and pull request, and a red cross on the pull request stops the merge. The helpdesk’s workflow, as data (the YAML file holds exactly this structure):

def workflow():
    """The helpdesk's CI workflow, as the data a .github/workflows/ci.yml file would hold."""
    return {
        "name": "CI",
        "on": {"push": {"branches": ["main"]}, "pull_request": {}},
        "permissions": {"contents": "read"},
        "jobs": {
            "test": {
                "runs-on": "ubuntu-latest",
                "timeout-minutes": 15,
                "strategy": {"matrix": {"python-version": ["3.12", "3.13"]}},
                "steps": [
                    {"uses": "actions/checkout@v4"},
                    {"uses": "actions/setup-python@v5", "with": {"python-version": "${{ matrix.python-version }}"}},
                    {"run": "pip install --require-hashes -r requirements.txt"},
                    {"run": "ruff check ."},
                    {"run": "pytest -q"},
                ],
            }
        },
    }

Each line carries a lesson from this phase: run on pull requests as well as pushes (the point is to catch problems before they’re merged); least privilege for the workflow’s token (contents: read, Week 79); a timeout so a hung test can’t run for six hours; a matrix of Python versions, because users run more than one; actions pinned to a version, never a moving branch like main (a compromised action is a supply-chain attack, Week 79); dependencies installed from the hashed lock file; and the linter as well as the tests (Week 74).

Because the workflow is data, it can be checked by code, which is how teams keep dozens of repositories’ workflows consistent.

Step 20: what the tests never reach

Line coverage (Week 72) tells you which lines ran. For an API there’s a simpler, very useful question: which routes did the test suite never call, and which never once succeeded? A route whose tests only ever see 403 and 422 has had its refusals tested but never its main job. Because the route table is data (Monday), a small after-request hook can record every call, and Monday’s match_route can tie each recorded path back to its route.

And here’s the idea behind Step 20 on a handful of recorded calls: each path is tied back to the route that served it, which is all a route-coverage report needs.
