Where we are

All the modules are in the files below, now including storage.py. Today’s code goes in spend.py, and it’s the only module that prints or reads the command line. It has three jobs:

  1. Parse: turn ["add", "12.50", "food", "lunch"] into named values.
  2. Dispatch: call the right core and storage functions, then save if the data changed.
  3. Present: print the results for a person to read, and turn errors into messages.

It holds no rules about money or dates. If you find yourself checking an amount in spend.py, that check belongs in the core.

money.py
"""Reading and showing money and categories."""
import re

AMOUNT = re.compile(r"£?\s*(\d{1,3}(?:,\d{3})+|\d+)(?:\.(\d{1,2}))?")


def parse_amount(text):
    """Text such as '£1,250.50' as whole pence. Raises ValueError with a readable message."""
    text = text.strip()
    if not text.lstrip("£").strip():
        raise ValueError("missing amount")
    m = AMOUNT.fullmatch(text)
    if m is None:
        raise ValueError(f"not an amount: {text!r}")
    total = int(m.group(1).replace(",", "")) * 100 + int((m.group(2) or "").ljust(2, "0"))
    if total == 0:
        raise ValueError("amount must be more than £0")
    return total


def format_pence(pence):
    """1250 as '£12.50', -350 as '-£3.50'."""
    sign = "-" if pence < 0 else ""
    pounds, rest = divmod(abs(pence), 100)
    return f"{sign}£{pounds:,}.{rest:02d}"


def clean_category(text):
    """' Eating   Out ' as 'eating out'. Raises ValueError if it isn't words of letters."""
    shown = text.strip()
    if not shown:
        raise ValueError("missing category")
    cleaned = " ".join(shown.lower().split())
    if len(cleaned) > 20 or not re.fullmatch(r"[a-z]+( [a-z]+)*", cleaned):
        raise ValueError(f"bad category: {shown!r}")
    return cleaned
dates.py
"""Reading dates and months that people type, and stepping through months."""
import re
from datetime import datetime, timedelta

FORMATS = ["%Y-%m-%d", "%d/%m/%Y"]


def parse_date(text, today):
    """'2026-11-02', '02/11/2026', 'today', 'yesterday' or '' as ISO text. No future dates."""
    shown = text.strip()
    word = shown.lower()
    if word in ("", "today"):
        when = today
    elif word == "yesterday":
        when = today - timedelta(days=1)
    else:
        when = None
        for fmt in FORMATS:
            try:
                when = datetime.strptime(word, fmt).date()
                break
            except ValueError:
                pass
        if when is None:
            raise ValueError(f"not a date: {shown!r}")
    if when > today:
        raise ValueError(f"date is in the future: {when.isoformat()}")
    return when.isoformat()


def parse_month(text):
    """'2026-11' or '11/2026' as '2026-11'. Raises ValueError otherwise."""
    shown = text.strip()
    m = re.fullmatch(r"(\d{4})-(\d{2})", shown)
    if m:
        year, month = int(m.group(1)), int(m.group(2))
    else:
        m = re.fullmatch(r"(\d{1,2})/(\d{4})", shown)
        if not m:
            raise ValueError(f"not a month: {shown!r}")
        month, year = int(m.group(1)), int(m.group(2))
    if not 1 <= month <= 12:
        raise ValueError(f"not a month: {shown!r}")
    return f"{year}-{month:02d}"


def next_month(month):
    """'2026-12' gives '2027-01'."""
    year, m = int(month[:4]), int(month[5:])
    if m == 12:
        return f"{year + 1}-01"
    return f"{year}-{m + 1:02d}"
core.py
"""The expense tracker's data and the operations on it."""
from money import parse_amount, clean_category
from dates import parse_date, next_month


def new_data():
    return {"expenses": [], "budgets": {}, "next_id": 1}


def add_expense(data, amount_text, category_text, note, date_text, today):
    """Check every field, then add the expense, or raise one ValueError listing every problem."""
    problems = []
    amount = category = when = None
    try:
        amount = parse_amount(amount_text)
    except ValueError as e:
        problems.append(str(e))
    try:
        category = clean_category(category_text)
    except ValueError as e:
        problems.append(str(e))
    try:
        when = parse_date(date_text, today)
    except ValueError as e:
        problems.append(str(e))
    note = note.strip()
    if len(note) > 60:
        problems.append("note is longer than 60 characters")
    if problems:
        raise ValueError("; ".join(problems))
    expense = {"id": data["next_id"], "date": when, "amount": amount, "category": category, "note": note}
    data["expenses"].append(expense)
    data["next_id"] += 1
    return expense


def remove_expense(data, expense_id):
    """Remove and return the expense with this id. KeyError if there isn't one."""
    for i, e in enumerate(data["expenses"]):
        if e["id"] == expense_id:
            return data["expenses"].pop(i)
    raise KeyError(f"no expense #{expense_id}")


def filter_expenses(data, month=None, category=None):
    """A new list of the matching expenses, newest first (by date, then id)."""
    chosen = [e for e in data["expenses"]
              if (month is None or e["date"][:7] == month)
              and (category is None or e["category"] == category)]
    return sorted(chosen, key=lambda e: (e["date"], e["id"]), reverse=True)


def total(expenses):
    return sum(e["amount"] for e in expenses)


def by_category(expenses):
    """(category, pence) pairs, largest first, alphabetical for ties."""
    totals = {}
    for e in expenses:
        totals[e["category"]] = totals.get(e["category"], 0) + e["amount"]
    return sorted(totals.items(), key=lambda pair: (-pair[1], pair[0]))


def month_summary(data, month):
    """Count, total, a row per category with spending or a budget, and the categories over budget."""
    expenses = filter_expenses(data, month=month)
    spent = dict(by_category(expenses))
    rows = []
    for category in set(spent) | set(data["budgets"]):
        budget = data["budgets"].get(category)
        used = spent.get(category, 0)
        rows.append({"category": category, "spent": used, "budget": budget,
                     "left": None if budget is None else budget - used})
    rows.sort(key=lambda r: (-r["spent"], r["category"]))
    return {"month": month, "count": len(expenses), "total": total(expenses), "rows": rows,
            "over": [r["category"] for r in rows if r["left"] is not None and r["left"] < 0]}


def monthly_totals(data):
    """(month, pence) for every month from the first with spending to the last, gaps as 0."""
    totals = {}
    for e in data["expenses"]:
        totals[e["date"][:7]] = totals.get(e["date"][:7], 0) + e["amount"]
    if not totals:
        return []
    out = []
    month, last = min(totals), max(totals)
    while month <= last:
        out.append((month, totals.get(month, 0)))
        month = next_month(month)
    return out


def set_budget(data, category_text, amount_text):
    """Check both, then set the category's monthly budget. Returns (category, pence)."""
    problems = []
    category = amount = None
    try:
        category = clean_category(category_text)
    except ValueError as e:
        problems.append(str(e))
    try:
        amount = parse_amount(amount_text)
    except ValueError as e:
        problems.append(str(e))
    if problems:
        raise ValueError("; ".join(problems))
    data["budgets"][category] = amount
    return category, amount


def remove_budget(data, category_text):
    """Remove a category's budget and return it. KeyError if it has none."""
    category = clean_category(category_text)
    if category not in data["budgets"]:
        raise KeyError(f"no budget for {category!r}")
    return data["budgets"].pop(category)
storage.py
"""Saving and loading the data, and importing and exporting CSV."""
import csv
import io
import json
import os
import re
from collections import Counter
from datetime import datetime

from core import new_data
from money import parse_amount

KEYS = ["id", "date", "amount", "category", "note"]


def load_data(path):
    """The data saved at path, or new_data() if there's no file. ValueError if it's damaged."""
    if not os.path.exists(path):
        return new_data()
    with open(path, encoding="utf-8") as f:
        try:
            data = json.load(f)
        except json.JSONDecodeError as e:
            raise ValueError(f"{path} is damaged: {e.msg} at line {e.lineno}") from None
    if not (isinstance(data, dict) and isinstance(data.get("expenses"), list)
            and isinstance(data.get("budgets"), dict) and isinstance(data.get("next_id"), int)):
        raise ValueError(f"{path} is not a spend data file")
    for e in data["expenses"]:
        if not (isinstance(e, dict) and list(e) == KEYS and isinstance(e["amount"], int)):
            raise ValueError(f"{path} has a damaged expense: {e!r}")
    return data


def save_data(path, data):
    """Save safely: write a temporary file, then replace the old file with it in one step."""
    tmp = path + ".tmp"
    try:
        with open(tmp, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2, ensure_ascii=False)
        os.replace(tmp, path)
    finally:
        if os.path.exists(tmp):
            os.remove(tmp)


STATEMENT_AMOUNT = re.compile(r"(-?)(£?[\d,]+(?:\.\d{1,2})?)")


def read_statement(text):
    """A bank CSV (Date, Description, Amount) as (rows, problems). Only money out is kept."""
    rows, problems = [], []
    reader = csv.DictReader(io.StringIO(text))
    for row in reader:
        line = reader.line_num
        try:
            when = datetime.strptime(row["Date"].strip(), "%d/%m/%Y").date().isoformat()
        except ValueError:
            problems.append(f"line {line}: bad date {row['Date'].strip()!r}")
            continue
        m = STATEMENT_AMOUNT.fullmatch(row["Amount"].strip())
        try:
            if m is None:
                raise ValueError
            pence = parse_amount(m.group(2))
        except ValueError:
            problems.append(f"line {line}: bad amount {row['Amount'].strip()!r}")
            continue
        if m.group(1) != "-":
            continue
        rows.append({"date": when, "amount": pence, "description": " ".join(row["Description"].split())})
    return rows, problems


RULES = [
    (r"tesco|sainsbury|aldi|lidl|co-?op", "food"),
    (r"\btfl\b|trainline|uber|bus", "travel"),
    (r"netflix|spotify|cinema", "fun"),
    (r"rent|lettings", "rent"),
]


def guess_category(description, rules=RULES):
    for pattern, category in rules:
        if re.search(pattern, description, re.IGNORECASE):
            return category
    return "other"


def import_statement(data, text, rules=RULES):
    rows, problems = read_statement(text)
    existing = Counter((e["date"], e["amount"], e["note"]) for e in data["expenses"])
    added = skipped = 0
    for r in rows:
        note = r["description"][:60]
        key = (r["date"], r["amount"], note)
        if existing[key] > 0:
            existing[key] -= 1
            skipped += 1
            continue
        data["expenses"].append({"id": data["next_id"], "date": r["date"], "amount": r["amount"],
                                 "category": guess_category(r["description"], rules), "note": note})
        data["next_id"] += 1
        added += 1
    return {"added": added, "skipped": skipped, "problems": problems}


def export_csv(data, path):
    with open(path, "w", newline="", encoding="utf-8") as f:
        w = csv.writer(f)
        w.writerow(["id", "date", "amount", "category", "note"])
        for e in sorted(data["expenses"], key=lambda e: (e["date"], e["id"])):
            w.writerow([e["id"], e["date"], f"{e['amount'] // 100}.{e['amount'] % 100:02d}", e["category"], e["note"]])

Step 23: the parser

argparse (Week 21) turns the command line into named values, writes the --help text, and rejects nonsense with a usage message and exit status 2. Subcommands, like Git’s git add and git commit, each get their own arguments:

Command Arguments
add amount, category, an optional note, and --date (default "", meaning today)
remove id, a whole number
list optional --month and --category
summary optional --month (default: this month)
budget optional category and amount (no category lists the budgets; none as the amount removes one)
import statement, a CSV file
export out, a CSV file to write

and before the command, --file (default spend.json) chooses the data file, so you can keep separate files or test without touching your real one.

Step 24: commands and main

Each command gets a small function taking (data, args, today). It does its job, prints the result, and returns True if it changed the data (so it needs saving) or False if it only looked. A dictionary from command names to functions replaces a long if/elif chain, as in Week 24’s refactoring day.

main ties it together. Two details matter:

  • It takes argv and today as parameters, defaulting to the real command line and the real date. Tests pass their own; the real program passes nothing. (parse_args(None) reads sys.argv.)
  • It returns an exit status instead of calling sys.exit itself: 0 for success, 1 for an error. Only the last two lines of the file, which run when it’s used as a script, pass that to sys.exit. Errors are printed to standard error as spend: ..., the convention for command-line tools (Week 21), so they don’t get mixed into output someone is saving to a file.

Step 24: the commands

Write the command functions and main(argv=None, today=None). The starter has everything from Steps 21 to 23. The outputs:

  • add: Added #14: ... (using describe); remove: Removed #14: ...;
  • list: the expense_table of the filtered expenses (month through parse_month, category through clean_category, when given);
  • summary: the summary_lines for the month given, or today’s month;
  • budget with no category: each budget as f" {category:<12} {format_pence(pence):>11} a month", alphabetically, or No budgets.; with a category and an amount: Budget for food: £60.00 a month; with none: Removed the budget for food; with a category but no amount: a ValueError("give an amount, or 'none' to remove the budget");

  • export: Exported 10 expenses to out.csv.

main parses the arguments, loads the data, runs the command, saves if it returns True, and returns 0. A ValueError, KeyError or OSError (a missing statement file, say) prints spend: and the message to standard error (for a KeyError, its args[0], which has no quotes round it) and returns 1, saving nothing.

Why main takes argv and today

Because main takes its inputs as parameters, a test can run the whole program, every layer from parsing to saving, without a terminal and on any date it likes. The real program passes nothing, so it gets the real command line and the real date. It’s the same idea as parse_date(text, today) on Day 1, applied to the whole program.

spend.py
"""spend: a command-line expense tracker."""
import argparse
import sys
from datetime import date, datetime

from core import (add_expense, remove_expense, filter_expenses, total, month_summary,
                  set_budget, remove_budget)
from dates import parse_month
from money import format_pence, clean_category
from storage import load_data, save_data, import_statement, export_csv


def describe(e):
    """'#14: £12.50 food (lunch with Tola) on 2026-11-02'"""
    note = f" ({e['note']})" if e["note"] else ""
    return f"#{e['id']}: {format_pence(e['amount'])} {e['category']}{note} on {e['date']}"


def expense_table(expenses):
    """The lines of a table of expenses, with a total line. No expenses gives one line saying so."""
    if not expenses:
        return ["No expenses."]
    lines = [f"{'#':>4}  {'Date':<10}  {'Amount':>11}  {'Category':<12}  Note"]
    for e in expenses:
        lines.append(f"{e['id']:>4}  {e['date']:<10}  {format_pence(e['amount']):>11}  {e['category']:<12}  {e['note']}".rstrip())
    count = "1 expense" if len(expenses) == 1 else f"{len(expenses)} expenses"
    lines.append(f"{'':>4}  {'Total':<10}  {format_pence(total(expenses)):>11}  {count}")
    return lines


def month_name(month):
    """'2026-11' as 'November 2026'."""
    return datetime.strptime(month, "%Y-%m").strftime("%B %Y")


def summary_lines(s):
    """The lines of a month's summary, from month_summary's result."""
    count = "1 expense" if s["count"] == 1 else f"{s['count']} expenses"
    lines = [f"{month_name(s['month'])}: {count}, {format_pence(s['total'])}"]
    for r in s["rows"]:
        line = f"  {r['category']:<12}  {format_pence(r['spent']):>11}"
        if r["budget"] is not None:
            line += f"  of {format_pence(r['budget']):>9}  "
            if r["left"] < 0:
                line += f"OVER by {format_pence(-r['left'])}"
            else:
                line += f"{format_pence(r['left'])} left"
        lines.append(line)
    return lines


def build_parser():
    parser = argparse.ArgumentParser(prog="spend", description="Track what you spend.")
    parser.add_argument("--file", default="spend.json", help="the data file (default: spend.json)")
    sub = parser.add_subparsers(dest="command", required=True)

    p = sub.add_parser("add", help="record an expense")
    p.add_argument("amount", help="for example 12.50 or £1,250")
    p.add_argument("category", help="for example food")
    p.add_argument("note", nargs="?", default="", help="what it was for")
    p.add_argument("--date", default="", help="YYYY-MM-DD, DD/MM/YYYY, today or yesterday")

    p = sub.add_parser("remove", help="remove an expense by its number")
    p.add_argument("id", type=int)

    p = sub.add_parser("list", help="list expenses, newest first")
    p.add_argument("--month", help="YYYY-MM or MM/YYYY")
    p.add_argument("--category")

    p = sub.add_parser("summary", help="a month's spending by category")
    p.add_argument("--month", help="YYYY-MM or MM/YYYY (default: this month)")

    p = sub.add_parser("budget", help="show, set or remove monthly budgets")
    p.add_argument("category", nargs="?")
    p.add_argument("amount", nargs="?", help="an amount, or 'none' to remove the budget")

    p = sub.add_parser("import", help="import a bank statement CSV")
    p.add_argument("statement")

    p = sub.add_parser("export", help="export everything as CSV")
    p.add_argument("out")
    return parser


def cmd_add(data, args, today):
    e = add_expense(data, args.amount, args.category, args.note, args.date, today)
    print("Added", describe(e))
    return True


def cmd_remove(data, args, today):
    print("Removed", describe(remove_expense(data, args.id)))
    return True


def cmd_list(data, args, today):
    month = parse_month(args.month) if args.month else None
    category = clean_category(args.category) if args.category else None
    for line in expense_table(filter_expenses(data, month, category)):
        print(line)
    return False


def cmd_summary(data, args, today):
    month = parse_month(args.month) if args.month else today.isoformat()[:7]
    for line in summary_lines(month_summary(data, month)):
        print(line)
    return False


def cmd_budget(data, args, today):
    if args.category is None:
        if not data["budgets"]:
            print("No budgets.")
        for category, pence in sorted(data["budgets"].items()):
            print(f"  {category:<12}  {format_pence(pence):>11} a month")
        return False
    if args.amount is None:
        raise ValueError("give an amount, or 'none' to remove the budget")
    if args.amount.strip().lower() == "none":
        category = clean_category(args.category)
        remove_budget(data, category)
        print(f"Removed the budget for {category}")
    else:
        category, pence = set_budget(data, args.category, args.amount)
        print(f"Budget for {category}: {format_pence(pence)} a month")
    return True


def cmd_import(data, args, today):
    with open(args.statement, newline="", encoding="utf-8") as f:
        result = import_statement(data, f.read())
    print(f"Imported {result['added']}, skipped {result['skipped']} already there")
    for problem in result["problems"]:
        print(f"  could not read {problem}")
    return result["added"] > 0


def cmd_export(data, args, today):
    export_csv(data, args.out)
    print(f"Exported {len(data['expenses'])} expenses to {args.out}")
    return False


COMMANDS = {"add": cmd_add, "remove": cmd_remove, "list": cmd_list, "summary": cmd_summary,
            "budget": cmd_budget, "import": cmd_import, "export": cmd_export}


def main(argv=None, today=None):
    """Run one command. Returns the exit status: 0 for success, 1 for an error."""
    args = build_parser().parse_args(argv)
    today = today or date.today()
    try:
        data = load_data(args.file)
        if COMMANDS[args.command](data, args, today):
            save_data(args.file, data)
    except (ValueError, KeyError, OSError) as e:
        message = e.args[0] if isinstance(e, KeyError) else str(e)
        print(f"spend: {message}", file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())
