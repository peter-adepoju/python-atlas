errors.py
"""Every error the library raises on purpose."""


class LibraryError(Exception):
    """Base class for the library's own errors."""


class InvalidInput(LibraryError, ValueError):
    """Something typed or loaded doesn't make sense."""


class NotFound(LibraryError, LookupError):
    """There's no book or member with that id."""


class NotAllowed(LibraryError):
    """The library's rules don't allow this right now."""


class StorageError(LibraryError):
    """The library's data couldn't be read or saved."""
values.py
"""Reading and checking the values people type, and showing money."""
from errors import InvalidInput

KINDS = {"B": "book", "M": "member"}


def clean_isbn(text):
    """'978-0-14-118776-1' as '9780141187761'. Raises InvalidInput."""
    shown = text.strip()
    digits = shown.replace("-", "").replace(" ", "")
    if len(digits) != 13 or not (digits.isascii() and digits.isdigit()):
        raise InvalidInput(f"an ISBN has 13 digits: {shown!r}")
    total = sum(int(d) * (3 if i % 2 else 1) for i, d in enumerate(digits))
    if total % 10 != 0:
        raise InvalidInput(f"wrong check digit: {shown!r}")
    return digits


def clean_text(text, what, longest):
    """'  Things   Fall Apart ' as 'Things Fall Apart'. Raises InvalidInput."""
    cleaned = " ".join(text.split())
    if not cleaned:
        raise InvalidInput(f"{what} is missing")
    if len(cleaned) > longest:
        raise InvalidInput(f"{what} is longer than {longest} characters")
    return cleaned


def parse_id(text, kind):
    """' b12 ' as 'B12', for kind 'B'. Raises InvalidInput."""
    cleaned = text.strip().upper()
    number = cleaned[1:]
    if not cleaned.startswith(kind) or not (number.isascii() and number.isdigit()) or number.startswith("0"):
        raise InvalidInput(f"not a {KINDS[kind]} id: {text.strip()!r}")
    return cleaned


def parse_pence(text):
    """'£1.20' or '0.6' as whole pence. Raises InvalidInput."""
    shown = text.strip()
    pounds, dot, pence = shown.removeprefix("£").partition(".")
    if (not (pounds.isascii() and pounds.isdigit())
            or (dot and not (pence.isascii() and pence.isdigit() and len(pence) <= 2))):
        raise InvalidInput(f"not an amount: {shown!r}")
    return int(pounds) * 100 + (int(pence.ljust(2, "0")) if dot else 0)


def format_pence(pence):
    """60 as '£0.60', -350 as '-£3.50'."""
    sign = "-" if pence < 0 else ""
    pounds, rest = divmod(abs(pence), 100)
    return f"{sign}£{pounds:,}.{rest:02d}"


def check_all(**checks):
    """Run each check (a function of no arguments); return their results by name, or raise one
    InvalidInput listing every problem, in order."""
    results, problems = {}, []
    for name, check in checks.items():
        try:
            results[name] = check()
        except InvalidInput as err:
            problems.append(str(err))
    if problems:
        raise InvalidInput("; ".join(problems))
    return results
domain.py
"""The library's things and rules. No input, output or storage here."""
from dataclasses import dataclass, field
from datetime import date, timedelta

from errors import InvalidInput, NotAllowed
from values import format_pence

LOAN_DAYS = 21
FINE_PER_DAY = 20      # pence for each day late
MAX_FINE = 500         # the most one loan can cost
FINE_LIMIT = 500       # owing this much stops a member borrowing


@dataclass
class Book:
    book_id: str
    isbn: str
    title: str
    author: str
    borrower: str | None = None
    holds: list[str] = field(default_factory=list)

    @property
    def on_loan(self):
        return self.borrower is not None

    def describe(self):
        return f"{self.book_id}: {self.title} by {self.author}"


@dataclass
class Member:
    member_id: str
    name: str
    max_loans: int = 3
    owes: int = 0


@dataclass
class Loan:
    loan_id: int
    book_id: str
    member_id: str
    out: date
    due: date
    returned: date | None = None

    @property
    def is_open(self):
        return self.returned is None

    def days_late(self, today):
        end = self.returned or today
        return max(0, (end - self.due).days)

    def fine(self, today):
        return min(self.days_late(today) * FINE_PER_DAY, MAX_FINE)


def check_can_borrow(member, book, open_loans, today):
    """Raise NotAllowed with the first reason the member can't borrow the book today."""
    if book.on_loan:
        raise NotAllowed(f"{book.book_id} is on loan")
    if book.holds and book.holds[0] != member.member_id:
        raise NotAllowed(f"{book.book_id} is being kept for someone else")
    if len(open_loans) >= member.max_loans:
        raise NotAllowed(f"{member.name} already has {len(open_loans)} books")
    if any(loan.days_late(today) for loan in open_loans):
        raise NotAllowed(f"{member.name} has an overdue book")
    if member.owes >= FINE_LIMIT:
        raise NotAllowed(f"{member.name} owes {format_pence(member.owes)}")


def lend(book, member, open_loans, today, loan_id):
    """Lend the book, or raise NotAllowed. Returns the new Loan."""
    check_can_borrow(member, book, open_loans, today)
    if book.holds:
        book.holds.pop(0)
    book.borrower = member.member_id
    return Loan(loan_id, book.book_id, member.member_id, today, today + timedelta(days=LOAN_DAYS))


def give_back(book, loan, member, today):
    """Close the loan and add any fine to what the member owes. Returns the fine."""
    assert loan.is_open and loan.book_id == book.book_id and loan.member_id == member.member_id
    loan.returned = today
    book.borrower = None
    fine = loan.fine(today)
    member.owes += fine
    return fine


def reserve(book, member):
    """Join the queue for a book. Returns the member's place in it."""
    if not book.on_loan and not book.holds:
        raise NotAllowed(f"{book.book_id} is on the shelf: borrow it instead")
    if book.borrower == member.member_id:
        raise NotAllowed(f"{member.name} already has {book.book_id}")
    if member.member_id in book.holds:
        raise NotAllowed(f"{member.name} is already waiting for {book.book_id}")
    book.holds.append(member.member_id)
    return len(book.holds)


def pay(member, pence):
    """Take a payment towards the member's fines. Returns what they still owe."""
    if pence <= 0:
        raise InvalidInput("a payment must be more than £0")
    if pence > member.owes:
        raise NotAllowed(f"{member.name} only owes {format_pence(member.owes)}")
    member.owes -= pence
    return member.owes
store.py
"""Where the library's data is kept: in memory, or in a JSON file."""
import json
import logging
import os
from dataclasses import asdict
from datetime import date

from domain import Book, Loan, Member
from errors import NotFound, StorageError

log = logging.getLogger("shelf.store")


class MemoryStore:
    """Keeps everything in dictionaries. commit() does nothing: there's nowhere to save."""

    def __init__(self):
        self.books = {}
        self.members = {}
        self.loans = []
        self.counters = {"B": 0, "M": 0, "L": 0}

    def next_id(self, kind):
        """'B1', 'B2', ... for books, 'M1', ... for members, and 1, 2, ... for loans. Never reused."""
        self.counters[kind] += 1
        number = self.counters[kind]
        return number if kind == "L" else f"{kind}{number}"

    def add_book(self, book):
        self.books[book.book_id] = book

    def add_member(self, member):
        self.members[member.member_id] = member

    def add_loan(self, loan):
        self.loans.append(loan)

    def book(self, book_id):
        try:
            return self.books[book_id]
        except KeyError:
            raise NotFound(f"no book {book_id}") from None

    def member(self, member_id):
        try:
            return self.members[member_id]
        except KeyError:
            raise NotFound(f"no member {member_id}") from None

    def open_loans(self, member_id=None):
        return [loan for loan in self.loans
                if loan.is_open and (member_id is None or loan.member_id == member_id)]

    def open_loan_for(self, book_id):
        for loan in self.loans:
            if loan.is_open and loan.book_id == book_id:
                return loan
        raise NotFound(f"{book_id} isn't on loan")

    def commit(self):
        pass


def to_data(store):
    """Everything in the store as plain dictionaries and lists, ready for JSON."""
    def loan_data(loan):
        d = asdict(loan)
        d["out"], d["due"] = loan.out.isoformat(), loan.due.isoformat()
        d["returned"] = loan.returned.isoformat() if loan.returned else None
        return d
    return {
        "books": [asdict(b) for b in store.books.values()],
        "members": [asdict(m) for m in store.members.values()],
        "loans": [loan_data(loan) for loan in store.loans],
        "counters": dict(store.counters),
    }


def fill(store, data):
    """Load data made by to_data into an empty store."""
    for d in data["books"]:
        store.add_book(Book(**d))
    for d in data["members"]:
        store.add_member(Member(**d))
    for d in data["loans"]:
        d = dict(d)
        d["out"], d["due"] = date.fromisoformat(d["out"]), date.fromisoformat(d["due"])
        d["returned"] = date.fromisoformat(d["returned"]) if d["returned"] else None
        store.add_loan(Loan(**d))
    store.counters = dict(data["counters"])


class JsonStore(MemoryStore):
    """A MemoryStore that loads from a JSON file when it's made, and saves on commit()."""

    def __init__(self, path):
        super().__init__()
        self.path = path
        self._load()

    def _load(self):
        try:
            with open(self.path, encoding="utf-8") as f:
                data = json.load(f)
        except FileNotFoundError:
            log.info("no file at %s yet: starting empty", self.path)
            return
        except (OSError, ValueError) as err:
            raise StorageError(f"can't read {self.path}") from err
        try:
            fill(self, data)
        except (KeyError, TypeError, ValueError) as err:
            raise StorageError(f"{self.path} is damaged") from err
        log.info("loaded %d books, %d members and %d loans from %s",
                 len(self.books), len(self.members), len(self.loans), self.path)

    def commit(self):
        """Save everything safely: write a temporary file, then replace the old file in one step."""
        temp = self.path + ".tmp"
        try:
            with open(temp, "w", encoding="utf-8") as f:
                json.dump(to_data(self), f, indent=1, ensure_ascii=False)
            os.replace(temp, self.path)
        except OSError as err:
            raise StorageError(f"can't save to {self.path}") from err
        finally:
            if os.path.exists(temp):
                os.remove(temp)
        log.debug("saved %s", self.path)

Each use case in the service follows the same four beats, which makes them short and easy to check:

  1. Read the typed values (parse_id, parse_pence, check_all), raising InvalidInput.
  2. Find the objects in the store, raising NotFound.
  3. Ask the domain to do the work, which raises NotAllowed or changes the objects.
  4. Commit, and log what happened.

The service never prints and never catches LibraryError: it passes refusals up to the interface, which decides how to show them (Week 37’s “log once, where it’s handled” applies to messages for people too).

service.py
"""The library's use cases. Takes and returns values; never prints."""
import logging

from domain import Book, Member, check_can_borrow, give_back, lend, pay, reserve
from values import check_all, clean_isbn, clean_text, parse_id, parse_pence

log = logging.getLogger("shelf.service")


class Library:
    def __init__(self, store, today):
        self.store = store
        self.today = today          # a function returning today's date

    def book(self, book_id):
        return self.store.book(parse_id(book_id, "B"))

    def member(self, member_id):
        return self.store.member(parse_id(member_id, "M"))

    def add_book(self, isbn, title, author):
        v = check_all(isbn=lambda: clean_isbn(isbn),
                      title=lambda: clean_text(title, "title", 100),
                      author=lambda: clean_text(author, "author", 60))
        book = Book(self.store.next_id("B"), v["isbn"], v["title"], v["author"])
        self.store.add_book(book)
        self.store.commit()
        log.info("added %s", book.book_id)
        return book

    def join(self, name):
        member = Member(self.store.next_id("M"), clean_text(name, "name", 60))
        self.store.add_member(member)
        self.store.commit()
        log.info("added %s", member.member_id)
        return member

    def lend(self, book_id, member_id):
        book, member = self.book(book_id), self.member(member_id)
        open_loans, today = self.store.open_loans(member.member_id), self.today()
        check_can_borrow(member, book, open_loans, today)    # first, so a refusal uses up no loan id
        loan = lend(book, member, open_loans, today, self.store.next_id("L"))
        self.store.add_loan(loan)
        self.store.commit()
        log.info("lent %s to %s", book.book_id, member.member_id)
        return loan

    def give_back(self, book_id):
        book = self.book(book_id)
        loan = self.store.open_loan_for(book.book_id)
        fine = give_back(book, loan, self.store.member(loan.member_id), self.today())
        self.store.commit()
        log.info("%s returned, fine %d", book.book_id, fine)
        return loan, fine

    def reserve(self, book_id, member_id):
        place = reserve(self.book(book_id), self.member(member_id))
        self.store.commit()
        return place

    def pay(self, member_id, amount):
        left = pay(self.member(member_id), parse_pence(amount))
        self.store.commit()
        return left

    def overdue(self):
        """(loan, days late) for every late loan, most days late first, then by book id."""
        today = self.today()
        late = [(loan, loan.days_late(today)) for loan in self.store.open_loans() if loan.days_late(today)]
        return sorted(late, key=lambda pair: (-pair[1], pair[0].book_id))

    def find(self, words):
        """Books whose title or author contains every word (in any case), by id number."""
        wanted = words.lower().split()
        found = [b for b in self.store.books.values()
                 if all(w in f"{b.title} {b.author}".lower() for w in wanted)]
        return sorted(found, key=lambda b: int(b.book_id[1:]))

Step 23: reading a line

Exercise20 points
Step 23: reading a line

execute is given. Write handle(line, library), which turns one typed line into a reply:

  1. Split the line with shlex.split (Week 21), so quoted titles stay together. An unclosed quote makes it raise ValueError: reply "error: " and its message.
  2. A blank line gets the reply "".
  3. The first word, in lower case, is the command. If it isn’t in USAGE, reply "unknown command: WORD".
  4. USAGE gives each command’s shape and how many arguments it takes (None means one or more). With the wrong number, reply "usage: COMMAND SHAPE" (with no space at the end when the shape is empty).
  5. Otherwise return execute’s reply, turning any LibraryError into "error: " and its message. Nothing else is caught.

Why catch LibraryError but nothing else? Every refusal the program makes on purpose is a LibraryError with a message written for people, so turning it into error: ... is always right. Anything else is a bug: catching it would show the volunteer a confusing message and hide the bug from whoever could fix it.

Step 24: wiring it together

The composition root (Week 36) is the one place that knows which store, which clock and which log level the program uses. Everything else receives them.

Exercise15 points
Step 24: wiring it together

Write the composition root:

  • build(env, today=date.today) returns a Library, wired from the settings in env (a dictionary, like os.environ). SHELF_LOG names the logging level (default "WARNING", any case; an unknown name means WARNING): call logging.basicConfig with that level, the format "%(levelname)s %(name)s: %(message)s" and stream=sys.stderr. SHELF_DATA is the data file (default "library.json"); the special value ":memory:" means a MemoryStore instead of a JsonStore.
  • main(lines, env, today=date.today) builds a library and prints the reply to each line, skipping empty replies.

"""The library's use cases. Takes and returns values; never prints."""
import logging

from domain import Book, Member, check_can_borrow, give_back, lend, pay, reserve
from values import check_all, clean_isbn, clean_text, parse_id, parse_pence

log = logging.getLogger("shelf.service")


class Library:
    def __init__(self, store, today):
        self.store = store
        self.today = today          # a function returning today's date

    def book(self, book_id):
        return self.store.book(parse_id(book_id, "B"))

    def member(self, member_id):
        return self.store.member(parse_id(member_id, "M"))

    def add_book(self, isbn, title, author):
        v = check_all(isbn=lambda: clean_isbn(isbn),
                      title=lambda: clean_text(title, "title", 100),
                      author=lambda: clean_text(author, "author", 60))
        book = Book(self.store.next_id("B"), v["isbn"], v["title"], v["author"])
        self.store.add_book(book)
        self.store.commit()
        log.info("added %s", book.book_id)
        return book

    def join(self, name):
        member = Member(self.store.next_id("M"), clean_text(name, "name", 60))
        self.store.add_member(member)
        self.store.commit()
        log.info("added %s", member.member_id)
        return member

    def lend(self, book_id, member_id):
        book, member = self.book(book_id), self.member(member_id)
        open_loans, today = self.store.open_loans(member.member_id), self.today()
        check_can_borrow(member, book, open_loans, today)    # first, so a refusal uses up no loan id
        loan = lend(book, member, open_loans, today, self.store.next_id("L"))
        self.store.add_loan(loan)
        self.store.commit()
        log.info("lent %s to %s", book.book_id, member.member_id)
        return loan

    def give_back(self, book_id):
        book = self.book(book_id)
        loan = self.store.open_loan_for(book.book_id)
        fine = give_back(book, loan, self.store.member(loan.member_id), self.today())
        self.store.commit()
        log.info("%s returned, fine %d", book.book_id, fine)
        return loan, fine

    def reserve(self, book_id, member_id):
        place = reserve(self.book(book_id), self.member(member_id))
        self.store.commit()
        return place

    def pay(self, member_id, amount):
        left = pay(self.member(member_id), parse_pence(amount))
        self.store.commit()
        return left

    def overdue(self):
        """(loan, days late) for every late loan, most days late first, then by book id."""
        today = self.today()
        late = [(loan, loan.days_late(today)) for loan in self.store.open_loans() if loan.days_late(today)]
        return sorted(late, key=lambda pair: (-pair[1], pair[0].book_id))

    def find(self, words):
        """Books whose title or author contains every word (in any case), by id number."""
        wanted = words.lower().split()
        found = [b for b in self.store.books.values()
                 if all(w in f"{b.title} {b.author}".lower() for w in wanted)]
        return sorted(found, key=lambda b: int(b.book_id[1:]))

"""The text interface: turns typed lines into calls, and results into text."""
import shlex

from errors import LibraryError
from values import format_pence

USAGE = {
    "add": ("ISBN TITLE AUTHOR", 3),
    "join": ("NAME", 1),
    "lend": ("BOOK MEMBER", 2),
    "return": ("BOOK", 1),
    "reserve": ("BOOK MEMBER", 2),
    "pay": ("MEMBER AMOUNT", 2),
    "overdue": ("", 0),
    "find": ("WORDS", None),
}


def handle(line, library):
    """One typed line in, one piece of text out."""
    try:
        words = shlex.split(line)
    except ValueError as err:
        return f"error: {err}"
    if not words:
        return ""
    command, args = words[0].lower(), words[1:]
    if command not in USAGE:
        return f"unknown command: {command}"
    shape, count = USAGE[command]
    if (count is None and not args) or (count is not None and len(args) != count):
        return f"usage: {command} {shape}".rstrip()
    try:
        return execute(command, args, library)
    except LibraryError as err:
        return f"error: {err}"


def execute(command, args, library):
    """Carry out one command, and describe the result."""
    if command == "add":
        return f"added {library.add_book(*args).describe()}"
    if command == "join":
        member = library.join(args[0])
        return f"joined {member.member_id}: {member.name}"
    if command == "lend":
        loan = library.lend(*args)
        return f"{loan.book_id} lent to {library.member(loan.member_id).name}, due {loan.due.isoformat()}"
    if command == "return":
        loan, fine = library.give_back(args[0])
        text = f"{loan.book_id} returned"
        if fine:
            text += f" {loan.days_late(loan.returned)} days late: fine {format_pence(fine)}"
        holds = library.book(loan.book_id).holds
        if holds:
            text += f"; keep it for {library.member(holds[0]).name}"
        return text
    if command == "reserve":
        place = library.reserve(*args)
        book, member = library.book(args[0]), library.member(args[1])
        return f"{member.name} is number {place} in the queue for {book.book_id}"
    if command == "pay":
        left = library.pay(*args)
        return f"{library.member(args[0]).name} now owes {format_pence(left)}"
    if command == "overdue":
        late = library.overdue()
        if not late:
            return "nothing is overdue"
        return "\n".join(f"{loan.book_id} {library.book(loan.book_id).title}: "
                         f"{library.member(loan.member_id).name}, {days} days late" for loan, days in late)
    if command == "find":
        books = library.find(" ".join(args))
        if not books:
            return "no books found"
        return "\n".join(b.describe() + (" (on loan)" if b.on_loan else "") for b in books)

Trying it out

Here’s the whole program: two sessions, as if a volunteer had closed shelf and opened it again three and a half weeks later. The data survives in library.json between them (open it after running). Change SHELF_DATA to ":memory:", or add "SHELF_LOG": "info" to the settings, to see the composition root at work.

One line, five layers

Follow lend b1 m2 through the program. handle splits it and checks the number of arguments; execute calls library.lend("b1", "m2"); the service reads the ids into "B1" and "M2", finds the book and member in the store, and asks the domain’s lend; the domain’s check_can_borrow finds the book on loan and raises NotAllowed("B1 is on loan"); nothing commits; the exception passes back up through the service and execute, untouched, until handle turns it into error: B1 is on loan. Each layer did one job, and only the top one decided what the volunteer sees.
