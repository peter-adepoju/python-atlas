Week 77, Day 3

REST design

Flask and FastAPI will let you build any API you like, including a confusing one. REST is the set of conventions that make HTTP APIs predictable: resources named by nouns, methods that mean what HTTP says they mean, status codes that tell the truth, and representations, errors, pagination and versions that work the same way across every endpoint. A well-designed API can be guessed: if GET /orders/7 exists, so does DELETE /orders/7. Today is the design side: naming, status codes, problem-details errors, safe retries with idempotency keys, conditional updates with ETags, opaque cursors, and changing an API without breaking its clients.

By the end of today you can

  • Name resources and map operations onto methods and status codes the REST way
  • Return consistent errors as problem details (RFC 9457)
  • Make retries safe with idempotency keys, and lost updates impossible with ETags and If-Match
  • Paginate with opaque cursors, and evolve an API without breaking its clients

import io
import json
import re
from contextvars import ContextVar
from http import HTTPStatus
from urllib.parse import parse_qs, urlencode, urlsplit

_current = ContextVar("request")


class HTTPException(Exception):
    def __init__(self, code, description=None):
        self.code = code
        self.name = HTTPStatus(code).phrase
        self.description = description or self.name
        super().__init__(f"{code} {self.name}: {self.description}")


def abort(code, description=None):
    raise HTTPException(code, description)


class Headers:
    """Case-insensitive headers."""
    def __init__(self, items=()):
        self._items = [(k, str(v)) for k, v in (items.items() if isinstance(items, dict) else items)]

    def get(self, name, default=None):
        for k, v in self._items:
            if k.lower() == name.lower():
                return v
        return default

    def __getitem__(self, name):
        value = self.get(name)
        if value is None:
            raise KeyError(name)
        return value

    def __contains__(self, name):
        return self.get(name) is not None

    def __setitem__(self, name, value):
        self._items = [(k, v) for k, v in self._items if k.lower() != name.lower()] + [(name, str(value))]

    def items(self):
        return list(self._items)


class Args(dict):
    """Query or form values: .get(name) gives the first value; .getlist(name) all of them."""
    def __init__(self, pairs):
        super().__init__()
        self._lists = {}
        for k, v in pairs:
            self._lists.setdefault(k, []).append(v)
            dict.setdefault(self, k, v)

    def get(self, key, default=None, type=None):
        if key not in self:
            return default
        value = self[key]
        if type is not None:
            try:
                return type(value)
            except (TypeError, ValueError):
                return default
        return value

    def getlist(self, key):
        return list(self._lists.get(key, []))


class Request:
    def __init__(self, environ):
        self.environ = environ
        self.method = environ["REQUEST_METHOD"].upper()
        self.path = environ.get("PATH_INFO", "/") or "/"
        self.args = Args(_pairs(environ.get("QUERY_STRING", "")))
        items = [(k[5:].replace("_", "-").title(), v) for k, v in environ.items() if k.startswith("HTTP_")]
        if environ.get("CONTENT_TYPE"):
            items.append(("Content-Type", environ["CONTENT_TYPE"]))
        self.headers = Headers(items)
        length = int(environ.get("CONTENT_LENGTH") or 0)
        self.data = environ["wsgi.input"].read(length) if length else b""
        self.content_type = environ.get("CONTENT_TYPE", "")
        self.form = Args(_pairs(self.data.decode("utf-8"))) if self.content_type.startswith("application/x-www-form-urlencoded") else Args([])
        self.view_args = {}
        cookie = environ.get("HTTP_COOKIE", "")
        self.cookies = dict(p.strip().split("=", 1) for p in cookie.split(";") if "=" in p)

    @property
    def is_json(self):
        return self.content_type.split(";")[0].strip() == "application/json"

    def get_json(self, silent=False, force=False):
        if not (self.is_json or force):
            if silent:
                return None
            abort(415, "expected a JSON body (Content-Type: application/json)")
        try:
            return json.loads(self.data or b"null")
        except ValueError:
            if silent:
                return None
            abort(400, "the body isn't valid JSON")

    @property
    def json(self):
        return self.get_json()


def _pairs(text):
    return [(k, v) for k, vs in parse_qs(text, keep_blank_values=True).items() for v in vs]


class _Proxy:
    def __init__(self, getter):
        object.__setattr__(self, "_getter", getter)

    def __getattr__(self, name):
        return getattr(self._getter(), name)

    def __setattr__(self, name, value):
        setattr(self._getter(), name, value)

    except LookupError:

def _g():
    try:

class _G:
    def get(self, name, default=None):
        return self.__dict__.get(name, default)


request = _Proxy(_request)
g = _Proxy(_g)


class Response:
    def __init__(self, response=b"", status=200, headers=None, mimetype=None, content_type=None):
        if isinstance(response, str):
            response = response.encode("utf-8")
        self.data = response
        self.status_code = status
        self.headers = Headers(headers or {})
        if content_type or mimetype or "Content-Type" not in self.headers:
            ctype = content_type or (f"{mimetype}; charset=utf-8" if mimetype and mimetype.startswith("text/") else mimetype) or "text/html; charset=utf-8"
            self.headers["Content-Type"] = ctype

    @property
    def status(self):
        return f"{self.status_code} {HTTPStatus(self.status_code).phrase}"

    @property
    def text(self):
        return self.data.decode("utf-8")

    def get_json(self):
        return json.loads(self.data)

    @property
    def json(self):
        return self.get_json()

    def set_cookie(self, key, value="", max_age=None, path="/", secure=False, httponly=False, samesite=None):
        parts = [f"{key}={value}"]
        if max_age is not None:
            parts.append(f"Max-Age={max_age}")
        parts.append(f"Path={path}")
        if secure:
            parts.append("Secure")
        if httponly:
            parts.append("HttpOnly")
        if samesite:
            parts.append(f"SameSite={samesite}")
        self.headers._items.append(("Set-Cookie", "; ".join(parts)))


def jsonify(*args, **kwargs):
    data = args[0] if len(args) == 1 else (list(args) if args else kwargs)
    return Response(json.dumps(data), headers={"Content-Type": "application/json"})


def redirect(location, code=302):
    return Response(b"", code, {"Location": location})


def make_response(*args):
    rv = args[0] if len(args) == 1 else args
    return _App._to_response(rv)


def url_for(endpoint, **values):
    return _current.get()[2].url_for(endpoint, **values)


_CONVERTERS = {"int": (r"\d+", int), "float": (r"\d+(?:\.\d+)?", float), "path": (r".+", str), "string": (r"[^/]+", str)}


class _Rule:
    def __init__(self, rule, endpoint, view, methods):
        self.rule, self.endpoint, self.view = rule, endpoint, view
        self.methods = {m.upper() for m in methods} | ({"HEAD"} if "GET" in {m.upper() for m in methods} else set())
        regex, self.converters = "", {}
        for piece in re.split(r"(<[^>]+>)", rule):
            m = re.fullmatch(r"<(?:(\w+):)?(\w+)>", piece)
            if m:
                kind, name = m.group(1) or "string", m.group(2)
                pattern, convert = _CONVERTERS[kind]
                regex += f"(?P<{name}>{pattern})"
                self.converters[name] = convert
            else:
                regex += re.escape(piece)
        self.regex = re.compile(regex)

    def match(self, path):
        m = self.regex.fullmatch(path)
        if not m:
            return None
        return {k: self.converters[k](v) for k, v in m.groupdict().items()}

    def build(self, values):
        def fill(m):
            return str(values.pop(m.group(2)))
        path = re.sub(r"<(?:(\w+):)?(\w+)>", fill, self.rule)
        return path + (f"?{urlencode(values)}" if values else "")


class _Setup:
    def route(self, rule, methods=("GET",), endpoint=None):
        def decorate(view):
            self._add(rule, endpoint or view.__name__, view, methods)
            return view
        return decorate

    def get(self, rule, **kw):
        return self.route(rule, methods=["GET"], **kw)

    def post(self, rule, **kw):
        return self.route(rule, methods=["POST"], **kw)

    def put(self, rule, **kw):
        return self.route(rule, methods=["PUT"], **kw)

    def patch(self, rule, **kw):
        return self.route(rule, methods=["PATCH"], **kw)

    def delete(self, rule, **kw):
        return self.route(rule, methods=["DELETE"], **kw)


class Blueprint(_Setup):
    def __init__(self, name, import_name, url_prefix=""):
        self.name, self.url_prefix = name, url_prefix
        self.deferred = []

    def _add(self, rule, endpoint, view, methods):
        self.deferred.append((rule, endpoint, view, methods))


class _App(_Setup):
    def __init__(self, import_name):
        self.import_name = import_name
        self.config = {"TESTING": False}
        self.rules = []
        self.before_funcs, self.after_funcs, self.error_handlers = [], [], {}

    def _add(self, rule, endpoint, view, methods):
        self.rules.append(_Rule(rule, endpoint, view, methods))

    def register_blueprint(self, bp, url_prefix=None):
        prefix = url_prefix if url_prefix is not None else bp.url_prefix
        for rule, endpoint, view, methods in bp.deferred:
            self._add(prefix + rule, f"{bp.name}.{endpoint}", view, methods)

    def before_request(self, f):
        self.before_funcs.append(f)
        return f

    def after_request(self, f):
        self.after_funcs.append(f)
        return f

    def errorhandler(self, code):
        def decorate(f):
            self.error_handlers[code] = f
            return f
        return decorate

    def url_for(self, endpoint, **values):
        for r in self.rules:
            if r.endpoint == endpoint:
                return r.build(dict(values))
        raise LookupError(f"no endpoint {endpoint!r}")

    @staticmethod
    def _to_response(rv):
        status, headers = None, None
        if isinstance(rv, tuple):
            if len(rv) == 3:
                rv, status, headers = rv
            elif isinstance(rv[1], (dict, list)):
                rv, headers = rv
            else:
                rv, status = rv
        if isinstance(rv, Response):
            response = rv
        elif isinstance(rv, (dict, list)):
            response = jsonify(rv)
        elif isinstance(rv, (str, bytes)):
            response = Response(rv)
        elif rv is None:
            raise TypeError("the view function did not return a valid response")
        else:
            raise TypeError(f"can't turn {type(rv).__name__} into a response")
        if status is not None:
            response.status_code = status
        for k, v in (headers.items() if isinstance(headers, dict) else headers or []):
            response.headers[k] = v
        return response

    def _error(self, e):
        handler = self.error_handlers.get(e.code)
        if handler:
            return self._to_response(handler(e))
        body = f"<h1>{e.code} {e.name}</h1><p>{e.description}</p>"
        return Response(body, e.code)

    def _dispatch(self, req):
        path = req.path
        allowed = set()
        for r in self.rules:
            values = r.match(path)
            if values is None:
                continue
            if req.method in r.methods:
                req.view_args = values
                return r.view(**values)
            allowed |= r.methods
        if allowed:
            raise _MethodNotAllowed(sorted(allowed | {"OPTIONS"}))
        abort(404, "The requested URL was not found on the server.")

    def full_dispatch(self, environ):
        req = Request(environ)
        token = _current.set((req, _G(), self))
        try:
            try:
                rv = None
                for f in self.before_funcs:
                    rv = f()
                    if rv is not None:
                        break
                if rv is None:
                    rv = self._dispatch(req)
                response = self._to_response(rv)
            except _MethodNotAllowed as e:
                response = self._error(HTTPException(405))
                response.headers["Allow"] = ", ".join(e.allowed)
            except HTTPException as e:
                response = self._error(e)
            except Exception as e:
                if self.config.get("TESTING") and 500 not in self.error_handlers:
                    raise
                response = self._error(HTTPException(500)) if 500 not in self.error_handlers else self._to_response(self.error_handlers[500](e))
                response.status_code = 500
            for f in self.after_funcs:
                response = f(response) or response
            if req.method == "HEAD":
                response.headers["Content-Length"] = len(response.data)
                response.data = b""
            return response
        finally:
            _current.reset(token)

    def __call__(self, environ, start_response):
        response = self.full_dispatch(environ)
        if "Content-Length" not in response.headers:
            response.headers["Content-Length"] = len(response.data)
        start_response(response.status, response.headers.items())
        return [response.data]

    def test_client(self):
        return _TestClient(self)


class _MethodNotAllowed(Exception):
    def __init__(self, allowed):
        self.allowed = allowed

class _TestClient:
    def __init__(self, app):
        self.app = app
        self.cookies = {}

    def open(self, path, method="GET", json=None, data=None, headers=None, query_string=None):
        url = urlsplit(path)
        query = url.query
        if query_string:
            query = (query + "&" if query else "") + urlencode(query_string, doseq=True)
        body, ctype = b"", None
        if json is not None:
            import json as _json
            body, ctype = _json.dumps(json).encode("utf-8"), "application/json"
        elif isinstance(data, dict):
            body, ctype = urlencode(data, doseq=True).encode("utf-8"), "application/x-www-form-urlencoded"
        elif data is not None:
            body = data.encode("utf-8") if isinstance(data, str) else data
        environ = {"REQUEST_METHOD": method.upper(), "PATH_INFO": url.path or "/", "QUERY_STRING": query,
                   "SERVER_NAME": "localhost", "SERVER_PORT": "80", "wsgi.url_scheme": "http",
                   "wsgi.input": io.BytesIO(body), "CONTENT_LENGTH": str(len(body))}
        headers = dict(headers or {})
        if ctype and not any(k.lower() == "content-type" for k in headers):
            environ["CONTENT_TYPE"] = ctype
        for k, v in headers.items():
            if k.lower() == "content-type":
                environ["CONTENT_TYPE"] = v
            else:
                environ["HTTP_" + k.upper().replace("-", "_")] = v
        if self.cookies:
            environ["HTTP_COOKIE"] = "; ".join(f"{k}={v}" for k, v in self.cookies.items())
        response = self.app.full_dispatch(environ)
        for k, v in response.headers.items():
            if k.lower() == "set-cookie":
                name, _, rest = v.partition("=")
                self.cookies[name] = rest.split(";")[0]
        return response

    def get(self, path, **kw):
        return self.open(path, "GET", **kw)

    def post(self, path, **kw):
        return self.open(path, "POST", **kw)

    def put(self, path, **kw):
        return self.open(path, "PUT", **kw)

    def patch(self, path, **kw):
        return self.open(path, "PATCH", **kw)

    def delete(self, path, **kw):
        return self.open(path, "DELETE", **kw)

Warm-up

Half of these routes put the action in the URL, where it duplicates (or contradicts) the method: POST /orders/7/delete is a DELETE in disguise. In REST the URL names a thing, the method says what to do to it, and the whole API becomes guessable.

Resources and methods

Resources are the nouns of an API: orders, customers, products. Each has a URL:

  • a collection: /orders; an item in it: /orders/7; plural nouns, lower case, with hyphens between words (/order-items);
  • related things nested one level: /customers/12/orders (Ada’s orders); deeper nesting gets unwieldy, so link instead;
  • no verbs, no file extensions (.json), no trailing slashes, and query strings for filtering, sorting and paging (/orders?status=shipped&sort=-placed).

Methods map onto operations:

Operation Request Success
list GET /orders 200 with the list
read GET /orders/7 200; 404 if it doesn’t exist
create POST /orders 201 with the new order and Location: /orders/42
replace PUT /orders/7 200 (or 201 if it created it)
update part PATCH /orders/7 200 with the result
delete DELETE /orders/7 204, no body

Some actions aren’t naturally create, read, update or delete: cancelling an order, sending an invoice. Model them as a resource of their own (POST /orders/7/cancellation, creating a cancellation) or as a state change (PATCH /orders/7 with {"status": "cancelled"}), rather than a verb in the path.

Responses and errors

Status codes must tell the truth. A 200 with {"error": "not found"} in the body breaks every client, cache and monitoring tool that trusts the status. Use the codes from Week 76 Day 2: 400 (malformed), 401 (who are you?), 403 (not allowed), 404, 409 (conflicts with the current state, such as a duplicate), 412 (a precondition failed, below), 422 (well-formed but invalid), 429, and 5xx only for the server’s own failures.

Consistent representations. One naming style for fields everywhere (snake_case or camelCase, never both), ISO 8601 dates and times with time zones (2026-11-10T09:30:00Z), money as integers in the smallest unit (or decimal strings) with a currency, and the same shape for a thing wherever it appears.

Errors as problem details. RFC 9457 standardises an error body with the media type application/problem+json:

{
  "type": "https://api.example.com/problems/insufficient-stock",
  "title": "Not enough stock",
  "status": 409,
  "detail": "Only 2 left of product 7, but 5 were requested.",
  "instance": "/orders/1042"
}

type identifies the kind of problem (a URL that documents it; about:blank when there’s nothing more than the status), title is a short summary of that kind, detail explains this occurrence, and extra fields (such as a list of validation errors) are allowed. Clients can handle every endpoint’s errors with one piece of code.

Safe retries and concurrent updates

Idempotency keys. A POST that timed out may or may not have worked (Week 76 Day 3). To make it safe to retry, the client sends a unique Idempotency-Key header (a random UUID) with each new operation; the server stores the response it gave for each key, and when the same key arrives again, it returns the stored response instead of doing the work twice. If the same key arrives with a different body, that’s a client bug: refuse it (422 or 409). Keys expire after a day or so. Payment APIs work this way.

Conditional updates. Two people edit the same order: both read it, both change it, the second save silently overwrites the first (Week 75 Day 5’s lost update, over HTTP). The fix is optimistic concurrency with ETags: every representation has an ETag (a version, often a hash of its content); a client updating it sends If-Match: "the-etag-it-read"; the server compares it with the current ETag and refuses with 412 Precondition Failed if they differ (someone changed it meanwhile). Servers can require it, answering 428 Precondition Required to updates without If-Match.

Pagination and versions

Pagination (Week 76 Day 3, from the client’s side). Offer a limit (with a maximum) and either page numbers or, better for large or changing data, cursors: an opaque string the client hands back to get the next page. Encode the cursor (for instance, base64 of a small JSON object holding the last id seen) so clients can’t depend on its insides, and validate it when it comes back: a tampered or malformed cursor is a 400, not a crash. Return the next cursor (or null) in the body, or a Link header.

Evolving an API. Clients you can’t update depend on every detail of your responses. Additive changes are safe: a new endpoint, a new optional parameter, a new field in a response (clients should ignore fields they don’t know). Breaking changes are not: removing or renaming a field or endpoint, changing a type, making an optional parameter required, changing what a status code means. Make breaking changes in a new version (/v2/orders, or a version header) while keeping the old one running for a published deprecation period, announced with headers such as Deprecation and Sunset.

Describe it. An OpenAPI document (FastAPI writes one for you at /openapi.json) lists every endpoint, parameter, body and response, and drives documentation, client generators and contract tests.

Key idea

REST URLs name resources (plural nouns, lower case, nested one level), never actions; methods say what to do: GET to list or read, POST to create (201 and Location), PUT to replace, PATCH to change part, DELETE (204); awkward actions become resources or state changes. Status codes must be honest (no 200 with an error inside). Keep representations consistent (one field-naming style, ISO dates, money as integers). Return errors as problem details (application/problem+json with type, title, status, detail, instance). Idempotency-Key headers make POST retries safe by replaying the stored response; ETags with If-Match prevent lost updates (412, or 428 when it’s missing). Paginate with limited, opaque, validated cursors. Additive changes are safe; breaking changes need a new version and a deprecation period; describe the API with OpenAPI.

Challenge: a well-behaved resource

Write create_app(clock=lambda: 0): a Flask (miniflask) app with an orders resource (orders have id, item (a non-empty string) and qty (an integer from 1 to 100)). Every error is a problem-details response (Content-Type: application/problem+json, with type about:blank, title, status, detail and instance set to the request’s path). GET /orders lists them (by id). POST /orders creates one (201, Location, and an ETag header); invalid bodies get 422 (detail "item must be a non-empty string and qty an integer from 1 to 100"); and an Idempotency-Key header makes retries safe: the same key and body replay the stored response (with the same status, body and Location, and a header Idempotent-Replayed: true), while the same key with a different body gets 422 with detail "Idempotency-Key reused with a different body". GET /orders/<id> returns the order with its ETag, or 304 for a matching If-None-Match. PATCH /orders/<id> requires If-Match (428, detail "If-Match is required"), refuses a stale one (412, detail "the order has changed"), validates the merged order (422), and returns it with its new ETag. Missing orders get 404 (detail "no order N").
