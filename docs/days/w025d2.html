money.py
"""Reading and showing money and categories."""
import re

AMOUNT = re.compile(r"£?\s*(\d{1,3}(?:,\d{3})+|\d+)(?:\.(\d{1,2}))?")


def parse_amount(text):
    """Text such as '£1,250.50' as whole pence. Raises ValueError with a readable message."""
    text = text.strip()
    if not text.lstrip("£").strip():
        raise ValueError("missing amount")
    m = AMOUNT.fullmatch(text)
    if m is None:
        raise ValueError(f"not an amount: {text!r}")
    total = int(m.group(1).replace(",", "")) * 100 + int((m.group(2) or "").ljust(2, "0"))
    if total == 0:
        raise ValueError("amount must be more than £0")
    return total


def format_pence(pence):
    """1250 as '£12.50', -350 as '-£3.50'."""
    sign = "-" if pence < 0 else ""
    pounds, rest = divmod(abs(pence), 100)
    return f"{sign}£{pounds:,}.{rest:02d}"


def clean_category(text):
    """' Eating   Out ' as 'eating out'. Raises ValueError if it isn't words of letters."""
    shown = text.strip()
    if not shown:
        raise ValueError("missing category")
    cleaned = " ".join(shown.lower().split())
    if len(cleaned) > 20 or not re.fullmatch(r"[a-z]+( [a-z]+)*", cleaned):
        raise ValueError(f"bad category: {shown!r}")
    return cleaned
dates.py
"""Reading dates that people type."""
from datetime import datetime, timedelta

FORMATS = ["%Y-%m-%d", "%d/%m/%Y"]


def parse_date(text, today):
    """'2026-11-02', '02/11/2026', 'today', 'yesterday' or '' as ISO text. No future dates."""
    shown = text.strip()
    word = shown.lower()
    if word in ("", "today"):
        when = today
    elif word == "yesterday":
        when = today - timedelta(days=1)
    else:
        when = None
        for fmt in FORMATS:
            try:
                when = datetime.strptime(word, fmt).date()
                break
            except ValueError:
                pass
        if when is None:
            raise ValueError(f"not a date: {shown!r}")
    if when > today:
        raise ValueError(f"date is in the future: {when.isoformat()}")
    return when.isoformat()
core.py
"""The expense tracker's data and the operations on it."""
from money import parse_amount, clean_category
from dates import parse_date


def new_data():
    return {"expenses": [], "budgets": {}, "next_id": 1}


def add_expense(data, amount_text, category_text, note, date_text, today):
    """Check every field, then add the expense, or raise one ValueError listing every problem."""
    problems = []
    amount = category = when = None
    try:
        amount = parse_amount(amount_text)
    except ValueError as e:
        problems.append(str(e))
    try:
        category = clean_category(category_text)
    except ValueError as e:
        problems.append(str(e))
    try:
        when = parse_date(date_text, today)
    except ValueError as e:
        problems.append(str(e))
    note = note.strip()
    if len(note) > 60:
        problems.append("note is longer than 60 characters")
    if problems:
        raise ValueError("; ".join(problems))
    expense = {"id": data["next_id"], "date": when, "amount": amount, "category": category, "note": note}
    data["expenses"].append(expense)
    data["next_id"] += 1
    return expense


def remove_expense(data, expense_id):
    for i, e in enumerate(data["expenses"]):
        if e["id"] == expense_id:
            return data["expenses"].pop(i)
    raise KeyError(f"no expense #{expense_id}")

Step 7: months

Lists and summaries work by month, written "2026-11", the first 7 characters of an ISO date. People will type months in two ways, so they need reading and checking like everything else.

Step 10: budgets

A budget is a monthly limit for one category, stored in data["budgets"] as pence. It reuses yesterday’s checks, so budget Food £60 works just like adding an expense.

Exercise15 points
Step 10: set and remove budgets

Write set_budget(data, category_text, amount_text): clean both with clean_category and parse_amount, collecting problems like add_expense (category first, joined by "; ", nothing changed), then store the budget and return (category, pence). Setting a budget again replaces it. And remove_budget(data, category_text): clean the category, remove its budget and return the amount, or raise KeyError("no budget for 'fun'") (with the cleaned name).

Step 11: a month’s summary

The summary command is the heart of the tool. Its function gathers everything the interface will print, as data, and the interface only has to lay it out.

Exercise25 points
Step 11: summarise a month

Write month_summary(data, month) returning a dictionary with:

  • "month": the month;
  • "count" and "total": how many expenses that month, and their total;
  • "rows": one dictionary per category that has spending that month or a budget, with keys "category", "spent" (0 if none), "budget" (None if none) and "left" (budget minus spent, or None with no budget); most spent first, alphabetical for ties;
  • "over": the categories whose "left" is below 0, in the same order.

Use filter_expenses and by_category, given in the starter.

Step 12: month by month

A trend needs every month in the range, including any with no spending at all. A month with nothing isn’t missing data; it’s a month where you spent £0, and leaving it out would make a chart lie.

Exercise15 points
Step 12: monthly totals

Write next_month(month) ("2026-12" gives "2027-01") and monthly_totals(data): a list of (month, pence) pairs from the earliest month with spending to the latest, in order, including months in between with a total of 0. No expenses at all gives [].

core.py
"""The expense tracker's data and the operations on it."""
from money import parse_amount, clean_category
from dates import parse_date


def new_data():
    return {"expenses": [], "budgets": {}, "next_id": 1}


def add_expense(data, amount_text, category_text, note, date_text, today):
    """Check every field, then add the expense, or raise one ValueError listing every problem."""
    problems = []
    amount = category = when = None
    try:
        amount = parse_amount(amount_text)
    except ValueError as e:
        problems.append(str(e))
    try:
        category = clean_category(category_text)
    except ValueError as e:
        problems.append(str(e))
    try:
        when = parse_date(date_text, today)
    except ValueError as e:
        problems.append(str(e))
    note = note.strip()
    if len(note) > 60:
        problems.append("note is longer than 60 characters")
    if problems:
        raise ValueError("; ".join(problems))
    expense = {"id": data["next_id"], "date": when, "amount": amount, "category": category, "note": note}
    data["expenses"].append(expense)
    data["next_id"] += 1
    return expense


def remove_expense(data, expense_id):
    """Remove and return the expense with this id. KeyError if there isn't one."""
    for i, e in enumerate(data["expenses"]):
        if e["id"] == expense_id:
            return data["expenses"].pop(i)
    raise KeyError(f"no expense #{expense_id}")


def filter_expenses(data, month=None, category=None):
    """A new list of the matching expenses, newest first (by date, then id)."""
    chosen = [e for e in data["expenses"]
              if (month is None or e["date"][:7] == month)
              and (category is None or e["category"] == category)]
    return sorted(chosen, key=lambda e: (e["date"], e["id"]), reverse=True)


def total(expenses):
    return sum(e["amount"] for e in expenses)


def by_category(expenses):
    """(category, pence) pairs, largest first, alphabetical for ties."""
    totals = {}
    for e in expenses:
        totals[e["category"]] = totals.get(e["category"], 0) + e["amount"]
    return sorted(totals.items(), key=lambda pair: (-pair[1], pair[0]))


def month_summary(data, month):
    """Count, total, a row per category with spending or a budget, and the categories over budget."""
    expenses = filter_expenses(data, month=month)
    spent = dict(by_category(expenses))
    rows = []
    for category in set(spent) | set(data["budgets"]):
        budget = data["budgets"].get(category)
        used = spent.get(category, 0)
        rows.append({"category": category, "spent": used, "budget": budget,
                     "left": None if budget is None else budget - used})
    rows.sort(key=lambda r: (-r["spent"], r["category"]))
    return {"month": month, "count": len(expenses), "total": total(expenses), "rows": rows,
            "over": [r["category"] for r in rows if r["left"] is not None and r["left"] < 0]}
