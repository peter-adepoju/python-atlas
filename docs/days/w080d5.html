"""A small subset of Flask that runs in the page (Week 77).

The names and behaviour follow Flask's own, so code written against this module also runs with
the real Flask (pip install flask): Flask, request, g, jsonify, abort, redirect, make_response,
url_for, Blueprint, and app.test_client(). It's a teaching copy, not the real thing: no templates,
sessions, streaming or debugger.

import io
import json
import re
from contextvars import ContextVar
from http import HTTPStatus
from urllib.parse import parse_qs, urlencode, urlsplit

_current = ContextVar("request")


class HTTPException(Exception):
    def __init__(self, code, description=None):
        self.code = code
        self.name = HTTPStatus(code).phrase
        self.description = description or self.name
        super().__init__(f"{code} {self.name}: {self.description}")


def abort(code, description=None):
    raise HTTPException(code, description)


class Headers:
    """Case-insensitive headers."""
    def __init__(self, items=()):
        self._items = [(k, str(v)) for k, v in (items.items() if isinstance(items, dict) else items)]

    def get(self, name, default=None):
        for k, v in self._items:
            if k.lower() == name.lower():
                return v
        return default

    def __getitem__(self, name):
        value = self.get(name)
        if value is None:
            raise KeyError(name)
        return value

    def __contains__(self, name):
        return self.get(name) is not None

    def __setitem__(self, name, value):
        self._items = [(k, v) for k, v in self._items if k.lower() != name.lower()] + [(name, str(value))]

    def items(self):
        return list(self._items)


class Args(dict):
    """Query or form values: .get(name) gives the first value; .getlist(name) all of them."""
    def __init__(self, pairs):
        super().__init__()
        self._lists = {}
        for k, v in pairs:
            self._lists.setdefault(k, []).append(v)
            dict.setdefault(self, k, v)

    def get(self, key, default=None, type=None):
        if key not in self:
            return default
        value = self[key]
        if type is not None:
            try:
                return type(value)
            except (TypeError, ValueError):
                return default
        return value

    def getlist(self, key):
        return list(self._lists.get(key, []))


class Request:
    def __init__(self, environ):
        self.environ = environ
        self.method = environ["REQUEST_METHOD"].upper()
        self.path = environ.get("PATH_INFO", "/") or "/"
        self.args = Args(_pairs(environ.get("QUERY_STRING", "")))
        items = [(k[5:].replace("_", "-").title(), v) for k, v in environ.items() if k.startswith("HTTP_")]
        if environ.get("CONTENT_TYPE"):
            items.append(("Content-Type", environ["CONTENT_TYPE"]))
        self.headers = Headers(items)
        length = int(environ.get("CONTENT_LENGTH") or 0)
        self.data = environ["wsgi.input"].read(length) if length else b""
        self.content_type = environ.get("CONTENT_TYPE", "")
        self.form = Args(_pairs(self.data.decode("utf-8"))) if self.content_type.startswith("application/x-www-form-urlencoded") else Args([])
        self.view_args = {}
        cookie = environ.get("HTTP_COOKIE", "")
        self.cookies = dict(p.strip().split("=", 1) for p in cookie.split(";") if "=" in p)

    @property
    def is_json(self):
        return self.content_type.split(";")[0].strip() == "application/json"

    def get_json(self, silent=False, force=False):
        if not (self.is_json or force):
            if silent:
                return None
            abort(415, "expected a JSON body (Content-Type: application/json)")
        try:
            return json.loads(self.data or b"null")
        except ValueError:
            if silent:
                return None
            abort(400, "the body isn't valid JSON")

    @property
    def json(self):
        return self.get_json()


def _pairs(text):
    return [(k, v) for k, vs in parse_qs(text, keep_blank_values=True).items() for v in vs]


class _Proxy:
    def __init__(self, getter):
        object.__setattr__(self, "_getter", getter)

    def __getattr__(self, name):
        return getattr(self._getter(), name)

    def __setattr__(self, name, value):
        setattr(self._getter(), name, value)

class Response:
    def __init__(self, response=b"", status=200, headers=None, mimetype=None, content_type=None):
        if isinstance(response, str):
            response = response.encode("utf-8")
        self.data = response
        self.status_code = status
        self.headers = Headers(headers or {})
        if content_type or mimetype or "Content-Type" not in self.headers:
            ctype = content_type or (f"{mimetype}; charset=utf-8" if mimetype and mimetype.startswith("text/") else mimetype) or "text/html; charset=utf-8"
            self.headers["Content-Type"] = ctype

    @property
    def status(self):
        return f"{self.status_code} {HTTPStatus(self.status_code).phrase}"

    @property
    def text(self):
        return self.data.decode("utf-8")

    def get_json(self):
        return json.loads(self.data)

    @property
    def json(self):
        return self.get_json()

    def set_cookie(self, key, value="", max_age=None, path="/", secure=False, httponly=False, samesite=None):
        parts = [f"{key}={value}"]
        if max_age is not None:
            parts.append(f"Max-Age={max_age}")
        parts.append(f"Path={path}")
        if secure:
            parts.append("Secure")
        if httponly:
            parts.append("HttpOnly")
        if samesite:
            parts.append(f"SameSite={samesite}")
        self.headers._items.append(("Set-Cookie", "; ".join(parts)))

_CONVERTERS = {"int": (r"\d+", int), "float": (r"\d+(?:\.\d+)?", float), "path": (r".+", str), "string": (r"[^/]+", str)}


class _Rule:
    def __init__(self, rule, endpoint, view, methods):
        self.rule, self.endpoint, self.view = rule, endpoint, view
        self.methods = {m.upper() for m in methods} | ({"HEAD"} if "GET" in {m.upper() for m in methods} else set())
        regex, self.converters = "", {}
        for piece in re.split(r"(<[^>]+>)", rule):
            m = re.fullmatch(r"<(?:(\w+):)?(\w+)>", piece)
            if m:
                kind, name = m.group(1) or "string", m.group(2)
                pattern, convert = _CONVERTERS[kind]
                regex += f"(?P<{name}>{pattern})"
                self.converters[name] = convert
            else:
                regex += re.escape(piece)
        self.regex = re.compile(regex)

    def match(self, path):
        m = self.regex.fullmatch(path)
        if not m:
            return None
        return {k: self.converters[k](v) for k, v in m.groupdict().items()}

    def build(self, values):
        def fill(m):
            return str(values.pop(m.group(2)))
        path = re.sub(r"<(?:(\w+):)?(\w+)>", fill, self.rule)
        return path + (f"?{urlencode(values)}" if values else "")


class _Setup:
    def route(self, rule, methods=("GET",), endpoint=None):
        def decorate(view):
            self._add(rule, endpoint or view.__name__, view, methods)
            return view
        return decorate

    def get(self, rule, **kw):
        return self.route(rule, methods=["GET"], **kw)

    def post(self, rule, **kw):
        return self.route(rule, methods=["POST"], **kw)

    def put(self, rule, **kw):
        return self.route(rule, methods=["PUT"], **kw)

    def patch(self, rule, **kw):
        return self.route(rule, methods=["PATCH"], **kw)

    def delete(self, rule, **kw):
        return self.route(rule, methods=["DELETE"], **kw)


class Blueprint(_Setup):
    def __init__(self, name, import_name, url_prefix=""):
        self.name, self.url_prefix = name, url_prefix
        self.deferred = []

    def _add(self, rule, endpoint, view, methods):
        self.deferred.append((rule, endpoint, view, methods))


class _App(_Setup):
    def __init__(self, import_name):
        self.import_name = import_name
        self.config = {"TESTING": False}
        self.rules = []
        self.before_funcs, self.after_funcs, self.error_handlers = [], [], {}

    def _add(self, rule, endpoint, view, methods):
        self.rules.append(_Rule(rule, endpoint, view, methods))

    def register_blueprint(self, bp, url_prefix=None):
        prefix = url_prefix if url_prefix is not None else bp.url_prefix
        for rule, endpoint, view, methods in bp.deferred:
            self._add(prefix + rule, f"{bp.name}.{endpoint}", view, methods)

    def before_request(self, f):
        self.before_funcs.append(f)
        return f

    def after_request(self, f):
        self.after_funcs.append(f)
        return f

    def errorhandler(self, code):
        def decorate(f):
            self.error_handlers[code] = f
            return f
        return decorate

    def url_for(self, endpoint, **values):
        for r in self.rules:
            if r.endpoint == endpoint:
                return r.build(dict(values))
        raise LookupError(f"no endpoint {endpoint!r}")

    @staticmethod
    def _to_response(rv):
        status, headers = None, None
        if isinstance(rv, tuple):
            if len(rv) == 3:
                rv, status, headers = rv
            elif isinstance(rv[1], (dict, list)):
                rv, headers = rv
            else:
                rv, status = rv
        if isinstance(rv, Response):
            response = rv
        elif isinstance(rv, (dict, list)):
            response = jsonify(rv)
        elif isinstance(rv, (str, bytes)):
            response = Response(rv)
        elif rv is None:
            raise TypeError("the view function did not return a valid response")
        else:
            raise TypeError(f"can't turn {type(rv).__name__} into a response")
        if status is not None:
            response.status_code = status
        for k, v in (headers.items() if isinstance(headers, dict) else headers or []):
            response.headers[k] = v
        return response

    def _error(self, e):
        handler = self.error_handlers.get(e.code)
        if handler:
            return self._to_response(handler(e))
        body = f"<h1>{e.code} {e.name}</h1><p>{e.description}</p>"
        return Response(body, e.code)

    def _dispatch(self, req):
        path = req.path
        allowed = set()
        for r in self.rules:
            values = r.match(path)
            if values is None:
                continue
            if req.method in r.methods:
                req.view_args = values
                return r.view(**values)
            allowed |= r.methods
        if allowed:
            raise _MethodNotAllowed(sorted(allowed | {"OPTIONS"}))
        abort(404, "The requested URL was not found on the server.")

    def full_dispatch(self, environ):
        req = Request(environ)
        token = _current.set((req, _G(), self))
        try:
            try:
                rv = None
                for f in self.before_funcs:
                    rv = f()
                    if rv is not None:
                        break
                if rv is None:
                    rv = self._dispatch(req)
                response = self._to_response(rv)
            except _MethodNotAllowed as e:
                response = self._error(HTTPException(405))
                response.headers["Allow"] = ", ".join(e.allowed)
            except HTTPException as e:
                response = self._error(e)
            except Exception as e:
                if self.config.get("TESTING") and 500 not in self.error_handlers:
                    raise
                response = self._error(HTTPException(500)) if 500 not in self.error_handlers else self._to_response(self.error_handlers[500](e))
                response.status_code = 500
            for f in self.after_funcs:
                response = f(response) or response
            if req.method == "HEAD":
                response.headers["Content-Length"] = len(response.data)
                response.data = b""
            return response
        finally:
            _current.reset(token)

    def __call__(self, environ, start_response):
        response = self.full_dispatch(environ)
        if "Content-Length" not in response.headers:
            response.headers["Content-Length"] = len(response.data)
        start_response(response.status, response.headers.items())
        return [response.data]

    def test_client(self):
        return _TestClient(self)

    def __init__(self, allowed):
        self.allowed = allowed

    def open(self, path, method="GET", json=None, data=None, headers=None, query_string=None):
        url = urlsplit(path)
        query = url.query
        if query_string:
            query = (query + "&" if query else "") + urlencode(query_string, doseq=True)
        body, ctype = b"", None

            import json as _json
            body, ctype = _json.dumps(json).encode("utf-8"), "application/json"
        elif isinstance(data, dict):

        environ = {"REQUEST_METHOD": method.upper(), "PATH_INFO": url.path or "/", "QUERY_STRING": query,
                   "SERVER_NAME": "localhost", "SERVER_PORT": "80", "wsgi.url_scheme": "http",

        headers = dict(headers or {})
        if ctype and not any(k.lower() == "content-type" for k in headers):
            environ["CONTENT_TYPE"] = ctype
        for k, v in headers.items():
            if k.lower() == "content-type":
                environ["CONTENT_TYPE"] = v

            environ["HTTP_COOKIE"] = "; ".join(f"{k}={v}" for k, v in self.cookies.items())

        for k, v in response.headers.items():
            if k.lower() == "set-cookie":
                name, _, rest = v.partition("=")
                self.cookies[name] = rest.split(";")[0]
        return response

"""The helpdesk's rules, with no web or database code (Week 80, Day 1).

Everything here is plain data and pure functions, so it can be tested on its own and used by the
database layer, the web layer and the tests alike.
"""
import re
from http import HTTPStatus

ROLES = ("customer", "agent", "admin")
PRIORITIES = ("low", "normal", "high", "urgent")
STATUSES = ("open", "in_progress", "waiting", "resolved", "closed")
STAFF = {"agent", "admin"}

# status -> {new status: roles that may make the change}
TRANSITIONS = {
    "open": {"in_progress": {"agent", "admin"}, "closed": {"customer", "admin"}},
    "in_progress": {"waiting": {"agent", "admin"}, "resolved": {"agent", "admin"}},
    "waiting": {"in_progress": {"customer", "agent", "admin"}},
    "resolved": {"closed": {"customer", "admin"}, "in_progress": {"customer", "agent", "admin"}},
    "closed": {},
}

# action -> (roles always allowed, roles allowed only on their own tickets)
PERMISSIONS = {
    "create_ticket": ({"customer"}, set()),
    "view_ticket": ({"agent", "admin"}, {"customer"}),
    "list_all_tickets": ({"agent", "admin"}, set()),
    "edit_ticket": ({"admin"}, {"customer"}),
    "comment": ({"agent", "admin"}, {"customer"}),
    "internal_note": ({"agent", "admin"}, set()),
    "change_priority": ({"agent", "admin"}, set()),
    "assign": ({"agent", "admin"}, set()),
    "delete_ticket": ({"admin"}, set()),
    "view_stats": ({"agent", "admin"}, set()),
}


class ValidationError(Exception):
    """Raised with a dict of field -> message."""

    def __init__(self, errors):
        super().__init__("; ".join(f"{k}: {v}" for k, v in sorted(errors.items())))
        self.errors = errors


def validate_ticket(data, partial=False):
    """Check a ticket body from a client. partial=True for PATCH: only the fields present are checked."""
    if not isinstance(data, dict):
        raise ValidationError({"_": "expected a JSON object"})
    errors, clean = {}, {}
    unknown = sorted(set(data) - {"title", "body", "priority"})
    if unknown:
        errors["_"] = f"unexpected field(s): {', '.join(unknown)}"
    for field in ("title", "body"):
        if field not in data:
            if not partial:
                errors[field] = "required"
            continue
        value = data[field]
        limit = (3, 120) if field == "title" else (1, 5000)
        if not isinstance(value, str) or not limit[0] <= len(value.strip()) <= limit[1]:
            errors[field] = f"{limit[0]} to {limit[1]} characters"
        else:
            clean[field] = value.strip()
    if "priority" in data:
        if data["priority"] not in PRIORITIES:
            errors["priority"] = f"one of {', '.join(PRIORITIES)}"
        else:
            clean["priority"] = data["priority"]
    elif not partial:
        clean["priority"] = "normal"
    if partial and not data:
        errors["_"] = "nothing to change"
    if errors:
        raise ValidationError(errors)
    return clean


def can_transition(current, new, role):
    """True if someone with this role may move a ticket from `current` to `new`."""
    return role in TRANSITIONS.get(current, {}).get(new, set())


def allowed(role, action, is_owner=False):
    """True if the role may do the action (is_owner: the ticket is the user's own)."""
    always, own = PERMISSIONS[action]
    return role in always or (is_owner and role in own)


def problem(status, detail, title=None, **extra):
    """An RFC 9457 problem-details response: (body, status, headers)."""
    body = {"type": "about:blank", "title": title or HTTPStatus(status).phrase, "status": status, "detail": detail}
    body.update(extra)
    return body, status, {"Content-Type": "application/problem+json"}

    """The routes as a Markdown table, for the README."""

Every query uses parameters; every change that touches more than one row runs in a transaction.

import sqlite3

MIGRATIONS = [
    (1, "users", """
        CREATE TABLE users (
            id INTEGER PRIMARY KEY,

        CREATE TABLE tickets (
            id INTEGER PRIMARY KEY,

def connect(path=":memory:"):
    conn = sqlite3.connect(path)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn


def migrate(conn, migrations=MIGRATIONS):
    """Apply the migrations the database hasn't had yet, each in its own transaction. Returns their versions."""
    conn.execute("CREATE TABLE IF NOT EXISTS schema_migrations (version INTEGER PRIMARY KEY, name TEXT NOT NULL)")
    conn.commit()
    versions = [v for v, _, _ in migrations]
    if versions != list(range(1, len(versions) + 1)):
        raise ValueError("migrations must be numbered 1, 2, 3... with no gaps")
    current = conn.execute("SELECT COALESCE(MAX(version), 0) FROM schema_migrations").fetchone()[0]
    if current > len(versions):
        raise RuntimeError(f"the database is at version {current}, newer than this code")
    applied = []
    for version, name, sql in migrations[current:]:
        try:
            conn.execute("BEGIN")
            for statement in sql.split(";"):
                if statement.strip():
                    conn.execute(statement)
            conn.execute("INSERT INTO schema_migrations (version, name) VALUES (?, ?)", (version, name))
            conn.commit()
        except Exception:
            conn.rollback()
            raise
        applied.append(version)
    return applied


def _dict(row):
    return dict(row) if row is not None else None

    conn.execute("INSERT INTO events (ticket_id, actor_id, kind, detail, at) VALUES (?, ?, ?, ?, ?)",
                 (ticket_id, actor_id, kind, detail, now))

def _like(text):
    return "%" + text.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_") + "%"


def list_tickets(conn, customer_id=None, status=None, priority=None, assignee_id=None, q=None, after_id=0, limit=20):
    """Tickets matching the filters, by id, after a cursor. Returns (items, next_after or None)."""
    where, values = ["id > ?"], [after_id]
    for column, value in (("customer_id", customer_id), ("status", status), ("priority", priority), ("assignee_id", assignee_id)):
        if value is not None:
            where.append(f"{column} = ?")
            values.append(value)
    if q:
        where.append("title LIKE ? ESCAPE '\\'")
        values.append(_like(q))
    rows = conn.execute(f"SELECT * FROM tickets WHERE {' AND '.join(where)} ORDER BY id LIMIT ?", values + [limit + 1]).fetchall()
    items = [dict(r) for r in rows[:limit]]
    return items, (items[-1]["id"] if len(rows) > limit else None)


def update_ticket(conn, ticket_id, changes, actor_id, now):
    if not changes:
        return get_ticket(conn, ticket_id)
    columns = sorted(changes)
    with conn:
        conn.execute(f"UPDATE tickets SET {', '.join(f'{c} = ?' for c in columns)}, updated_at = ? WHERE id = ?",
                     [changes[c] for c in columns] + [now, ticket_id])
        _event(conn, ticket_id, actor_id, "edited", ",".join(columns), now)
    return get_ticket(conn, ticket_id)


def change_status(conn, ticket_id, expected, new, actor_id, now):
    """Move a ticket from `expected` to `new`, unless someone changed its status first."""
    with conn:
        cur = conn.execute("UPDATE tickets SET status = ?, updated_at = ? WHERE id = ? AND status = ?",
                           (new, now, ticket_id, expected))
        if cur.rowcount != 1:
            raise ConflictError(f"ticket {ticket_id} is no longer {expected}")
        _event(conn, ticket_id, actor_id, "status", f"{expected}->{new}", now)
    return get_ticket(conn, ticket_id)

"""Passwords and tokens for the helpdesk (Week 79's tools, gathered in one place)."""
import base64
import hashlib
import hmac
import secrets


def pbkdf2_sha256(password, salt, iterations, dklen=32):
    base = hmac.new(password, digestmod=hashlib.sha256)
    out, block = b"", 1
    while len(out) < dklen:
        h = base.copy()
        h.update(salt + block.to_bytes(4, "big"))
        u = h.digest()
        total = int.from_bytes(u, "big")
        for _ in range(iterations - 1):
            h = base.copy()
            h.update(u)
            u = h.digest()
            total ^= int.from_bytes(u, "big")
        out += total.to_bytes(32, "big")
        block += 1
    return out[:dklen]


def hash_password(password, iterations):
    salt = secrets.token_bytes(16)
    digest = pbkdf2_sha256(password.encode(), salt, iterations)
    return f"pbkdf2_sha256${iterations}${base64.b64encode(salt).decode()}${base64.b64encode(digest).decode()}"


def check_password(password, stored):
    try:
        algorithm, iterations, salt, digest = stored.split("$")
        expected = pbkdf2_sha256(password.encode(), base64.b64decode(salt), int(iterations))
        return algorithm == "pbkdf2_sha256" and hmac.compare_digest(expected, base64.b64decode(digest))
    except ValueError:
        return False


def issue_token(user_id, expires, key):
    """A signed bearer token: USER_ID.EXPIRES.SIGNATURE."""
    payload = f"{user_id}.{expires}"
    return f"{payload}.{hmac.new(key, payload.encode(), hashlib.sha256).hexdigest()}"


def verify_token(token, key, now):
    """The user id in a valid, unexpired token, else None."""
    parts = token.split(".")
    if len(parts) != 3 or not parts[0].isdigit() or not parts[1].isdigit():
        return None
    payload = f"{parts[0]}.{parts[1]}"
    expected = hmac.new(key, payload.encode(), hashlib.sha256).hexdigest()
    if not hmac.compare_digest(parts[2], expected) or now >= int(parts[1]):
        return None
    return int(parts[0])

class ApiError(HTTPException):

        super().__init__(code, detail)
        self.extra = extra


def iso(t):
    return datetime.fromtimestamp(t, timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


def create_app(settings, conn, clock=time.time, log=print):

    key = settings["SECRET_KEY"].encode()
    token_seconds = settings.get("TOKEN_SECONDS", 3600)
    iterations = settings.get("PASSWORD_ITERATIONS", 600_000)
    dummy = hash_password(secrets.token_hex(8), iterations)   # checked for unknown emails, so they take as long as real ones

    # ---- step 11
    for code in ERROR_CODES:
        @app.errorhandler(code)
        def handle(e, code=code):
            if code == 500 and not isinstance(e, HTTPException):
                incident = secrets.token_hex(4)
                log(f"incident {incident}: {type(e).__name__}: {e}")
                return problem(500, "something went wrong", incident=incident)
            return problem(e.code, e.description, **getattr(e, "extra", {}))

    @app.get("/healthz")
    def healthz():
        return {"status": "ok"}

    @app.get("/readyz")
    def readyz():
        try:
            version = conn.execute("SELECT MAX(version) FROM schema_migrations").fetchone()[0]
        except Exception:
            version = None
        if version != len(db.MIGRATIONS):
            raise ApiError(503, "the database isn't ready")
        return {"status": "ready", "schema": version}

    # ---- step 12
    @app.before_request
    def authenticate():
        g.user = None
        header = request.headers.get("Authorization", "")
        if header.startswith("Bearer "):
            user_id = verify_token(header[7:], key, int(clock()))

        email = data.get("email", "").strip().lower() if isinstance(data.get("email"), str) else ""

Step 22: request ids and logs

When a customer reports “it said something went wrong at about ten past nine”, you need to find that request among thousands. So every request gets a request id:

  • if the request arrives with an X-Request-ID header (load balancers and other services often add one), use it, but only if it looks like an id (letters, digits and hyphens, 8 to 64 characters), because it will be written into logs and headers (Week 79);
  • otherwise make one with secrets.token_hex(8);
  • send it back in the response’s X-Request-ID header, so a client (or the support person) can quote it;
  • write one JSON log line per request with the id, method, path, status, duration in milliseconds and user id. The path only, never the query string or headers, which can hold tokens and personal data.

The duration uses an injected timer (in production time.perf_counter, which measures elapsed time and never jumps the way the wall clock can), so tests can control it. Week 77’s 500 handler already logs an incident id; with request ids in the same log, the two can be matched.

Step 23: the container

The helpdesk ships as a container image (Week 77): a Dockerfile describes how to build it, and every instance runs the same image. A good Dockerfile for a Python service:

FROM python:3.13-slim
WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir --require-hashes -r requirements.txt
COPY . .
RUN useradd --create-home helpdesk
USER helpdesk
EXPOSE 8000
HEALTHCHECK CMD python -c "import urllib.request; urllib.request.urlopen('http://localhost:8000/healthz')"
CMD ["gunicorn", "--bind", "0.0.0.0:8000", "helpdesk_wsgi:app"]

The habits it follows, each one a common review comment:

  • Pin the base image (python:3.13-slim, not python or python:latest), so a rebuild next month gets the same Python. (Pinning to an image digest is stricter still.)
  • No secrets in ENV or ARG: they’re stored in the image’s layers, readable by anyone who can pull it. Secrets arrive at run time from the platform (Week 79).
  • Copy the requirements and install them before copying the code, so Docker’s layer cache reuses the slow install step whenever only the code changed.
  • pip install with --require-hashes (Week 79) and --no-cache-dir (no point keeping pip’s cache in the image).
  • Don’t run as root: create a user and switch to it with USER, so a compromised app can’t change the container’s own files.
  • A HEALTHCHECK, so the container runtime knows when the process is stuck.
  • CMD in exec form (a JSON list), so the server is the container’s main process and receives the stop signal directly, shutting down cleanly instead of being killed.

Step 24: changing a live database

During a release, old and new instances run at the same time against one database, and the migration runs before any new instance starts. So every migration must leave a database that the old code can still use. That rules out, in a single release:

  • dropping a table or column the old code reads;
  • renaming a table or column (to the old code, that’s a drop);
  • adding a NOT NULL column without a default (the old code’s INSERTs don’t mention it, so they all fail).

Adding tables, adding nullable columns or columns with defaults, and creating indexes are all safe.

So how do you rename a column? In steps, across releases, a pattern called expand and contract:

  1. Expand: add the new column; release code that writes both columns and still reads the old one.
  2. Migrate: copy the old values into the new column for existing rows; release code that reads the new column (still writing both, in case of a rollback).
  3. Contract: once no running code reads the old column, release code that uses only the new one, then drop the old column.

Each step is safe on its own, and you can stop or roll back between any two. It takes three releases instead of one; that’s the price of never being down.

Step 25: the release

Two last decisions before a release goes out: what it’s called, and whether it may go.

Semantic versioning (Week 73), for an API: MAJOR.MINOR.PATCH. A breaking change (removing an endpoint or field, changing what a status means) bumps MAJOR and resets the others; a new feature that existing clients can ignore bumps MINOR; only fixes bump PATCH. The changelog’s entries are labelled with their kind, so the next version can be worked out rather than argued about.

A release gate is the list of things that must be true before anything ships, checked by code so nobody skips one under pressure: CI is green; no migration would break the running version (Step 24); no serious vulnerability (CVSS 7.0 or more, Week 79) in the dependencies; the changelog has an entry for this version (Week 73); and the Dockerfile is clean (Step 23). The gate returns every reason at once, like every other check this week.

Exercise10 points
Step 25: the release

Write next_version(current, changes): changes is a list of (kind, description) with kinds "breaking", "feature" or "fix"; return the next version text (a breaking change bumps MAJOR, else a feature bumps MINOR, else PATCH; lower numbers reset to 0); an empty list raises ValueError("nothing to release"). Then release_gate(facts): a list of reasons the release can’t go, in this order: "CI isn't green" (unless facts["ci_green"] is true); "N migration statement(s) would break the running version" (if migration_problems is non-empty); "N serious vulnerability(ies) in dependencies" (findings in audit with severity of at least 7.0); "no changelog entry for VERSION" (if version isn’t in changelog_versions); "the Dockerfile has problems" (if dockerfile_problems is non-empty). Missing keys count as empty or false.
